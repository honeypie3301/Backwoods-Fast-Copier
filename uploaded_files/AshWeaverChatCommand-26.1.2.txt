		static String match(String rawText) {
			if (rawText == null || rawText.length() > 96) return null;
			String text = rawText.trim().toLowerCase(Locale.ROOT);
			int end = text.length();
			while (end > 0 && ".!?".indexOf(text.charAt(end - 1)) >= 0) end--;
			StringBuilder normalized = new StringBuilder(end);
			boolean space = true;
			for (int i = 0; i < end; i++) {
				char ch = text.charAt(i);
				if (ch >= 'a' && ch <= 'z') { normalized.append(ch); space = false; }
				else if ((ch == '\'' || ch == '\u2019') && i > 0 && i + 1 < end
						&& text.charAt(i - 1) >= 'a' && text.charAt(i - 1) <= 'z'
						&& text.charAt(i + 1) >= 'a' && text.charAt(i + 1) <= 'z') { /* I'm -> im */ }
				else if (ch == ' ' || ch == '\t' || ch == ',' || ch == ':') {
					if (!space) { normalized.append(' '); space = true; }
				} else return null; // Quoted commands, hidden characters, slash commands and other scripts are not guessed.
			}
			String command = normalized.toString().trim();
			if (command.isEmpty()) return null;
			String[] tokens = command.split(" ");
			int skip = 0;
			if (tokens.length > 2 && tokens[0].equals("ash") && isAddress(tokens[1])) skip = 2;
			else if (tokens.length > 1 && (isAddress(tokens[0]) || tokens[0].equals("ashweaver"))) skip = 1;
			if (skip > 0) command = String.join(" ", java.util.Arrays.copyOfRange(tokens, skip, tokens.length));
			if (isApology(command, skip > 0)) return "apologize";
			for (String[] phrase : PHRASES) if (phrase[0].equals(command)) return phrase[1];
			// Exact unaddressed commands remain compatible; typo correction requires a clear addressee.
			if (skip == 0) return null;
			String[] input = java.util.Arrays.copyOfRange(tokens, skip, tokens.length);
			int best = Integer.MAX_VALUE;
			String winner = null;
			boolean ambiguous = false;
			for (int i = 0; i < PHRASES.length; i++) {
				String[] expected = WORDS[i];
				if (input.length != expected.length) continue;
				String action = PHRASES[i][1];
				if (action.equals("forget home") || action.equals("restore home")) continue; // Memory changes must be exact.
				boolean home = action.contains("home");
				int allowance = home || PHRASES[i][0].length() < 14 ? 1 : 2;
				int score = 0;
				for (int word = 0; word < input.length; word++) {
					String wanted = expected[word];
					String actual = input[word];
					if (wanted.equals(actual)) continue;
					// Short words carry meaning: never turn "say" into "stay" or "some" into "home".
					if (wanted.length() <= 3 || wanted.length() == 4 && !shortTypo(actual, wanted)) {
						score = allowance + 1; break;
					}
					int limit = wanted.length() >= 8 && !home ? 2 : 1;
					int edits = distance(actual, wanted, limit);
					if (edits > limit) { score = allowance + 1; break; }
					score += edits;
					if (score > allowance) break;
				}
				if (score > allowance) continue;
				if (score < best) { best = score; winner = action; ambiguous = false; }
				else if (score == best && !action.equals(winner)) ambiguous = true;
			}
			return ambiguous ? null : winner;
		}


		private static boolean isApology(String command, boolean addressed) {
			String[] input = command.split(" ");
			int first = 0, last = input.length;
			if (first < last && (input[first].equals("hey") || input[first].equals("hi"))) first++;
			if (last - first >= 3 && input[first].equals("ash") && isAddress(input[first + 1])) {
				first += 2; addressed = true;
			} else if (last - first >= 2 && (isAddress(input[first]) || input[first].equals("ashweaver"))) {
				first++; addressed = true;
			}
			if (last - first >= 3 && input[last - 2].equals("ash") && isAddress(input[last - 1])) {
				last -= 2; addressed = true;
			} else if (last - first >= 2 && (isAddress(input[last - 1]) || input[last - 1].equals("ashweaver"))) {
				last--; addressed = true;
			}
			if (first < last && input[first].equals("please")) first++;
			if (first < last && input[last - 1].equals("please")) last--;
			if (first < last && (input[last - 1].equals("buddy") || input[last - 1].equals("friend")
					|| input[last - 1].equals("dude"))) last--;
			if (last - first <= 0 || last - first > 10) return false;
			// Whole phrases only: "not sorry", quotes and unrelated sentences cannot become apologies.
			if (addressed && last - first == 1 && input[first].equals("mb")) return true;
			for (String[] phrase : APOLOGIES) {
				if (last - first != phrase.length) continue;
				int budget = phrase.length >= 3 ? 2 : 1, score = 0;
				boolean matches = true;
				for (int i = 0; i < phrase.length; i++) {
					String actual = input[first + i], wanted = phrase[i];
					if (actual.equals(wanted)) continue;
					if (wanted.length() <= 3) {
						if (wanted.length() == 3 && shortTypo(actual, wanted)) score++;
						else { matches = false; break; }
					} else {
						// Unaddressed one-word typos need an apology-like initial, not e.g. "worry".
						if (!addressed && phrase.length == 1 && (actual.isEmpty()
								|| actual.charAt(0) != wanted.charAt(0))) { matches = false; break; }
						int limit = wanted.length() >= 8 ? 2 : 1;
						int edits = distance(actual, wanted, limit);
						if (edits > limit) { matches = false; break; }
						score += edits;
					}
					if (score > budget) { matches = false; break; }
				}
				if (matches) return true;
			}
			return false;
		}

		private static boolean isAddress(String word) {
			return word.equals("weaver") || word.length() >= 5 && word.length() <= 7
					&& distance(word, "weaver", 1) == 1;
		}

		private static boolean shortTypo(String actual, String wanted) {
			if (actual.length() == wanted.length()) {
				for (int i = 0; i + 1 < actual.length(); i++)
					if (actual.charAt(i) == wanted.charAt(i + 1) && actual.charAt(i + 1) == wanted.charAt(i)
							&& actual.substring(0, i).equals(wanted.substring(0, i))
							&& actual.substring(i + 2).equals(wanted.substring(i + 2))) return true;
			} else if (actual.length() == wanted.length() + 1) {
				for (int i = 1; i < actual.length(); i++)
					if (actual.charAt(i) == actual.charAt(i - 1)
							&& (actual.substring(0, i) + actual.substring(i + 1)).equals(wanted)) return true;
			}
			return false;
		}

		// Bounded edit distance with adjacent-key transpositions treated as one typo.
		private static int distance(String actual, String wanted, int limit) {
			if (actual.length() > 16 || Math.abs(actual.length() - wanted.length()) > limit) return limit + 1;
			int[] before = new int[wanted.length() + 1];
			int[] previous = new int[wanted.length() + 1];
			int[] current = new int[wanted.length() + 1];
			for (int j = 0; j <= wanted.length(); j++) previous[j] = j;
			for (int i = 1; i <= actual.length(); i++) {
				current[0] = i;
				for (int j = 1; j <= wanted.length(); j++) {
					int change = actual.charAt(i - 1) == wanted.charAt(j - 1) ? 0 : 1;
					current[j] = Math.min(previous[j] + 1, Math.min(current[j - 1] + 1, previous[j - 1] + change));
					if (i > 1 && j > 1 && actual.charAt(i - 1) == wanted.charAt(j - 2)
							&& actual.charAt(i - 2) == wanted.charAt(j - 1))
						current[j] = Math.min(current[j], before[j - 2] + 1);
				}
				int[] reuse = before; before = previous; previous = current; current = reuse;
			}
			return previous[wanted.length()];
		}
	}

	// One transient reply per Weaver; weak keys and UUID-only values avoid retaining unloaded entities.
	private static final java.util.Map<AshWeaverEntity, Conversation> CONVERSATIONS =
			java.util.Collections.synchronizedMap(new java.util.WeakHashMap<>());

	private static final class Conversation {
		java.util.UUID playerId;
		String dimension;
		String text;
		long due;
		long expires;
		long noticeAt;
		long lookUntil;
		long nextSpeechAt;
		boolean urgent;
		boolean silent;
	}

	private static boolean speechBusy(AshWeaverEntity weaver, boolean urgent) {
		var data = weaver.getPersistentData();
		int state = chatInt(data, KEY_STATE, 0);
		long now = weaver.level().getGameTime();
		boolean usingDoor = data.contains("ashWeaverDoorPending")
				&& now - chatLong(data, "ashWeaverDoorFaceAt", now) <= 20L
				|| data.contains("ashWeaverDoorPos") && data.contains("ashWeaverDoorCloseAt")
					&& now - chatLong(data, "ashWeaverDoorCloseAt", now) <= 20L;
		return !weaver.onGround() || weaver.isInWater() || weaver.hurtTime > 0
				|| state == 4 || state == 8 || state == 9 || state == 12
				|| usingDoor
				|| !urgent && (state == 2 || state == 7 || state == 10);
	}

	public static void processPendingReply(AshWeaverEntity weaver) {
		Conversation conversation = CONVERSATIONS.get(weaver);
		if (conversation == null) return;
		if (!(weaver.level() instanceof ServerLevel level) || !weaver.isAlive()) {
			CONVERSATIONS.remove(weaver);
			return;
		}
		long now = level.getGameTime();
		String owner = weaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer);
		ServerPlayer player = level.getServer().getPlayerList().getPlayer(conversation.playerId);
		if (player == null || !player.isAlive() || player.level() != level
				|| !owner.equals(conversation.playerId.toString())
				|| !level.dimension().toString().equals(conversation.dimension)
				|| weaver.distanceToSqr(player) > 32.0 * 32.0
				|| now > conversation.expires) {
			CONVERSATIONS.remove(weaver);
			return;
		}
		if (conversation.text == null || now < conversation.due || now < conversation.nextSpeechAt) return;
		if (conversation.silent) {
			conversation.text = null;
			conversation.lookUntil = 0L;
			return;
		}
		if (speechBusy(weaver, conversation.urgent)) return;
		String text = currentFeedback(weaver, conversation.text);
		conversation.text = null;
		conversation.nextSpeechAt = now + 8L;
		conversation.lookUntil = now + 6L;
		sendReply(player, weaver, text);
	}

	private static String currentFeedback(AshWeaverEntity weaver, String text) {
		var data = weaver.getPersistentData();
		boolean movementReply = text.equals("Following.") || text.equals("Waiting here.")
				|| text.equals("Staying home.") || text.equals("Watching for threats.") || text.equals("I'll protect you.");
		if (!movementReply) return text;
		if (chatBoolean(data, "ashWeaverAvoidOwner", false)
				|| AshWeaverOnEntityTickUpdateProcedure.relationship(weaver) <= -25.0) return "Keep your distance.";
		if (chatBoolean(data, "ashWeaverHomeStay", false)) return "Staying home.";
		if (chatBoolean(data, KEY_CHAT_STAY, false)) return "Waiting here.";
		return text.equals("Waiting here.") || text.equals("Staying home.") ? "Following." : text;
	}

	// Invoked after movement/look arbitration; never steals attention from an action or danger.
	public static void updateConversationLook(AshWeaverEntity weaver) {
		Conversation conversation = CONVERSATIONS.get(weaver);
		if (conversation == null || !(weaver.level() instanceof ServerLevel level)) return;
		long now = level.getGameTime();
		if (now < conversation.noticeAt || now > conversation.lookUntil || now > conversation.expires) return;
		var data = weaver.getPersistentData();
		int state = chatInt(data, KEY_STATE, 0);
		if (state != 0 && state != 1 && state != 11 || speechBusy(weaver, false)
				|| chatBoolean(data, "ashWeaverAvoidOwner", false) || conversation.silent
				|| weaver.swinging
				|| !weaver.getNavigation().isDone() || weaver.getDeltaMovement().horizontalDistanceSqr() > 0.01) return;
		ServerPlayer player = level.getServer().getPlayerList().getPlayer(conversation.playerId);
		if (player == null || player.level() != level || !player.isAlive()
				|| !player.getUUID().toString().equals(weaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))
				|| weaver.distanceToSqr(player) > 8.0 * 8.0 || !weaver.hasLineOfSight(player)) return;
		weaver.getLookControl().setLookAt(player, 14.0F, 10.0F);
	}

	@SubscribeEvent
	public static void onChat(ServerChatEvent event) {
		if (event == null || event.getPlayer() == null) return;
		ServerPlayer player = event.getPlayer();
		handleChat(player, (ServerLevel) player.level(), event.getRawText());
	}

	public static void cancelConversation(AshWeaverEntity weaver) {
		CONVERSATIONS.remove(weaver);
	}

	private static void handleChat(ServerPlayer player, ServerLevel level, String rawText) {
		String command = ChatCommandMatcher.match(rawText);
		if (command == null) return;
		if (!player.isAlive() || player.isSpectator()) return;
		long chatTime = level.getGameTime();
		var playerData = player.getPersistentData();
		if (chatTime < chatLong(playerData, "ashWeaverCommandAt", 0L)) return;
		playerData.putLong("ashWeaverCommandAt", chatTime
				+ (command.equals("help me") || command.equals("watch") ? 4L : 10L));

		boolean follow = command.equals("follow") || command.equals("follow me") || command.equals("come here");
		boolean stay = command.equals("stay") || command.equals("stay here") || command.equals("wait here");
		boolean watch = command.equals("watch") || command.equals("watch my back");
		boolean help = command.equals("help me") || command.equals("protect me");
		boolean status = command.equals("status");
		boolean apology = command.equals("apologize");
		String homeAction = switch (command) {
			case "this is home", "this is our home", "remember this place" -> "set";
			case "this is also home", "remember this area too" -> "extend";
			case "stay home", "stay at home" -> "stay";
			case "forget home", "forget this home" -> "forget";
			case "home status" -> "status";
			case "restore home" -> "restore";
			default -> "";
		};
		if (!follow && !stay && !watch && !help && !status && !apology && homeAction.isEmpty()) return;

		AshWeaverEntity weaver = findOwnedWeaver(player);
		if (weaver == null) return;

		CompoundTag data = weaver.getPersistentData();
		if (weaver.level() != level || weaver.distanceToSqr(player) > 32.0 * 32.0) return;
		long gameTime = level.getGameTime();
		if (apology) {
			// Reconciliation is heard even when wary; it never earns normal chat rewards or changes an order.
			reply(player, weaver, AshWeaverOnEntityTickUpdateProcedure.apologize(weaver, player));
			return;
		}
		AshWeaverOnEntityTickUpdateProcedure.noteConversation(weaver, gameTime);
		if (data.getBooleanOr("ashWeaverAvoidOwner", false) || AshWeaverOnEntityTickUpdateProcedure.relationship(weaver) <= -25.0) {
			if (stay) data.putBoolean(KEY_CHAT_STAY, true);
			reply(player, weaver, "Keep your distance.");
			return;
		}
		if (!homeAction.isEmpty()) {
			reply(player, weaver, AshWeaverOnEntityTickUpdateProcedure.homeCommand(weaver, player, homeAction));
			return;
		}
		if (follow || stay || watch || help) AshWeaverOnEntityTickUpdateProcedure.acceptMovementOrder(weaver);
		if (follow) {
			data.remove("ashWeaverHomeStay");
			data.remove(KEY_CHAT_STAY);
			if (data.getIntOr(KEY_STATE, 0) == 0) setState(data, 1, gameTime, 0L);
			reply(player, weaver, "Following.");
		} else if (stay) {
			data.remove("ashWeaverHomeStay");
			data.putBoolean(KEY_CHAT_STAY, true);
			reply(player, weaver, "Waiting here.");
		} else if (watch || help) {
			data.remove("ashWeaverHomeStay");
			data.remove(KEY_CHAT_STAY);
			data.putLong(KEY_NEXT_WARNING, 0L);
			if (data.getIntOr(KEY_STATE, 0) == 0) setState(data, 1, gameTime, 0L);
			reply(player, weaver, watch ? "Watching for threats." : "I'll protect you.");
		} else {
			reply(player, weaver, data.getBooleanOr("ashWeaverHomeStay", false) ? "Staying home."
					: data.getBooleanOr(KEY_CHAT_STAY, false) ? "Waiting here." : "Following.");
		}
	}



	private static void reply(ServerPlayer player, AshWeaverEntity weaver, String text) {
		if (text == null || text.isBlank() || !(weaver.level() instanceof ServerLevel level)) return;
		long now = level.getGameTime();
		var data = weaver.getPersistentData();
		double bond = AshWeaverOnEntityTickUpdateProcedure.relationship(weaver);
		boolean urgent = text.equals("I'll protect you.");
		int state = chatInt(data, KEY_STATE, 0);
		boolean wary = bond < -10.0 || chatBoolean(data, "ashWeaverAvoidOwner", false)
				|| state == 2 || state == 7 || state == 8 || state == 9 || state == 10
				|| weaver.getHealth() <= weaver.getMaxHealth() * 0.4F
				|| now < chatLong(data, "ashWeaverDistractionUntil", 0L);
		boolean unfamiliar = AshWeaverOnEntityTickUpdateProcedure.speechObfuscation(weaver) > 0.55;
		int min = urgent ? 4 : wary ? 20 : unfamiliar ? 18 : 8;
		int max = urgent ? 8 : wary ? 50 : unfamiliar ? 36 : 18;
		// Averaging two draws favors a natural middle rather than abrupt extreme delays.
		int delay = min + (weaver.getRandom().nextInt(max - min + 1)
				+ weaver.getRandom().nextInt(max - min + 1)) / 2;
		Conversation conversation = CONVERSATIONS.get(weaver);
		if (conversation == null) {
			conversation = new Conversation();
			CONVERSATIONS.put(weaver, conversation);
		}
		conversation.playerId = player.getUUID();
		conversation.dimension = level.dimension().toString();
		conversation.text = text; // A newer recognized request replaces, rather than queues behind, this reply.
		conversation.due = Math.max(now + delay, conversation.nextSpeechAt);
		conversation.expires = now + 120L;
		conversation.noticeAt = now + 2L + weaver.getRandom().nextInt(3);
		conversation.lookUntil = conversation.due + 6L;
		conversation.urgent = urgent;
		conversation.silent = !urgent && bond < -25.0 && weaver.getRandom().nextFloat() < 0.2F;
	}

	private static void sendReply(ServerPlayer player, AshWeaverEntity weaver, String text) {
		if (weaver.getPersistentData().getIntOr("ashWeaverState", 0) == 10) text = "Taking cover.";
		double fraction = AshWeaverOnEntityTickUpdateProcedure.speechObfuscation(weaver);
		int letters = 0;
		for (int i = 0; i < text.length(); i++) if (Character.isLetterOrDigit(text.charAt(i))) letters++;
		int hidden = (int) Math.round(letters * fraction);
		MutableComponent message = Component.empty();
		for (int i = 0; i < text.length(); i++) {
			char ch = text.charAt(i);
			boolean obscure = false;
			if (Character.isLetterOrDigit(ch)) {
				obscure = letters > 0 && weaver.getRandom().nextInt(letters) < hidden;
				letters--;
				if (obscure) hidden--;
			}
			MutableComponent part = Component.literal(String.valueOf(ch));
			if (obscure) part.withStyle(ChatFormatting.OBFUSCATED);
			message.append(part);
		}
		player.sendSystemMessage(message, true);
	}


	private static int chatInt(net.minecraft.nbt.CompoundTag data, String key, int fallback) {
		return data.getIntOr(key, fallback);
	}

	private static long chatLong(net.minecraft.nbt.CompoundTag data, String key, long fallback) {
		return data.getLongOr(key, fallback);
	}

	private static boolean chatBoolean(net.minecraft.nbt.CompoundTag data, String key, boolean fallback) {
		return data.getBooleanOr(key, fallback);
	}

	private static AshWeaverEntity findOwnedWeaver(ServerPlayer player) {
		MinecraftServer server = ((ServerLevel) player.level()).getServer();
		if (server == null) return null;
		String ownerId = player.getUUID().toString();
		AshWeaverEntity otherDimensionMatch = null;
		for (ServerLevel dimension : server.getAllLevels()) {
			for (Entity entity : dimension.getAllEntities()) {
				if (entity instanceof AshWeaverEntity candidate && candidate.isAlive()
						&& ownerId.equals(candidate.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))
						&& AshWeaverOnEntityTickUpdateProcedure.isCanonicalCompanion(candidate, ownerId)) {
					if (candidate.level() == player.level()) return candidate;
					if (otherDimensionMatch == null) otherDimensionMatch = candidate;
				}
			}
		}
		return otherDimensionMatch;
	}

	private static void setState(CompoundTag data, int state, long gameTime, long until) {
		data.putInt(KEY_STATE, state);
		data.putLong(KEY_STATE_STARTED, gameTime);
		data.putLong(KEY_STATE_UNTIL, until);
	}
