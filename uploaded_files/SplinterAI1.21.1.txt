	@SubscribeEvent
	public static void onEntityTickAfterGoals(EntityTickEvent.Post event) {
		if (event.getEntity() instanceof SplinterEntity member) {
			if (!member.level().isClientSide() && Math.floorMod(member.tickCount + member.getId(), 5) == 0)
				member.getEntityData().set(SplinterEntity.DATA_frozenByRose,
						checkNearbyRoseBlocks(member.level(), member) ? 1 : 0);
			if (member.getEntityData().get(SplinterEntity.DATA_frozenByRose) == 1) {
				setSpeed(member, 0);
				stopNav(member);
				Vec3 motion = member.getDeltaMovement();
				member.setDeltaMovement(0.0, motion.y, 0.0);
				return;
			}
			if (SplinterOnEntityTickUpdateProcedure.AdaptiveHive.finishAction(member, member.getEntityData().get(SplinterEntity.DATA_mineProgress))) return;
			SplinterHivemind.followFormationPost(member, member.getEntityData().get(SplinterEntity.DATA_mineProgress));
		}
	}


	@SubscribeEvent(priority = net.neoforged.bus.api.EventPriority.LOWEST)
	public static void onHiveNavigationTick(EntityTickEvent.Pre event) {
		if (event.getEntity() instanceof Mob member && member.level() instanceof ServerLevel
				&& SplinterHivemind.member(member)) {
            if (member instanceof BlindspotSplinterEntity && member.getPersistentData().contains("htMirrorUntil")) {
                Player target = member.getTarget() instanceof Player p ? p : null;
                if (!AdaptiveHive.validPlayer(target, (ServerLevel) member.level())
                        || AdaptiveHive.progress(member) != 0 || SplinterHivemind.frozen(member))
                    AdaptiveHive.endMirror(member, false);
            }
			AdaptiveHive.installNavigationGoals(member);
            AdaptiveHive.advanceCover(member);
            // Global fallback is variant-independent, including paths that returned before ordinary tactics.
            // Mining, roses and the afterimage lease remain exclusive; defendFor deduplicates each entity tick.
            if (member.getTarget() instanceof Player target && !AdaptiveHive.mirrorHidden(member)
                    && !member.getPersistentData().contains("htMirrorAttackUntil"))
                AdaptiveHive.defend(member, target, AdaptiveHive.progress(member));
			AdaptiveHive.updateVisuals(member, (ServerLevel) member.level(), false);
		}
	}


	@SubscribeEvent(priority = net.neoforged.bus.api.EventPriority.LOWEST)
	public static void onHiveVisualPost(EntityTickEvent.Post event) {
		if (event.getEntity() instanceof Mob member && member.level() instanceof ServerLevel level
				&& SplinterHivemind.member(member)) AdaptiveHive.updateVisuals(member, level, true);
	}

	@SubscribeEvent
	public static void onHiveVisualLeave(net.neoforged.neoforge.event.entity.EntityLeaveLevelEvent event) {
		if (event.getEntity() instanceof Mob member && event.getLevel() instanceof ServerLevel level
				&& SplinterHivemind.member(member)) {
			AdaptiveHive.endMirror(member, false);
			AdaptiveHive.clearMiningVisual(member, level);
			AdaptiveHive.BUILDING_HANDS.remove(member);
		}
	}


	@SubscribeEvent
	public static void onHiveVisualCleanup(net.neoforged.neoforge.event.tick.ServerTickEvent.Post event) {
		if (event.getServer().overworld().getGameTime() % 10 != 0) return;
		for (Mob member : new ArrayList<>(AdaptiveHive.MINING_VISUALS.keySet())) {
			if (member.level() instanceof ServerLevel level && level.getServer() == event.getServer()) {
				AdaptiveHive.MiningVisual visual = AdaptiveHive.MINING_VISUALS.get(member);
				if (visual != null && (member.isRemoved() || !member.isAlive()
						|| level.getGameTime() - visual.tick > 2)) AdaptiveHive.clearMiningVisual(member, level);
			}
		}
		for (Mob member : new ArrayList<>(AdaptiveHive.BUILDING_HANDS.keySet())) {
			if (member.level() instanceof ServerLevel level && level.getServer() == event.getServer())
				AdaptiveHive.updateVisuals(member, level, true);
		}
	}

	private static void execute(@Nullable Event event, LevelAccessor world, double x, double y, double z, Entity entity) {
		if (!(entity instanceof SplinterEntity splinter)) return;

		updatePlacedBlockCleanup(world, splinter, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState());

		if (IsRotNearbyConditionProcedure.execute(world, splinter)) {
			splinter.setTarget(null);
			splinter.setLastHurtByMob(null);
			setSpeed(splinter, ACTIVE_MOVE_SPEED);
			return;
		}

		if (splinter.getTarget() instanceof Player p && (p.isCreative() || p.isSpectator())) {
			splinter.setTarget(null);
		}
		if (splinter.getLastHurtByMob() instanceof Player p && (p.isCreative() || p.isSpectator())) {
			splinter.setLastHurtByMob(null);
		}

		if (!splinter.getPersistentData().getBoolean(K_PERSONALITY_INIT)) {
			int p = (splinter.getRandom().nextInt(100) < ANGEL_CHANCE_PERCENT) ? PERSONALITY_ANGEL : PERSONALITY_STALK_HUNT;
			splinter.getPersistentData().putInt(K_PERSONALITY, p);
			splinter.getPersistentData().putBoolean(K_PERSONALITY_INIT, true);
		}
		int personality = splinter.getPersistentData().getInt(K_PERSONALITY);

		if (splinter.isPassenger()) {
			Entity vehicle = splinter.getVehicle();
			if (vehicle instanceof net.minecraft.world.entity.vehicle.Boat || vehicle instanceof net.minecraft.world.entity.vehicle.ChestBoat) {
				splinter.stopRiding();
				splinter.setDeltaMovement(splinter.getDeltaMovement().add(0, 0.2, 0));
			}
		}

		int age = splinter.getEntityData().get(SplinterEntity.DATA_Age);
		if (age >= AGE_LAST) {
			splinter.discard();
			return;
		}

		boolean isDegraded = age >= AGE_THIRD_TO_LAST;
		boolean isCritical = age >= AGE_SECOND_TO_LAST;

		Player foundPlayerDirect = findNearestPlayerInRange(world, splinter.getX(), splinter.getY(), splinter.getZ(), TARGET_RANGE);
		if (world instanceof Level level && level.isClientSide()) {
			return;
		}
		LivingEntity foundPlayer = foundPlayerDirect;

		LivingEntity prioritizedMob = null;
		String attackerUUIDStr = splinter.getPersistentData().getString("splinter_attacker_uuid");
		if (!attackerUUIDStr.isEmpty() && world instanceof net.minecraft.server.level.ServerLevel serverLevel) {
			try {
				java.util.UUID attackerUUID = java.util.UUID.fromString(attackerUUIDStr);
				net.minecraft.world.entity.Entity foundEntity = serverLevel.getEntity(attackerUUID);
				if (foundEntity instanceof LivingEntity livingAttacker && livingAttacker.isAlive() && splinter.distanceToSqr(livingAttacker) < TARGET_RANGE * TARGET_RANGE) {
					boolean isWoodbound = livingAttacker.getType().is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.ResourceLocation.parse("the_backwoods:woodbound_entities")));
					if (!isWoodbound && !(livingAttacker instanceof Player)) {
						prioritizedMob = livingAttacker;
					}
				}
			} catch (Exception e) {}
		}

		LivingEntity attacker = splinter.getLastHurtByMob();
		if (attacker != null) {
			boolean isWoodbound = attacker.getType().is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.ResourceLocation.parse("the_backwoods:woodbound_entities")));
			if (isWoodbound) {
				if (splinter.getLastHurtByMob() == attacker) {
					splinter.setLastHurtByMob(null);
				}
				attacker = null;
			} else if (attacker.isAlive() && splinter.distanceToSqr(attacker) < TARGET_RANGE * TARGET_RANGE) {
				if (!(attacker instanceof Player)) {
					prioritizedMob = attacker;
					splinter.getPersistentData().putString("splinter_attacker_uuid", attacker.getUUID().toString());
				}
			}
		}

		boolean hadPrioritizedMob = !attackerUUIDStr.isEmpty();
		if (hadPrioritizedMob && prioritizedMob == null) {
			splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 0);
			splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
			splinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
			splinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
			splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
			splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
			splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			if (splinter.getLastHurtByMob() != null) {
				splinter.setLastHurtByMob(null);
			}
			splinter.getPersistentData().remove("splinter_attacker_uuid");
		}

		if (prioritizedMob != null) {
			foundPlayer = prioritizedMob;
			splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 1);
			if (splinter.getTarget() != prioritizedMob) {
				splinter.setTarget(prioritizedMob);
			}
			if (splinter.getLastHurtByMob() != prioritizedMob) {
				splinter.setLastHurtByMob(prioritizedMob);
			}
		} else {
			if (attacker instanceof Player p && p.isAlive() && !p.isCreative() && !p.isSpectator() && splinter.distanceToSqr(attacker) < TARGET_RANGE * TARGET_RANGE) {
				foundPlayer = p;
			}
			if (foundPlayer != null) {
				if (splinter.getTarget() != foundPlayer) {
					splinter.setTarget(foundPlayer);
				}
			} else {
				if (splinter.getTarget() != null) {
					splinter.setTarget(null);
				}
			}
		}

		if (foundPlayer != null) {
			boolean isWoodbound = foundPlayer.getType().is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.ResourceLocation.parse("the_backwoods:woodbound_entities")));
			if (isWoodbound) {
				foundPlayer = null;
			}
		}

		if (foundPlayer == null) {
			if (SplinterHivemind.search(splinter)) return;
			int remainingForcedHunt = splinter.getPersistentData().getInt(K_FORCED_HUNT_TICKS);

			if (remainingForcedHunt <= 0) {
				splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
				splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 0);
				splinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
				splinter.getPersistentData().putInt(K_HM_PHASE, PHASE_SILENT);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 0);
				splinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
				splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
				stopNav(splinter);
			} else {
				splinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, remainingForcedHunt - 1);
				splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 1);
				splinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, HUNT_TICKS);
			}
			return;
		}

		int frozenByRose = splinter.getEntityData().get(SplinterEntity.DATA_frozenByRose);
		int isEnraged = splinter.getEntityData().get(SplinterEntity.DATA_isEnraged);
		int watchTimer = splinter.getEntityData().get(SplinterEntity.DATA_watchTimer);

		int forcedHuntTicks = splinter.getPersistentData().getInt(K_FORCED_HUNT_TICKS);
		if (forcedHuntTicks > 0) {
			splinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, forcedHuntTicks - 1);
			splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 1);
			isEnraged = 1;
		}

		if (splinter.getLastHurtByMob() != null) {
			boolean isWoodbound = splinter.getLastHurtByMob().getType().is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.ResourceLocation.parse("the_backwoods:woodbound_entities")));
			if (!isWoodbound) {
				int lastSeenHurtTime = splinter.getPersistentData().getInt(K_LAST_HURT_TIME);
				int currentHurtTime = splinter.hurtTime;
				if (currentHurtTime > 0 && currentHurtTime != lastSeenHurtTime) {
					int bonus = splinter.getPersistentData().getInt(K_RAGE_BONUS) + RAGE_THRESHOLD_HIT_BONUS;
					splinter.getPersistentData().putInt(K_RAGE_BONUS, bonus);
					splinter.getPersistentData().putInt(K_LAST_HURT_TIME, currentHurtTime);
				}
			}
		}

		int effectiveRageThreshold = Math.max(1, RAGE_WATCH_THRESHOLD - splinter.getPersistentData().getInt(K_RAGE_BONUS));

		boolean isWatched = false;
		if (foundPlayerDirect != null) {
			Vec3 toSplinter = splinter.getEyePosition().subtract(foundPlayerDirect.getEyePosition());
			if (toSplinter.lengthSqr() > 1.0e-8) toSplinter = toSplinter.normalize();
			double dot = foundPlayerDirect.getLookAngle().normalize().dot(toSplinter);
			isWatched = (dot > WATCH_DOT_THRESHOLD) && foundPlayerDirect.hasLineOfSight(splinter);
		}

		double distToPlayer = foundPlayerDirect != null ? splinter.position().distanceTo(foundPlayerDirect.position()) : Double.MAX_VALUE;
		if (isEnraged == 1 && distToPlayer > RAGE_ESCAPE_RANGE) {
			if (splinter.getPersistentData().getInt(K_FORCED_HUNT_TICKS) <= 0) {
				splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 0);
				splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
				isEnraged = 0;
			}
		}

		if (isWatched && isEnraged == 0) {
			watchTimer++;
			splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, watchTimer);
			if (watchTimer >= effectiveRageThreshold) {
				splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 1);
				splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
				isEnraged = 1;
			}
		} else if (!isWatched && isEnraged == 0) {
			splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
		}

		boolean foundRose = false;
		if (foundPlayerDirect != null) {
			foundRose = checkHeldRose(foundPlayerDirect, splinter, world, foundPlayerDirect.getX(), foundPlayerDirect.getY(), foundPlayerDirect.getZ());
		}
		if (foundRose || frozenByRose == 1) {
			setSpeed(splinter, 0);
			splinter.getEntityData().set(SplinterEntity.DATA_isEnraged, 0);
			splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
			splinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
			splinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
			splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
			splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
			splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			stopNav(splinter);
			return;
		}

		SplinterHivemind.observe(splinter, foundPlayerDirect, splinter.getLastHurtByMob());
        if (personality == PERSONALITY_STALK_HUNT && SplinterHivemind.beginGroupHunt(splinter, foundPlayerDirect)) {
            splinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
            splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 160);
            splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
            splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
        }

		double heightDiff = foundPlayer.getY() - splinter.getY();
		double horizontalDist = splinter.position().distanceTo(foundPlayer.position());
		boolean isTowering = !isCritical && (heightDiff >= 1.4 && horizontalDist < 12);

		// One bounded cover move owns navigation while the archer is drawing.
		if (foundPlayerDirect != null && splinter.getTarget() == foundPlayerDirect
				&& SplinterHivemind.handleBowThreat(splinter, foundPlayerDirect,
					splinter.getEntityData().get(SplinterEntity.DATA_mineProgress),
					isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED)) return;
		if (personality == PERSONALITY_STALK_HUNT
				&& SplinterHivemind.bowCommitActive(splinter, foundPlayerDirect)
				&& splinter.getPersistentData().getInt(K_HM_PHASE) != PHASE_HUNT) {
			splinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
			splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 90);
		}

		if (personality == PERSONALITY_ANGEL) {
			if (isWatched && isEnraged == 0 && !SplinterHivemind.bowCommitActive(splinter, foundPlayerDirect)) {
				setSpeed(splinter, 0);
				stopNav(splinter);
				splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
				return;
			}

			if (splinter.tickCount % 3 == 0) {
				splinter.lookAt(EntityAnchorArgument.Anchor.EYES, new Vec3(foundPlayer.getX(), foundPlayer.getEyeY(), foundPlayer.getZ()));
			}

			setSpeed(splinter, isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED);
			if (splinter instanceof Mob angelMob) {
				navMoveToEntityThrottled(splinter, angelMob, foundPlayer, 1.0, 3);
			}
		} else {
			int phase = splinter.getPersistentData().getInt(K_HM_PHASE);
			int phaseTimer = splinter.getPersistentData().getInt(K_HM_PHASE_TIMER);
			int silentCd = splinter.getPersistentData().getInt(K_HM_SILENT_CD);

			if (phase == PHASE_SILENT && isEnraged == 0 && forcedHuntTicks <= 0) {
				phase = PHASE_STALK;
				phaseTimer = randomBetween(splinter, STALK_MIN_TICKS, STALK_MAX_TICKS);
				splinter.getPersistentData().putInt(K_HM_PHASE, phase);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
				splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			}

			if (silentCd > 0) {
				silentCd--;
				splinter.getPersistentData().putInt(K_HM_SILENT_CD, silentCd);
			}

			if (isEnraged == 1 || splinter.getPersistentData().getInt(K_FORCED_HUNT_TICKS) > 0) {
				phase = PHASE_HUNT;
				phaseTimer = HUNT_TICKS + 40;
				splinter.getPersistentData().putInt(K_HM_PHASE, phase);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
				splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			} else {
				phaseTimer--;
				if (phaseTimer <= 0) {
					if (phase == PHASE_HUNT) {
						phase = PHASE_SILENT;
						phaseTimer = randomBetween(splinter, SILENT_MIN_TICKS, SILENT_MAX_TICKS);
						splinter.getPersistentData().putInt(K_HM_SILENT_CD, POST_HUNT_SILENT_CD);
						splinter.getEntityData().set(SplinterEntity.DATA_watchTimer, 0);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					} else if (phase == PHASE_SILENT) {
						phase = PHASE_STALK;
						phaseTimer = randomBetween(splinter, STALK_MIN_TICKS, STALK_MAX_TICKS);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					} else {
						phase = PHASE_SILENT;
						phaseTimer = randomBetween(splinter, SILENT_MIN_TICKS, SILENT_MAX_TICKS);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					}
				}
				splinter.getPersistentData().putInt(K_HM_PHASE, phase);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
			}

			if (phase == PHASE_SILENT) {
				setSpeed(splinter, 0);
				stopNav(splinter);
				splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
				return;
			}

			if (splinter.tickCount % 3 == 0) {
				splinter.lookAt(EntityAnchorArgument.Anchor.EYES, new Vec3(foundPlayer.getX(), foundPlayer.getEyeY(), foundPlayer.getZ()));
			}

			if (phase == PHASE_STALK) {
				setSpeed(splinter, STALK_SPEED);

				int watchTicks = splinter.getPersistentData().getInt(K_STALK_WATCH_TICKS);
				if (watchTicks > 0) {
					splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, watchTicks - 1);
					stopNav(splinter);
					splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
					return;
				}

				int flankTicks = splinter.getPersistentData().getInt(K_FLANK_TICKS);
				if (flankTicks > 0) {
					splinter.getPersistentData().putInt(K_FLANK_TICKS, flankTicks - 1);
					if (splinter instanceof net.minecraft.world.entity.Mob mob2) {
						double fx = splinter.getPersistentData().getDouble(K_FLANK_X);
						double fy = splinter.getPersistentData().getDouble(K_FLANK_Y);
						double fz = splinter.getPersistentData().getDouble(K_FLANK_Z);
						navMoveToPosThrottled(splinter, mob2, fx, fy, fz, 0.8, 1);
					}
					splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
					return;
				}

				if (splinter.getRandom().nextInt(STALK_WATCH_CHANCE) == 0) {
					int flankStreak = splinter.getPersistentData().getInt(K_FLANK_STREAK);
					boolean doFlank = !isWatched && flankStreak < 2 && splinter.getRandom().nextBoolean();

					if (doFlank) {
						splinter.getPersistentData().putInt(K_FLANK_STREAK, flankStreak + 1);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, randomBetween(splinter, STALK_WATCH_MIN, STALK_WATCH_MAX));
						double dist = STALK_FOLLOW_DIST - 2.0;
						double angleOffset = (splinter.getRandom().nextDouble() * 30.0 + 60.0) * (splinter.getRandom().nextBoolean() ? 1 : -1);
						double angleRad = Math.toRadians(foundPlayer.getYRot() + angleOffset);
						double fx = foundPlayer.getX() - Math.sin(angleRad) * dist;
						double fz = foundPlayer.getZ() + Math.cos(angleRad) * dist;
						double fy = foundPlayer.getY();
						splinter.getPersistentData().putDouble(K_FLANK_X, fx);
						splinter.getPersistentData().putDouble(K_FLANK_Y, fy);
						splinter.getPersistentData().putDouble(K_FLANK_Z, fz);
					} else {
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, randomBetween(splinter, STALK_WATCH_MIN, STALK_WATCH_MAX));
						stopNav(splinter);
					}
					splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
					return;
				}

				if (splinter instanceof Mob mob) {
					Vec3 hiveStalkPoint = SplinterHivemind.stalkPoint(mob, foundPlayer);
					if (hiveStalkPoint != null) {
						SplinterHivemind.allowPost(splinter);
						navMoveToPosThrottled(splinter, mob, hiveStalkPoint.x, hiveStalkPoint.y, hiveStalkPoint.z, 0.8, 1);
					} else if (distToPlayer > STALK_FOLLOW_DIST) {
						navMoveToEntityThrottled(splinter, mob, foundPlayer, 0.8, 1);
					} else {
						if (!mob.getNavigation().isDone() && splinter.tickCount % 10 == 0) {
							stopNav(splinter);
						}
					}
				}
				splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
				return;
			}

			setSpeed(splinter, isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED);
			if (splinter instanceof Mob huntMob) {
				navMoveToEntityThrottled(splinter, huntMob, foundPlayer, 1.0, 3);
			}
		}

		Vec3 splinterEyes = splinter.getEyePosition(1f);
		Vec3 splinterView = splinter.getViewVector(1f);
		Vec3 blockCheckTarget = splinterEyes.add(splinterView.scale(MINE_RAY_DISTANCE));

		HitResult hit = world.clip(new ClipContext(splinterEyes, blockCheckTarget, ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, splinter));

		BlockPos facePos;
		BlockPos feetPos;

		if (hit.getType() == HitResult.Type.BLOCK) {
			facePos = ((BlockHitResult) hit).getBlockPos();
			feetPos = new BlockPos(facePos.getX(), Mth.floor(splinter.getY()), facePos.getZ());
		} else {
			Vec3 look = splinter.getLookAngle().normalize();
			int fx = Mth.floor(splinter.getX() + look.x);
			int fz = Mth.floor(splinter.getZ() + look.z);
			feetPos = new BlockPos(fx, Mth.floor(splinter.getY()), fz);
			facePos = new BlockPos(fx, Mth.floor(splinter.getEyeY()), fz);
		}

		if (SplinterOnEntityTickUpdateProcedure.AdaptiveHive.detourAroundCover(world, splinter, foundPlayer, feetPos, facePos)) {
                splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
                return;
            }

		boolean canMineFeet = canMine(world, feetPos, foundPlayer);
		boolean canMineFace = canMine(world, facePos, foundPlayer);
		
		BlockPos downPos = splinter.blockPosition().below();
		boolean canMineDown = (heightDiff <= -2.0 && horizontalDist <= 3.0) && canMine(world, downPos, foundPlayer);
		
		int prevProgress = splinter.getEntityData().get(SplinterEntity.DATA_mineProgress);
		BlockPos prevTrackPos = new BlockPos(
			splinter.getEntityData().get(SplinterEntity.DATA_mineX),
			splinter.getEntityData().get(SplinterEntity.DATA_mineY),
			splinter.getEntityData().get(SplinterEntity.DATA_mineZ)
		);
		BlockPos trackPos = canMineDown ? downPos : (canMineFeet ? feetPos : facePos);
		if (prevProgress > 0 && trackPos != null && !trackPos.equals(prevTrackPos)) {
			boolean lockPrev = canMine(world, prevTrackPos, foundPlayer) && prevTrackPos.closerToCenterThan(splinter.position(), 4.0) && !isPositionClaimed(world, prevTrackPos, splinter);
			if (lockPrev) {
				trackPos = prevTrackPos;
				canMineDown = trackPos.equals(downPos);
				canMineFeet = trackPos.equals(feetPos);
				canMineFace = trackPos.equals(facePos);
				if (!canMineDown && !canMineFeet && !canMineFace) {
					if (trackPos.getY() < splinter.getY()) { canMineDown = true; downPos = trackPos; }
					else { canMineFeet = true; feetPos = trackPos; }
				}
			} else {
				prevProgress = 0;
			}
		}
		if (canMineDown && isPositionClaimed(world, downPos, splinter)) canMineDown = false;
		if (canMineFeet && isPositionClaimed(world, feetPos, splinter)) canMineFeet = false;
		if (canMineFace && isPositionClaimed(world, facePos, splinter)) canMineFace = false;

		if (canMineFeet || canMineFace || canMineDown) {
			splinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
			int mineProgress = prevProgress + 1;
			splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, mineProgress);

			if (splinter.tickCount % 6 == 0) {
				splinter.swing(InteractionHand.MAIN_HAND);
			}

			splinter.getEntityData().set(SplinterEntity.DATA_mineX, trackPos.getX());
			splinter.getEntityData().set(SplinterEntity.DATA_mineY, trackPos.getY());
			splinter.getEntityData().set(SplinterEntity.DATA_mineZ, trackPos.getZ());

			float speedRef;
			if (canMineDown) {
				speedRef = world.getBlockState(downPos).getDestroySpeed(world, downPos);
			} else if (canMineFeet) {
				speedRef = world.getBlockState(feetPos).getDestroySpeed(world, feetPos);
			} else {
				speedRef = world.getBlockState(facePos).getDestroySpeed(world, facePos);
			}

			float mineThreshold = isDegraded
					? speedRef * DEGRADED_MINE_SPEED_MULTIPLIER + DEGRADED_MINE_SPEED_BASE
					: speedRef * MINE_SPEED_MULTIPLIER + MINE_SPEED_BASE;

			SplinterOnEntityTickUpdateProcedure.AdaptiveHive.showMiningCracks(splinter, mineProgress, mineThreshold,
					canMineDown ? downPos : null, canMineFeet ? feetPos : null, canMineFace ? facePos : null);
			if (mineProgress > mineThreshold) {
				if (canMineDown) world.destroyBlock(downPos, false);
				if (canMineFeet) world.destroyBlock(feetPos, false);
				if (canMineFace) world.destroyBlock(facePos, false);
				stopNav(splinter);
				splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);
			}
		} else {
			splinter.getEntityData().set(SplinterEntity.DATA_mineProgress, 0);

			if (!isCritical && !splinter.isSwimming() && !splinter.isInWater()) {
				if (heightDiff >= 1.4 && horizontalDist < 12) {
					splinter.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get()));

					BlockPos under = BlockPos.containing(splinter.getX(), splinter.getY() - 1, splinter.getZ());
					BlockPos body = BlockPos.containing(splinter.getX(), splinter.getY(), splinter.getZ());
					BlockPos bodyUp = body.above();

					if (world.getBlockState(under).isAir()) {
						if (world.setBlock(under, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState(), 3)) {
							SplinterOnEntityTickUpdateProcedure.AdaptiveHive.swingBuild(splinter);
							trackPlacedBlock(splinter, under);
						}
					}
					if (world.getBlockState(body).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) {
						world.setBlock(body, Blocks.AIR.defaultBlockState(), 3);
					}
					if (world.getBlockState(bodyUp).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) {
						world.setBlock(bodyUp, Blocks.AIR.defaultBlockState(), 3);
					}

					world.destroyBlock(BlockPos.containing(splinter.getX(), splinter.getY() + 2, splinter.getZ()), false);
					world.destroyBlock(BlockPos.containing(splinter.getX(), splinter.getY() + 3, splinter.getZ()), false);

					if (splinter.onGround()) {
						splinter.setDeltaMovement(new Vec3(splinter.getDeltaMovement().x(), 0.4, splinter.getDeltaMovement().z()));
					}
					splinter.fallDistance = 0;
				} else if (heightDiff > -0.5 && heightDiff < 2.5) {
					Vec3 look = splinter.getLookAngle();
					BlockPos bridgePos = BlockPos.containing(splinter.getX() + look.x, splinter.getY() - 1, splinter.getZ() + look.z);
					BlockPos farBridgePos = BlockPos.containing(splinter.getX() + look.x * 2.0, splinter.getY() - 1, splinter.getZ() + look.z * 2.0);
					boolean isNearGap = !(world.getBlockFloorHeight(bridgePos) > 0) || !(world.getBlockFloorHeight(farBridgePos) > 0);

					if (isNearGap) {
						splinter.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get()));
						if (!(world.getBlockFloorHeight(bridgePos) > 0)) {
							if (world.setBlock(bridgePos, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState(), 3)) {
								SplinterOnEntityTickUpdateProcedure.AdaptiveHive.swingBuild(splinter);
								trackPlacedBlock(splinter, bridgePos);
							}
						}
					} else {
						splinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
					}
				} else {
					splinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
				}
			} else {
				splinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
			}
		}

		// Melee reach range increase for non-players only (by 50%)
		LivingEntity target = splinter.getTarget();
		if (target != null && target.isAlive() && !(target instanceof Player)) {
			double reach = splinter.getBbWidth() + target.getBbWidth() + 0.8;
			double increasedReach = reach * 1.5;
			double distSqr = splinter.distanceToSqr(target);
			if (distSqr <= increasedReach * increasedReach) {
				int lastAttackTick = splinter.getPersistentData().getInt("last_melee_attack_tick");
				if (splinter.tickCount - lastAttackTick >= 20) {
					splinter.doHurtTarget(target);
					splinter.swing(InteractionHand.MAIN_HAND);
					splinter.getPersistentData().putInt("last_melee_attack_tick", splinter.tickCount);
				}
			}
		}
	}

	private static int randomBetween(SplinterEntity entity, int min, int max) {
		if (max <= min) return min;
		return min + entity.getRandom().nextInt(max - min + 1);
	}

	private static void setSpeed(LivingEntity entity, double speed) {
		if (entity.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)) {
			entity.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(speed);
		}

	}

private static boolean isPositionClaimed(LevelAccessor world, BlockPos pos, SplinterEntity thisEntity) {
		net.minecraft.world.phys.AABB box = new net.minecraft.world.phys.AABB(pos).inflate(3);
		java.util.List<SplinterEntity> others = world.getEntitiesOfClass(SplinterEntity.class, box, e -> e != thisEntity && e.isAlive());
		int ourProgress = thisEntity.getEntityData().get(SplinterEntity.DATA_mineProgress);
		for (SplinterEntity other : others) {
			int otherProgress = other.getEntityData().get(SplinterEntity.DATA_mineProgress);
			if (otherProgress > 0 &&
				other.getEntityData().get(SplinterEntity.DATA_mineX) == pos.getX() &&
				other.getEntityData().get(SplinterEntity.DATA_mineY) == pos.getY() &&
				other.getEntityData().get(SplinterEntity.DATA_mineZ) == pos.getZ()) {
				
				if (ourProgress == 0) return true;
				if (otherProgress > ourProgress) return true;
				if (otherProgress == ourProgress && other.getId() < thisEntity.getId()) return true;
			}
		}
		return false;
	}

	private static boolean canMine(LevelAccessor world, BlockPos pos, LivingEntity player) {
		BlockState state = world.getBlockState(pos);
		if (state.isAir()) return false;
		float speed = state.getDestroySpeed(world, pos);
		if (speed < 0 || speed >= MAX_BREAKABLE_HARDNESS) return false;
		if (pos.getY() == (int) (player.getY() - 2)) return false;
		return true;
	}

	private static boolean checkHeldRose(Entity holder, SplinterEntity splinter, LevelAccessor world, double x, double y, double z) {
		if (!(holder instanceof LivingEntity living)) return false;

		ItemStack main = living.getMainHandItem();
		ItemStack off = living.getOffhandItem();

		boolean mainIsRose = main.getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem();
		boolean offIsRose = off.getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem();

		if (!mainIsRose && !offIsRose) return false;

		setSpeed(splinter, 0);
		stopNav(splinter);

		if (mainIsRose) tickRoseItem(main, world, x, y, z);
		if (offIsRose) tickRoseItem(off, world, x, y, z);

		return true;
	}

	private static void tickRoseItem(ItemStack rose, LevelAccessor world, double x, double y, double z) {
		double wiltTimer = rose.getOrDefault(DataComponents.CUSTOM_DATA, CustomData.EMPTY).copyTag().getDouble("wilt_timer") + 1.0;
		CustomData.update(DataComponents.CUSTOM_DATA, rose, tag -> tag.putDouble("wilt_timer", wiltTimer));

		if (wiltTimer > ROSE_WILT_TICKS) {
			if (world instanceof Level level) {
				if (!level.isClientSide()) {
					level.playSound(null, BlockPos.containing(x, y, z), SoundEvents.FIRE_EXTINGUISH, SoundSource.NEUTRAL, 1f, 1f);
				} else {
					level.playLocalSound(x, y, z, SoundEvents.FIRE_EXTINGUISH, SoundSource.NEUTRAL, 1f, 1f, false);
				}
			}
			rose.shrink(1);
			CustomData.update(DataComponents.CUSTOM_DATA, rose, tag -> tag.putDouble("wilt_timer", 0));
		}
	}

	private static boolean checkNearbyRoseBlocks(LevelAccessor world, SplinterEntity splinter) {
		BlockPos base = splinter.blockPosition();
		BlockPos.MutableBlockPos check = new BlockPos.MutableBlockPos();
		for (int sx = -ROSE_SCAN_XZ; sx < ROSE_SCAN_XZ; sx++) {
			for (int sz = -ROSE_SCAN_XZ; sz < ROSE_SCAN_XZ; sz++) {
				check.set(base.getX() + sx, base.getY(), base.getZ() + sz);
				if (world instanceof Level level && !level.hasChunkAt(check)) continue;
				for (int sy = -ROSE_SCAN_Y; sy < ROSE_SCAN_Y; sy++) {
					check.set(base.getX() + sx, base.getY() + sy, base.getZ() + sz);
					BlockState roseState = world.getBlockState(check);
					if (roseState.is(TheBackwoodsModBlocks.ASH_ROSE.get())
							|| roseState.is(TheBackwoodsModBlocks.POTTED_ASH_ROSE.get())) {
						return true;
					}
				}
			}
		}
		return false;
	}

	private static Player findNearestPlayerInRange(LevelAccessor world, double x, double y, double z, double range) {
		AABB box = AABB.ofSize(new Vec3(x, y, z), range, range, range);
		List<Player> players = world.getEntitiesOfClass(Player.class, box, e -> !e.isCreative() && !e.isSpectator());

		Player nearest = null;
		double best = range * range;
		for (Player p : players) {
			double d = p.distanceToSqr(x, y, z);
			if (d < best) {
				best = d;
				nearest = p;
			}
		}
		return nearest;
	}

	private static void stopNav(SplinterEntity splinter) {
		if (splinter instanceof Mob mob) {
			mob.getNavigation().stop();
		}
		splinter.getPersistentData().putInt(K_NAV_MODE, 0);
	}

	private static void navMoveToEntityThrottled(SplinterEntity splinter, Mob mob, Entity target, double speed, int mode) {
		Vec3 hiveApproach = SplinterHivemind.approach(mob, target);
		if (target instanceof Player && hiveApproach.distanceToSqr(target.position()) > 1.0) SplinterHivemind.allowPost(mob);
		navMoveToPosThrottled(splinter, mob, hiveApproach.x, hiveApproach.y, hiveApproach.z, speed, mode);
	}

	private static void navMoveToPosThrottled(SplinterEntity splinter, Mob mob, double tx, double ty, double tz, double speed, int mode) {
		int now = splinter.tickCount;
		int lastTick = splinter.getPersistentData().getInt(K_NAV_LAST_TICK);
		int lastMode = splinter.getPersistentData().getInt(K_NAV_MODE);

		double lastTx = splinter.getPersistentData().getDouble(K_NAV_TX);
		double lastTy = splinter.getPersistentData().getDouble(K_NAV_TY);
		double lastTz = splinter.getPersistentData().getDouble(K_NAV_TZ);

		boolean timerReady = (now - lastTick) >= PATH_RECALC_INTERVAL_TICKS;
		boolean modeChanged = lastMode != mode;
		boolean noLast = (lastTick == 0 && lastMode == 0 && lastTx == 0.0 && lastTy == 0.0 && lastTz == 0.0);

		double movedSqr = noLast ? Double.MAX_VALUE : sqr(tx - lastTx) + sqr(ty - lastTy) + sqr(tz - lastTz);
		boolean targetMovedEnough = movedSqr >= TARGET_MOVE_REPATH_DIST_SQR;

		if (modeChanged || timerReady || targetMovedEnough || mob.getNavigation().isDone()) {
			mob.getNavigation().moveTo(tx, ty, tz, speed);
			splinter.getPersistentData().putInt(K_NAV_LAST_TICK, now);
			splinter.getPersistentData().putInt(K_NAV_MODE, mode);
			splinter.getPersistentData().putDouble(K_NAV_TX, tx);
			splinter.getPersistentData().putDouble(K_NAV_TY, ty);
			splinter.getPersistentData().putDouble(K_NAV_TZ, tz);
		}
	}

	private static double sqr(double v) {
		return v * v;
	} // 1.21.1

	private static void trackPlacedBlock(Entity splinter, BlockPos pos) {
		net.minecraft.nbt.CompoundTag tag = splinter.getPersistentData();
		int size = tag.getInt("pb_size");

		for (int i = 0; i < size; i++) {
			int x = tag.getInt("pb_x_" + i);
			int y = tag.getInt("pb_y_" + i);
			int z = tag.getInt("pb_z_" + i);
			if (x == pos.getX() && y == pos.getY() && z == pos.getZ()) {
				return;
			}
		}

		if (size < 64) {
			tag.putInt("pb_x_" + size, pos.getX());
			tag.putInt("pb_y_" + size, pos.getY());
			tag.putInt("pb_z_" + size, pos.getZ());
			tag.putInt("pb_size", size + 1);
		}
	}

	private static void updatePlacedBlockCleanup(LevelAccessor world, Entity splinter, BlockState targetBlockState) {
		net.minecraft.nbt.CompoundTag tag = splinter.getPersistentData();
		int size = tag.getInt("pb_size");
		if (size == 0) return;

		boolean standingOnAny = false;
		for (int i = 0; i < size; i++) {
			int x = tag.getInt("pb_x_" + i);
			int y = tag.getInt("pb_y_" + i);
			int z = tag.getInt("pb_z_" + i);

			double dx = Math.abs(splinter.getX() - (x + 0.5));
			double dz = Math.abs(splinter.getZ() - (z + 0.5));
			double dy = splinter.getY() - y;

			if (dx <= 1.0 && dz <= 1.0 && dy >= -0.2 && dy <= 2.2) {
				standingOnAny = true;
				break;
			}
		}

		if (standingOnAny) {
			tag.putInt("pb_delay", 0);
		} else {
			int delay = tag.getInt("pb_delay") + 1;
			tag.putInt("pb_delay", delay);

			if (delay % 2 == 0) {
				int x0 = tag.getInt("pb_x_0");
				int y0 = tag.getInt("pb_y_0");
				int z0 = tag.getInt("pb_z_0");
				BlockPos pos0 = new BlockPos(x0, y0, z0);

				BlockState state = world.getBlockState(pos0);
				if (state.is(targetBlockState.getBlock())) {
					world.destroyBlock(pos0, false);
				}

				for (int i = 0; i < size - 1; i++) {
					tag.putInt("pb_x_" + i, tag.getInt("pb_x_" + (i + 1)));
					tag.putInt("pb_y_" + i, tag.getInt("pb_y_" + (i + 1)));
					tag.putInt("pb_z_" + i, tag.getInt("pb_z_" + (i + 1)));
				}
				size--;
				tag.putInt("pb_size", size);
				if (size == 0) {
					tag.putInt("pb_delay", 0);
				}
			}
		}
	}

		private static final class NavigationAwareGoal extends net.minecraft.world.entity.ai.goal.Goal {
			private final Mob self;
			private final net.minecraft.world.entity.ai.goal.Goal original;
			NavigationAwareGoal(Mob self, net.minecraft.world.entity.ai.goal.Goal original) {
				this.self = self; this.original = original;
				setFlags(original.getFlags().clone());
			}
			@Override public boolean canUse() { return !(original instanceof net.minecraft.world.entity.ai.goal.MeleeAttackGoal && mirrorAttackBlocked(self)) && !navigationOwned(self) && original.canUse(); }

			@Override public boolean isInterruptable() { return original.isInterruptable(); }
			@Override public boolean requiresUpdateEveryTick() { return original.requiresUpdateEveryTick(); }
			@Override public void start() { original.start(); }
			@Override public void tick() { if (!navigationOwned(self)) original.tick(); }
			@Override public void stop() {
				boolean preserve = navigationOwned(self);
				net.minecraft.world.level.pathfinder.Path path = preserve ? self.getNavigation().getPath() : null;
				original.stop();
				if (path != null && !path.isDone()) self.getNavigation().moveTo(path,
						self.getPersistentData().getInt("htState") == WITHDRAW ? 0.9 : 1.0);
			}
		}

                    && now(level) >= actor.getPersistentData().getLong("ahPlacePose")

        private static void endMirror(Mob actor, boolean attack) {
            if (!(actor.level() instanceof ServerLevel level)) return;
            net.minecraft.nbt.CompoundTag data = actor.getPersistentData();
            if (data.contains("htMirrorUntil")) {
                var effect = actor.getEffect(net.minecraft.world.effect.MobEffects.INVISIBILITY);
                long remaining = data.getLong("htMirrorUntil") - now(level);
                // Remove only our short amplifier-zero effect; preserve an externally replaced stronger/longer effect.
                var ownedEffect = MIRROR_EFFECTS.remove(actor);
                if (effect != null && (effect == ownedEffect
                        || effect.getAmplifier() == 0 && effect.getDuration() <= Math.max(8, remaining + 8)))
                    actor.removeEffect(net.minecraft.world.effect.MobEffects.INVISIBILITY);
                actor.setInvisible(actor.hasEffect(net.minecraft.world.effect.MobEffects.INVISIBILITY));
                data.remove("htMirrorUntil"); data.remove("htMirrorStart"); data.remove("htMirrorOwner");
                data.remove("htMirrorDimension"); data.remove("htMirrorNextRoute");
                actor.setSilent(data.getBoolean("htMirrorOldSilent"));
                data.remove("htMirrorOldSilent"); data.remove("htMirrorCloakAt");
                data.remove("htMirrorGroup"); data.remove("htMirrorRearTicks"); data.remove("htMirrorConfirmed");
                clearTactic(actor, now(level));
            }
            if (attack) data.putLong("htMirrorAttackUntil", now(level) + 40);
            else data.remove("htMirrorAttackUntil");
        }

        /** Called before normal Blindspot decisions; parkour, mining, cover and melee cannot compete with this lease. */
        public static boolean handleMirrorPhase(Mob actor, Player player, int mineProgress, double speed) {
            if (!(actor.level() instanceof ServerLevel level)) return false;
            net.minecraft.nbt.CompoundTag data = actor.getPersistentData();
            long time = now(level);
            boolean has = data.contains("htMirrorUntil");
            if (!has && !data.contains("htMirrorAttackUntil")) return false;
            if (!validPlayer(player, level) || !actor.isAlive() || actor.getTarget() != player
                    || mineProgress != 0 || actor.isPassenger() || actor.isInWater()
                    || SplinterHivemind.frozen(actor) || roseHeld(player)
                    || actor.distanceToSqr(player) > 48 * 48
                    || has && (!player.getUUID().toString().equals(data.getString("htMirrorOwner"))
                        || !level.dimension().location().toString().equals(data.getString("htMirrorDimension"))
                        || time < data.getLong("htMirrorStart"))) {
                endMirror(actor, false); return false;
            }
            long until = data.getLong("htMirrorUntil");
            if (has && (time >= until || until - time > 180)) {
                endMirror(actor, until - time <= 180);
                has = false;
            }
            if (!has) {
                long release = data.getLong("htMirrorAttackUntil");
                if (release <= time || release - time > 40) { data.remove("htMirrorAttackUntil"); return false; }
                SplinterOnEntityTickUpdateProcedure.setSpeed(actor, speed);
                if (data.getInt("htState") != MELEE_PRESSURE || time >= data.getLong("htUntil"))
                    meleePressure(level, actor, player, new net.minecraft.nbt.CompoundTag(), time);
                data.putLong("htLastTick", time); data.putLong("htPost", level.getGameTime());
                actor.getLookControl().setLookAt(player, 25, 20);
                return true; // Vanilla melee goal is the only attack source, including its normal cooldown.
            }
            data.putInt("htState", MIRROR); data.putLong("htPost", level.getGameTime()); data.putLong("htLastTick", time);
            SplinterOnEntityTickUpdateProcedure.setSpeed(actor, speed);
            actor.setSilent(true);
            if (time < data.getLong("htMirrorCloakAt")) {
                actor.getNavigation().stop(); point(actor, actor.position()); return true;
            }
            if (!actor.hasEffect(net.minecraft.world.effect.MobEffects.INVISIBILITY)) {
                // Reapplication never extends the fixed deadline; a dispel cannot enable premature attacks.
                actor.addEffect(new net.minecraft.world.effect.MobEffectInstance(
                        net.minecraft.world.effect.MobEffects.INVISIBILITY, (int) (until - time) + 4, 0, false, false, false));
            }
            actor.setInvisible(true);
            Vec3 look = Vec3.directionFromRotation(0.0F, player.getYRot()).multiply(1, 0, 1).normalize();
            Vec3 relative = actor.position().subtract(player.position()).multiply(1, 0, 1);
            double distance = relative.length();
            // Stable reservations like Fractus: translation follows the player, angles never orbit with time.
            int slot = data.getInt("htMirrorSlot");
            boolean surround = slot % 4 != 0;
            int rank = surround ? slot - slot / 4 - 1 : slot / 4;
            int members = Math.max(1, data.getInt("htMirrorCount"));
            int roleCount = surround ? members - (members + 3) / 4 : (members + 3) / 4;
            int perRing = surround ? 24 : 12;
            int ringCount = Math.max(1, Math.min(perRing, roleCount - rank / perRing * perRing));
            double radius = (surround ? 10.0 : 8.5) + 3.0 * (rank / perRing);
            double angle = surround
                    ? data.getDouble("htMirrorBearing") + Math.PI * 2.0 * Math.floorMod(rank, 24) / ringCount
                    : Math.toRadians(player.getYRot()) + Math.PI + (Math.floorMod(rank, 12) - (ringCount - 1) * 0.5) * 0.12;
            Vec3 direction = new Vec3(-Math.sin(angle), 0, Math.cos(angle));
            Vec3 prediction = player.getDeltaMovement().multiply(1, 0, 1);
            if (prediction.lengthSqr() > 0.25) prediction = prediction.normalize().scale(0.5);
            Vec3 desired = player.position().add(prediction.scale(3)).add(direction.scale(radius));
            double clearance = 6.0 + Math.min(2.0, prediction.length() * 4.0);
            boolean retreat = distance < clearance || distance < clearance + 2.0
                    && prediction.dot(relative.normalize()) > 0.08;
            if (retreat) {
                // Rot-style threat spacing, but navigate on verified ground rather than teleporting.
                Vec3 away = distance > 0.01 ? relative.normalize() : direction;
                desired = actor.position().add(away.scale(5.0)).add(prediction.scale(2));
                if (tacticPoint(actor).distanceToSqr(player.position()) < distance * distance) {
                    actor.getNavigation().stop(); TACTICAL_ROUTES.remove(actor);
                    point(actor, actor.position());
                }
            }
            data.putDouble("htMirrorClearance", clearance);
            boolean rear = distance >= clearance && distance <= radius + 3.0
                    && relative.normalize().dot(look) <= -0.55;
            boolean settled = actor.onGround() && !retreat
                    && (surround ? actor.position().distanceToSqr(desired) <= 6.25 : rear
                        && actor.position().distanceToSqr(desired) <= 9.0);
            int rearTicks = settled ? data.getInt("htMirrorRearTicks") + 1 : 0;
            data.putInt("htMirrorRearTicks", Math.min(16, rearTicks));
            if (rearTicks >= 16) data.putBoolean("htMirrorConfirmed", true);
            actor.getLookControl().setLookAt(player, 20, 15);
            long nextRoute = data.getLong("htMirrorNextRoute");
            if (time >= nextRoute) {
                // At 200 participants, 32 stable time slots share the existing global ten-path/tick budget.
                data.putLong("htMirrorNextRoute", time + (retreat ? 8 : Math.min(32, 6 + members / 8)));
                Vec3 old = tacticPoint(actor);
                if (retreat || !settled && (TACTICAL_ROUTES.get(actor) == null
                        || old.distanceToSqr(desired) > 4 || actor.getNavigation().isDone())) {
                    boolean routed = false;
                    int side = (slot & 1) == 0 ? -1 : 1;
                    Vec3 tangent = player.position().add(relative.normalize().scale(Math.max(clearance + 2, distance)))
                            .add(-relative.normalize().z * side * 5, 0, relative.normalize().x * side * 5);
                    routeSearch: for (Vec3 destination : new Vec3[] {desired, tangent}) for (int dy : new int[] {0, -1, 1, -2}) {
                        Vec3 step = destination.subtract(actor.position()).multiply(1, 0, 1);
                        Vec3 waypoint = step.lengthSqr() > 14 * 14
                                ? actor.position().add(step.normalize().scale(14)) : destination;
                        Vec3 candidate = Vec3.atBottomCenterOf(BlockPos.containing(waypoint).offset(0, dy, 0));
                        if (candidate.distanceToSqr(player.position()) < clearance * clearance
                                || !meleeRoute(level, actor, candidate)) continue;
                        if (SplinterHivemind.safeRoute(actor, level, candidate)) {
                            var path = TACTICAL_ROUTES.get(actor);
                            double minimum = Math.min(clearance * clearance, Math.max(0, distance * distance - 0.25));
                            boolean avoidsPlayer = path != null && path.getNodeCount() > 0;
                            for (int i = 0; avoidsPlayer && i < path.getNodeCount(); i++)
                                avoidsPlayer = path.getEntityPosAtNode(actor, i).distanceToSqr(player.position()) >= minimum;
                            if (!avoidsPlayer) { TACTICAL_ROUTES.remove(actor); continue; }
                            point(actor, path.getEntityPosAtNode(actor, path.getNodeCount() - 1));
                            routed = true; break routeSearch;
                        }
                    }
                    if (!routed && retreat) {
                        actor.getNavigation().stop(); TACTICAL_ROUTES.remove(actor); point(actor, actor.position());
                    }
                }
            }
            finishTactic(actor, mineProgress);
            return true;
        }

				if (other != self && other.isAlive() && SplinterHivemind.member(other) && other.getTarget() == player
						&& !SplinterHivemind.frozen(other) && progress(other) == 0
						&& other.getHealth() > other.getMaxHealth() * 0.35 && other.hasLineOfSight(player)
						&& other.isWithinMeleeAttackRange(player)
						&& other.getAttributeValue(Attributes.MOVEMENT_SPEED) > 0.01) ready[0] = true;
				return ready[0] || ++checked[0] >= 24 ? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
						: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
			});
			return ready[0];
		}

        private static boolean coverNote(Mob self, String reason) {

        /** Weaving requires a presently readied weapon aimed through this mob's body; cover reacts to incoming shots separately. */
        private static boolean rangedThreatNow(Mob self, Player player) {
            if (!(self.level() instanceof ServerLevel level) || !validPlayer(player, level)
                    || !SplinterHivemind.bowDrawn(self, player) || !self.hasLineOfSight(player)) return false;
            Vec3 eye = player.getEyePosition();
            double range = Math.min(64.0, self.distanceTo(player) + self.getBbWidth() + 1.0);
            Vec3 end = eye.add(player.getLookAngle().normalize().scale(range));
            // Intersect the actual body, not a broad angular cone; released arrows do not sustain weaving.
            var hit = self.getBoundingBox().inflate(0.10).clip(eye, end);
            return hit.isPresent() && SplinterHivemind.loadedLine(level, eye, hit.get())
                    && level.clip(new ClipContext(eye, hit.get(), ClipContext.Block.COLLIDER,
                            ClipContext.Fluid.NONE, player)).getType() == HitResult.Type.MISS;
        }

        private static double weaveWidth(double distance, int learned) {
            double near = Mth.clamp((28.0 - distance) / 23.0, 0.0, 1.0);
            return (0.85 + near * 1.85) * (0.85 + Mth.clamp(learned, 0, 100) * 0.0015);
        }

        /** Prepared cover still uses normal griefing permissions, two-stage placement, range and squad budgets. */
        private static boolean prepareAimedCover(ServerLevel level, Mob self, Player player,
                net.minecraft.nbt.CompoundTag record, long time, boolean anticipated) {
            int learned = skill(record);
            if (!anticipated || learned < 35 || self.distanceToSqr(player) < 5 * 5
                    || self.distanceToSqr(player) > 48 * 48
                    || time - (self.getPersistentData().contains("ahAimAt") ? self.getPersistentData().getLong("ahAimAt") : time) < Math.max(5, 14 - learned / 10))
                return coverNote(self, "waiting for an incoming shot or sustained aim");
            if (requiresThickCover(self, player)) return coverNote(self, "wood failed against this weapon; seeking terrain cover");
            if (time < self.getPersistentData().getLong("ahNextBuild")
                    || time < record.getLong("NextBuild")) return coverNote(self, "build cooldown");
            boolean freeSlot = false;
            for (int c = 0; c < COVER_SLOTS; c++) if (!record.contains("Cover" + c)) { freeSlot = true; break; }
            if (!freeSlot) return coverNote(self, "all tracked cover slots occupied");
            if (time - record.getLong("BuildWindow") < 1200L
                    && record.getInt("BuildCount") + 2 > 8 + learned * 32 / 100)
                return coverNote(self, "per-player minute build budget exhausted");
            Vec3 toward = player.getEyePosition().subtract(self.getEyePosition()).multiply(1, 0, 1).normalize();
            Vec3 eye = self.getEyePosition();
            for (double distance : new double[] {2.2, 1.5, 2.8}) {
                BlockPos pos = BlockPos.containing(eye.add(toward.scale(distance)));
                if (placeCover(level, self, player, null, pos, eye, record, time, null)) {
                    self.getPersistentData().putLong("ahNextBuild", time + Math.max(10, 80 - learned * 7 / 10));
                    self.getPersistentData().remove("hive_bow_post_time");
                    self.getPersistentData().remove("hive_post_time");
                    self.getPersistentData().putInt("hive_bow_state", 0);
                    self.getPersistentData().putLong("hive_bow_cooldown", level.getGameTime() + 20L);
                    self.getNavigation().stop();
                    swingBuild(self, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState());
                    coverNote(self, "building prepared cover"); return true;
                }
            }
            return coverNote(self, "no safe supported two-block column on the firing line");
        }

        private static boolean trackedCover(ServerLevel level, LivingEntity target, BlockPos pos) {
            if (!(target instanceof Player player) || !level.hasChunkAt(pos)
                    || !level.getBlockState(pos).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) return false;

        }

        /** Only own/shared temporary cover gets this route-first exception; ordinary mining rules are unchanged. */
        public static boolean detourAroundCover(LevelAccessor world, Mob self, LivingEntity target, BlockPos... obstruction) {
            if (!(world instanceof ServerLevel level) || target == null || !target.isAlive()
                    || self.getTarget() != target || SplinterHivemind.frozen(self) || mirrorAttackBlocked(self)
                    || target instanceof Player player && roseHeld(player)) return false;
            BlockPos cover = null;
            for (BlockPos pos : obstruction) if (pos != null && trackedCover(level, target, pos)) { cover = pos; break; }
            if (cover == null) return false;
            var data = self.getPersistentData();
            long time = now(level);
            var path = COVER_DETOURS.get(self);
            boolean same = data.getLong("ahDetourCover") == cover.asLong()
                    && target.getUUID().toString().equals(data.getString("ahDetourOwner"))
                    && time < data.getLong("ahDetourUntil")
                    && target.distanceToSqr(new Vec3(data.getDouble("ahDetourTargetX"), target.getY(),
                            data.getDouble("ahDetourTargetZ"))) < 4;
            if (!same || path == null || path.isDone()) {
                if (time < data.getLong("ahDetourRetry")) return false;
                data.putLong("ahDetourRetry", time + 8);
                path = null;
                Vec3 toward = self.position().subtract(target.position()).multiply(1, 0, 1).normalize();
                Vec3 anchor = self.distanceToSqr(target) > 10 * 10
                        ? self.position().subtract(toward.scale(6)) : target.position();
                Vec3[] goals = {anchor.add(toward.scale(1.8)),
                        anchor.add(-toward.z * 2, 0, toward.x * 2),
                        anchor.add(toward.z * 2, 0, -toward.x * 2)};
                for (Vec3 goal : goals) {
                    for (int dy : new int[] {0, -1, 1}) {
                        path = SplinterHivemind.tacticalPath(self, level, BlockPos.containing(goal).offset(0, dy, 0));
                        if (path != null) break;
                    }
                    if (path != null) break;
                }
                if (path == null) return false; // A genuinely blocked route may still use the original mining fallback.
                COVER_DETOURS.put(self, path);
                data.putLong("ahDetourCover", cover.asLong()); data.putString("ahDetourOwner", target.getUUID().toString());
                data.putLong("ahDetourUntil", time + 30);
                data.putDouble("ahDetourTargetX", target.getX()); data.putDouble("ahDetourTargetZ", target.getZ());
            }
            clearTactic(self, time);
            data.remove("hive_post_time"); data.remove("hive_bow_post_time"); data.remove("blindspot_flank_post");
            data.putInt("hive_bow_state", 0); data.putLong("ahDetourPost", level.getGameTime());
            self.getNavigation().moveTo(path, 1.0);
            int next = Math.min(path.getNextNodeIndex(), path.getNodeCount() - 1);
            Vec3 step = path.getEntityPosAtNode(self, next);
            self.getLookControl().setLookAt(step.x, self.getEyeY(), step.z, 25, 20);
            clearMiningVisual(self, level);
            return true;
        }

        public static boolean finishCoverDetour(Mob self, int mineProgress) {

            return true;
        }

            });

        }

		private static boolean rangedPressure(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time, int learned) {
			if (self instanceof PetrifiedLogSplinterEntity || learned < 35 || !self.hasLineOfSight(player) || self.distanceToSqr(player) > 28 * 28
					|| !rangedThreatNow(self, player)) return false;
			if (self.distanceToSqr(player) < 5 * 5) return meleePressure(level, self, player, record, time);
			if (time < self.getPersistentData().getLong("htWeaveRetry")) return false;
			self.getPersistentData().putLong("htWeaveRetry", time + 12);
			int side = self.getPersistentData().contains("htWeaveSide")
                    ? -self.getPersistentData().getInt("htWeaveSide") : (self.getUUID().hashCode() & 1) == 0 ? 1 : -1;
			return weaveStep(level, self, player, time, learned, side, true)
					|| weaveStep(level, self, player, time, learned, -side, true);
		}

		private static boolean weaveStep(ServerLevel level, Mob self, Player player, long time, int learned,
                int side, boolean start) {
            if (!rangedThreatNow(self, player)) return false;
            var data = self.getPersistentData();
            Vec3 toward = player.position().subtract(self.position()).multiply(1, 0, 1).normalize();
            if (toward.lengthSqr() < 0.01) return false;
            Vec3 center = start ? self.position() : new Vec3((data.contains("htWeaveCenterX") ? data.getDouble("htWeaveCenterX") : self.getX()),
                    self.getY(), (data.contains("htWeaveCenterZ") ? data.getDouble("htWeaveCenterZ") : self.getZ()));
            Vec3 oldDirection = new Vec3((data.contains("htWeaveDirX") ? data.getDouble("htWeaveDirX") : toward.x), 0,
                    (data.contains("htWeaveDirZ") ? data.getDouble("htWeaveDirZ") : toward.z));
            if (!start && oldDirection.dot(toward) > 0) toward = oldDirection.lerp(toward, 0.25).normalize();
            double width = weaveWidth(self.distanceTo(player), learned);
            double advance = self.distanceToSqr(player) < 10 * 10 ? 0.9 : 1.6;
            center = center.add(toward.scale(advance));
            Vec3 goal = center.add(-toward.z * side * width, 0, toward.x * side * width);
            // Stay on the local terrain rather than oscillating toward a point above a descending slope.
            Vec3 grounded = null;
            for (int dy : new int[] {0, -1, 1}) {
                Vec3 candidate = Vec3.atBottomCenterOf(BlockPos.containing(goal).offset(0, dy, 0));
                if (candidate.distanceToSqr(player.position()) >= 4 * 4 && meleeRoute(level, self, candidate)) {
                    grounded = candidate; break;
                }
            }
            if (grounded == null) return false;
            if (start) {
                if (!startTactic(level, self, player, SERPENTINE, grounded, 80, time)) return false;
            } else {
                if (!SplinterHivemind.safeRoute(self, level, grounded)) return false;
                var route = TACTICAL_ROUTES.get(self);
                point(self, route != null && route.getNodeCount() > 0
                        ? route.getEntityPosAtNode(self, route.getNodeCount() - 1) : grounded);
            }
            data.putDouble("htWeaveCenterX", center.x); data.putDouble("htWeaveCenterZ", center.z);
            data.putDouble("htWeaveDirX", toward.x); data.putDouble("htWeaveDirZ", toward.z);
            data.putInt("htWeaveSide", side);
            data.putLong("htWeaveNext", time + 4);
            double speed = Math.max(0.08, self.getAttributeValue(Attributes.MOVEMENT_SPEED) * 0.6);
            data.putLong("htWeaveDeadline", time + Mth.clamp((int) Math.ceil(self.position().distanceTo(grounded) / speed) + 12, 16, 60));
            return true;
        }

		// Use normal movement control, never a velocity shove or expanded attack hitbox.
		public static boolean closeForMelee(Mob self, int mineProgress) {
			if (!(self.level() instanceof ServerLevel level) || mineProgress != 0 || SplinterHivemind.frozen(self)
					|| !self.isAlive() || self.isPassenger() || self.isInWater() || self.hurtTime > 0 || !self.onGround()
					|| !(self.getTarget() instanceof Player player) || !validPlayer(player, level)
					|| self.distanceToSqr(player) > 4 * 4 || Math.abs(self.getY() - player.getY()) > 1
					|| !self.hasLineOfSight(player) || (self instanceof PetrifiedLogSplinterEntity
                            ? self.distanceToSqr(player) < 1.44 : self.isWithinMeleeAttackRange(player))
					|| self.getAttributeValue(Attributes.MOVEMENT_SPEED) <= 0.01
					|| player.getMainHandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())
					|| player.getOffhandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())) return false;
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			int mode = data.getInt("htState");
			if (mode != 0 && mode != MELEE_PRESSURE || navigationOwned(self)
					|| now(level) < data.getLong("ahPlacePose")) return false;
			if (!self.getNavigation().isDone()) return false;
			Vec3 forward = player.position().subtract(self.position()).multiply(1, 0, 1).normalize().scale(0.25);
			Vec3 next = self.position().add(forward);
			AABB body = self.getBoundingBox().move(forward);
			if (!level.hasChunkAt(BlockPos.containing(body.minX, body.minY, body.minZ))
					|| !level.hasChunkAt(BlockPos.containing(body.maxX, body.maxY, body.maxZ))
					|| !level.getWorldBorder().isWithinBounds(body) || !level.noCollision(self, body)) return false;
			for (double x : new double[] {body.minX + 0.02, body.maxX - 0.02})
				for (double z : new double[] {body.minZ + 0.02, body.maxZ - 0.02}) {
					BlockPos floor = BlockPos.containing(x, self.getY() - 0.1, z);
					if (!level.getBlockState(floor).isFaceSturdy(level, floor, net.minecraft.core.Direction.UP)
							|| !level.getFluidState(floor).isEmpty()) return false;
				}
			self.getMoveControl().setWantedPosition(next.x, self.getY(), next.z, 1.0);
			return true;
		}

		private static final class MiningVisual {
			final BlockPos[] cells = new BlockPos[3];
			final int[] stages = {-1, -1, -1};
			long tick;
		}
		private static final java.util.Map<Mob, MiningVisual> MINING_VISUALS = new java.util.WeakHashMap<>();

		public static void showMiningCracks(Mob self, int progress, double threshold, BlockPos... positions) {
			if (!(self.level() instanceof ServerLevel level) || !self.isAlive() || progress <= 0
					|| SplinterHivemind.frozen(self) || !Double.isFinite(threshold)) return;
			if (!MINING_VISUALS.containsKey(self) && MINING_VISUALS.size() >= 512) return;
			MiningVisual visual = MINING_VISUALS.computeIfAbsent(self, key -> new MiningVisual());
			visual.tick = level.getGameTime();
			int stage = Mth.clamp((int) (progress * 10.0 / Math.max(1.0, threshold)), 0, 9);
			for (int i = 0; i < 3; i++) {
				BlockPos pos = i < positions.length ? positions[i] : null;
				if (pos != null && (!level.hasChunkAt(pos) || level.getBlockState(pos).isAir())) pos = null;
				int breaker = -1 - self.getId() * 4 - i;
				if (visual.cells[i] != null && !visual.cells[i].equals(pos)) {
					level.destroyBlockProgress(breaker, visual.cells[i], -1); visual.stages[i] = -1;
				}
				if (pos != null && (!pos.equals(visual.cells[i]) || visual.stages[i] != stage))
					level.destroyBlockProgress(breaker, pos, stage);
				visual.cells[i] = pos; visual.stages[i] = pos == null ? -1 : stage;
			}
		}

		private static void clearMiningVisual(Mob self, ServerLevel level) {
			MiningVisual visual = MINING_VISUALS.remove(self);
			if (visual == null) return;
			for (int i = 0; i < 3; i++) if (visual.cells[i] != null)
				level.destroyBlockProgress(-1 - self.getId() * 4 - i, visual.cells[i], -1);
		}

		private static final class BuildingHand {
			ItemStack display = ItemStack.EMPTY, actual = ItemStack.EMPTY;
			long until, post = -1;
		}
		private static final java.util.Map<Mob, BuildingHand> BUILDING_HANDS = new java.util.WeakHashMap<>();

		public static void showBuildingHand(Mob self, BlockState material) {
			if (!(self.level() instanceof ServerLevel level) || !self.isAlive()) return;
			if (!BUILDING_HANDS.containsKey(self) && BUILDING_HANDS.size() >= 512) return;
			BuildingHand visual = BUILDING_HANDS.computeIfAbsent(self, key -> new BuildingHand());
			visual.display = new ItemStack(material.getBlock());
			visual.until = level.getGameTime() + 8;
			visual.post = -1; // Reapply after vanilla's equipment synchronization on this placement tick.

		}

		private static void sendHand(ServerLevel level, Mob self, ItemStack item) {
			// Client-only equipment view: no fake inventory item, changed combat damage, or generated item drops.
			level.getChunkSource().broadcastAndSend(self,
					new net.minecraft.network.protocol.game.ClientboundSetEquipmentPacket(self.getId(),
							java.util.List.of(com.mojang.datafixers.util.Pair.of(
									net.minecraft.world.entity.EquipmentSlot.MAINHAND, item.copy()))));
		}

		private static void updateVisuals(Mob self, ServerLevel level, boolean post) {
			MiningVisual cracks = MINING_VISUALS.get(self);
			if (cracks != null && (!self.isAlive() || SplinterHivemind.frozen(self)
					|| progress(self) == 0 || cracks.tick < level.getGameTime()))
				clearMiningVisual(self, level);
			if (!post) return;
			BuildingHand visual = BUILDING_HANDS.get(self);
			if (visual == null) return;
			if (!self.isAlive() || self.isRemoved() || progress(self) > 0 || level.getGameTime() >= visual.until) {
				sendHand(level, self, self.getMainHandItem()); BUILDING_HANDS.remove(self); return;
			}
			if (visual.post == level.getGameTime()) return;
			if (visual.post < 0 || (level.getGameTime() & 1) == 0
					|| !ItemStack.matches(visual.actual, self.getMainHandItem())) {
				sendHand(level, self, visual.display);
				visual.actual = self.getMainHandItem().copy();
			}
			visual.post = level.getGameTime();
		}

        private static boolean learningHere(ServerLevel level) {

                case "the_backwoods:backwoods", "the_backwoods:the_grain",

		private static long now(ServerLevel level) { return level.getServer().overworld().getGameTime(); }

		private static boolean hasCover(net.minecraft.nbt.CompoundTag record) {
			for (int c = 0; c < COVER_SLOTS; c++) if (record.contains("Cover" + c)) return true;
			return false;
		}

		private static int slot(net.minecraft.nbt.CompoundTag root, Player player, long now, boolean create) {
			int free = -1;
			long oldest = Long.MAX_VALUE;
			int evict = -1;
			for (int i = 0; i < PLAYERS; i++) {
				String key = "Player" + i;
				if (!root.contains(key)) { if (free < 0) free = i; continue; }
				net.minecraft.nbt.CompoundTag record = root.getCompound(key);
				if (player.getUUID().toString().equals(record.getString("UUID"))) return i;
				if (!hasCover(record) && record.getLong("Seen") < oldest
						&& now - record.getLong("Seen") > 24000L) {
					oldest = record.getLong("Seen"); evict = i;
				}
			}
			if (!create) return -1;
			int chosen = free >= 0 ? free : evict;
			if (chosen < 0) return -1;
			net.minecraft.nbt.CompoundTag record = new net.minecraft.nbt.CompoundTag();
			record.putString("UUID", player.getUUID().toString());
			record.putLong("Seen", now);
			record.putLong("Decay", now);
			root.put("Player" + chosen, record);
			return chosen;
		}

		private static void decay(net.minecraft.nbt.CompoundTag record, long now) {
			long elapsed = Math.max(0L, now - record.getLong("Decay"));
			long halfLife = 24000L * Math.max(1, MEMORY_HALF_LIFE_DAYS);
			if (elapsed < halfLife) return;
			int periods = (int) Math.min(12L, elapsed / halfLife);
			record.putDouble("Knowledge", Math.scalb(knowledge(record), -periods));
			for (String key : new String[] {"Shots", "Hits", "Blocks", "Left", "Right"})
				record.putInt(key, record.getInt(key) >> periods);
			for (int mode = 1; mode <= SERPENTINE; mode++) {
				record.putInt("Loss" + mode, record.getInt("Loss" + mode) >> periods);
				record.putInt("Win" + mode, record.getInt("Win" + mode) >> periods);
			}
			record.putInt("WoodFailures", record.getInt("WoodFailures") >> periods);
			for (int i = 0; i < 3; i++) {
				net.minecraft.nbt.CompoundTag perch = record.getCompound("Perch" + i);
				if (!perch.isEmpty()) perch.putInt("Uses", perch.getInt("Uses") >> periods);
			}
			for (int i = 0; i < 2; i++) {
				net.minecraft.nbt.CompoundTag cadence = record.getCompound("Cadence" + i);
				if (!cadence.isEmpty()) cadence.putInt("Samples", cadence.getInt("Samples") >> periods);
				net.minecraft.nbt.CompoundTag melee = record.getCompound("Melee" + i);
				if (!melee.isEmpty()) for (String key : new String[] {"Samples", "Hits", "Swings"})
					melee.putInt(key, melee.getInt(key) >> periods);
			}
			record.putLong("Decay", now);
		}

		private static double knowledge(net.minecraft.nbt.CompoundTag record) {
			if (record.contains("Knowledge")) {
				double saved = record.getDouble("Knowledge");
				if (Double.isFinite(saved)) return Mth.clamp(saved, 0.0, 100.0);
			}
			// Migrate existing worlds at their current experience, without applying the new rate retroactively.
			return Math.min(100, record.getInt("Shots") * 0.5 + record.getInt("Hits")
					+ record.getInt("Blocks") * 2);
		}

		private static int skill(net.minecraft.nbt.CompoundTag record) {
			return (int) Math.floor(knowledge(record));
		}

		private static void gain(net.minecraft.nbt.CompoundTag record, double points) {
			double rate = Double.isFinite(LEARNING_RATE) ? Mth.clamp(LEARNING_RATE, 0.0, 16.0) : 1.0;
			if (!Double.isFinite(points)) return;
			record.putDouble("Knowledge", Math.min(100.0, knowledge(record) + Math.max(0.0, points) * rate));
		}

		public static int experience(Mob self, Player player) {
			if (player == null || !(self.level() instanceof ServerLevel level)) return 0;
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			long now = now(level);
			if (player.getUUID().toString().equals(data.getString("ahOwner"))
					&& now < data.getLong("ahSkillUntil")) return data.getInt("ahSkill");
			net.minecraft.nbt.CompoundTag root = root(level);
			int slot = slot(root, player, now, false);
			int value = 0;
			if (slot >= 0) {
				net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
				long before = record.getLong("Decay");
				decay(record, now);
				value = skill(record);
				if (record.getLong("Decay") != before) save(level, root);
			}
			data.putString("ahOwner", player.getUUID().toString());
			data.putInt("ahSkill", value);
			data.putLong("ahSkillUntil", now + 100L);
			return value;
		}


		public static boolean afterimageDebugActive(net.minecraft.server.level.ServerPlayer player) {
			ServerLevel level = (ServerLevel) player.level();
			long time = now(level);
			long until = player.getPersistentData().getLong("hiveDebugAfterimageUntil");
			return until > time && until - time <= 6000L && hiveDebugPermission(player);
		}

		/** Siblings keep complementary sides; learned player strafing biases their approach. */
		public static int flankSide(Mob self, Player player) {
			int split = (self.getUUID().hashCode() & 1) == 0 ? 1 : -1;
			if (experience(self, player) < 25 || !(self.level() instanceof ServerLevel level)) return split;
			net.minecraft.nbt.CompoundTag root = root(level);
			int s = slot(root, player, now(level), false);
			if (s < 0) return split;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
			int bias = record.getInt("Right") - record.getInt("Left");
			if (Math.abs(bias) < 6 || Math.floorMod(self.getUUID().hashCode(), 3) == 0) return split;
			return bias > 0 ? -1 : 1;
		}

		private static boolean validPlayer(Player player, ServerLevel level) {
			return player != null && player.isAlive() && player.level() == level
					&& !player.isCreative() && !player.isSpectator();
		}


		private static final java.util.Map<net.minecraft.server.MinecraftServer, long[]> COVER_RAY_BUDGET =
				new java.util.WeakHashMap<>();

		private static int coverReactionTicks(int skill) { return Math.max(1, 9 - Mth.clamp(skill, 0, 100) / 12); }
		private static int projectileHorizon(int skill) { return 18 + Mth.clamp(skill, 0, 100) * 42 / 100; }
		private static double projectileScanRange(int skill) { return 24.0 + Mth.clamp(skill, 0, 100) * 0.4; }

		private static boolean coverRayBudget(ServerLevel level) {
			long time = now(level);
			long[] budget = COVER_RAY_BUDGET.computeIfAbsent(level.getServer(), key -> new long[] {-1L, 0L});
			if (budget[0] != time) { budget[0] = time; budget[1] = 0L; }
			return budget[1]++ < 96L;
		}

		/** Retain the original entry point; learned defense uses the bounded extended forecast below. */
		private static double[] incoming(Mob self, net.minecraft.world.entity.projectile.Projectile shot) {
			return incoming(self, shot, 18);
		}

		private static double[] incoming(Mob self, net.minecraft.world.entity.projectile.Projectile shot, int horizon) {
			Vec3 velocity = shot.getDeltaMovement();
			Vec3 point = shot.position();
			Vec3 toward = self.getBoundingBox().getCenter().subtract(point);
			if (!Double.isFinite(velocity.lengthSqr()) || velocity.lengthSqr() < 0.015
					|| !Double.isFinite(point.x) || !Double.isFinite(point.y) || !Double.isFinite(point.z)
					|| velocity.dot(toward) <= 0.0) return null;
			AABB body = self.getBoundingBox().inflate(0.12 + Math.min(0.35, shot.getBbWidth() * 0.5));
			if (body.contains(point)) return null; // Already inside the victim: no retroactive interception.
			double gravity = shot.isNoGravity() ? 0.0 : shot.getGravity();
			if (!Double.isFinite(gravity)) return null;
			horizon = Mth.clamp(horizon, 1, 60);
			// Header: impact ETA/position/velocity; then the actual predicted polyline, including its start.
			double[] path = new double[7 + (horizon + 1) * 3];
			path[7] = point.x; path[8] = point.y; path[9] = point.z;
			for (int tick = 1; tick <= horizon; tick++) {
				Vec3 next = point.add(velocity);
				java.util.Optional<Vec3> hit = body.clip(point, next);
				Vec3 step = hit.orElse(next);
				int offset = 7 + tick * 3;
				path[offset] = step.x; path[offset + 1] = step.y; path[offset + 2] = step.z;
				if (hit.isPresent()) {
					path[0] = tick; path[1] = step.x; path[2] = step.y; path[3] = step.z;
					path[4] = velocity.x; path[5] = velocity.y; path[6] = velocity.z;
					return java.util.Arrays.copyOf(path, offset + 3);
				}
				point = next;
				velocity = velocity.scale(shot.isInWater() ? 0.6 : 0.99).add(0.0, -gravity, 0.0);
			}
			return null;
		}

		private static boolean predictedCoverPathClear(ServerLevel level,
				net.minecraft.world.entity.projectile.Projectile shot, BlockPos pos, double[] prediction) {
			AABB cell = new AABB(pos).inflate(0.12);
			Vec3 previous = new Vec3(prediction[7], prediction[8], prediction[9]);
			Vec3 entry = null;
			int end = -1;
			for (int offset = 10; offset + 2 < prediction.length; offset += 3) {
				Vec3 next = new Vec3(prediction[offset], prediction[offset + 1], prediction[offset + 2]);
				java.util.Optional<Vec3> hit = cell.clip(previous, next);
				if (cell.contains(previous) || hit.isPresent()) { entry = hit.orElse(previous); end = offset; break; }
				previous = next;
			}
			if (entry == null) return false;
			previous = new Vec3(prediction[7], prediction[8], prediction[9]);
			// Two-tick chords closely follow the curve, instead of testing a false straight line to impact.
			for (int offset = 10; offset <= end;) {
				int last = Math.min(end, offset + 3);
				Vec3 next = last == end ? entry : new Vec3(prediction[last], prediction[last + 1], prediction[last + 2]);
				if (!SplinterHivemind.loadedLine(level, previous, next) || !coverRayBudget(level)
						|| level.clip(new ClipContext(previous, next, ClipContext.Block.COLLIDER,
								ClipContext.Fluid.NONE, shot)).getType() != HitResult.Type.MISS) return false;
				previous = next;
				offset = last + 3;
			}
			return true;
		}

		private static boolean alternateCover(ServerLevel level, Mob self, Player player,
				net.minecraft.world.entity.projectile.Projectile shot, Vec3 impact, Vec3 flight, BlockPos original,
				net.minecraft.nbt.CompoundTag record, long now, double[] prediction, int skill) {
			if (skill < 30) return false;
			java.util.Set<BlockPos> candidates = new java.util.LinkedHashSet<>();
			for (double distance : new double[] {2.8, 2.2, 1.6, 1.0, 0.6})
				candidates.add(BlockPos.containing(impact.subtract(flight.scale(distance))));
			for (int offset = 7; offset + 2 < prediction.length; offset += 3) {
				Vec3 point = new Vec3(prediction[offset], prediction[offset + 1], prediction[offset + 2]);
				if (self.distanceToSqr(point) <= 3.2 * 3.2) candidates.add(BlockPos.containing(point));
				if (candidates.size() >= 16) break;
			}
			candidates.remove(original);
			int tried = 0, limit = skill >= 90 ? 10 : skill >= 60 ? 6 : 3;
			for (BlockPos pos : candidates) {
				if (self.distanceToSqr(Vec3.atCenterOf(pos)) > 3.2 * 3.2
						|| pos.getY() < self.blockPosition().getY() || pos.getY() > self.blockPosition().getY() + 1) continue;
				if (++tried > limit) break;
				if (placeCover(level, self, player, shot, pos, impact, record, now, prediction)) return true;
			}
			return false;
		}

		private static void noteShot(ServerLevel level, Player player,
				net.minecraft.world.entity.projectile.Projectile shot, net.minecraft.nbt.CompoundTag record, long now) {
			if (!learningHere(level) || shot.getPersistentData().getBoolean("ahWitnessed")
					|| now < record.getLong("NextSample")) return;
			shot.getPersistentData().putBoolean("ahWitnessed", true);
			decay(record, now);
			gain(record, SHOT_LEARNING_GAIN);
			record.putInt("Shots", Math.min(240, record.getInt("Shots") + 1));
			Vec3 look = player.getLookAngle();
			Vec3 movement = player.getDeltaMovement();
			double lateral = movement.x * look.z - movement.z * look.x;
			if (Math.abs(lateral) > 0.06) {
				String key = lateral > 0 ? "Right" : "Left";
				record.putInt(key, Math.min(240, record.getInt(key) + 1));
			}
			record.putLong("NextSample", now + Math.max(1, SHOT_SAMPLE_INTERVAL_TICKS));
			record.putLong("Seen", now);
		}

		public static boolean defend(Mob self, Player player, int mineProgress) {
			return defendFor(self, player, mineProgress, self);
		}

		private static boolean defendFor(Mob self, Player player, int mineProgress, Mob ward) {
            if (!(self.level() instanceof ServerLevel level)) return false;
            if (self instanceof PetrifiedLogSplinterEntity || ward instanceof PetrifiedLogSplinterEntity) return coverNote(self, "projectile-immune; cover unnecessary");
            net.minecraft.nbt.CompoundTag data = self.getPersistentData();
            if (self == ward) {
                if (data.contains("ahDefenseTick") && data.getLong("ahDefenseTick") == level.getGameTime()) return false;
                data.putLong("ahDefenseTick", level.getGameTime());
            }
			if (!validPlayer(player, level)
					|| !self.isAlive() || !ward.isAlive() || ward.level() != level || !ward.onGround()
					|| self.distanceToSqr(ward) > 6 * 6 || mineProgress != 0 || SplinterHivemind.frozen(self)
					|| self.getTarget() != player || self.isInWater() || !self.onGround()
					|| self.distanceToSqr(player) > 96.0 * 96.0
					|| player.getMainHandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())
					|| player.getOffhandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())) return false;
			long now = now(level);
            if (!net.neoforged.neoforge.event.EventHooks.canEntityGrief(level, self))
                return coverNote(self, "mob griefing disabled/denied");
			if (now < data.getLong("ahPlacePose")) {
				self.getNavigation().stop();
				return true;
			}
			boolean drawn = SplinterHivemind.bowDrawn(self, player);
			Vec3 aim = ward.getEyePosition().subtract(player.getEyePosition());
			boolean anticipated = drawn && aim.lengthSqr() > 0.01
					&& player.getLookAngle().dot(aim.normalize()) > 0.85 && self.hasLineOfSight(player);
			if (anticipated) {
				if (!player.getUUID().toString().equals(data.getString("ahAimOwner"))
						|| now - data.getLong("ahAimSeen") > 4L) {
					data.putLong("ahAimAt", now);
					data.putString("ahAimOwner", player.getUUID().toString());
				}
				data.putLong("ahAimSeen", now);
            } else {
                data.remove("ahAimAt"); data.remove("ahAimOwner"); data.remove("ahAimSeen");
            }
			int learned = experience(self, player);
			if (Math.floorMod(self.tickCount + self.getId(), learned >= 80 ? 1 : 2) != 0
					|| now < data.getLong("ahNextScan")) return false;
			net.minecraft.nbt.CompoundTag root = root(level);
			int s = slot(root, player, now, learningHere(level));
			if (s < 0) return false;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
			data.putInt("ahCoverPlayerSlot", s);
			int skill = skill(record);
			// At most two spatial projectile scans per player per server tick, shared by all variants.
			if (record.getLong("ScanTick") != now) { record.putLong("ScanTick", now); record.putInt("Scans", 0); }
			if (record.getInt("Scans") >= 2) return false;
			record.putInt("Scans", record.getInt("Scans") + 1);
			data.putLong("ahNextScan", now + (data.contains("ahProjectile") || anticipated ? (skill >= 80 ? 1L : 2L) : Math.max(2, 6 - skill / 25)));
			List<net.minecraft.world.entity.projectile.Projectile> shots = new ArrayList<>();
			int[] examined = {0};
			level.getEntities().get(net.minecraft.world.level.entity.EntityTypeTest.forClass(
					net.minecraft.world.entity.projectile.Projectile.class), ward.getBoundingBox().inflate(projectileScanRange(skill)), shot -> {
				if (shot.isAlive() && shot.getOwner() == player && shot.getDeltaMovement().lengthSqr() > 0.015)
					shots.add(shot);
				return ++examined[0] >= 32 + skill / 3 || shots.size() >= 8 + skill / 25 ? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
						: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
			});
			net.minecraft.world.entity.projectile.Projectile best = null;
			double[] prediction = null;
			for (net.minecraft.world.entity.projectile.Projectile shot : shots) {
				double[] predicted = incoming(ward, shot, projectileHorizon(skill));
				if (predicted == null || !self.hasLineOfSight(shot)) continue;
				if (prediction == null || predicted[0] < prediction[0]) { best = shot; prediction = predicted; }
			}
			if (best == null) {
                data.remove("ahProjectile"); data.remove("ahThreatUntil");
                boolean prepared = ward == self && prepareAimedCover(level, self, player, record, now, anticipated);
                save(level, root);
                return prepared;
            }
            data.putLong("ahThreatUntil", now + 6);
            data.putString("ahThreatOwner", player.getUUID().toString());
            data.putInt("ahThreatId", best.getId()); data.putString("ahThreatUUID", best.getUUID().toString());
			noteShot(level, player, best, record, now);
			self.getLookControl().setLookAt(best.getX(), best.getY(), best.getZ(), 30.0F, 20.0F);
			if (!best.getUUID().toString().equals(data.getString("ahProjectile"))) {
				data.putString("ahProjectile", best.getUUID().toString());
				boolean expected = player.getUUID().toString().equals(data.getString("ahAimOwner"))
						&& now - data.getLong("ahAimSeen") <= 8L;
				long observed = expected ? data.getLong("ahAimAt") : now;
				data.putLong("ahReactAt", Math.max(now, observed + coverReactionTicks(skill)));
			}
			data.putLong("ahNextScan", now + (skill >= 80 ? 1L : 2L));
			// Commit only after a real observation delay; immediate shots can still hit.
			if (now < data.getLong("ahReactAt") || prediction[0] < (skill >= 90 ? 1.0 : 2.0)
					|| now < data.getLong("ahNextBuild") || now < record.getLong("NextBuild")) {
				save(level, root); return coverNote(self, "observing projectile / build cooldown");
			}
			// Every variant may defend itself; shared NextBuild/scan limits still coordinate the squad.
			Vec3 flight = new Vec3(prediction[4], prediction[5], prediction[6]).normalize();
			Vec3 impact = new Vec3(prediction[1], prediction[2], prediction[3]);
			Vec3 intercept = impact.subtract(flight.scale(skill >= 80 ? 2.4 : 1.6));
			BlockPos pos = BlockPos.containing(intercept);
			if (requiresThickCover(self, player)) { save(level, root); return coverNote(self, "wood failed against this weapon; seeking terrain cover"); }
			if (placeCover(level, self, player, best, pos, impact, record, now, prediction)
					|| alternateCover(level, self, player, best, impact, flight, pos, record, now, prediction, skill)) {
				data.putLong("ahNextBuild", now + Math.max(10, 80 - skill * 7 / 10));
				data.putLong("ahPlacePose", now + coverStepDelay(skill) + 2);
				data.remove("hive_bow_post_time");
				data.remove("hive_post_time");
				data.putInt("hive_bow_state", 0);
				data.putLong("hive_bow_cooldown", level.getGameTime() + 20L);
				self.getNavigation().stop();
				swingBuild(self, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState());
				coverNote(self, "building interception cover");
                save(level, root);
				return true;
			}
            coverNote(self, "no safe supported column along the observed projectile path");
			data.putLong("ahNextBuild", now + Math.max(3, 12 - skill / 12));
			save(level, root);
			return false;
		}


		// A tiny server-scoped budget prevents a large squad from issuing an A* search per member per tick.
		private static final java.util.Map<net.minecraft.server.MinecraftServer, long[]> PATH_BUDGET =
				new java.util.WeakHashMap<>();
		private static final java.util.Map<Mob, net.minecraft.world.level.pathfinder.Path> TACTICAL_ROUTES =
				new java.util.WeakHashMap<>();

		private static boolean pathBudget(ServerLevel level) {
			long time = now(level);
			long[] budget = PATH_BUDGET.computeIfAbsent(level.getServer(), key -> new long[] {-1, 0});
			if (budget[0] != time) { budget[0] = time; budget[1] = 0; }
			return budget[1]++ < 10;
		}

		private static Vec3 lastSeenEye(Mob self, Player player) {
			if (!(self.level() instanceof ServerLevel level) || !validPlayer(player, level)) return null;
			if (self.hasLineOfSight(player)) return player.getEyePosition();
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			if (player.getUUID().toString().equals(data.getString("htSeenOwner"))
					&& now(level) - data.getLong("htSeenAt") <= 60)
				return new Vec3(data.getDouble("htLookX"), data.getDouble("htLookY"), data.getDouble("htLookZ"));
			if (SplinterHivemind.current(self, player, level))
				return new Vec3(data.getDouble("hive_x"), data.getDouble("hive_y") + player.getEyeHeight(), data.getDouble("hive_z"));
			return null;
		}

		private static net.minecraft.nbt.CompoundTag meleeMemory(net.minecraft.nbt.CompoundTag record,
				Player player, boolean create) {
			String held = net.minecraft.core.registries.BuiltInRegistries.ITEM.getKey(player.getMainHandItem().getItem()).toString();
			int chosen = 0;
			for (int i = 0; i < 2; i++) {
				net.minecraft.nbt.CompoundTag entry = record.getCompound("Melee" + i);
				if (held.equals(entry.getString("Weapon"))) return entry;
				if (entry.getLong("Seen") < record.getCompound("Melee" + chosen).getLong("Seen")) chosen = i;
			}
			net.minecraft.nbt.CompoundTag entry = new net.minecraft.nbt.CompoundTag();
			if (create) { entry.putString("Weapon", held); record.put("Melee" + chosen, entry); }
			return entry;
		}

		// Surface distance, not centre-to-centre distance; this learns reach without giving mobs extra reach.
		private static double observedReach(Player player, Mob victim) {
			AABB box = victim.getBoundingBox();
			Vec3 eye = player.getEyePosition();
			double dx = Math.max(box.minX - eye.x, Math.max(0, eye.x - box.maxX));
			double dy = Math.max(box.minY - eye.y, Math.max(0, eye.y - box.maxY));
			double dz = Math.max(box.minZ - eye.z, Math.max(0, eye.z - box.maxZ));
			return Math.sqrt(dx * dx + dy * dy + dz * dz);
		}

		private static void observeMelee(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time) {
			if (!self.hasLineOfSight(player) || self.distanceToSqr(player) > 16 * 16
					|| time == record.getLong("MeleeObserveAt")) return;
			record.putLong("MeleeObserveAt", time);
			net.minecraft.nbt.CompoundTag memory = meleeMemory(record, player, learningHere(level));
			String dimension = level.dimension().location().toString();
			boolean continuous = time - record.getLong("MeleeVisibleAt") <= 3
					&& dimension.equals(record.getString("MeleeDimension"));
			boolean began = player.swinging && (!continuous || !record.getBoolean("MeleeAnimating")
					|| player.swingTime < record.getInt("MeleePhase"));
			record.putLong("MeleeVisibleAt", time);
			record.putString("MeleeDimension", dimension);
			record.putBoolean("MeleeAnimating", player.swinging);
			record.putInt("MeleePhase", player.swingTime);
			if (!began || player.swingTime > 1 || player.isUsingItem()
					|| player.swingingArm != InteractionHand.MAIN_HAND || !learningHere(level)
					|| self.distanceToSqr(player) > Math.pow(Math.max(3.0, memory.getDouble("Reach")) + 3, 2)) return;
			Vec3 toward = self.getBoundingBox().getCenter().subtract(player.getEyePosition());
			if (toward.lengthSqr() < 0.01 || player.getLookAngle().dot(toward.normalize()) < 0.35) return;
			// Mining/building hand animations are not evidence of a missed melee attack.
			Vec3 end = player.getEyePosition().add(player.getLookAngle().scale(
					Math.max(3.0, memory.getDouble("Reach")) + 1.0));
			if (!SplinterHivemind.loadedLine(level, player.getEyePosition(), end)
					|| level.clip(new ClipContext(player.getEyePosition(), end, ClipContext.Block.COLLIDER,
							ClipContext.Fluid.NONE, player)).getType() != HitResult.Type.MISS) return;
			long previous = memory.getLong("Swing");
			if (previous > 0 && time - previous >= 4 && time - previous <= 80) {
				double interval = time - previous;
				double mean = memory.getInt("Samples") == 0 ? interval : memory.getDouble("Interval");
				memory.putDouble("Interval", mean * 0.75 + interval * 0.25);
				memory.putDouble("Deviation", memory.getDouble("Deviation") * 0.75 + Math.abs(interval - mean) * 0.25);
				memory.putInt("Samples", Math.min(30, memory.getInt("Samples") + 1));
			}
			memory.putLong("Swing", time);
			memory.putLong("Seen", time);
			memory.putInt("Swings", Math.min(100, memory.getInt("Swings") + 1));
			boolean descending = !player.onGround() && player.getDeltaMovement().y < -0.05 && !player.isInWater();
			memory.putDouble("JumpBias", memory.getDouble("JumpBias") * 0.85 + (descending ? 0.15 : 0));
			Vec3 motion = player.getDeltaMovement();
			Vec3 look = player.getLookAngle().multiply(1, 0, 1).normalize();
			double side = look.x * motion.z - look.z * motion.x;
			memory.putDouble("StrafeBias", memory.getDouble("StrafeBias") * 0.8 + Math.signum(side) * 0.2);
			memory.putDouble("RetreatBias", memory.getDouble("RetreatBias") * 0.85
					+ (motion.dot(toward.normalize()) < -0.04 ? 0.15 : 0));
			record.putLong("MeleeContact", time);
			record.putLong("Seen", time);
			if (time >= record.getLong("MeleeGainAt")) {
				gain(record, SHOT_LEARNING_GAIN);
				record.putLong("MeleeGainAt", time + Math.max(1, SHOT_SAMPLE_INTERVAL_TICKS));
			}
		}

		private static void learnMeleeHit(ServerLevel level, Mob victim, Player player) {
			if (!learningHere(level) || !validPlayer(player, level)) return;
			double reach = observedReach(player, victim);
			if (!Double.isFinite(reach) || reach > 32) return;
			long time = now(level);
			net.minecraft.nbt.CompoundTag root = root(level);
			int slot = slot(root, player, time, true);
			if (slot < 0) return;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
			decay(record, time);
			net.minecraft.nbt.CompoundTag memory = meleeMemory(record, player, true);
			record.putLong("Seen", time);
			memory.putDouble("Reach", Math.max(memory.getDouble("Reach") * 0.98, reach));
			memory.putInt("Hits", Math.min(100, memory.getInt("Hits") + 1));
			memory.putLong("Hit", time); memory.putLong("Seen", time);
			record.putLong("MeleeContact", time);
			record.putString("MeleeDimension", level.dimension().location().toString());
			if (time >= record.getLong("MeleeHitGainAt")) {
				gain(record, HIT_LEARNING_GAIN);
				record.putLong("MeleeHitGainAt", time + Math.max(1, OUTCOME_SAMPLE_INTERVAL_TICKS));
			}
			save(level, root);
		}

		private static boolean meleeOpening(net.minecraft.nbt.CompoundTag memory, long time) {
			if (memory.getInt("Samples") < Math.max(2, CADENCE_SAMPLES_REQUIRED)) return false;
			long swing = memory.getLong("Swing"), since = time - swing;
			double window = Mth.clamp(memory.getDouble("Interval")
					- memory.getDouble("Deviation") * 2.0 - 3.0, 0, 16);
			// Wait long enough to observe a miss, never inspect the player's hidden attack cooldown.
			return swing > 0 && memory.getLong("Hit") < swing - 2 && since >= 6 && since <= window;
		}

		private static boolean meleeRoute(ServerLevel level, Mob self, Vec3 point) {
			if (!level.hasChunkAt(BlockPos.containing(point))) return false;
			AABB space = self.getBoundingBox().move(point.subtract(self.position())).inflate(0.15);
			if (!level.getEntitiesOfClass(Mob.class, space,
					other -> other != self && other.isAlive() && SplinterHivemind.member(other)).isEmpty()) return false;
			return SplinterHivemind.clearStandingRoom(self, level, BlockPos.containing(point));
		}

		private static boolean meleePressure(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time) {
			String dimension = level.dimension().location().toString(), owner = self.getUUID().toString();
			// Leader fields are hints only; never suppress a member's ordinary melee attack.
			record.putString("MeleeLeader", owner); record.putString("MeleeLeadDimension", dimension);
			record.putLong("MeleeLeadUntil", time + 20);
			record.putString("MeleePrevious", owner); record.putLong("MeleeRotateUntil", time + 32);
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			data.putInt("htState", MELEE_PRESSURE); data.putLong("htUntil", time + 20);
			data.putLong("htStarted", time); data.putLong("htLastTick", time);
			data.putString("htOwner", player.getUUID().toString()); data.putString("htDimension", dimension);
			data.putInt("htTrial", MELEE_PRESSURE); data.putLong("htTrialUntil", time + 60);
			data.putLong("htPost", level.getGameTime());
			data.remove("hive_bow_post_time"); data.remove("hive_post_time");
			data.putInt("hive_bow_state", 0); data.putLong("hive_route_until", 0);
			TACTICAL_ROUTES.remove(self);
			self.getNavigation().stop();
			// One hand-off path; vanilla melee is the sole continuing movement/attack owner.
			if (pathBudget(level)) self.getNavigation().moveTo(player, 1.0);
			return true;
		}


		private static boolean meleeTactics(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time, int learned) {
			if (learned < 20 || !self.hasLineOfSight(player) || self.distanceToSqr(player) > 12 * 12
					|| SplinterHivemind.bowDrawn(self, player)
					|| !level.dimension().location().toString().equals(record.getString("MeleeDimension"))
					|| time - record.getLong("MeleeContact") > 100) return false;
			net.minecraft.nbt.CompoundTag data = self.getPersistentData(), memory = meleeMemory(record, player, false);
			if (memory.getInt("Hits") == 0 && memory.getInt("Swings") == 0) return false;
			boolean supported = allyStriking(level, self, player);
			// Never surrender the player's immediate pressure to a distant or merely assigned teammate.
			if (!supported || self.distanceToSqr(player) < 3 * 3)
				return meleePressure(level, self, player, record, time);
			if (time < data.getLong("htDecision")) return false;
			data.putLong("htDecision", time + 12 + Math.floorMod(self.getId(), 7));
			Vec3 toward = player.position().subtract(self.position()).multiply(1, 0, 1).normalize();
			int side = memory.getDouble("StrafeBias") > 0.2 ? -1
					: memory.getDouble("StrafeBias") < -0.2 ? 1 : (self.getUUID().hashCode() & 1) == 0 ? 1 : -1;
			if (learned >= 50 && player.isBlocking()) {
				Vec3 look = player.getLookAngle().multiply(1, 0, 1).normalize();
				Vec3 goal = player.position().subtract(look.scale(1.8))
						.add(-look.z * side * 1.4, 0, look.x * side * 1.4);
				if (permitTactic(record, MELEE_FLANK, self, time) && meleeRoute(level, self, goal))
					return startTactic(level, self, player, MELEE_FLANK, goal, 20, time);
			}
			if (learned >= 35 && permitTactic(record, MELEE_FLANK, self, time)) {
				Vec3 lead = player.getDeltaMovement().multiply(1, 0, 1).scale(
						memory.getDouble("RetreatBias") > 0.35 ? 3.0 : 0.0);
				Vec3 goal = self.position().add(toward.scale(0.8))
						.add(-toward.z * side * 1.4 + lead.x, 0, toward.x * side * 1.4 + lead.z);
				if (meleeRoute(level, self, goal))
					return startTactic(level, self, player, MELEE_FLANK, goal, 12, time);
			}
			return meleePressure(level, self, player, record, time);
		}

		private static int progress(Mob mob) {
			if (mob instanceof SplinterEntity s) return s.getEntityData().get(SplinterEntity.DATA_mineProgress);
			if (mob instanceof BlindspotSplinterEntity s) return s.getEntityData().get(BlindspotSplinterEntity.DATA_mineProgress);
			if (mob instanceof LogSplinterEntity s) return s.getEntityData().get(LogSplinterEntity.DATA_mineProgress);
			if (mob instanceof PetrifiedLogSplinterEntity s) return s.getEntityData().get(PetrifiedLogSplinterEntity.DATA_mineProgress);
			return 1;
		}

		private static String weapon(Player player) {
			ItemStack held = player.isUsingItem() ? player.getUseItem() : player.getMainHandItem();
			if (held.isEmpty()) held = player.getOffhandItem();
			return net.minecraft.core.registries.BuiltInRegistries.ITEM.getKey(held.getItem()).toString();
		}

		private static List<Mob> squad(ServerLevel level, Mob self, Player player) {
			List<Mob> members = new ArrayList<>();
			int[] checked = {0};
			level.getEntities().get(net.minecraft.world.level.entity.EntityTypeTest.forClass(Mob.class),
					self.getBoundingBox().inflate(24, 10, 24), member -> {
				if (member.isAlive() && SplinterHivemind.member(member) && member.getTarget() == player
						&& !SplinterHivemind.frozen(member) && progress(member) == 0) members.add(member);
				return ++checked[0] >= 48 || members.size() >= 12
						? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
						: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
			});
			if (!members.contains(self)) members.add(self);
			return members;
		}

		private static int role(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time) {
			if (!record.getString("TeamDimension").equals(level.dimension().location().toString())
					|| time >= record.getLong("TeamUntil")) {
				List<Mob> team = squad(level, self, player);
				team.sort(Comparator.comparing(Entity::getUUID));
				Mob tank = team.stream().filter(e -> e.getHealth() > e.getMaxHealth() * 0.35)
						.min(Comparator.comparingDouble(e -> e.distanceToSqr(player)
								- (e instanceof PetrifiedLogSplinterEntity ? 150 : 0))).orElse(self);
				Mob builder = team.stream().filter(e -> e != tank && e.getHealth() > e.getMaxHealth() * 0.35)
						.min(Comparator.comparingDouble(e -> e.distanceToSqr(tank)
								- (e instanceof LogSplinterEntity ? 40 : 0))).orElse(tank);
				record.putString("Team0", builder.getUUID().toString());
				record.putString("Team1", tank.getUUID().toString());
				team.remove(builder); team.remove(tank);
				for (int i = 2; i < 8; i++) record.remove("Team" + i);
				team.sort(Comparator.comparingDouble(e -> e instanceof BlindspotSplinterEntity ? -1 : 0));
				for (int i = 0; i < Math.min(6, team.size()); i++)
					record.putString("Team" + (i + 2), team.get(i).getUUID().toString());
				Mob wounded = squad(level, self, player).stream()
						.filter(e -> e.getHealth() < e.getMaxHealth() * 0.35)
						.min(Comparator.comparingDouble(e -> e.getHealth() / e.getMaxHealth())).orElse(null);
				record.putString("Wounded", wounded == null ? "" : wounded.getUUID().toString());
				record.putString("TeamDimension", level.dimension().location().toString());
				record.putLong("TeamUntil", time + 40L);
			}
			// The front member retains pressure if there is only one available sibling.
			if (self.getUUID().toString().equals(record.getString("Team1"))) return 1;
			for (int i = 0; i < 8; i++) if (self.getUUID().toString().equals(record.getString("Team" + i))) return i;
			return 4 + Math.floorMod(self.getUUID().hashCode(), 2);
		}

		private static void rememberPerch(ServerLevel level, Player player,
				net.minecraft.nbt.CompoundTag record, long time) {
			if (!player.onGround() || time < record.getLong("PerchSample")
					|| player.getDeltaMovement().horizontalDistanceSqr() > 0.025) return;
			record.putLong("PerchSample", time + 40L);
			int chosen = -1, oldest = 0;
			long oldestTime = Long.MAX_VALUE;
			for (int i = 0; i < 3; i++) {
				net.minecraft.nbt.CompoundTag perch = record.getCompound("Perch" + i);
				if (perch.isEmpty()) { if (chosen < 0) chosen = i; continue; }
				BlockPos pos = BlockPos.of(perch.getLong("Pos"));
				if (perch.getString("Dimension").equals(level.dimension().location().toString())
						&& pos.distToCenterSqr(player.getX(), player.getY(), player.getZ()) < 4 * 4) {
					chosen = i; break;
				}
				if (perch.getLong("Seen") < oldestTime) { oldestTime = perch.getLong("Seen"); oldest = i; }
			}
			if (chosen < 0) chosen = oldest;
			net.minecraft.nbt.CompoundTag perch = record.getCompound("Perch" + chosen);
			boolean same = !perch.isEmpty()
					&& perch.getString("Dimension").equals(level.dimension().location().toString())
					&& BlockPos.of(perch.getLong("Pos")).distToCenterSqr(player.getX(), player.getY(), player.getZ()) < 16;
			if (!same) perch = new net.minecraft.nbt.CompoundTag();
			perch.putString("Dimension", level.dimension().location().toString());
			perch.putLong("Pos", player.blockPosition().asLong());
			perch.putLong("Seen", time);
			perch.putInt("Uses", Math.min(30, perch.getInt("Uses") + 1));
			record.put("Perch" + chosen, perch);
		}

		private static void observeCombat(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time) {
			if (time < record.getLong("CombatScan") || !self.hasLineOfSight(player)) return;
			record.putLong("CombatScan", time + 4L);
			List<net.minecraft.world.entity.projectile.Projectile> shots = new ArrayList<>();
			int[] examined = {0};
			level.getEntities().get(net.minecraft.world.level.entity.EntityTypeTest.forClass(
					net.minecraft.world.entity.projectile.Projectile.class),
					player.getBoundingBox().inflate(12), shot -> {
				if (shot.isAlive() && shot.getOwner() == player && shot.tickCount <= 8
						&& shot.getDeltaMovement().lengthSqr() > 0.015) shots.add(shot);
				return ++examined[0] >= 24 || shots.size() >= 4
						? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
						: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
			});
			for (net.minecraft.world.entity.projectile.Projectile shot : shots) {
				if (shot.getPersistentData().getBoolean("htNoted") || !self.hasLineOfSight(shot)) continue;
				shot.getPersistentData().putBoolean("htNoted", true);
				noteShot(level, player, shot, record, time);
				long fired = Math.max(0L, time - Math.min(8, shot.tickCount));
				String held = weapon(player);
				int chosen = 0;
				for (int i = 0; i < 2; i++) {
					net.minecraft.nbt.CompoundTag cadence = record.getCompound("Cadence" + i);
					if (held.equals(cadence.getString("Weapon"))) { chosen = i; break; }
					if (cadence.isEmpty() || cadence.getLong("Last") < record.getCompound("Cadence" + chosen).getLong("Last"))
						chosen = i;
				}
				net.minecraft.nbt.CompoundTag cadence = record.getCompound("Cadence" + chosen);
				if (!held.equals(cadence.getString("Weapon"))) {
					cadence = new net.minecraft.nbt.CompoundTag(); cadence.putString("Weapon", held);
				}
				long last = cadence.getLong("Last");
				if (learningHere(level) && last > 0 && fired - last >= 4 && fired - last <= 120) {
					double interval = fired - last;
					double mean = cadence.getInt("Samples") == 0 ? interval : cadence.getDouble("Mean");
					cadence.putDouble("Deviation", cadence.getDouble("Deviation") * 0.75 + Math.abs(interval - mean) * 0.25);
					cadence.putDouble("Mean", mean * 0.75 + interval * 0.25);
					cadence.putInt("Samples", Math.min(30, cadence.getInt("Samples") + 1));
				}
				if (last == 0 || fired - last >= 4) cadence.putLong("Last", fired);
				record.put("Cadence" + chosen, cadence);
				record.putLong("LastFire", Math.max(fired, record.getLong("LastFire")));
				record.putString("ObservedWeapon", held);
				if (learningHere(level)) rememberPerch(level, player, record, time);
			}
		}

		private static boolean opening(net.minecraft.nbt.CompoundTag record, long time) {
			for (int i = 0; i < 2; i++) {
				net.minecraft.nbt.CompoundTag cadence = record.getCompound("Cadence" + i);
				if (!cadence.getString("Weapon").equals(record.getString("ObservedWeapon"))
						|| cadence.getInt("Samples") < Math.max(2, Math.min(30, CADENCE_SAMPLES_REQUIRED))) continue;
				double window = Math.min(18, cadence.getDouble("Mean") * 0.65 - cadence.getDouble("Deviation") * 1.5);
				long since = time - cadence.getLong("Last");
				return since >= 2 && since <= window && since <= 20;
			}
			return false;
		}

		private static boolean permitTactic(net.minecraft.nbt.CompoundTag record, int mode, Mob self, long time) {
			int failures = record.getInt("Loss" + mode), wins = record.getInt("Win" + mode);
			return failures <= wins + 2
					|| Math.floorMod(self.getUUID().hashCode() ^ (int) (time / 100) ^ mode, 8) == 0;
		}

		private static boolean rangedContext(net.minecraft.nbt.CompoundTag record, long time) {
			return record.getLong("LastFire") > 0 && time - record.getLong("LastFire") < 100;
		}

		private static void clearTactic(Mob self, long time) {
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			int previousMode = data.getInt("htState");
			boolean ours = previousMode != 0;
			data.remove("htState"); data.remove("htPost"); TACTICAL_ROUTES.remove(self);
			data.putLong("htDecision", previousMode >= MELEE_FEINT ? time : time + 16L);
			if (ours) self.getNavigation().stop();
		}

		private static Vec3 tacticPoint(Mob self) {
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			return new Vec3(data.getDouble("htX"), data.getDouble("htY"), data.getDouble("htZ"));
		}

		private static void point(Mob self, Vec3 point) {
			self.getPersistentData().putDouble("htX", point.x);
			self.getPersistentData().putDouble("htY", point.y);
			self.getPersistentData().putDouble("htZ", point.z);
		}

		private static boolean startTactic(ServerLevel level, Mob self, Player player, int mode, Vec3 goal,
				long duration, long time) {
			if (!SplinterHivemind.safeRoute(self, level, goal)) return false;
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			data.remove("htEchoHoldUntil"); data.remove("htEchoGazeUntil"); data.remove("htEchoHoldPost");
			data.putInt("htState", mode);
			data.putLong("htUntil", time + duration);
			data.putLong("htStarted", time);
			data.putLong("htProgressAt", time + 16L);
			data.putDouble("htRemaining", self.distanceToSqr(goal));
			data.putInt("htStalls", 0);
			data.putString("htOwner", player.getUUID().toString());
			data.putString("htDimension", level.dimension().location().toString());
			data.putLong("htLastTick", time);
			Vec3 observed = lastSeenEye(self, player);
			if (observed == null) observed = goal.add(0, self.getEyeHeight(), 0);
			data.putDouble("htLookX", observed.x); data.putDouble("htLookY", observed.y); data.putDouble("htLookZ", observed.z);
			data.putInt("htStage", 0);
			data.remove("htFeintRewarded");
			data.putInt("htTrial", mode);
			data.putLong("htTrialUntil", time + duration + 60L);
			data.remove("hive_bow_post_time"); data.remove("hive_post_time");
			data.putInt("hive_bow_state", 0);
			data.putLong("hive_bow_pressure_until", 0);
			data.putLong("hive_route_until", 0);
			if (mode >= MELEE_FEINT) {
				var route = TACTICAL_ROUTES.get(self);
				if (route != null && route.getNodeCount() > 0)
					goal = route.getEntityPosAtNode(self, route.getNodeCount() - 1);
			}
			point(self, goal);
			data.putLong("htPost", level.getGameTime());
			net.minecraft.world.level.pathfinder.Path path = TACTICAL_ROUTES.get(self);
			if (path != null) self.getNavigation().moveTo(path, mode == WITHDRAW ? 0.9 : 1.0);
			return true;
		}

		private static Vec3 retreatPoint(ServerLevel level, Mob self, Player player, boolean covered) {
			Vec3 known = lastSeenEye(self, player);
			if (known == null) return null;
			Vec3 shelter = covered ? SplinterHivemind.nearbyCover(self, player, level, null) : null;
			if (shelter != null && shelter.distanceToSqr(known) >= self.position().distanceToSqr(known) - 1) return shelter;
			Vec3 eye = lastSeenEye(self, player);
			if (eye == null) return null;
			Vec3 away = self.position().subtract(eye).multiply(1, 0, 1).normalize();
			if (away.lengthSqr() < 0.01) return null;
			for (int side : new int[] {1, -1}) {
				Vec3 goal = self.position().add(away.scale(4)).add(-away.z * side * 2, 0, away.x * side * 2);
				if (SplinterHivemind.safeRoute(self, level, goal)) return goal;
			}
			return null;
		}

		private static Vec3 peekPoint(ServerLevel level, Mob self, Player player) {
			Vec3 eye = lastSeenEye(self, player);
			if (eye == null) return null;
			Vec3 along = eye.subtract(self.position()).multiply(1, 0, 1).normalize();
			for (int side : new int[] {1, -1}) for (int distance = 1; distance <= 2; distance++) {
				Vec3 point = self.position().add(-along.z * side * distance, 0, along.x * side * distance);
				if (SplinterHivemind.clearStandingRoom(self, level, BlockPos.containing(point))
						&& !SplinterHivemind.blockedShot(self, player, level, point)
						&& SplinterHivemind.safeRoute(self, level, point)) return point;
			}
			return null;
		}

		private static Vec3 repeatedPerchFlank(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, int side) {
			for (int i = 0; i < 3; i++) {
				net.minecraft.nbt.CompoundTag perch = record.getCompound("Perch" + i);
				if (perch.getInt("Uses") < Math.max(2, Math.min(30, PERCH_OBSERVATIONS_REQUIRED)) || !perch.getString("Dimension").equals(level.dimension().location().toString())) continue;
				BlockPos pos = BlockPos.of(perch.getLong("Pos"));
				if (pos.distToCenterSqr(player.getX(), player.getY(), player.getZ()) > 6 * 6) continue;
				Vec3 toward = Vec3.atBottomCenterOf(pos).subtract(self.position()).multiply(1, 0, 1).normalize();
				Vec3 goal = Vec3.atBottomCenterOf(pos).add(-toward.z * side * 6 - toward.x * 2,
						0, toward.x * side * 6 - toward.z * 2);
				if (SplinterHivemind.safeRoute(self, level, goal)) return goal;
			}
			return null;
		}

		private static Vec3 explosiveThreat(ServerLevel level, Mob self, long time) {
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			if (time < data.getLong("htHazardScan")) {
				if (time >= data.getLong("htHazardUntil")) return null;
				Entity cached = level.getEntity(data.getInt("htHazardId"));
				return cached != null && cached.isAlive() && self.hasLineOfSight(cached) ? cached.position() : null;
			}
			data.putLong("htHazardScan", time + 6L);
			List<Entity> hazards = new ArrayList<>();
			int[] seen = {0};
			level.getEntities().get(net.minecraft.world.level.entity.EntityTypeTest.forClass(Entity.class),
					self.getBoundingBox().inflate(10), entity -> {
				String id = net.minecraft.core.registries.BuiltInRegistries.ENTITY_TYPE.getKey(entity.getType()).getPath();
				boolean active = entity instanceof net.minecraft.world.entity.item.PrimedTnt tnt && tnt.getFuse() < 50
						|| entity instanceof net.minecraft.world.entity.monster.Creeper creeper && creeper.getSwellDir() > 0
						|| entity instanceof net.minecraft.world.entity.projectile.Projectile
							&& entity.getDeltaMovement().dot(self.position().subtract(entity.position())) > 0
							&& (id.contains("fireball") || id.contains("rocket") || id.contains("grenade")
								|| id.contains("missile") || id.contains("wither_skull"));
				if (active && entity.isAlive() && self.hasLineOfSight(entity)) hazards.add(entity);
				return ++seen[0] >= 48 || hazards.size() >= 4
						? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
						: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
			});
			Entity nearest = hazards.stream().min(Comparator.comparingDouble(self::distanceToSqr)).orElse(null);
			if (nearest == null) { data.remove("htHazardUntil"); return null; }
			data.putInt("htHazardId", nearest.getId());
			data.putLong("htHazardUntil", time + 8L);
			return nearest.position();
		}

		private static boolean runTactic(ServerLevel level, Mob self, Player player,
				net.minecraft.nbt.CompoundTag record, long time, double speed) {
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			int mode = data.getInt("htState");
			if (mode == 0) return false;
			if (!player.getUUID().toString().equals(data.getString("htOwner"))
					|| !level.dimension().location().toString().equals(data.getString("htDimension"))
					|| time >= data.getLong("htUntil") || time - data.getLong("htLastTick") > 2) {
				clearTactic(self, time); return false;
			}
			data.putLong("htLastTick", time);
			if (mode == MELEE_FEINT || mode == MELEE_WAIT
					|| mode == MELEE_FLANK && Math.floorMod(self.tickCount, 4) == 0 && !allyStriking(level, self, player)) {
				clearTactic(self, time);
				return meleePressure(level, self, player, record, time)
						&& runTactic(level, self, player, record, time, speed);
			}
			if (mode == MELEE_PRESSURE) {
				if (!self.hasLineOfSight(player) || self.distanceToSqr(player) > 16 * 16) {
					clearTactic(self, time); return false;
				}
				SplinterOnEntityTickUpdateProcedure.setSpeed(self, speed);
				data.putLong("htPost", level.getGameTime());
				self.getLookControl().setLookAt(player, 25, 20);
				// Vanilla melee owns movement and actual hits here; no duplicated attacks or reach extension.
				return true;
			}
			Vec3 goal = tacticPoint(self);
			boolean arrived = self.distanceToSqr(goal) < arrivalRadiusSqr(mode);
			boolean sees = self.hasLineOfSight(player);
			if (sees && time >= data.getLong("htEchoGazeUntil")) {
				data.putDouble("htLookX", player.getX()); data.putDouble("htLookY", player.getEyeY());
				data.putDouble("htLookZ", player.getZ());
			}
			if (mode == SERPENTINE) {
				if (!sees || !rangedThreatNow(self, player)) {
					clearTactic(self, time); return false;
				}
				if (self.distanceToSqr(player) < 5 * 5) {
					clearTactic(self, time);
					return meleePressure(level, self, player, record, time)
							&& runTactic(level, self, player, record, time, speed);
				}
				if (!arrived && time >= data.getLong("htWeaveDeadline")) {
                    clearTactic(self, time); return false;
                }
                if (arrived && time >= data.getLong("htWeaveNext")) {
					int side = data.getInt("htWeaveSide") >= 0 ? -1 : 1;
					if (!weaveStep(level, self, player, time, skill(record), side, false)
							&& !weaveStep(level, self, player, time, skill(record), -side, false)) {
						clearTactic(self, time); return false;
					}
					goal = tacticPoint(self); arrived = false;
				}
			}
			if (mode == MELEE_FEINT && sees && learningHere(level)
					&& meleeOpening(meleeMemory(record, player, false), time)
					&& meleeMemory(record, player, false).getLong("Swing") >= data.getLong("htStarted")
					&& !data.getBoolean("htFeintRewarded")) {
				record.putInt("Win" + MELEE_FEINT, Math.min(24, record.getInt("Win" + MELEE_FEINT) + 1));
				data.putBoolean("htFeintRewarded", true);
			}
			if (mode >= MELEE_FEINT && mode <= MELEE_WAIT && sees
					&& (meleeOpening(meleeMemory(record, player, false), time) || mode == MELEE_FLANK && arrived)
					&& meleePressure(level, self, player, record, time)) {
				SplinterOnEntityTickUpdateProcedure.setSpeed(self, speed);
				return true;
			}
			if (mode == BAIT) {
				int stage = data.getInt("htStage");
				if (stage == 0 && arrived) { data.putInt("htStage", 1); data.putLong("htPeekUntil", time + 6L); }
				if ((stage <= 1 && record.getLong("LastFire") > data.getLong("htStarted"))
						|| stage == 1 && time >= data.getLong("htPeekUntil")) {
					Vec3 shelter = new Vec3(data.getDouble("htCoverX"), data.getDouble("htCoverY"), data.getDouble("htCoverZ"));
					if (!SplinterHivemind.safeRoute(self, level, shelter)) { clearTactic(self, time); return false; }
					point(self, shelter); goal = shelter; arrived = false; data.putInt("htStage", 2);
				} else if (stage == 2 && arrived) {
					if (sees && opening(record, time)) {
						Vec3 toward = player.position().subtract(self.position()).multiply(1, 0, 1).normalize();
						Vec3 advance = self.position().add(toward.scale(Math.min(5, self.distanceTo(player) - 3)));
						if (startTactic(level, self, player, PRESSURE, advance, 18, time)) return true;
					}
					clearTactic(self, time); return false;
				}
			}
			if (mode == AMBUSH && sees && self.distanceToSqr(player) < 4 * 4) {
				clearTactic(self, time);
				data.putLong("hive_bow_commit_until", level.getGameTime() + 30L);
				data.putString("hive_bow_target", player.getUUID().toString());
				return false;
			}
			if (mode == PRESSURE && (self.distanceToSqr(player) < 3 * 3 || arrived)) {
				clearTactic(self, time); return false;
			}
			if (mode == FLANK && arrived) { clearTactic(self, time); return false; }
			if (!arrived && time >= data.getLong("htProgressAt")) {
				double remaining = self.distanceToSqr(goal);
				int stalls = remaining >= data.getDouble("htRemaining") - 0.3 ? data.getInt("htStalls") + 1 : 0;
				data.putInt("htStalls", stalls); data.putDouble("htRemaining", remaining);
				data.putLong("htProgressAt", time + 16L);
				if (stalls >= 2 || !SplinterHivemind.safeRoute(self, level, goal)) {
					clearTactic(self, time); return false;
				}
			}
			SplinterOnEntityTickUpdateProcedure.setSpeed(self, speed);
			data.putLong("htPost", level.getGameTime());
			self.getLookControl().setLookAt(data.getDouble("htLookX"), data.getDouble("htLookY"),
					data.getDouble("htLookZ"), 25, 20);
			if (arrived) self.getNavigation().stop();
			else {
				net.minecraft.world.level.pathfinder.Path path = TACTICAL_ROUTES.get(self);
				if (path != null) self.getNavigation().moveTo(path, mode == WITHDRAW ? 0.9 : 1.0);
			}
			return true;
		}


		// A one-shot request from the server-only visual element; all movement still uses the shared tactical owner.
		public static boolean afterimageManeuver(Mob self, Player player, boolean retreat, boolean ambush, Vec3 attention) {
			if (!(self instanceof BlindspotSplinterEntity) || !(self.level() instanceof ServerLevel level)
					|| !validPlayer(player, level) || self.getTarget() != player || !self.isAlive()
					|| self.isPassenger() || self.isInWater() || progress(self) != 0 || SplinterHivemind.frozen(self)
					|| self.distanceToSqr(player) > 36 * 36 || self.distanceToSqr(player) < 4 * 4
					|| !Double.isFinite(attention.x) || !Double.isFinite(attention.y) || !Double.isFinite(attention.z)
					|| player.getMainHandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())
					|| player.getOffhandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())) return false;
			long time = now(level);
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			int current = data.getInt("htState");
			if (time < data.getLong("ahPlacePose") || current != 0 && current != FLANK
					&& current != AMBUSH && current != WITHDRAW && current != MELEE_FLANK && current != SERPENTINE) return false;
			retreat |= self.getHealth() < self.getMaxHealth() * 0.35;
			Vec3 destination = retreat ? retreatPoint(level, self, player, true)
					: ambush ? SplinterHivemind.nearbyCover(self, player, level, null)
					: SplinterHivemind.screenedFlank(self, player, level);
			if (!retreat && !ambush) {
				var route = self.getNavigation().getPath();
				if (route != null && !route.isDone()) destination = route.getEntityPosAtNode(self, route.getNodeCount() - 1);
			}
			int mode = retreat ? WITHDRAW : ambush ? AMBUSH : FLANK;
			if (destination == null || !startTactic(level, self, player, mode, destination, ambush ? 90 : 80, time)) return false;
			data.putDouble("htLookX", attention.x); data.putDouble("htLookY", attention.y); data.putDouble("htLookZ", attention.z);
			data.putLong("htEchoGazeUntil", time + (ambush ? 90 : 6));
			data.putLong("htEchoHoldUntil", time + (retreat || self.hurtTime > 0 ? 0 : 6));
			// No teleport, extra hit, speed multiplier, or pathfinder outside the existing global budget.
			return true;
		}

		private static boolean holdAfterimage(ServerLevel level, Mob self, Player player, long time, double speed) {
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			if (data.getInt("htState") == 0 || time >= data.getLong("htEchoHoldUntil")
					|| !player.getUUID().toString().equals(data.getString("htOwner"))
					|| !level.dimension().location().toString().equals(data.getString("htDimension"))) return false;
			if (self.hurtTime > 0 || self.distanceToSqr(player) < 6 * 6) {
				data.remove("htEchoHoldUntil"); return false;
			}
			data.putLong("htLastTick", time); data.putLong("htPost", level.getGameTime());
			data.putLong("htEchoHoldPost", level.getGameTime());
			SplinterOnEntityTickUpdateProcedure.setSpeed(self, speed);
			self.getNavigation().stop();
			self.getLookControl().setLookAt(data.getDouble("htLookX"), data.getDouble("htLookY"),
					data.getDouble("htLookZ"), 20, 15);
			return true;
		}

		public static boolean finishTactic(Mob self, int mineProgress) {
			if (!(self.level() instanceof ServerLevel level) || mineProgress != 0 || SplinterHivemind.frozen(self)
					|| self.getPersistentData().getLong("htPost") != level.getGameTime()
					|| self.getPersistentData().getInt("htState") == 0
					|| !(self.getTarget() instanceof Player owner) || !validPlayer(owner, level)
					|| !owner.getUUID().toString().equals(self.getPersistentData().getString("htOwner"))
					|| !level.dimension().location().toString().equals(self.getPersistentData().getString("htDimension"))) return false;
			if (self.getPersistentData().getInt("htState") == MELEE_PRESSURE) {
				closeForMelee(self, mineProgress); return true;
			}
			if (self.getPersistentData().contains("htEchoHoldPost")
					&& self.getPersistentData().getLong("htEchoHoldPost") == level.getGameTime()) {
				self.getNavigation().stop();
				if (self.onGround() && self.hurtTime == 0) self.setDeltaMovement(0, self.getDeltaMovement().y, 0);
				return true;
			}
            if (self.getPersistentData().getInt("htState") == MIRROR) {
                var route = TACTICAL_ROUTES.get(self);
                if (route != null && !route.isDone()) {
                    int next = route.getNextNodeIndex();
                    for (int i = next; i < Math.min(route.getNodeCount(), next + 4); i++) {
                        Vec3 step = route.getEntityPosAtNode(self, i);
                        double clearance = Math.max(6.0, self.getPersistentData().getDouble("htMirrorClearance"));
                        if (step.distanceToSqr(owner.position()) < clearance * clearance
                                && step.distanceToSqr(owner.position()) < self.distanceToSqr(owner)) {
                            self.getNavigation().stop(); TACTICAL_ROUTES.remove(self); point(self, self.position());
                            return true;
                        }
                    }
                }
            }
			Vec3 goal = tacticPoint(self);
			if (self.distanceToSqr(goal) < arrivalRadiusSqr(self.getPersistentData().getInt("htState"))) {
				self.getNavigation().stop();
				if (self.onGround() && self.hurtTime == 0) {
					Vec3 motion = self.getDeltaMovement();
					self.setDeltaMovement(0.0, motion.y, 0.0);
				}
			} else {
				net.minecraft.world.level.pathfinder.Path path = TACTICAL_ROUTES.get(self);
				if (path == null) {
                    if (self.getPersistentData().getInt("htState") == MIRROR) {
                        self.getNavigation().stop(); return true;
                    }
                    return false;
                }
				self.getNavigation().moveTo(path, self.getPersistentData().getInt("htState") == WITHDRAW ? 0.9 : 1.0);
			}
			return true;
		}

		public static boolean tactics(Mob self, Player player, int mineProgress, double speed) {
			if (!(self.level() instanceof ServerLevel level)) return false;
			long time = now(level);
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			if (!validPlayer(player, level) || !self.isAlive() || self.isPassenger() || self.isInWater()
					|| mineProgress != 0 || SplinterHivemind.frozen(self) || self.getTarget() != player
					|| self.distanceToSqr(player) > 48 * 48
					|| player.getMainHandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())
					|| player.getOffhandItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())) {
				// Mining and rose immobilization retain their own navigation and speed controls.
				data.remove("htState"); data.remove("htPost"); TACTICAL_ROUTES.remove(self); PENDING_COVER.remove(self); return false;
			}
			net.minecraft.nbt.CompoundTag root = root(level);
			int slot = slot(root, player, time, learningHere(level));
			if (slot < 0) return defend(self, player, mineProgress);
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
			observeCombat(level, self, player, record, time);
			observeMelee(level, self, player, record, time);
			decay(record, time);
			int learned = skill(record);
			boolean sees = self.hasLineOfSight(player);
			if (sees) {
				data.putString("htSeenOwner", player.getUUID().toString()); data.putLong("htSeenAt", time);
				if (time >= data.getLong("htEchoGazeUntil")) {
					data.putDouble("htLookX", player.getX()); data.putDouble("htLookY", player.getEyeY());
					data.putDouble("htLookZ", player.getZ());
				}
			}
			if (time < data.getLong("ahPlacePose")) {
				data.remove("htState"); data.remove("htPost"); TACTICAL_ROUTES.remove(self);
				self.getNavigation().stop(); save(level, root); return true;
			}
			boolean known = sees || lastSeenEye(self, player) != null;
			Vec3 hazard = explosiveThreat(level, self, time);
			if (hazard != null && time >= data.getLong("htHazardRetry")
					&& (data.getInt("htState") != EVADE || tacticPoint(self).distanceToSqr(hazard) < 4 * 4)) {
				data.putLong("htHazardRetry", time + 16L);
				Vec3 away = self.position().subtract(hazard).multiply(1, 0, 1).normalize();
				int side = (self.getUUID().hashCode() & 1) == 0 ? 1 : -1;
				for (int sign : new int[] {side, -side}) {
					Vec3 goal = self.position().add(away.scale(6)).add(-away.z * sign * 3, 0, away.x * sign * 3);
					if (startTactic(level, self, player, EVADE, goal, 36L, time)) break;
				}
			}
			if (data.getInt("htState") == EVADE) {
				save(level, root); return runTactic(level, self, player, record, time, speed);
			}
			int role = learned >= 15 && sees ? role(level, self, player, record, time) : data.getInt("htRole");
			if (sees) data.putInt("htRole", role);
			// Injured members withdraw; a designated sibling screens them instead of forming a pile.
			if (learned >= 15 && sees && self.getHealth() < self.getMaxHealth() * 0.35
					&& data.getInt("htState") != WITHDRAW && time >= data.getLong("htDecision")) {
				Vec3 retreat = retreatPoint(level, self, player, true);
				if (retreat != null) startTactic(level, self, player, WITHDRAW, retreat, 60L, time);
			}
			if (data.getInt("htState") == WITHDRAW) {
				save(level, root); return runTactic(level, self, player, record, time, speed);
			}
			if ((role == 0 || role == 1) && learned >= 20 && !record.getString("Wounded").isEmpty()) {
				Entity entity;
				try { entity = level.getEntity(UUID.fromString(record.getString("Wounded"))); }
				catch (IllegalArgumentException ex) { entity = null; }
				if (entity instanceof Mob ward && ward != self && ward.isAlive() && ward.getTarget() == player
						&& ward.getHealth() < ward.getMaxHealth() * 0.35 && self.distanceToSqr(ward) < 6 * 6
						&& self.hasLineOfSight(ward) && !SplinterHivemind.frozen(ward)) {
					save(level, root);
					if (defendFor(self, player, mineProgress, ward)) { data.remove("htState"); TACTICAL_ROUTES.remove(self); return true; }
					root = root(level); record = root.getCompound("Player" + slot);
					if (sees && time >= data.getLong("htDecision") && data.getInt("htState") == 0) {
						Vec3 between = player.position().subtract(ward.position()).multiply(1, 0, 1).normalize();
						Vec3 screen = ward.position().add(between.scale(2));
						if (startTactic(level, self, player, ESCORT, screen, 32L, time)) {
							save(level, root); return runTactic(level, self, player, record, time, speed);
						}
					}
				}
			}
			save(level, root);
			if (defend(self, player, mineProgress)) { data.remove("htState"); TACTICAL_ROUTES.remove(self); return true; }
			root = root(level); record = root.getCompound("Player" + slot);
			if (holdAfterimage(level, self, player, time, speed)) { save(level, root); return true; }
			if (data.getInt("htState") != 0) {
				boolean active = runTactic(level, self, player, record, time, speed);
				if (active) { save(level, root); return true; }
			}
			if (rangedPressure(level, self, player, record, time, learned)
					|| meleeTactics(level, self, player, record, time, learned)) {
				boolean active = runTactic(level, self, player, record, time, speed);
				save(level, root); return active;
			}
			if (learned < 20 || !known || time < data.getLong("htDecision") || !rangedContext(record, time)) {
				save(level, root); return false;
			}
			data.putLong("htDecision", time + 30L + Math.floorMod(self.getId(), 11));
			boolean protectedNow = SplinterHivemind.blockedShot(self, player, level, self.position());
			boolean started = false;
			if (sees && role >= 2 && !(role == 3 && learned >= 55) && permitTactic(record, FLANK, self, time)) {
				int side = role % 2 == 0 ? 1 : -1;
				Vec3 flank = learned >= 30 ? repeatedPerchFlank(level, self, player, record, side) : null;
				if (flank == null) flank = SplinterHivemind.screenedFlank(self, player, level);
				if (flank != null) started = startTactic(level, self, player, FLANK, flank, 55L, time);
			}
			if (!started && role == 3 && learned >= 55 && permitTactic(record, AMBUSH, self, time)) {
				Vec3 shelter = SplinterHivemind.nearbyCover(self, player, level, null);
				if (shelter != null) started = startTactic(level, self, player, AMBUSH, shelter, 80L, time);
			}
			if (!started && sees && opening(record, time) && permitTactic(record, PRESSURE, self, time)) {
				Vec3 toward = player.position().subtract(self.position()).multiply(1, 0, 1).normalize();
				Vec3 goal = self.position().add(toward.scale(Math.min(5.0, Math.max(0.0, self.distanceTo(player) - 3))));
				started = startTactic(level, self, player, PRESSURE, goal, 18L, time);
			}
			if (!started && protectedNow && role <= 1 && learned >= 25 && permitTactic(record, BAIT, self, time)) {
				Vec3 peek = peekPoint(level, self, player);
				if (peek != null) {
					data.putDouble("htCoverX", self.getX()); data.putDouble("htCoverY", self.getY()); data.putDouble("htCoverZ", self.getZ());
					started = startTactic(level, self, player, BAIT, peek, 42L, time);
				}
			}
			if (!started && sees && role == 1 && learned >= 55 && permitTactic(record, WITHDRAW, self, time)) {
				Mob partner = squad(level, self, player).stream().filter(e -> e != self
						&& e.getPersistentData().getInt("htState") == AMBUSH && e.distanceToSqr(tacticPoint(e)) < 2.25
						&& e.distanceToSqr(self) < 10 * 10).findFirst().orElse(null);
				if (partner != null) {
					Vec3 lure = tacticPoint(partner);
					Vec3 away = self.position().subtract(player.position()).multiply(1, 0, 1).normalize();
					Vec3 retreat = lure.add(away.scale(3));
					if (retreat.distanceToSqr(player.position()) > self.distanceToSqr(player) + 4)
						started = startTactic(level, self, player, WITHDRAW, retreat, 35L, time);
				}
			}
			save(level, root);
			return started && runTactic(level, self, player, record, time, speed);
		}

		private static void recordTacticResult(ServerLevel level, Mob member, Player player, boolean success) {
			if (!learningHere(level) || player == null || !validPlayer(player, level)) return;
			net.minecraft.nbt.CompoundTag data = member.getPersistentData();
			long time = now(level);
			int mode = data.getInt("htTrial");
			if (mode < 1 || mode > SERPENTINE || time >= data.getLong("htTrialUntil")
					|| !player.getUUID().toString().equals(data.getString("htOwner"))
					|| time < data.getLong("htOutcomeAt")) return;
			net.minecraft.nbt.CompoundTag root = root(level);
			int slot = slot(root, player, time, true);
			if (slot < 0) return;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
			String key = (success ? "Win" : "Loss") + mode;
			record.putInt(key, Math.min(24, record.getInt(key) + 1));
			data.putLong("htOutcomeAt", time + 80L);
			save(level, root);
		}

		private static void learnPenetration(ServerLevel level, Mob victim, Player player,
				net.minecraft.world.entity.projectile.Projectile shot) {
			if (!learningHere(level) || shot.getDeltaMovement().lengthSqr() < 0.015) return;
			Vec3 end = shot.position();
			Vec3 start = end.subtract(shot.getDeltaMovement().normalize().scale(6));
			if (!SplinterHivemind.loadedLine(level, start, end)) return;
			BlockHitResult obstruction = level.clip(new ClipContext(start, end,
					ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, victim));
			if (obstruction.getType() != HitResult.Type.BLOCK) return;
			BlockState material = level.getBlockState(obstruction.getBlockPos());
			if (!material.is(net.minecraft.tags.BlockTags.PLANKS) && !material.is(net.minecraft.tags.BlockTags.LOGS)) return;
			net.minecraft.nbt.CompoundTag root = root(level);
			long time = now(level);
			int slot = slot(root, player, time, true);
			if (slot < 0) return;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
			if (time < record.getLong("PenetrationAt")) return;
            boolean trackedCover = false;
            for (int c = 0; c < COVER_SLOTS; c++)
                if (record.getCompound("Cover" + c).contains("PlacedAt")
                        && record.getCompound("Cover" + c).getLong("PlacedAt") <= time - Math.max(0, shot.tickCount)
                        && coverCell(record.getCompound("Cover" + c), level, obstruction.getBlockPos())) {
                    trackedCover = true; break;
                }
            // Grain terrain is wood too; hitting a mob beside a wooden slope is NOT evidence of penetration.
            if (!trackedCover) return;
            if (record.getInt("WoodEvidenceVersion") < 1) record.putInt("WoodFailures", 0);
            record.putInt("WoodEvidenceVersion", 1);
			String held = record.getString("ObservedWeapon");
			if (held.isEmpty()) return;
			if (!held.equals(record.getString("WeakWeapon"))) record.putInt("WoodFailures", 0);
			record.putString("WeakWeapon", held);
			record.putInt("WoodFailures", Math.min(8, record.getInt("WoodFailures") + 1));
			record.putLong("PenetrationAt", time + 20L);
			save(level, root);
		}

		public static boolean requiresThickCover(Mob self, Player player) {
			if (!(self.level() instanceof ServerLevel level)) return false;
			net.minecraft.nbt.CompoundTag data = self.getPersistentData();
			long time = now(level);
			if (time < data.getLong("htMaterialAt")
					&& player.getUUID().toString().equals(data.getString("htMaterialOwner")))
				return data.getBoolean("htThick");
			net.minecraft.nbt.CompoundTag root = root(level);
			int slot = slot(root, player, time, false);
			net.minecraft.nbt.CompoundTag record = slot < 0 ? new net.minecraft.nbt.CompoundTag() : root.getCompound("Player" + slot);
			boolean thick = record.getInt("WoodEvidenceVersion") >= 1 && record.getInt("WoodFailures") >= 2
					&& record.getString("WeakWeapon").equals(record.getString("ObservedWeapon"));
			data.putLong("htMaterialAt", time + 60L); data.putString("htMaterialOwner", player.getUUID().toString());
			data.putBoolean("htThick", thick);
			return thick;
		}

		public static boolean finishAction(Mob self, int mineProgress) {
			if (!(self.level() instanceof ServerLevel level) || mineProgress != 0
					|| SplinterHivemind.frozen(self) || self.hurtTime > 0 || !self.onGround()
					|| now(level) >= self.getPersistentData().getLong("ahPlacePose")) return false;
			self.getNavigation().stop();
			Vec3 motion = self.getDeltaMovement();
			self.setDeltaMovement(0.0, motion.y, 0.0);
			return true;
		}

		private static boolean safeExit(ServerLevel level, Mob self, BlockPos obstruction) {
			int exits = 0;
			for (net.minecraft.core.Direction direction : net.minecraft.core.Direction.Plane.HORIZONTAL) {
				BlockPos floor = self.blockPosition().relative(direction);
				if (floor.getX() == obstruction.getX() && floor.getZ() == obstruction.getZ()) continue;
				if (!level.hasChunkAt(floor) || !level.getBlockState(floor.below()).isFaceSturdy(
						level, floor.below(), net.minecraft.core.Direction.UP)) continue;
				AABB body = self.getBoundingBox().move(direction.getStepX(), 0.0, direction.getStepZ());
				if (level.noCollision(self, body)) exits++;
			}
			return exits >= 1;
		}

		private static boolean placementCell(ServerLevel level, Mob self, Player player, BlockPos pos) {
			if (!level.hasChunkAt(pos) || !level.getWorldBorder().isWithinBounds(pos)
					|| level.isOutsideBuildHeight(pos) || !level.getBlockState(pos).isAir()
					|| !level.getFluidState(pos).isEmpty() || !player.mayInteract(level, pos)
					|| self.distanceToSqr(Vec3.atCenterOf(pos)) > 3.2 * 3.2) return false;
			AABB box = new AABB(pos).inflate(0.05);
			if (!level.getEntitiesOfClass(LivingEntity.class, box, Entity::isAlive).isEmpty()
					|| !level.getEntitiesOfClass(net.minecraft.world.entity.vehicle.AbstractMinecart.class, box).isEmpty()) return false;
			for (int x = -1; x <= 1; x++) for (int z = -1; z <= 1; z++) {
				BlockPos nearby = pos.offset(x, 0, z);
				if (!level.hasChunkAt(nearby)) return false;
				for (int y = -1; y <= 1; y++) {
					BlockState state = level.getBlockState(nearby.offset(0, y, 0));
					if (state.getBlock() instanceof net.minecraft.world.level.block.DoorBlock
							|| state.getBlock() instanceof net.minecraft.world.level.block.TrapDoorBlock
							|| state.getBlock() instanceof net.minecraft.world.level.block.FenceGateBlock
							|| state.getBlock() instanceof net.minecraft.world.level.block.Portal) return false;
				}
			}
			return true;
		}

		private static boolean placeCover(ServerLevel level, Mob self, Player player,
				net.minecraft.world.entity.projectile.Projectile shot, BlockPos pos, Vec3 impact,
				net.minecraft.nbt.CompoundTag record, long now) {
			return placeCover(level, self, player, shot, pos, impact, record, now, null);
		}

		private static boolean placeCover(ServerLevel level, Mob self, Player player,
				net.minecraft.world.entity.projectile.Projectile shot, BlockPos pos, Vec3 impact,
				net.minecraft.nbt.CompoundTag record, long now, double[] prediction) {
			if (!net.neoforged.neoforge.event.EventHooks.canEntityGrief(level, self)
					|| PENDING_COVER.containsKey(self) || PENDING_COVER.size() >= PLAYERS * COVER_SLOTS
					|| pos.getY() < self.blockPosition().getY() || pos.getY() > self.blockPosition().getY() + 1
					|| !safeExit(level, self, pos) || !placementCell(level, self, player, pos)) return false;
			if (prediction != null) {
				if (!predictedCoverPathClear(level, shot, pos, prediction)) return false;
			} else {
				Vec3 start = shot == null ? player.getEyePosition() : shot.position();
                if (shot == null && (!SplinterHivemind.bowDrawn(self, player) || skill(record) < 35)) return false;
				BlockHitResult occlusion = level.clip(new ClipContext(start, Vec3.atCenterOf(pos),
						ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, shot == null ? player : shot));
				if (occlusion.getType() != HitResult.Type.MISS) return false;
				if (new AABB(pos).inflate(0.15).clip(start, impact).isEmpty()) return false;
			}
			BlockPos base = pos;
			if (!level.getBlockState(pos.below()).isFaceSturdy(level, pos.below(), net.minecraft.core.Direction.UP))
				base = pos.below();
			int delay = coverStepDelay(skill(record)), count = 2;
			if (!level.hasChunkAt(base) || !level.getBlockState(base.below()).isFaceSturdy(
						level, base.below(), net.minecraft.core.Direction.UP)
					|| !placementCell(level, self, player, base) || !placementCell(level, self, player, base.above())
					// A high shot must not be "blocked" before the upper block can actually exist.
					|| prediction != null && pos.getY() > base.getY() && prediction[0] <= delay + 1) return false;
			int free = -1;
			for (int c = 0; c < COVER_SLOTS; c++) if (!record.contains("Cover" + c)) { free = c; break; }
			if (free < 0) return coverNote(self, "all tracked cover slots occupied");
			if (now - record.getLong("BuildWindow") >= 1200L) {
				record.putLong("BuildWindow", now); record.putInt("BuildCount", 0);
			}
			if (record.getInt("BuildCount") + count > 8 + skill(record) * 32 / 100) return coverNote(self, "per-player minute build budget exhausted");
			BlockState material = TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState();
			List<net.neoforged.neoforge.common.util.BlockSnapshot> snapshots = new ArrayList<>();
			net.neoforged.neoforge.common.util.BlockSnapshot snapshot =
					net.neoforged.neoforge.common.util.BlockSnapshot.create(level.dimension(), level, base);
			snapshots.add(snapshot);
			if (!level.setBlock(base, material, 2)
					|| net.neoforged.neoforge.event.EventHooks.onBlockPlace(self, snapshot, net.minecraft.core.Direction.UP)
					|| !level.getBlockState(base).equals(material)) {
				rollback(level, snapshots, material); return false;
			}
			net.minecraft.nbt.CompoundTag cover = new net.minecraft.nbt.CompoundTag();
			cover.putString("Dimension", level.dimension().location().toString());
			cover.putLong("Pos", base.asLong());
			cover.putInt("Count", count);
			cover.putInt("Mask", 1); // Upper cell is reserved, NOT placed or eligible for drops/cleanup yet.
			cover.putLong("Until", now + 120L);
            cover.putLong("PlacedAt", now); // Penetration evidence excludes cover erected after this shot was fired.
			record.put("Cover" + free, cover);
			record.putInt("BuildCount", record.getInt("BuildCount") + count);
			record.putLong("NextBuild", now + Math.max(6, 40 - skill(record) * 34 / 100));
			record.putLong("Seen", now);
			// The caller saves this record synchronously before another entity can observe it.
			PENDING_COVER.put(self, new PendingCover(player.getUUID().toString(),
					level.dimension().location().toString(), base.asLong(),
					self.getPersistentData().getInt("ahCoverPlayerSlot"), free, now + delay));
			self.getPersistentData().putLong("ahPlacePose", now + delay + 2);
			self.getLookControl().setLookAt(base.getX() + 0.5, base.getY() + 0.5, base.getZ() + 0.5, 30, 25);
			level.updateNeighborsAt(base, material.getBlock());
			return true;
		}

		private static void rollback(ServerLevel level,
				List<net.neoforged.neoforge.common.util.BlockSnapshot> snapshots, BlockState material) {
			for (int n = snapshots.size() - 1; n >= 0; n--) {
				net.neoforged.neoforge.common.util.BlockSnapshot snapshot = snapshots.get(n);
				if (level.getBlockState(snapshot.getPos()).equals(material)) snapshot.restore();
			}
		}

		private static void releaseCell(net.minecraft.nbt.CompoundTag record, int slot, BlockPos pos) {
			net.minecraft.nbt.CompoundTag cover = record.getCompound("Cover" + slot);
			BlockPos base = BlockPos.of(cover.getLong("Pos"));
			int bit = pos.getY() - base.getY();
			if (bit < 0 || bit > 1) return;
			int mask = cover.getInt("Mask") & ~(1 << bit);
			if (mask == 0) record.remove("Cover" + slot);
			else cover.putInt("Mask", mask);
		}

		private static boolean coverCell(net.minecraft.nbt.CompoundTag cover, ServerLevel level, BlockPos pos) {
			if (!cover.getString("Dimension").equals(level.dimension().location().toString())) return false;
			BlockPos base = BlockPos.of(cover.getLong("Pos"));
			return pos.getX() == base.getX() && pos.getZ() == base.getZ()
					&& pos.getY() >= base.getY() && pos.getY() < base.getY() + cover.getInt("Count")
					&& (cover.getInt("Mask") & (1 << (pos.getY() - base.getY()))) != 0;
		}
	}



	// Exact, OP-only debug chat. No fuzzy matching and no changes to another player's memory.
	private static boolean hiveDebugPermission(net.minecraft.server.level.ServerPlayer player) {

	}

	private static void invalidateNearbyHiveExperience(ServerLevel level, Player player) {
		String owner = player.getUUID().toString();
		int[] examined = {0};
		level.getEntities().get(net.minecraft.world.level.entity.EntityTypeTest.forClass(Mob.class),
				player.getBoundingBox().inflate(64.0), member -> {
			if (SplinterHivemind.member(member) && owner.equals(member.getPersistentData().getString("ahOwner")))
				member.getPersistentData().remove("ahSkillUntil");
			return ++examined[0] >= 128 ? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
					: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
		});
	}

	@SubscribeEvent(priority = net.neoforged.bus.api.EventPriority.HIGHEST)
	public static void onHiveDebugChat(net.neoforged.neoforge.event.ServerChatEvent event) {
		String raw = event.getRawText().trim();
		if (!(raw.equalsIgnoreCase("!hive") || raw.regionMatches(true, 0, "!hive ", 0, 6))) return;
		event.setCanceled(true); // Debug requests and responses are never broadcast to other players.
		net.minecraft.server.level.ServerPlayer player = event.getPlayer();
		ServerLevel level = (ServerLevel) player.level();
		long now = AdaptiveHive.now(level);
		net.minecraft.nbt.CompoundTag data = player.getPersistentData();
		if (data.contains("hiveDebugChatAt") && now >= data.getLong("hiveDebugChatAt")
				&& now - data.getLong("hiveDebugChatAt") < 4L) return;
		data.putLong("hiveDebugChatAt", now);
		if (!hiveDebugPermission(player)) {
			hiveDebugReply(player, "Requires cheats / operator permission level 2.");
			return;
		}
		if (raw.length() > 80) { hiveDebugReply(player, "Request too long."); return; }
		String[] words = raw.toLowerCase(java.util.Locale.ROOT).split(" +");
		if (words.length == 1 || words.length == 2 && words[1].equals("help")) {
			hiveDebugReply(player, "!hive xp | !hive xp <0-100> | !hive cover | !hive afterimage [on|off|preview|status]");
			return;
		}
		if (words.length == 2 && words[1].equals("cover")) {
            hiveDebugReply(player, AdaptiveHive.coverDebug(player));
            return;
        }
		if (words[1].equals("afterimage") && words.length <= 3) {
			if (words.length == 3) {
				if (words[2].equals("status")) {
					hiveDebugReply(player, net.mcreator.thebackwoods.BlindspotAfterimage.debugStatus(player));
					return;
				}
				if (words[2].equals("preview")) {
					hiveDebugReply(player, net.mcreator.thebackwoods.BlindspotAfterimage.debugPreview(player));
					return;
				}
				if (words[2].equals("on")) data.putLong("hiveDebugAfterimageUntil", now + 6000L);
				else if (words[2].equals("off")) data.remove("hiveDebugAfterimageUntil");
				else { hiveDebugReply(player, "Use !hive afterimage on, off, preview or status."); return; }
			}
			hiveDebugReply(player, "Afterimage test " + (AdaptiveHive.afterimageDebugActive(player) ? "ON (up to 5 minutes)" : "OFF")
					+ ": ON skips chance and cooldowns only; 60+ XP, a survival target, a seen Blindspot and a real flank outside your view are still required.");
			return;
		}
		if (words.length < 2 || words.length > 3 || !words[1].equals("xp")) {
			hiveDebugReply(player, "Use !hive help.");
			return;
		}
		boolean change = words.length == 3;
		int value = -1;
		if (change) {
			if (!words[2].matches("[0-9]{1,3}")) { hiveDebugReply(player, "XP must be an integer from 0 to 100."); return; }
			value = Integer.parseInt(words[2]);
			if (value > 100) { hiveDebugReply(player, "XP must be from 0 to 100."); return; }
		}
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
		int slot = AdaptiveHive.slot(root, player, now, change);
		if (slot < 0) {
			hiveDebugReply(player, change ? "No free player-memory slot; nothing changed." : "Your shared hivemind experience: 0 / 100 (no memory yet).");
			return;
		}
		net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
		long before = record.getLong("Decay");
		AdaptiveHive.decay(record, now);
		if (change) {
			record.putDouble("Knowledge", value);
			record.putLong("Seen", now);
			record.putLong("Decay", now);
		}
		if (change || record.getLong("Decay") != before) {
			root.put("Player" + slot, record);
			AdaptiveHive.save(level, root);
		}
		if (change) invalidateNearbyHiveExperience(level, player);
		hiveDebugReply(player, "Your shared hivemind experience: "
				+ String.format(java.util.Locale.ROOT, "%.1f", AdaptiveHive.knowledge(record)) + " / 100"
				+ (change ? " (saved for all four splinters; observed combat patterns were not fabricated)." : "."));
	}

	@SubscribeEvent
	public static void onHiveTacticDamage(net.neoforged.neoforge.event.entity.living.LivingDamageEvent.Post event) {
		if (event.getNewDamage() <= 0 || !(event.getEntity().level() instanceof ServerLevel level)) return;
		Entity attacker = event.getSource().getEntity();
		if (event.getEntity() instanceof Mob victim && SplinterHivemind.member(victim)
				&& attacker instanceof Player player && event.getSource().getDirectEntity() == player
				&& !event.getSource().is(net.minecraft.tags.DamageTypeTags.IS_EXPLOSION))
			AdaptiveHive.learnMeleeHit(level, victim, player);
		if (event.getEntity() instanceof Mob victim && SplinterHivemind.member(victim)
				&& attacker instanceof Player player)
			AdaptiveHive.recordTacticResult(level, victim, player, false);
		if (event.getEntity() instanceof Player player && attacker instanceof Mob member && SplinterHivemind.member(member))
			AdaptiveHive.recordTacticResult(level, member, player, true);
		if (event.getEntity() instanceof Mob victim && SplinterHivemind.member(victim)
				&& attacker instanceof Player player && AdaptiveHive.validPlayer(player, level)
				&& !event.getSource().is(net.minecraft.tags.DamageTypeTags.IS_EXPLOSION)
				&& event.getSource().getDirectEntity() instanceof net.minecraft.world.entity.projectile.Projectile shot) {
			AdaptiveHive.learnPenetration(level, victim, player, shot);
		}
	}


    @SubscribeEvent(priority = net.neoforged.bus.api.EventPriority.LOWEST)
    public static void onHiveLearningDeath(net.neoforged.neoforge.event.entity.living.LivingDeathEvent event) {
        if (event.isCanceled() || !(event.getEntity() instanceof Mob victim) || !SplinterHivemind.member(victim)
                || !(victim.level() instanceof ServerLevel level) || !AdaptiveHive.learningHere(level)
                || victim.getPersistentData().getBoolean("ahKillLearned")) return;
        Entity attacker = event.getSource().getEntity();
        if (!(attacker instanceof Player) && event.getSource().getDirectEntity()
                instanceof net.minecraft.world.entity.projectile.Projectile shot) attacker = shot.getOwner();
        if (!(attacker instanceof Player) && victim.getLastHurtByMob() instanceof Player recent
                && victim.tickCount - victim.getLastHurtByMobTimestamp() <= 100) attacker = recent;
        if (!(attacker instanceof Player player) || !AdaptiveHive.validPlayer(player, level)) return;
        victim.getPersistentData().putBoolean("ahKillLearned", true);
        long time = AdaptiveHive.now(level);
        net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
        int slot = AdaptiveHive.slot(root, player, time, true);
        if (slot < 0) return;
        net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + slot);
        AdaptiveHive.decay(record, time);
        AdaptiveHive.gain(record, AdaptiveHive.KILL_LEARNING_GAIN);
        record.putLong("Seen", time); root.put("Player" + slot, record); AdaptiveHive.save(level, root);
    }

	@SubscribeEvent
	public static void onHiveTacticDeath(net.neoforged.neoforge.event.entity.living.LivingDeathEvent event) {
		if (event.isCanceled() || !(event.getEntity() instanceof Mob member) || !SplinterHivemind.member(member)
				|| !(member.level() instanceof ServerLevel level) || !AdaptiveHive.learningHere(level)) return;
		Player player = null;
		try {
			Entity remembered = level.getEntity(UUID.fromString(member.getPersistentData().getString("htOwner")));
			if (remembered instanceof Player owner) player = owner;
		} catch (IllegalArgumentException ignored) { }
		if (!AdaptiveHive.validPlayer(player, level)) return;
		boolean[] witnessed = {false};
		int[] checked = {0};
		level.getEntities().get(net.minecraft.world.level.entity.EntityTypeTest.forClass(Mob.class),
				member.getBoundingBox().inflate(24), sibling -> {
			if (sibling != member && sibling.isAlive() && SplinterHivemind.member(sibling)
					&& !SplinterHivemind.frozen(sibling) && sibling.hasLineOfSight(member)) witnessed[0] = true;
			return witnessed[0] || ++checked[0] >= 32
					? net.minecraft.util.AbortableIterationConsumer.Continuation.ABORT
					: net.minecraft.util.AbortableIterationConsumer.Continuation.CONTINUE;
		});
		if (witnessed[0]) {
			member.getPersistentData().remove("htOutcomeAt");
			AdaptiveHive.recordTacticResult(level, member, player, false);
		}
	}

	@SubscribeEvent
	public static void onHiveProjectileImpact(net.neoforged.neoforge.event.entity.ProjectileImpactEvent event) {
		if (!(event.getProjectile().level() instanceof ServerLevel level) || !AdaptiveHive.learningHere(level)
				|| !(event.getProjectile().getOwner() instanceof Player player)
				|| !AdaptiveHive.validPlayer(player, level) || event.isCanceled()
				|| event.getProjectile().getPersistentData().getBoolean("ahScored")
				|| !(event.getRayTraceResult() instanceof BlockHitResult hit)) return;
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
		int s = AdaptiveHive.slot(root, player, AdaptiveHive.now(level), false);
		if (s < 0) return;
		net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
		for (int c = 0; c < AdaptiveHive.COVER_SLOTS; c++) {
			if (!record.contains("Cover" + c) || !AdaptiveHive.coverCell(record.getCompound("Cover" + c), level, hit.getBlockPos())
					|| !level.getBlockState(hit.getBlockPos()).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) continue;
			event.getProjectile().getPersistentData().putBoolean("ahScored", true);
			long now = AdaptiveHive.now(level);
			if (now >= record.getLong("NextOutcome")) {
				AdaptiveHive.decay(record, now);
				AdaptiveHive.gain(record, AdaptiveHive.BLOCK_LEARNING_GAIN);
				record.putInt("Blocks", Math.min(80, record.getInt("Blocks") + 1));
				record.putLong("NextOutcome", now + Math.max(1, AdaptiveHive.OUTCOME_SAMPLE_INTERVAL_TICKS));
				record.putLong("Seen", now);
				AdaptiveHive.save(level, root);
			}
			return;
		}
	}

	@SubscribeEvent
	public static void onHiveRangedDamage(net.neoforged.neoforge.event.entity.living.LivingDamageEvent.Post event) {
		if (!(event.getEntity() instanceof Mob self) || !SplinterHivemind.member(self)
				|| !(self.level() instanceof ServerLevel level) || !AdaptiveHive.learningHere(level)
				|| event.getNewDamage() <= 0.0F || !(event.getSource().getEntity() instanceof Player player)
				|| !AdaptiveHive.validPlayer(player, level)
				|| !(event.getSource().getDirectEntity() instanceof net.minecraft.world.entity.projectile.Projectile shot)
				|| shot.getOwner() != player) return;
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
		long now = AdaptiveHive.now(level);
		int s = AdaptiveHive.slot(root, player, now, true);
		if (s < 0) return;
		net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
		if (now < record.getLong("NextOutcome") || shot.getPersistentData().getBoolean("ahScored")) return;
		shot.getPersistentData().putBoolean("ahScored", true);
		AdaptiveHive.decay(record, now);
		AdaptiveHive.gain(record, AdaptiveHive.HIT_LEARNING_GAIN);
		record.putInt("Hits", Math.min(100, record.getInt("Hits") + 1));
		record.putLong("NextOutcome", now + Math.max(1, AdaptiveHive.OUTCOME_SAMPLE_INTERVAL_TICKS));
		record.putLong("Seen", now);
		AdaptiveHive.save(level, root);
	}

	@SubscribeEvent
	public static void onHiveCoverDrops(net.neoforged.neoforge.event.level.BlockDropsEvent event) {
		ServerLevel level = event.getLevel();
		if (!event.getState().is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) return;
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
		for (int s = 0; s < AdaptiveHive.PLAYERS; s++) {
			if (!root.contains("Player" + s)) continue;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
			for (int c = 0; c < AdaptiveHive.COVER_SLOTS; c++) {
				if (record.contains("Cover" + c) && AdaptiveHive.coverCell(record.getCompound("Cover" + c), level, event.getPos())) {
					event.getDrops().clear();
					event.setDroppedExperience(0);
					AdaptiveHive.releaseCell(record, c, event.getPos());
					AdaptiveHive.save(level, root);
					return;
				}
			}
		}
	}


	@SubscribeEvent
	public static void onHiveCoverPiston(net.neoforged.neoforge.event.level.PistonEvent.Pre event) {
		if (!(event.getLevel() instanceof ServerLevel level) || event.isCanceled()) return;
		net.minecraft.world.level.block.piston.PistonStructureResolver resolver = event.getStructureHelper();
		if (resolver == null || !resolver.resolve()) return;
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
		for (int s = 0; s < AdaptiveHive.PLAYERS; s++) {
			if (!root.contains("Player" + s)) continue;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
			for (int c = 0; c < AdaptiveHive.COVER_SLOTS; c++) {
				if (!record.contains("Cover" + c)) continue;
				net.minecraft.nbt.CompoundTag cover = record.getCompound("Cover" + c);
				for (BlockPos pos : resolver.getToPush()) if (AdaptiveHive.coverCell(cover, level, pos)) {
					event.setCanceled(true); return;
				}
			}
		}
	}

	@SubscribeEvent(priority = net.neoforged.bus.api.EventPriority.LOWEST)
	public static void onHiveCoverReplaced(net.neoforged.neoforge.event.level.BlockEvent.EntityPlaceEvent event) {
		if (!(event.getLevel() instanceof ServerLevel level) || event.isCanceled()) return;
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(level);
		boolean changed = false;
		for (int s = 0; s < AdaptiveHive.PLAYERS; s++) {
			if (!root.contains("Player" + s)) continue;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
			for (int c = 0; c < AdaptiveHive.COVER_SLOTS; c++) {
				if (record.contains("Cover" + c)
						&& AdaptiveHive.coverCell(record.getCompound("Cover" + c), level, event.getPos())) {
					AdaptiveHive.releaseCell(record, c, event.getPos());
					changed = true;
				}
			}
		}
		if (changed) AdaptiveHive.save(level, root);
	}

	@SubscribeEvent
	public static void onHiveCoverExpiry(net.neoforged.neoforge.event.tick.ServerTickEvent.Post event) {
		ServerLevel overworld = event.getServer().overworld();
		long now = overworld.getGameTime();
		if (now % 10L != 0L) return;
		net.minecraft.nbt.CompoundTag root = AdaptiveHive.root(overworld);
		boolean changed = false;
		for (int s = 0; s < AdaptiveHive.PLAYERS; s++) {
			if (!root.contains("Player" + s)) continue;
			net.minecraft.nbt.CompoundTag record = root.getCompound("Player" + s);
			for (int c = 0; c < AdaptiveHive.COVER_SLOTS; c++) {
				String key = "Cover" + c;
				if (!record.contains(key)) continue;
				net.minecraft.nbt.CompoundTag cover = record.getCompound(key);
				if (now < cover.getLong("Until")) continue;
				net.minecraft.resources.ResourceLocation id = net.minecraft.resources.ResourceLocation.tryParse(cover.getString("Dimension"));
				if (id == null) { record.remove(key); changed = true; continue; }
				ServerLevel level = event.getServer().getLevel(net.minecraft.resources.ResourceKey.create(
						net.minecraft.core.registries.Registries.DIMENSION, id));
				BlockPos pos = BlockPos.of(cover.getLong("Pos"));
				if (level == null || !level.hasChunkAt(pos)) continue;
				int count = Mth.clamp(cover.getInt("Count"), 1, 2);
				AABB occupied = new AABB(pos).expandTowards(0, count + 1, 0).inflate(0.1);
				if (!level.getEntitiesOfClass(LivingEntity.class, occupied, Entity::isAlive).isEmpty()) continue;
				for (int n = count - 1; n >= 0; n--) {
					BlockPos cell = pos.above(n);
					if (AdaptiveHive.coverCell(cover, level, cell)
							&& level.getBlockState(cell).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get()))
						level.destroyBlock(cell, false);
				}
				record.remove(key); changed = true;
			}
		}
		if (changed) AdaptiveHive.save(overworld, root);
	}

	/** Short-lived, local coordination across the four Splinter variants. Server side only. */
	public static final class SplinterHivemind {
	    private static final int ALERT_TICKS = 120;
	    private static double getAlertRadius(Level level) {
	        if (level == null) return 40.0;
	        if (level.getLevelData().isHardcore()) {
	            return 160.0;
	        }
	        net.minecraft.world.Difficulty diff = level.getDifficulty();
	        if (diff == net.minecraft.world.Difficulty.EASY || diff == net.minecraft.world.Difficulty.PEACEFUL) {
	            return 28.0;
	        } else if (diff == net.minecraft.world.Difficulty.HARD) {
	            return 90.0;
	        }
	        return 40.0;
	    }
	    private static final String ID = "hive_target_uuid";
	    private static final String DIM = "hive_dimension";
	    private static final String UNTIL = "hive_until";
	    private static final String X = "hive_x";
	    private static final String Y = "hive_y";
	    private static final String Z = "hive_z";

	    private SplinterHivemind() { }

	    public static void allowPost(Mob self) {
	        self.getPersistentData().putLong("hive_post_time", self.level().getGameTime());
	    }

	    /** Reassert a flanking route after the vanilla melee goal updates navigation. */
	    public static void followFormationPost(Mob self, int mineProgress) {
	        if (AdaptiveHive.finishCoverDetour(self, mineProgress)) return;
            if (AdaptiveHive.finishTactic(self, mineProgress)) return;
	        if (AdaptiveHive.closeForMelee(self, mineProgress)) return;
	        if (!(self.level() instanceof ServerLevel) || frozen(self) || mineProgress != 0
	                || (self.getPersistentData().getLong("hive_post_time") != self.level().getGameTime()
	                    && self.getPersistentData().getLong("hive_bow_post_time") != self.level().getGameTime())
	                || !self.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)
	                || self.getAttribute(Attributes.MOVEMENT_SPEED).getBaseValue() <= 0.01) return;
	        if (!(self.getTarget() instanceof Player target) || !target.isAlive()
	                || target.isCreative() || target.isSpectator()) return;
	        if (self.getPersistentData().getLong("hive_bow_post_time") == self.level().getGameTime()) {
	            if (self.getPersistentData().getInt("hive_bow_state") == BOW_SEEK || self.getPersistentData().getInt("hive_bow_state") == BOW_FLANK) {
	                Vec3 point = bowPoint(self);
	                self.getNavigation().moveTo(point.x, point.y, point.z, 1.0);
	            } else {
	                self.getNavigation().stop();
	            }
	            return;
	        }
	        Vec3 destination = approach(self, target);
	        if (destination.distanceToSqr(target.position()) < 1.0
	                || self.distanceToSqr(target) < 16.0) return;
	        if ((self.tickCount + self.getId()) % 3 == 0 || self.getNavigation().isDone()) {
	            self.getNavigation().moveTo(destination.x, destination.y, destination.z, 1.05);
	        }
	    }

	    private static boolean member(Entity e) {
	        return e instanceof SplinterEntity || e instanceof BlindspotSplinterEntity
	            || e instanceof LogSplinterEntity || e instanceof PetrifiedLogSplinterEntity;
	    }

	    private static boolean frozen(Mob e) {
	        if (e instanceof SplinterEntity s) return s.getEntityData().get(SplinterEntity.DATA_frozenByRose) == 1;
	        if (e instanceof BlindspotSplinterEntity s) return s.getEntityData().get(BlindspotSplinterEntity.DATA_frozenByRose) == 1;
	        if (e instanceof LogSplinterEntity s) return s.getEntityData().get(LogSplinterEntity.DATA_frozenByRose) == 1;
	        if (e instanceof PetrifiedLogSplinterEntity s) return s.getEntityData().get(PetrifiedLogSplinterEntity.DATA_frozenByRose) == 1;
	        return true;
	    }

	    private static String dimension(ServerLevel level) {
	        return level.dimension().location().toString();
	    }

	    private static boolean current(Mob e, Player player, ServerLevel level) {
	        return dimension(level).equals(e.getPersistentData().getString(DIM))
	            && player.getUUID().toString().equals(e.getPersistentData().getString(ID))
	            && level.getGameTime() < e.getPersistentData().getLong(UNTIL);
	    }

	    private static void remember(Mob e, Player target, ServerLevel level) {
	        e.getPersistentData().putString(ID, target.getUUID().toString());
	        e.getPersistentData().putString(DIM, dimension(level));
	        e.getPersistentData().putLong(UNTIL, level.getGameTime() + ALERT_TICKS);
	        e.getPersistentData().putDouble(X, target.getX());
	        e.getPersistentData().putDouble(Y, target.getY());
	        e.getPersistentData().putDouble(Z, target.getZ());
	    }

	    /** A true sighting or attack shares the last known position with loaded neighbors. */
	    public static void observe(Mob source, Player visible, Entity attacker) {
	        if (!(source.level() instanceof ServerLevel level) || frozen(source)) return;
	        if ((source.tickCount + source.getId()) % 10 != 0) return;
	        Player target = null;
	        if (attacker instanceof Player p && p.isAlive() && !p.isCreative() && !p.isSpectator()
	                && source.distanceToSqr(p) <= 32 * 32 && source.hurtTime > 0) target = p;
	        else if (visible != null && visible.isAlive() && !visible.isCreative() && !visible.isSpectator()
	                && source.distanceToSqr(visible) <= 48 * 48
	                && source.hasLineOfSight(visible)) target = visible;
	        if (target == null) return;
	        if (target.getMainHandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()
	                || target.getOffhandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) return;

	        double alertRadius = getAlertRadius(level);
	        AABB area = source.getBoundingBox().inflate(alertRadius, 12, alertRadius);
	        List<Mob> neighbors = level.getEntitiesOfClass(Mob.class, area,
	            e -> e != source && e.isAlive() && member(e) && !frozen(e)
	                && e.distanceToSqr(source) <= alertRadius * alertRadius);
	        if (neighbors.isEmpty()) { remember(source, target, level); return; }

	        boolean newAlarm = !current(source, target, level);
	        if (newAlarm) source.getPersistentData().putLong("hive_engage_until", level.getGameTime() + 160);
	        remember(source, target, level);
	        for (Mob neighbor : neighbors) {
	            // Sharing a report never sets rage, personality, melee timing, or a live target.
	            if (!current(neighbor, target, level)) {
	                newAlarm = true;
	                neighbor.getPersistentData().putLong("hive_engage_until", level.getGameTime() + 160);
	            }
	            remember(neighbor, target, level);
	        }
	        long lastCue = source.getPersistentData().getLong("hive_last_cue");
	        if (newAlarm && (lastCue == 0 || level.getGameTime() - lastCue >= 160)) {
	            source.getPersistentData().putLong("hive_last_cue", level.getGameTime());
	            level.playSound(null, source.blockPosition(), SoundEvents.FIRE_EXTINGUISH,
	                SoundSource.HOSTILE, 1.2f, 0.55f);
	        }
	    }

	    /** Apply the group alert once; leave rage and individual attack timing alone. */
	    public static boolean beginGroupHunt(Mob self, Entity target) {
	        if (!(target instanceof Player player) || !(self.level() instanceof ServerLevel level)
	                || !current(self, player, level) || frozen(self)) return false;
	        long epoch = self.getPersistentData().getLong("hive_engage_until");
	        if (epoch <= level.getGameTime() || self.getPersistentData().getLong("hive_hunt_used") == epoch) return false;
	        self.getPersistentData().putLong("hive_hunt_used", epoch);
	        return true;
	    }

	    /** A lost target yields a brief, fallible search around the last reported position. */
	    public static boolean search(Mob self) {
	        if (!(self.level() instanceof ServerLevel level) || frozen(self)) return false;
	        long until = self.getPersistentData().getLong(UNTIL);
	        if (until <= level.getGameTime() || !dimension(level).equals(self.getPersistentData().getString(DIM))) return false;
	        try {
	            Entity target = level.getEntity(UUID.fromString(self.getPersistentData().getString(ID)));
	            if (!(target instanceof Player p) || !p.isAlive() || p.isCreative() || p.isSpectator()) return false;
	        } catch (IllegalArgumentException ex) {
	            return false;
	        }
	        double x = self.getPersistentData().getDouble(X);
	        double y = self.getPersistentData().getDouble(Y);
	        double z = self.getPersistentData().getDouble(Z);
	        if (self.distanceToSqr(x, y, z) > 48 * 48) return false;
	        if (self.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)) {
	            self.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(0.23);
	        }
	        int slot = Math.floorMod(self.getUUID().hashCode(), 4);
	        double sx = x + (slot == 0 ? 4 : slot == 1 ? -4 : 0);
	        double sz = z + (slot == 2 ? 4 : slot == 3 ? -4 : 0);
	        if ((self.tickCount + self.getId()) % 12 == 0 || self.getNavigation().isDone()) {
	            self.getNavigation().moveTo(sx, y, sz, 1.0);
	        }
	        return true;
	    }


	    private static double facing(Mob self, Player player) {
	        Vec3 toward = self.getEyePosition().subtract(player.getEyePosition());
	        if (toward.lengthSqr() < 0.01) return 1.0;
	        return player.getLookAngle().normalize().dot(toward.normalize());
	    }

	    /** Strafe around a raised shield, then commit to the ordinary melee goal. */
	    private static Vec3 shieldRoute(Mob self, Player player, ServerLevel level) {
	        long now = level.getGameTime();
	        if (self.getPersistentData().getLong("hive_shield_strike") > now) return player.position();
	        double distance = self.distanceToSqr(player);
	        if (distance < 4 || distance > 100 || facing(self, player) < 0.25
	                || !player.isBlocking() || !player.getUseItem().is(Items.SHIELD)) return null;
	        Vec3 forward = player.getLookAngle();
	        double length = Math.hypot(forward.x, forward.z);
	        if (length < 0.01) return null;
	        double side = (self.getUUID().hashCode() & 1) == 0 ? 1 : -1;
	        Vec3 goal = player.position().add(-forward.x / length * 2 + side * forward.z / length * 3,
	            0, -forward.z / length * 2 - side * forward.x / length * 3);
	        long deadline = self.getPersistentData().getLong("hive_shield_side_until");
	        if (deadline <= now) {
	            self.getPersistentData().putLong("hive_shield_side_until", now + 32);
	            deadline = now + 32;
	        }
	        if (self.distanceToSqr(goal) < 6.25 || deadline - now <= 1) {
	            self.getPersistentData().putLong("hive_shield_strike", now + 18);
	            self.getPersistentData().putLong("hive_shield_side_until", 0);
	            return player.position();
	        }
	        return goal;
	    }

	    // Bow and ranged projectile movement: seek cover, assess it, then take a screened exit.
	    private static final int BOW_SEEK = 1;
	    private static final int BOW_COVER = 2;
	    private static final int BOW_FLANK = 3;
	    private static final int BOW_COMMIT_TICKS = 45;
	    private static final int BOW_COOLDOWN_TICKS = 120;

	    private static float estimateRangedDamage(ItemStack stack) {
	        if (stack.isEmpty()) return 0.0f;
	        if (stack.is(Items.BOW)) return 9.0f;
	        if (stack.is(Items.CROSSBOW)) return 10.0f;
	        if (stack.is(Items.TRIDENT)) return 8.0f;
	        if (stack.is(Items.SPLASH_POTION) || stack.is(Items.LINGERING_POTION)) return 6.0f;
	        if (stack.is(Items.WIND_CHARGE)) return 2.5f;
	        return 0.0f;
	    }

	    private static boolean isRangedWeaponDrawn(Player player, ItemStack stack) {
	        if (stack.isEmpty()) return false;
	        if (stack.is(Items.CROSSBOW)) {
	            return net.minecraft.world.item.CrossbowItem.isCharged(stack)
	                || (player.isUsingItem() && player.getUseItem() == stack);
	        }
	        if (stack.is(Items.BOW) || stack.is(Items.TRIDENT)
	                || stack.is(Items.SPLASH_POTION) || stack.is(Items.LINGERING_POTION)
	                || stack.is(Items.WIND_CHARGE)) {
	            return player.isUsingItem() && player.getUseItem() == stack;
	        }
	        return false;
	    }

	    private static boolean bowDrawn(Mob self, Player player) {
	        if (player == null || !player.isAlive() || player.isCreative() || player.isSpectator()) return false;
	        ItemStack main = player.getMainHandItem();
	        ItemStack off = player.getOffhandItem();
	        ItemStack threatStack = ItemStack.EMPTY;
	        if (isRangedWeaponDrawn(player, main)) threatStack = main;
	        else if (isRangedWeaponDrawn(player, off)) threatStack = off;
	        if (threatStack.isEmpty()) return false;
	        float dmg = estimateRangedDamage(threatStack);
	        if (dmg <= 0.0f) return false;
	        return true;
	    }

	    private static boolean bowDrawn(Player player) {
	        return bowDrawn(null, player);
	    }

	    public static boolean bowCommitActive(Mob self, Player player) {
	        return self.level() instanceof ServerLevel level && bowDrawn(self, player)
	            && level.getGameTime() < self.getPersistentData().getLong("hive_bow_commit_until")
	            && player.getUUID().toString().equals(self.getPersistentData().getString("hive_bow_target"));
	    }

	    private static void commitBow(Mob self, Player player, long now) {
	        int previousState = self.getPersistentData().getInt("hive_bow_state");
	        self.getPersistentData().putInt("hive_bow_state", 0);
	        self.getPersistentData().putLong("hive_bow_commit_until", now + BOW_COMMIT_TICKS);
	        self.getPersistentData().putLong("hive_bow_cooldown", now + BOW_COOLDOWN_TICKS - SplinterOnEntityTickUpdateProcedure.AdaptiveHive.experience(self, player) / 2);
	        self.getPersistentData().putString("hive_bow_target", player.getUUID().toString());
	        self.getPersistentData().putLong("hive_bow_pressure_until", 0);
	        if (previousState != BOW_FLANK && self.level() instanceof ServerLevel level
	                && bowDrawn(self, player) && facing(self, player) >= 0.70
	                && blockedShot(self, player, level, self.position())) {
	            Vec3 exit = pressureExit(self, player, level);
	            if (exit != null) {
	                self.getPersistentData().putDouble("hive_bow_pressure_x", exit.x);
	                self.getPersistentData().putDouble("hive_bow_pressure_y", exit.y);
	                self.getPersistentData().putDouble("hive_bow_pressure_z", exit.z);
	                self.getPersistentData().putLong("hive_bow_pressure_until", now + BOW_COMMIT_TICKS);
	            }
	        }
	    }

	    private static Vec3 bowPoint(Mob self) {
	        return new Vec3(self.getPersistentData().getDouble("hive_bow_x"),
	            self.getPersistentData().getDouble("hive_bow_y"), self.getPersistentData().getDouble("hive_bow_z"));
	    }

	    /** Check the whole actual collision box, including headroom and solid footing. */

	    private static boolean clearStandingRoom(Mob self, ServerLevel level, BlockPos pos) {
	        Vec3 point = new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	        AABB box = self.getBoundingBox().move(point.subtract(self.position()));
	        if (!level.getWorldBorder().isWithinBounds(box)) return false;
	        for (int x = Mth.floor(box.minX); x <= Mth.floor(box.maxX - 0.001); x++) {
	            for (int z = Mth.floor(box.minZ); z <= Mth.floor(box.maxZ - 0.001); z++) {
	                BlockPos floor = new BlockPos(x, pos.getY() - 1, z);
	                if (!level.hasChunkAt(floor) || level.isOutsideBuildHeight(floor)
	                        || !level.getFluidState(floor).isEmpty() || danger(level.getBlockState(floor))
	                        || !level.getBlockState(floor).isFaceSturdy(level, floor, net.minecraft.core.Direction.UP)) return false;
	                for (int y = pos.getY(); y <= Mth.floor(box.maxY - 0.001); y++) {
	                    BlockPos cell = new BlockPos(x, y, z);
	                    if (level.isOutsideBuildHeight(cell) || !level.getFluidState(cell).isEmpty()
	                            || danger(level.getBlockState(cell))) return false;
	                }
	            }
	        }
	        return level.noCollision(self, box)
	            && level.getEntitiesOfClass(LivingEntity.class, box.deflate(0.05), e -> e != self && e.isAlive()).isEmpty();
	    }

	    private static boolean danger(BlockState state) {
	        return state.is(Blocks.CACTUS) || state.is(Blocks.MAGMA_BLOCK) || state.is(Blocks.FIRE)
	            || state.is(Blocks.SOUL_FIRE) || state.is(Blocks.POWDER_SNOW) || state.is(Blocks.WITHER_ROSE)
	            || state.is(Blocks.SWEET_BERRY_BUSH) || state.is(Blocks.COBWEB)
	            || state.getBlock() instanceof net.minecraft.world.level.block.CampfireBlock
	            || state.getBlock() instanceof net.minecraft.world.level.block.Portal;
	    }

	    private static boolean loadedLine(ServerLevel level, Vec3 start, Vec3 end) {
	        if (start.distanceToSqr(end) > 72 * 72) return false;
	        int minX = Mth.floor(Math.min(start.x, end.x) - 1) >> 4;
	        int maxX = Mth.floor(Math.max(start.x, end.x) + 1) >> 4;
	        int minZ = Mth.floor(Math.min(start.z, end.z) - 1) >> 4;
	        int maxZ = Mth.floor(Math.max(start.z, end.z) + 1) >> 4;
	        for (int x = minX; x <= maxX; x++) for (int z = minZ; z <= maxZ; z++)
	            if (!level.hasChunkAt(new BlockPos(x << 4, Mth.floor(start.y), z << 4))) return false;
	        return true;
	    }

	    private static net.minecraft.world.level.pathfinder.Path tacticalPath(Mob self, ServerLevel level, BlockPos pos) {
	        if (self.distanceToSqr(Vec3.atCenterOf(pos)) > 16 * 16 || !clearStandingRoom(self, level, pos)
	                || !AdaptiveHive.pathBudget(level)) return null;
	        int minX = Math.min(self.blockPosition().getX(), pos.getX()) - 8;
	        int maxX = Math.max(self.blockPosition().getX(), pos.getX()) + 8;
	        int minZ = Math.min(self.blockPosition().getZ(), pos.getZ()) - 8;
	        int maxZ = Math.max(self.blockPosition().getZ(), pos.getZ()) + 8;
	        for (int x = minX >> 4; x <= maxX >> 4; x++) for (int z = minZ >> 4; z <= maxZ >> 4; z++)
	            if (!level.hasChunkAt(new BlockPos(x << 4, pos.getY(), z << 4))) return null;
	        net.minecraft.world.level.pathfinder.Path path = self.getNavigation().createPath(pos, 0);
	        if (path == null || !path.canReach() || path.getNodeCount() > 40) return null;
	        BlockPos previous = self.blockPosition();
	        for (int n = path.getNextNodeIndex(); n < path.getNodeCount(); n++) {
	            net.minecraft.world.level.pathfinder.Node node = path.getNode(n);
	            BlockPos step = new BlockPos(node.x, node.y, node.z);
	            if (Math.abs(step.getY() - previous.getY()) > 1 || !clearStandingRoom(self, level, step)) return null;
	            previous = step;
	        }
	        return path;
	    }

	    private static boolean safeRoute(Mob self, ServerLevel level, Vec3 point) {
	        net.minecraft.world.level.pathfinder.Path path = tacticalPath(self, level, BlockPos.containing(point));
	        if (path == null) return false;
	        AdaptiveHive.TACTICAL_ROUTES.put(self, path);
	        return true;
	    }

	    private static boolean blockedShot(Mob self, Player player, ServerLevel level, Vec3 point) {
	        Vec3 start = AdaptiveHive.lastSeenEye(self, player);
	        if (start == null || !loadedLine(level, start, point)) return false;
	        double height = self.getBbHeight();
	        double width = self.getBbWidth() * 0.35;
	        Vec3 toward = point.subtract(start);
	        double span = Math.hypot(toward.x, toward.z);
	        if (span < 0.01) return false;
	        double lateralX = -toward.z / span * width;
	        double lateralZ = toward.x / span * width;
	        boolean thick = AdaptiveHive.requiresThickCover(self, player);
	        Vec3[] silhouette = {
	            point.add(0, Math.min(height - 0.12, self.getEyeHeight()), 0),
	            point.add(0, height * 0.55, 0),
	            point.add(0, 0.22, 0),
	            point.add(lateralX, height * 0.55, lateralZ),
	            point.add(-lateralX, height * 0.55, -lateralZ)
	        };
	        for (Vec3 sample : silhouette) {
	            HitResult hit = level.clip(new ClipContext(start, sample,
	                ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, self));
	            if (hit.getType() != HitResult.Type.BLOCK
	                    || hit.getLocation().distanceToSqr(sample) < 0.16) return false;
	            if (thick && !thickScreen(level, start, sample, (BlockHitResult) hit)) return false;
	        }
	        return true;
	    }


	    private static boolean thickScreen(ServerLevel level, Vec3 start, Vec3 end, BlockHitResult hit) {
	        Vec3 direction = end.subtract(start).normalize();
	        BlockPos previous = null;
	        int layers = 0;
	        for (double depth : new double[] {0.2, 0.75, 1.35}) {
	            BlockPos pos = BlockPos.containing(hit.getLocation().add(direction.scale(depth)));
	            if (!level.hasChunkAt(pos)) return false;
	            BlockState state = level.getBlockState(pos);
	            if (state.is(net.minecraft.tags.BlockTags.PLANKS) || state.is(net.minecraft.tags.BlockTags.LOGS)
	                    || state.getCollisionShape(level, pos).isEmpty()) return false;
	            if (!pos.equals(previous)) layers++;
	            previous = pos;
	        }
	        return layers >= 2;
	    }

	    /** Scan local terrain; discard the last shelter when changing cover. */

	    private static Vec3 nearbyCover(Mob self, Player player, ServerLevel level, Vec3 avoid) {
	        Vec3 origin = self.position();
	        int examined = 0, paths = 0;
	        int rotation = Math.floorMod(self.getId() + (int) (level.getGameTime() / 20), 8);
	        for (int radius : new int[] {3, 5, 7}) {
	            for (int vertical : new int[] {0, -1, 1}) for (int direction = 0; direction < 8; direction++) {
	                if (++examined > 40) return null;
	                double angle = ((direction + rotation) & 7) * Math.PI / 4;
	                BlockPos pos = BlockPos.containing(origin.x + Math.cos(angle) * radius,
	                    origin.y + vertical, origin.z + Math.sin(angle) * radius);
	                Vec3 point = Vec3.atBottomCenterOf(pos);
	                if (avoid != null && point.distanceToSqr(avoid) < 9) continue;
	                if (!clearStandingRoom(self, level, pos) || !blockedShot(self, player, level, point)) continue;
	                if (++paths > 2) return null;
	                if (tacticalPath(self, level, pos) != null) return point;
	            }
	        }
	        return null;
	    }

	    private static Vec3 shortDodge(Mob self, Player player, ServerLevel level) {
	        Vec3 away = self.position().subtract(player.position());
	        double span = Math.hypot(away.x, away.z);
	        if (span < 0.01) return null;
	        double side = (self.getUUID().hashCode() & 1) == 0 ? 1.0 : -1.0;
	        for (int attempt = 0; attempt < 2; attempt++, side = -side) {
	            BlockPos pos = BlockPos.containing(self.getX() - away.z / span * side * 3,
	                self.getY(), self.getZ() + away.x / span * side * 3);
	            if (!clearStandingRoom(self, level, pos)) continue;
	            net.minecraft.world.level.pathfinder.Path path = tacticalPath(self, level, pos);
	            if (path != null && path.canReach())
	                return new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	        }
	        return null;
	    }

	    /** Choose a flank with a screened approach, favoring a different exit next time. */
	    private static Vec3 screenedFlank(Mob self, Player player, ServerLevel level) {
	        int attempts = 0;
	        Vec3 toward = player.position().subtract(self.position());
	        double span = Math.hypot(toward.x, toward.z);
	        if (span < 1.0) return null;
	        Vec3 along = new Vec3(toward.x / span, 0, toward.z / span);
	        Vec3 lateral = new Vec3(-along.z, 0, along.x);
	        int lastSide = self.getPersistentData().getInt("hive_bow_last_side");
	        int variation = self.getUUID().hashCode() ^ player.getUUID().hashCode()
	            ^ (int) (level.getGameTime() / 80);
	        int preferred = lastSide != 0 && (variation & 3) != 0
	            ? -lastSide : (variation & 4) == 0 ? 1 : -1;
	        for (int order = 0; order < 2; order++) {
	            int side = order == 0 ? preferred : -preferred;
	            for (int offset = 2; offset <= 6; offset++) {
	                for (int advance = 1; advance <= 3; advance++) {
	                    Vec3 desired = self.position().add(lateral.scale(side * offset)).add(along.scale(advance));
	                    for (int vertical : new int[] {0, -1, 1}) {
	                        BlockPos pos = BlockPos.containing(desired.x, desired.y + vertical, desired.z);
	                        Vec3 point = new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	                        if (point.distanceToSqr(player.position()) >= self.distanceToSqr(player) - 2.0
	                                || !blockedShot(self, player, level, point)
	                                || !clearStandingRoom(self, level, pos)) continue;
	                        boolean screened = true;
	                        for (double step : new double[] {0.25, 0.5, 0.75}) {
	                            if (!blockedShot(self, player, level, self.position().lerp(point, step))) {
	                                screened = false;
	                                break;
	                            }
	                        }
	                        if (!screened) continue;
	                        if (++attempts > 2) return null;
	                        net.minecraft.world.level.pathfinder.Path path = tacticalPath(self, level, pos);
	                        if (path != null && path.canReach() && mostlyScreenedPath(self, player, level, path)) {
	                            self.getPersistentData().putInt("hive_bow_last_side", side);
	                            return point;
	                        }
	                    }
	                }
	            }
	        }
	        return null;
	    }

	    /** The actual navigation path matters; a straight line can hide an exposed detour. */
	    private static boolean mostlyScreenedPath(Mob self, Player player, ServerLevel level,
	            net.minecraft.world.level.pathfinder.Path path) {
	        int visible = 0;
	        int checked = 0;
	        int stride = Math.max(1, path.getNodeCount() / 7);
	        for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i += stride) {
	            net.minecraft.world.level.pathfinder.Node node = path.getNode(i);
	            Vec3 point = new Vec3(node.x + 0.5, node.y, node.z + 0.5);
	            if (!blockedShot(self, player, level, point)) visible++;
	            checked++;
	        }
	        return checked > 0 && visible <= Math.max(1, checked / 4);
	    }

	    /** If safety is impossible, leave cover around a side rather than retracing entry. */
	    private static Vec3 pressureExit(Mob self, Player player, ServerLevel level) {
	        Vec3 toward = player.position().subtract(self.position());
	        double span = Math.hypot(toward.x, toward.z);
	        if (span < 1.0) return null;
	        Vec3 along = new Vec3(toward.x / span, 0, toward.z / span);
	        Vec3 lateral = new Vec3(-along.z, 0, along.x);
	        int previous = self.getPersistentData().getInt("hive_bow_last_side");
	        int variation = self.getUUID().hashCode() ^ player.getUUID().hashCode()
	            ^ (int) (level.getGameTime() / 80);
	        int first = previous != 0 && (variation & 3) != 0
	            ? -previous : (variation & 4) == 0 ? 1 : -1;
	        for (int order = 0; order < 2; order++) {
	            int side = order == 0 ? first : -first;
	            for (int offset = 3; offset <= 6; offset++) {
	                Vec3 goal = self.position().add(lateral.scale(side * offset)).add(along.scale(2));
	                for (int vertical : new int[] {0, -1, 1}) {
	                    BlockPos pos = BlockPos.containing(goal.x, goal.y + vertical, goal.z);
	                    if (!clearStandingRoom(self, level, pos)) continue;
	                    net.minecraft.world.level.pathfinder.Path path = tacticalPath(self, level, pos);
	                    if (path == null || !path.canReach()) continue;
	                    boolean usesSide = false;
	                    for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i++) {
	                        net.minecraft.world.level.pathfinder.Node node = path.getNode(i);
	                        Vec3 step = new Vec3(node.x + 0.5, node.y, node.z + 0.5).subtract(self.position());
	                        if (step.dot(lateral) * side > 1.0) { usesSide = true; break; }
	                    }
	                    if (!usesSide) continue;
	                    self.getPersistentData().putInt("hive_bow_last_side", side);
	                    return new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	                }
	            }
	        }
	        return null;
	    }

	    /** Wide Screened Flank Arc: curves around the player towards their blindspot, avoiding direct line-of-fire charge (derived from Rot AI avoidance). */
	    private static Vec3 wideScreenedFlankArc(Mob self, Player player, ServerLevel level) {
	        Vec3 playerPos = player.position();
	        Vec3 toward = playerPos.subtract(self.position());
	        double dist = Math.hypot(toward.x, toward.z);
	        if (dist < 3.0 || dist > 32.0) return playerPos;

	        float tYaw = player.getYRot() * ((float) Math.PI / 180.0F);
	        int lastSide = self.getPersistentData().getInt("hive_bow_last_side");
	        int side = lastSide != 0 ? lastSide : SplinterOnEntityTickUpdateProcedure.AdaptiveHive.flankSide(self, player);

	        double arcAngle = tYaw + (side * (float) (Math.PI / 2.3));
	        double arcDist = Math.max(3.2, Math.min(dist * 0.75, 7.5));
	        double arcX = playerPos.x - Math.sin(arcAngle) * arcDist;
	        double arcZ = playerPos.z + Math.cos(arcAngle) * arcDist;

	        BlockPos arcPos = BlockPos.containing(arcX, self.getY(), arcZ);
	        for (int yOff : new int[] {0, 1, -1}) {
	            BlockPos testPos = arcPos.above(yOff);
	            if (clearStandingRoom(self, level, testPos)) {
	                net.minecraft.world.level.pathfinder.Path path = tacticalPath(self, level, testPos);
	                if (path != null && path.canReach()) {
	                    return new Vec3(testPos.getX() + 0.5, testPos.getY(), testPos.getZ() + 0.5);
	                }
	            }
	        }
	        return playerPos;
	    }

	    private static void bowDestination(Mob self, Vec3 point, int state, long now) {
	        self.getPersistentData().putInt("hive_bow_state", state);
	        self.getPersistentData().putDouble("hive_bow_x", point.x);
	        self.getPersistentData().putDouble("hive_bow_y", point.y);
	        self.getPersistentData().putDouble("hive_bow_z", point.z);
	        self.getPersistentData().putDouble("hive_bow_progress", self.distanceToSqr(point));
	        self.getPersistentData().putInt("hive_bow_stalls", 0);
	        self.getPersistentData().putLong("hive_bow_path_check", now + 12);
	    }

	    /** The main AI yields navigation to this state before its watched/phase rules. */
	    public static boolean handleBowThreat(Mob self, Player player, int mineProgress, double speed) {
            if (self instanceof PetrifiedLogSplinterEntity) return AdaptiveHive.petrifiedPressure(self, player, mineProgress, speed);
	        // Extend perception beyond the close tactical range without starving the existing wounded-ally scan.
	        if (player != null && self.distanceToSqr(player) > 48.0 * 48.0 && AdaptiveHive.defend(self, player, mineProgress)) {
	            self.getPersistentData().remove("htState");
	            self.getPersistentData().remove("htPost");
	            AdaptiveHive.TACTICAL_ROUTES.remove(self);
	            return true;
	        }
	        if (AdaptiveHive.tactics(self, player, mineProgress, speed)) return true;
	        if (!(self.level() instanceof ServerLevel level) || frozen(self) || mineProgress != 0
	                || player == null || self.getTarget() != player) return false;
	        long now = level.getGameTime();
	        int state = self.getPersistentData().getInt("hive_bow_state");
	        if (!bowDrawn(self, player)) {
	            if (state != 0) commitBow(self, player, now);
	            return false;
	        }
	        if (player.getMainHandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()
	                || player.getOffhandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) return false;
	        if (state != 0 && !player.getUUID().toString().equals(self.getPersistentData().getString("hive_bow_target"))) {
	            self.getPersistentData().putInt("hive_bow_state", 0);
	            state = 0;
	        }
	        if (state == 0) {
	            if (self.getPersistentData().getLong("hive_bow_cooldown") > now || self.distanceToSqr(player) < 16
	                    || self.distanceToSqr(player) > 28 * 28 || player.getTicksUsingItem() < Math.max(4, 9 - SplinterOnEntityTickUpdateProcedure.AdaptiveHive.experience(self, player) / 20)
	                    || facing(self, player) < 0.84 || !self.hasLineOfSight(player)) return false;
	            Vec3 lastCover = self.getPersistentData().getInt("hive_bow_has_last_cover") != 0
	                ? new Vec3(self.getPersistentData().getDouble("hive_bow_last_cover_x"), self.getPersistentData().getDouble("hive_bow_last_cover_y"), self.getPersistentData().getDouble("hive_bow_last_cover_z")) : null;
	            Vec3 point = nearbyCover(self, player, level, lastCover);
	            if (point == null && lastCover != null) point = nearbyCover(self, player, level, null);
	            boolean hasCover = point != null;
	            if (!hasCover) point = shortDodge(self, player, level);
	            if (point == null) {
	                commitBow(self, player, now);
	                return false;
	            }
	            self.getPersistentData().putString("hive_bow_target", player.getUUID().toString());
	            self.getPersistentData().putInt("hive_bow_changes", 0);
	            self.getPersistentData().putInt("hive_bow_failed_choices", 0);
	            self.getPersistentData().putInt("hive_bow_cover", hasCover ? 1 : 0);
	            bowDestination(self, point, BOW_SEEK, now);
	            state = BOW_SEEK;
	        }

	        if (self.distanceToSqr(player) < 9) {
	            commitBow(self, player, now);
	            return false;
	        }
	        Vec3 point = bowPoint(self);
	        boolean protectedNow = blockedShot(self, player, level, self.position());
	        if (state == BOW_SEEK && protectedNow) {
	            bowDestination(self, self.position(), BOW_COVER, now);
	            self.getPersistentData().putInt("hive_bow_has_last_cover", 1);
	            self.getPersistentData().putDouble("hive_bow_last_cover_x", self.getX());
	            self.getPersistentData().putDouble("hive_bow_last_cover_y", self.getY());
	            self.getPersistentData().putDouble("hive_bow_last_cover_z", self.getZ());
	            state = BOW_COVER;
	        }
	        if (state == BOW_FLANK && !protectedNow && facing(self, player) >= 0.84
	                && self.getPersistentData().getInt("hive_bow_changes") < 2 && now >= self.getPersistentData().getLong("hive_bow_next_choice")) {
	            self.getPersistentData().putLong("hive_bow_next_choice", now + 12);
	            Vec3 other = nearbyCover(self, player, level, point);
	            if (other != null) {
	                self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getInt("hive_bow_changes") + 1);
	                bowDestination(self, other, BOW_SEEK, now);
	                state = BOW_SEEK;
	                point = other;
	            }
	        }
	        if (state == BOW_SEEK || state == BOW_FLANK) {
	            if (state == BOW_FLANK && self.distanceToSqr(point) < 2.25) {
	                commitBow(self, player, now);
	                return false;
	            }
	            if (state == BOW_SEEK && self.distanceToSqr(point) < 2.25 && !protectedNow) {
	                // The spot was screened when picked but the player moved: reconsider.
	                if (self.getPersistentData().getInt("hive_bow_changes") >= 2) {
	                    commitBow(self, player, now);
	                    return false;
	                }
	                Vec3 other = nearbyCover(self, player, level, point);
	                if (other == null) {
	                    commitBow(self, player, now);
	                    return false;
	                }
	                self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getInt("hive_bow_changes") + 1);
	                bowDestination(self, other, BOW_SEEK, now);
	            }
	            if (now >= self.getPersistentData().getLong("hive_bow_path_check")) {
	                double remaining = self.distanceToSqr(bowPoint(self));
	                int stalls = remaining >= self.getPersistentData().getDouble("hive_bow_progress") - 0.4 ? self.getPersistentData().getInt("hive_bow_stalls") + 1 : 0;
	                self.getPersistentData().putInt("hive_bow_stalls", stalls);
	                self.getPersistentData().putDouble("hive_bow_progress", remaining);
	                self.getPersistentData().putLong("hive_bow_path_check", now + 12);
	                if (stalls >= 2) {
	                    if (state == BOW_SEEK && self.getPersistentData().getInt("hive_bow_changes") < 2) {
	                        Vec3 other = nearbyCover(self, player, level, bowPoint(self));
	                        if (other != null) {
	                            self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getInt("hive_bow_changes") + 1);
	                            bowDestination(self, other, BOW_SEEK, now);
	                        } else {
	                            commitBow(self, player, now);
	                            return false;
	                        }
	                    } else {
	                        commitBow(self, player, now);
	                        return false;
	                    }
	                }
	            }
	        } else if (state == BOW_COVER) {
	            if (facing(self, player) < 0.70) {
	                commitBow(self, player, now); // The archer looks away: exploit the opening.
	                return false;
	            }
	            if (!protectedNow) {
	                Vec3 other = self.getPersistentData().getInt("hive_bow_changes") < 2 ? nearbyCover(self, player, level, self.position()) : null;
	                if (other == null) {
	                    commitBow(self, player, now);
	                    return false;
	                }
	                self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getInt("hive_bow_changes") + 1);
	                bowDestination(self, other, BOW_SEEK, now);
	            } else if (now >= self.getPersistentData().getLong("hive_bow_next_choice")) {
	                self.getPersistentData().putLong("hive_bow_next_choice", now + 12 + (self.getId() & 7));
	                Vec3 flank = screenedFlank(self, player, level);
	                if (flank != null) {
	                    bowDestination(self, flank, BOW_FLANK, now);
	                } else {
	                    int failures = self.getPersistentData().getInt("hive_bow_failed_choices") + 1;
	                    self.getPersistentData().putInt("hive_bow_failed_choices", failures);
	                    Vec3 other = failures == 1 && self.getPersistentData().getInt("hive_bow_changes") < 2
	                        ? nearbyCover(self, player, level, self.position()) : null;
	                    if (other != null) {
	                        self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getInt("hive_bow_changes") + 1);
	                        bowDestination(self, other, BOW_SEEK, now);
	                    } else if (failures >= 2) {
	                        // Patience Trap Fix: Never rush out into an aimed ranged weapon while safely occluded in cover!
	                        self.getPersistentData().putInt("hive_bow_failed_choices", 1);
	                        self.getPersistentData().putLong("hive_bow_next_choice", now + 16 + (self.getId() & 7));
	                    }
	                }
	            }
	        }
	        setSpeed(self, speed);
	        self.getPersistentData().putLong("hive_bow_post_time", now);
	        return true;
	    }

	    /** Preserve ordinary movement for lone members and for non-player targets. */
	    public static Vec3 approach(Mob self, Entity target) {
	        Vec3 direct = target.position();
	        if (!(target instanceof Player player) || !(self.level() instanceof ServerLevel level)
	                || frozen(self)) return direct;
	        Vec3 shield = shieldRoute(self, player, level);
	        if (shield != null) return shield;
	        if (bowCommitActive(self, player)) {
	            if (self.getPersistentData().getLong("hive_bow_pressure_until") > level.getGameTime()) {
	                Vec3 exit = new Vec3(self.getPersistentData().getDouble("hive_bow_pressure_x"), self.getPersistentData().getDouble("hive_bow_pressure_y"), self.getPersistentData().getDouble("hive_bow_pressure_z"));
	                if (self.distanceToSqr(exit) > 2.25) return exit;
	            }
	            Vec3 arc = wideScreenedFlankArc(self, player, level);
	            if (arc != null && self.distanceToSqr(arc) > 2.25) return arc;
	            return direct;
	        }
	        if (!current(self, player, level) || self.distanceToSqr(player) < 16) return direct;
	        if (self.getPersistentData().getLong("hive_commit_until") > level.getGameTime()) return direct;
	        long huntUntil = self.getPersistentData().getLong("hive_engage_until");
	        if (huntUntil > level.getGameTime() && huntUntil - level.getGameTime() < 110) {
	            self.getPersistentData().putLong("hive_commit_until", level.getGameTime() + 70);
	            return direct;
	        }

	        long cachedUntil = self.getPersistentData().getLong("hive_route_until");
	        if (cachedUntil > level.getGameTime()) {
	            Vec3 cachedPoint = new Vec3(self.getPersistentData().getDouble("hive_route_x"),
	                self.getPersistentData().getDouble("hive_route_y"),
	                self.getPersistentData().getDouble("hive_route_z"));
	            if (self.distanceToSqr(cachedPoint) < 9.0) {
	                self.getPersistentData().putLong("hive_commit_until", level.getGameTime() + 70);
	                return direct;
	            }
	            return cachedPoint;
	        }

	        List<Mob> team = new ArrayList<>();
	        team.add(self);
	        for (Mob e : level.getEntitiesOfClass(Mob.class, self.getBoundingBox().inflate(24, 12, 24),
	                other -> other != self && other.isAlive() && member(other) && !frozen(other))) {
	            if (current(e, player, level) && e.distanceToSqr(player) <= 32 * 32
	                    && e.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)
	                    && e.getAttribute(Attributes.MOVEMENT_SPEED).getBaseValue() > 0.01) team.add(e);
	        }
	        if (team.size() < 2) return direct;

	        // Tank takes pressure when present; otherwise the closest member does.
	        Mob pursuer = team.stream().min(Comparator.comparingDouble(e ->
	            e.distanceToSqr(player) - (e instanceof PetrifiedLogSplinterEntity ? 100.0 : 0.0))).orElse(self);
	        if (self == pursuer) return direct;
	        team.remove(pursuer);
	        team.sort(Comparator.comparing(Entity::getUUID));
	        int index = team.indexOf(self);
	        if (index < 0) return direct;

	        Vec3 facing = player.getLookAngle();
	        double length = Math.hypot(facing.x, facing.z);
	        if (length < 0.01) return direct;
	        double fx = facing.x / length;
	        double fz = facing.z / length;
	        double side = index % 2 == 0 ? 1.0 : -1.0;
	        Vec3 approach;
	        if (index >= 2 && player.getDeltaMovement().horizontalDistanceSqr() > 0.003) {
	            Vec3 velocity = player.getDeltaMovement();
	            approach = direct.add(Math.max(-6, Math.min(6, velocity.x * 12)) + side * fz * 3,
	                0, Math.max(-6, Math.min(6, velocity.z * 12)) - side * fx * 3);
	        } else {
	            approach = direct.add(-fx * 3 + side * fz * 6, 0, -fz * 3 - side * fx * 6);
	        }
	        self.getPersistentData().putLong("hive_route_until", level.getGameTime() + 10);
	        self.getPersistentData().putDouble("hive_route_x", approach.x);
	        self.getPersistentData().putDouble("hive_route_y", approach.y);
	        self.getPersistentData().putDouble("hive_route_z", approach.z);
	        if (self.distanceToSqr(approach) < 9.0) {
	            self.getPersistentData().putLong("hive_commit_until", level.getGameTime() + 70);
	            return direct;
	        }
	        return approach;
	    }

	    public static Vec3 stalkPoint(Mob self, Entity target) {
	        Vec3 approach = approach(self, target);
	        return approach.distanceToSqr(target.position()) > 1 ? approach : null;
	    }
	}
