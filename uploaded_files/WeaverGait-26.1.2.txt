/** Client-side, tick-clocked tripod gait; rendering only samples committed steps. */
@EventBusSubscriber(modid = "the_backwoods", value = Dist.CLIENT)
public class WeaverGait {
    public static double FOOT_SPREAD = 4.0;
    public static double STEP_DISTANCE = 2.0;
    public static double STEP_HEIGHT = 0.55;
    public static double STANCE_HEIGHT = 3.1;
    public static double POSE_BLEND_TICKS = 4.0;
    private static final Map<UUID, EntityGaitState> GAIT_STATES = new HashMap<>();
    private static Level gaitLevel;

    public static EntityGaitState getOrCreateState(UUID uuid) {
        if (!GAIT_STATES.containsKey(uuid) && GAIT_STATES.size() >= 512) {
            GAIT_STATES.remove(GAIT_STATES.keySet().iterator().next());
        }
        return GAIT_STATES.computeIfAbsent(uuid, k -> new EntityGaitState());
    }

    public static EntityGaitState getState(UUID uuid) {
        return GAIT_STATES.get(uuid);
    }

    public static float transitionWeight(EntityGaitState state, double time, boolean active) {
        double elapsed = Math.max(0, time - state.blendStartedAt);
        float current = (float) Mth.lerp(Mth.clamp(elapsed / Math.max(1, POSE_BLEND_TICKS), 0, 1),
            state.blendFrom, state.poseActive ? 1.0 : 0.0);
        if (state.poseActive != active) {
            state.blendFrom = current;
            state.blendStartedAt = time;
            state.poseActive = active;
        }
        return current;
    }

    /** Continuous leg weight, independent of the authored body/root blend. */
    public static float transitionLegWeight(EntityGaitState state, double time, float target) {
        double elapsed = Double.isNaN(state.legBlendTime) ? 0 : Math.max(0, time - state.legBlendTime);
        state.legBlendTime = Double.isNaN(state.legBlendTime) ? time : Math.max(state.legBlendTime, time);
        float step = (float) (elapsed / Math.max(1, POSE_BLEND_TICKS));
        state.legWeight += Mth.clamp(target - state.legWeight, -step, step);
        return state.legWeight;
    }

    /** Fade authored foot motion without reducing terrain IK authority. */
    public static float transitionActionWeight(EntityGaitState state, double time, boolean active) {
        double elapsed = Double.isNaN(state.actionBlendTime) ? 0 : Math.max(0, time - state.actionBlendTime);
        state.actionBlendTime = Double.isNaN(state.actionBlendTime) ? time : Math.max(state.actionBlendTime, time);
        float step = (float) (elapsed / Math.max(1, POSE_BLEND_TICKS));
        state.actionWeight += Mth.clamp((active ? 1f : 0f) - state.actionWeight, -step, step);
        return state.actionWeight;
    }

    public static void cleanUp(UUID uuid) {
        GAIT_STATES.remove(uuid);
    }

    @SubscribeEvent
    public static void onEntityLeave(EntityLeaveLevelEvent event) {
        if (event.getLevel().isClientSide()) cleanUp(event.getEntity().getUUID());
    }

    @SubscribeEvent
    public static void onLevelUnload(LevelEvent.Unload event) {
        if (event.getLevel() == gaitLevel) {
            GAIT_STATES.clear();
            gaitLevel = null;
        }
    }

    public static class LegState {
        public Vec3 currentPlantedPos = Vec3.ZERO;
        public Vec3 lastPlantedPos = Vec3.ZERO;
        public Vec3 targetHomePos = Vec3.ZERO;
        public Vec3 landingPos = Vec3.ZERO;
        public float stepProgress = 1.0f;
        public boolean isStepping;
        public boolean grounded;
        public boolean contactAvailable, narrow;
        public boolean targetHasContact, landingHasContact;
        public Vec3 contactNormal = new Vec3(0, 1, 0);
        public Vec3 targetContactNormal = new Vec3(0, 1, 0);
        public Vec3 takeoffNormal = new Vec3(0, 1, 0);
        public boolean targetNarrow;
        public Vec3 liftUp = new Vec3(0, 1, 0);
        public double stepStart;
        public double stepDuration = 6.0;
        public double lift = STEP_HEIGHT;
        public boolean recoveryRequested;
        public float kneeSwivel, previousKneeSwivel;
        public int clearanceTick = Integer.MIN_VALUE;
        public boolean clearanceBlocked;
        public Vec3 clearanceHip, clearanceKnee, clearanceAnkle, avoidContactPos;
        public int avoidContactUntil;
        public long lastRecoveryTick = -1000;
        public Vec3 renderedAnkle;
        public Vec3 renderedNormal;
        public Vec3 plantedTangent;
        public double renderedAt = Double.NaN;
        public float contactWeight;
        public float minYaw = Float.NaN, maxYaw = Float.NaN;
        public long plantedAtTick = -100;

        public void init(Vec3 initialPos) {
            currentPlantedPos = lastPlantedPos = targetHomePos = landingPos = initialPos;
            stepProgress = 1.0f;
            isStepping = false;
            recoveryRequested = false;
            kneeSwivel = previousKneeSwivel = 0;
            clearanceTick = Integer.MIN_VALUE;
            clearanceBlocked = false;
            clearanceHip = clearanceKnee = clearanceAnkle = avoidContactPos = null;
            avoidContactUntil = 0;
            lastRecoveryTick = -1000;
            plantedAtTick = -100;
            renderedAnkle = renderedNormal = null;
            plantedTangent = null;
            renderedAt = Double.NaN;
            contactWeight = 1.0f;
            minYaw = maxYaw = Float.NaN;
        }

        public Vec3 sampleNormal(double time) {
            if (!isStepping) return contactNormal;
            double p = Mth.clamp((time - stepStart) / stepDuration, 0, 1);
            double ease = p * p * (3 - 2 * p);
            org.joml.Vector3f n = new org.joml.Vector3f((float)takeoffNormal.x,
                (float)takeoffNormal.y, (float)takeoffNormal.z);
            Quaternionf turn = new Quaternionf().rotationTo(n,
                new org.joml.Vector3f((float)contactNormal.x, (float)contactNormal.y, (float)contactNormal.z));
            new Quaternionf().slerp(turn, (float)ease).transform(n).normalize();
            return new Vec3(n.x, n.y, n.z);
        }

        public Vec3 sample(double time) {
            if (!isStepping) return currentPlantedPos;
            if (time < stepStart) return lastPlantedPos;
            double p = Mth.clamp((time - stepStart) / stepDuration, 0.0, 1.0);
            double ease = p * p * p * (p * (p * 6 - 15) + 10);
            double arc = 64 * p * p * p * (1 - p) * (1 - p) * (1 - p);
            return lastPlantedPos.lerp(landingPos, ease).add(liftUp.scale(lift * arc));
        }
    }

    public static boolean canUseGroundIK(Entity entity) {
        EntityGaitState state = getOrCreateState(entity.getUUID());
        if (state.contactCheckTick != entity.tickCount) {
            state.contactCheckTick = entity.tickCount;
            state.wasTerrainSupported = state.terrainSupported;
            BlockHitResult floor = entity.getDeltaMovement().y <= 0.03
                ? WoodweaverSurfaceNavigation.clipLoaded(entity.level(), entity,
                    entity.position().add(0, 0.35, 0), entity.position().add(0, -Math.min(12.0,
                        Math.max(1.25, -entity.getDeltaMovement().y * 6 + 1.5)), 0)) : null;
            state.nearLanding = floor != null;
            double distance = floor == null ? Double.POSITIVE_INFINITY
                : Math.max(0, entity.getY() - floor.getLocation().y);
            state.landingApproach = landingWeight(distance, Math.max(0, -entity.getDeltaMovement().y));
            // A stair/drop can clear onGround while several feet still support the body.
            // Validate the existing anchors, rather than replacing them with moving homes.
            state.terrainSupported = entity.onGround() || WoodweaverSurfaceNavigation.attached(entity);
            if (!state.terrainSupported && state.initialized && !isLeapPose(entity)
                    && entity.getDeltaMovement().y <= 0.03) {
                int supports = 0;
                Vec3 root = entity.position().add(0, WoodweaverSurfaceNavigation.ROOT_HEIGHT, 0);
                for (LegState leg : state.legs) {
                    if (!leg.grounded || leg.isStepping || root.distanceToSqr(leg.currentPlantedPos) > 6.0 * 6.0) continue;
                    if (validSupport(entity, leg) && ++supports >= 2) break;
                }
                state.terrainSupported = supports >= 2
                    || (distance <= 1.25 && entity.tickCount - state.lastGroundedTick <= 3);
            }
        }
        // Keep the gait clock warm through ascent/apex; leg weight still follows contact/landing approach.
        return state.terrainSupported || state.nearLanding || isLeapPose(entity);
    }

    private static boolean isLeapPose(Entity entity) {
        return entity instanceof net.mcreator.thebackwoods.entity.WoodweaverEntity woodweaver
            && (woodweaver.getEntityData().get(net.mcreator.thebackwoods.entity.WoodweaverEntity.DATA_isLeaping)
                || woodweaver.getEntityData().get(net.mcreator.thebackwoods.entity.WoodweaverEntity.DATA_isLanding));
    }

    public static boolean hasTerrainSupport(Entity entity) {
        canUseGroundIK(entity);
        return getOrCreateState(entity.getUUID()).terrainSupported;
    }

    private static boolean validSupport(Entity entity, LegState leg) {
        if (!entity.level().hasChunkAt(BlockPos.containing(leg.currentPlantedPos))) return false;
        BlockHitResult hit = WoodweaverSurfaceNavigation.clipLoaded(entity.level(), entity,
            leg.currentPlantedPos.add(leg.contactNormal.scale(0.35)),
            leg.currentPlantedPos.subtract(leg.contactNormal.scale(0.35)));
        return hit != null && hit.getLocation().distanceToSqr(leg.currentPlantedPos) <= 0.35 * 0.35;
    }

    /** Finish extending the feet two ticks before predicted impact. */
    public static float landingWeight(double distance, double downwardSpeed) {
        if (!Double.isFinite(distance)) return 0;
        double gap = Math.max(0, distance - 0.15), speed = Math.max(0, downwardSpeed);
        double ticks = (Math.sqrt(speed * speed + 0.16 * gap) - speed) / 0.08;
        float approach = (float) Mth.clamp((6 - ticks) / 4, 0, 1);
        return approach * approach * (3 - 2 * approach);
    }

    /** Model X/Z to world X/Z, matching MobRenderer's 180 - bodyYaw rotation. */
    public static Vec3 rotateHorizontal(Vec3 offset, float yaw) {
        double radians = Math.toRadians(yaw), cos = Math.cos(radians), sin = Math.sin(radians);
        return new Vec3(offset.x * cos + offset.z * sin, offset.y, offset.x * sin - offset.z * cos);
    }

    public record Contact(Vec3 position, Vec3 normal, boolean narrow) {}

    /** Surface-frame probes cover floor, wall and ceiling collision shapes, never unloaded columns. */
    public static Contact findContact(Level level, Entity entity, Vec3 relativeOffset, float yaw, Vec3 prediction) {
        Quaternionf rotation = WoodweaverSurfaceNavigation.rotation(entity, 1);
        Vec3 up = WoodweaverSurfaceNavigation.up(entity, 1);
        Vec3 offset = WoodweaverSurfaceNavigation.WeaverFrame.offset(rotation, yaw, relativeOffset.x, 0, relativeOffset.z);
        Vec3 tangentPrediction = prediction.subtract(up.scale(prediction.dot(up)));
        Vec3 point = WoodweaverSurfaceNavigation.footPlane(entity).add(offset).add(tangentPrediction);

        // ArachnoMod obstacle / slope detection: raise scan start if moving toward a rise or step
        double heightBias = 0.0;
        if (tangentPrediction.lengthSqr() > 0.01) {
            Vec3 forwardProbe = point.add(tangentPrediction.normalize().scale(0.6));
            BlockPos probePos = BlockPos.containing(forwardProbe.add(up.scale(0.3)));
            if (!level.getBlockState(probePos).isAir()) {
                heightBias = 0.85; // Raised scan start to cleanly detect higher step/slope terraces
            }
        }

        double startHeight = Math.min(7.0, Math.max(1.5, relativeOffset.y) + 1.0 + heightBias);
        BlockHitResult hit = WoodweaverSurfaceNavigation.clipLoaded(level, entity,
            point.add(up.scale(startHeight)), point.subtract(up.scale(12.0)));

        // Multi-candidate scan on slopes and uneven ground (ArachnoMod candidate grid)
        // Accepts both vertical (top) and horizontal (sides of blocks/cliffs) surfaces
        if (hit == null) {
            double[] gridD = {-0.4, 0.4};
            for (double gx : gridD) {
                for (double gz : gridD) {
                    Vec3 candPoint = point.add(WoodweaverSurfaceNavigation.WeaverFrame.offset(rotation, yaw, gx, 0, gz));
                    BlockHitResult candHit = WoodweaverSurfaceNavigation.clipLoaded(level, entity,
                        candPoint.add(up.scale(startHeight)), candPoint.subtract(up.scale(12.0)));
                    if (candHit != null) {
                        hit = candHit;
                        point = candPoint;
                        break;
                    }
                }
                if (hit != null) break;
            }
        }

        if (hit == null) {
            Vec3 hipOrigin = entity.position().add(0, WoodweaverSurfaceNavigation.ROOT_HEIGHT, 0).add(offset.scale(0.35));
            Vec3 angledDown = point.subtract(up.scale(9.5)).subtract(offset.scale(0.25));
            hit = WoodweaverSurfaceNavigation.clipLoaded(level, entity, hipOrigin, angledDown);
        }

        // Side-wall & cliff grip probe: raycast inward from the overhang toward the cliff wall
        if (hit == null) {
            Vec3 inward = entity.position().subtract(point);
            inward = inward.subtract(up.scale(inward.dot(up)));
            if (inward.lengthSqr() > 0.01) {
                Vec3 inNorm = inward.normalize();
                for (int step = 1; step <= 4; step++) {
                    Vec3 probe = point.add(inNorm.scale(step * 0.6));
                    hit = WoodweaverSurfaceNavigation.clipLoaded(level, entity,
                        probe.add(up.scale(startHeight)), probe.subtract(up.scale(10.0)));
                    if (hit != null) break;
                }
                if (hit == null) {
                    // Direct horizontal wall scan for cliff faces
                    Vec3 wallScanStart = point.add(inNorm.scale(1.8)).subtract(up.scale(0.5));
                    hit = WoodweaverSurfaceNavigation.clipLoaded(level, entity, wallScanStart, point.subtract(up.scale(0.5)));
                }
            }
        }

        if (hit == null) return null;
        return new Contact(hit.getLocation(), WoodweaverSurfaceNavigation.normal(hit.getDirection()), Math.abs(relativeOffset.x) < 2.0);
    }

    public static Vec3 findGroundHeight(Level level, Entity entity, Vec3 relativeOffset, float yaw, Vec3 worldPredict) {
        Contact contact = findContact(level, entity, relativeOffset, yaw, worldPredict);
        return contact == null ? null : contact.position;
    }

    private static Contact reachableGround(Level level, Entity entity, Vec3 offset, float yaw, Vec3 prediction, int leg) {
        return reachableGround(level, entity, offset, yaw, prediction, leg, Vec3.ZERO);
    }

    private static Contact reachableGround(Level level, Entity entity, Vec3 offset, float yaw, Vec3 prediction, int leg, Vec3 bodyPrediction) {
        Contact contact = findContact(level, entity, offset, yaw, prediction);
        if (contact == null) return null;
        LegState state = getOrCreateState(entity.getUUID()).legs[leg];
        if (state.avoidContactPos != null && entity.tickCount < state.avoidContactUntil
            && (!state.grounded || contact.position.distanceToSqr(state.currentPlantedPos) >= 0.25 * 0.25)
            && contact.position.distanceToSqr(state.avoidContactPos) < 0.65 * 0.65) return null;
        double hx = leg % 2 == 0 ? -0.5 : 0.5;
        double hz = leg < 2 ? -0.625 : leg < 4 ? 0.0625 : 0.75;
        Vec3 hip = entity.position().add(0, WoodweaverSurfaceNavigation.ROOT_HEIGHT, 0).add(
            WoodweaverSurfaceNavigation.WeaverFrame.offset(WoodweaverSurfaceNavigation.rotation(entity, 1), yaw, hx, 0.5, hz));
        if (!WoodweaverSurfaceNavigation.attached(entity)) {
            EntityGaitState gaitState = getOrCreateState(entity.getUUID());
            double pitchOffset = -Math.sin(gaitState.pitch) * hz;
            double heightOffset = gaitState.bodyHeight - (entity.getY() + STANCE_HEIGHT);
            hip = hip.add(0, offset.y - 3.15 + pitchOffset + heightOffset, 0);
        }
        if (!entity.onGround() && entity.getDeltaMovement().y < 0)
            hip = hip.add(0, -Math.min(2.0, Math.max(0, entity.getY() - contact.position.y)), 0);
        hip = hip.add(bodyPrediction);
        double reach = contact.position.add(contact.normal.scale(0.1875)).distanceToSqr(hip);
        Vec3 difference = contact.position.subtract(hip);
        org.joml.Vector3f local = WoodweaverSurfaceNavigation.rotation(entity, 1).conjugate()
            .transform(new org.joml.Vector3f((float)difference.x, (float)difference.y, (float)difference.z));
        Vec3 model = rotateHorizontal(new Vec3(local.x, local.y, local.z), yaw);
        double side = leg % 2 == 0 ? -1 : 1;
        double angle = Math.atan2(-side * model.z, side * model.x);
        double min = leg < 2 ? (side < 0 ? -1.30 : 0.35) : leg < 4 ? -0.25 : (side < 0 ? 0.35 : -1.30);
        double max = leg < 2 ? (side < 0 ? -0.35 : 1.30) : leg < 4 ? 0.25 : (side < 0 ? 1.30 : -0.35);
        if (contact.narrow) {
            min = leg < 2 ? (side < 0 ? -1.30 : 0.30) : leg < 4 ? -0.20 : (side < 0 ? 0.30 : -1.30);
            max = leg < 2 ? (side < 0 ? -0.30 : 1.30) : leg < 4 ? 0.20 : (side < 0 ? 1.30 : -0.30);
        }
        double normalDistance = difference.dot(WoodweaverSurfaceNavigation.up(entity, 1));
        if (model.x * side < (normalDistance < -0.8 ? -0.55 : -0.20) || angle < min - 0.20 || angle > max + 0.20) return null;
        // Two 48-unit links: match the analytical solver extension limit.
        double maxReach = 6.0 * 0.975;
        return reach >= 2.7 * 2.7 && reach <= maxReach * maxReach ? contact : null;
    }

    public static void updateGait(Entity entity, float partialTicks, double[] hipY) {
        updateGait(entity, partialTicks, hipY, false);
    }

    public static void updateGait(Entity entity, float partialTicks, double[] hipY, boolean preserveRoot) {
        updateGait(entity, partialTicks, hipY, preserveRoot, false);
    }

    public static void updateGait(Entity entity, float partialTicks, double[] hipY, boolean preserveRoot, boolean landingPose) {
        Level level = entity.level();
        if (!level.isClientSide()) return;
        if (level != gaitLevel) {
            GAIT_STATES.clear();
            gaitLevel = level;
        }
        EntityGaitState state = getOrCreateState(entity.getUUID());
        state.preserveAuthoredRoot = preserveRoot;
        long tick = entity.tickCount;
        if (state.lastTick == tick) return;
        boolean reset = !state.initialized || tick < state.lastTick
            || entity.position().distanceToSqr(state.lastBodyPos) > 64;
        state.lastTick = tick;
        boolean supported = hasTerrainSupport(entity);
        boolean touchdown = entity.onGround() && tick - state.lastGroundedTick > 1 && !state.wasTerrainSupported;
        if (entity.onGround()) state.lastGroundedTick = tick;
        state.lastBodyPos = entity.position();
        float yaw = entity instanceof LivingEntity living ? living.yBodyRot : entity.getYRot();
        Vec3 surfaceUp = WoodweaverSurfaceNavigation.up(entity, 1);
        Vec3 motion = new Vec3(entity.getX() - entity.xo, entity.getY() - entity.yo, entity.getZ() - entity.zo);
        motion = motion.subtract(surfaceUp.scale(motion.dot(surfaceUp)));
        double speed = motion.length();
        boolean isDashing = false;
        if (entity instanceof LivingEntity living) {
            try {
                isDashing = living.getPersistentData().getBooleanOr("is_dashing", false);
            } catch (Throwable ignored) {}
        }
        boolean fastGait = isDashing || speed > 0.45;
        if ((isDashing || speed > 0.40) && speed > 0.05) {
            yaw = (float) Math.toDegrees(Math.atan2(motion.z, motion.x)) - 90.0F;
        }
        float yawDelta = entity instanceof LivingEntity living ? Mth.wrapDegrees(living.yBodyRot - living.yBodyRotO) : 0;
        double yawDeltaRad = Math.toRadians(yawDelta);
        double maxPredict = fastGait ? Math.min(8.0, 3.2 + speed * 4.8) : 3.2;
        double landingTicks = fastGait ? Mth.clamp(2.2 - speed * 0.8, 1.25, 2.2) : 0;
        Vec3 bodyPrediction = motion.scale(landingTicks);
        Vec3 linearPredict = motion.scale(fastGait ? landingTicks + 0.75 : Math.min(maxPredict, 2.0 + speed * 3.8));
        double time = tick;
        Quaternionf surfaceRotation = WoodweaverSurfaceNavigation.rotation(entity, 1);
        boolean airborneApproach = !supported && entity.getDeltaMovement().y < 0 && state.nearLanding;
        boolean anyStepping = false;
        double[] groupError = new double[2];
        for (int i = 0; i < 6; i++) {
            LegState leg = state.legs[i];
            double x = (i % 2 == 0 ? -1 : 1) * (i >= 2 && i < 4 ? FOOT_SPREAD * 1.15 : FOOT_SPREAD);
            double z = i < 2 ? -2.4 : i < 4 ? 0.0 : 2.5;
            Vec3 turnOffset = WoodweaverSurfaceNavigation.WeaverFrame.offset(surfaceRotation, yaw, -z * yawDeltaRad * 2.5, 0, x * yawDeltaRad * 2.5);
            Vec3 predict = linearPredict.add(turnOffset);
            if (predict.length() > maxPredict) predict = predict.scale(maxPredict / predict.length());
            Vec3 offset = new Vec3(x, hipY[i], z);
            Contact contact = reachableGround(level, entity, offset, yaw, predict, i, bodyPrediction);
            // Keep useful forward lead when the full landing target is obstructed.
            if (contact == null && fastGait) for (double lead : new double[]{0.75, 0.5, 0.25}) {
                contact = reachableGround(level, entity, offset, yaw, predict.scale(lead), i, bodyPrediction);
                if (contact != null) break;
            }
            if (contact == null && predict.lengthSqr() > 0.001)
                contact = reachableGround(level, entity, offset, yaw, Vec3.ZERO, i);
            // Search around a ledge before folding the leg inward toward the body.
            if (contact == null) for (double dz : new double[]{-0.75, 0.75, -1.5, 1.5}) {
                contact = reachableGround(level, entity, new Vec3(x, hipY[i], z + dz), yaw, Vec3.ZERO, i);
                if (contact != null) break;
            }
            if (contact == null) for (double scale : new double[]{0.9, 0.8, 0.7}) {
                double probeZ = scale < 0.8 && i >= 2 && i < 4 ? (i == 2 ? -1.1 : 1.1) : z;
                contact = reachableGround(level, entity, new Vec3(x * scale, hipY[i], probeZ), yaw, Vec3.ZERO, i);
                if (contact != null) break;
            }
            // A slanted probe can grip a slope or drop face below the ledge:
            if (contact == null) {
                Vec3 hip = entity.position().add(0, WoodweaverSurfaceNavigation.ROOT_HEIGHT, 0).add(
                    WoodweaverSurfaceNavigation.WeaverFrame.offset(surfaceRotation, yaw, i % 2 == 0 ? -0.5 : 0.5, 0.5, z * 0.3));
                for (double downDist : new double[]{0.8, 1.8, 3.2, 5.0}) {
                    Vec3 edge = WoodweaverSurfaceNavigation.footPlane(entity).add(
                        WoodweaverSurfaceNavigation.WeaverFrame.offset(surfaceRotation, yaw, (i % 2 == 0 ? -0.25 : 0.25), 0, z));
                    BlockHitResult grip = WoodweaverSurfaceNavigation.clipLoaded(level, entity, hip, edge.subtract(surfaceUp.scale(downDist)));
                    if (grip != null && hip.distanceToSqr(grip.getLocation()) >= 1.8 * 1.8
                        && hip.distanceToSqr(grip.getLocation()) < 5.6 * 5.6) {
                        contact = new Contact(grip.getLocation(), WoodweaverSurfaceNavigation.normal(grip.getDirection()), true);
                        break;
                    }
                }
            }
            Vec3 hanging = WoodweaverSurfaceNavigation.footPlane(entity).add(
                WoodweaverSurfaceNavigation.WeaverFrame.offset(surfaceRotation, yaw, x, 0.2, z));
            Vec3 home = contact != null ? contact.position : hanging;
            leg.targetHasContact = contact != null;
            if (contact != null) {
                leg.targetContactNormal = contact.normal;
                leg.targetNarrow = contact.narrow;
                if (reset) { leg.contactNormal = contact.normal; leg.narrow = contact.narrow; }
            } else {
                leg.targetContactNormal = surfaceUp;
                leg.targetNarrow = false;
                if (reset) { leg.contactNormal = surfaceUp; leg.narrow = false; }
            }
            if (!leg.isStepping) leg.liftUp = surfaceUp;
            if (reset) {
                leg.init(home);
                leg.grounded = contact != null;
                leg.landingHasContact = contact != null;
                leg.plantedAtTick = tick;
                leg.contactWeight = 1.0f;
            }
            leg.targetHomePos = home;
            if (!supported) {
                // In air, feet track the body frame (or anticipate landing floor) rather than lagging at takeoff
                leg.plantedTangent = null;
                leg.grounded = false;
                leg.isStepping = false;
                leg.stepProgress = 1;
                leg.currentPlantedPos = leg.lastPlantedPos = leg.landingPos = home;
                leg.takeoffNormal = leg.sampleNormal(time);
                leg.contactNormal = leg.targetContactNormal;
                leg.narrow = leg.targetNarrow;
            }
            if ((touchdown && !leg.grounded) || (landingPose && entity.onGround() && !leg.grounded)) {
                leg.plantedTangent = null;
                Vec3 touchdownNormal = leg.sampleNormal(time);
                if (leg.currentPlantedPos != null && leg.currentPlantedPos.distanceToSqr(home) > 0.04) {
                    leg.lastPlantedPos = leg.currentPlantedPos;
                    leg.landingPos = home;
                    leg.stepStart = time;
                    leg.stepDuration = 2.0;
                    leg.lift = 0.0;
                    leg.isStepping = true;
                } else {
                    leg.currentPlantedPos = leg.lastPlantedPos = leg.landingPos = home;
                    leg.isStepping = false;
                }
                leg.stepProgress = 0;
                leg.takeoffNormal = touchdownNormal;
                leg.contactNormal = leg.targetContactNormal;
                leg.narrow = leg.targetNarrow;
                leg.grounded = contact != null;
                leg.landingHasContact = contact != null;
                leg.plantedAtTick = tick;
            }
            if (leg.isStepping) {
                leg.currentPlantedPos = leg.sample(time);
                leg.stepProgress = (float) Mth.clamp((time - leg.stepStart) / leg.stepDuration, 0, 1);
                if (leg.stepProgress >= 1.0f) {
                    leg.currentPlantedPos = leg.landingPos;
                    leg.isStepping = false;
                    leg.grounded = leg.landingHasContact && validSupport(entity, leg);
                    // A swing's recovery request must not immediately lift its new plant.
                    leg.recoveryRequested = false;
                    leg.plantedAtTick = tick;
                }
            }
            // Validate planted contacts cheaply, including blocks removed under an idle foot.
            if (!reset && !leg.isStepping && leg.grounded) {
                leg.grounded = validSupport(entity, leg);
            }
            if (supported && !leg.grounded && !leg.isStepping && contact == null)
                leg.currentPlantedPos = leg.lastPlantedPos = leg.landingPos = hanging;
            leg.contactAvailable = contact != null || leg.grounded || leg.isStepping;
        }
        if (reset) {
            state.initialized = true;
            state.enabledAtTick = tick;
            state.poseActive = true;
            state.hasPose = true;
            state.blendFrom = 1.0f;
            state.legWeight = 1.0f;
            state.legBlendTime = tick;
            state.actionWeight = 0;
            state.actionBlendTime = tick;
            state.renderedRoot = null;
            state.renderedRootAt = Double.NaN;
            state.blendStartedAt = tick;
            state.bodyHeight = state.previousBodyHeight = entity.getY() + STANCE_HEIGHT;
            state.pitch = state.previousPitch = state.roll = state.previousRoll = 0;
            state.terrainOrigin = null;
            state.terrainGradientX = state.terrainGradientZ = 0;
            state.terrainPlaneTick = Long.MIN_VALUE;
        } else if (supported || (!airborneApproach && !landingPose)) {
            double baseThreshold = STEP_DISTANCE * STEP_DISTANCE * (speed < 0.04 ? 2.0 : 0.85);
            double comfortThreshold = baseThreshold * 1.6; // ArachnoMod asynchronous comfort boundary
            int supports = 0;
            for (LegState support : state.legs) {
                if (support.grounded && !support.isStepping) supports++;
            }

            int group = state.activeGroup;
            int otherGroup = 1 - group;
            int[] checkOrder = new int[6];
            int idx = 0;
            for (int i : GAIT_GROUPS[group]) checkOrder[idx++] = i;
            for (int i : GAIT_GROUPS[otherGroup]) checkOrder[idx++] = i;

            boolean anyStarted = false;
            for (int i : checkOrder) {
                LegState leg = state.legs[i];
                if (leg.isStepping || leg.targetHomePos == null || !leg.targetHasContact) continue;

                Vec3 diff = leg.currentPlantedPos.subtract(leg.targetHomePos);
                Vec3 tangentDiff = diff.subtract(surfaceUp.scale(diff.dot(surfaceUp)));
                double error = tangentDiff.lengthSqr() + Math.pow(diff.dot(surfaceUp) * 0.35, 2);

                boolean isComfortBreached = error > comfortThreshold;
                boolean isGroupTurn = (i == GAIT_GROUPS[group][0] || i == GAIT_GROUPS[group][1] || i == GAIT_GROUPS[group][2]);
                boolean wantsToMove = !leg.grounded || isComfortBreached || (isGroupTurn && error > baseThreshold) || leg.recoveryRequested;

                if (speed < 0.04 && leg.grounded && tick - leg.plantedAtTick < 8 && !leg.recoveryRequested && !isComfortBreached) {
                    continue;
                }

                if (wantsToMove) {
                    boolean neighborStepping = false;
                    for (int adj : ADJACENT_LEGS[i]) {
                        if (state.legs[adj].isStepping) {
                            neighborStepping = true;
                            break;
                        }
                    }
                    boolean urgent = leg.recoveryRequested && speed > 0.04;
                    if (neighborStepping && leg.grounded && !isComfortBreached && !urgent) continue;
                    int minSupports = urgent || ((isDashing || speed > 0.45) && isComfortBreached) ? 2 : 3;
                    if (leg.grounded && (supports <= minSupports || (!urgent && !isComfortBreached && !stableSupport(state, i, surfaceUp)))) continue;

                    if (leg.grounded) supports--;
                    leg.lastPlantedPos = leg.currentPlantedPos;
                    leg.landingPos = leg.targetHomePos;
                    leg.landingHasContact = leg.targetHasContact;
                    leg.takeoffNormal = leg.sampleNormal(time);
                    leg.contactNormal = leg.targetContactNormal;
                    leg.narrow = leg.targetNarrow;
                    leg.stepStart = time;
                    leg.plantedTangent = null;
                    double stepDist = leg.lastPlantedPos.distanceTo(leg.landingPos);
                    double minStepDuration = isDashing || speed > 0.45 ? 1.25 : 2.4;
                    double maxStepDuration = isDashing || speed > 0.45 ? 2.2 : 4.0;
                    leg.stepDuration = Mth.clamp(2.2 + stepDist * 0.25 - speed * 1.6, minStepDuration, maxStepDuration);
                    double maxLift = isDashing || speed > 0.45 ? 0.85 : 1.25;
                    leg.lift = Math.max(STEP_HEIGHT, Math.min(maxLift,
                        Math.abs(leg.landingPos.subtract(leg.lastPlantedPos).dot(surfaceUp)) * 0.5 + 0.3));
                    BlockHitResult obstacle = WoodweaverSurfaceNavigation.clipLoaded(level, entity,
                        leg.lastPlantedPos.add(surfaceUp.scale(0.2)), leg.landingPos.add(surfaceUp.scale(0.2)));
                    if (obstacle != null) leg.lift = Math.max(leg.lift, 1.0);
                    leg.isStepping = true;
                    leg.grounded = false;
                    leg.recoveryRequested = false;
                    leg.plantedAtTick = tick;
                    anyStarted = true;
                }
            }
            if (anyStarted) state.activeGroup = 1 - group;
        }
        if (surfaceUp.y > 0) updateBody(state, entity);
        else {
            state.previousBodyHeight = state.bodyHeight = entity.getY() + STANCE_HEIGHT;
            state.previousPitch = state.pitch = state.previousRoll = state.roll = 0;
        }
    }

    private static boolean stableSupport(EntityGaitState state, int lifted, Vec3 up) {
        // Require a spread of supporting contacts, not merely a contact count.
        double bestArea = 0;
        for (int a = 0; a < 6; a++) {
            if (a == lifted || !state.legs[a].grounded || state.legs[a].isStepping) continue;
            for (int b = a + 1; b < 6; b++) {
                if (b == lifted || !state.legs[b].grounded || state.legs[b].isStepping) continue;
                for (int c = b + 1; c < 6; c++) {
                    if (c == lifted || !state.legs[c].grounded || state.legs[c].isStepping) continue;
                    Vec3 ab = state.legs[b].currentPlantedPos.subtract(state.legs[a].currentPlantedPos);
                    Vec3 ac = state.legs[c].currentPlantedPos.subtract(state.legs[a].currentPlantedPos);
                    bestArea = Math.max(bestArea, Math.abs(ab.cross(ac).dot(up)));
                }
            }
        }
        return bestArea > 0.5;
    }

    /** Least-squares height plane in entity-relative world coordinates. */
    private static final class TerrainPlane {
        int count;
        double x, z, y, xx, xz, zz, xy, zy;
        double gradientX, gradientZ, height;

        void add(Vec3 point, Vec3 origin) {
            double px = point.x - origin.x, pz = point.z - origin.z;
            double py = point.y - origin.y;
            count++;
            x += px; z += pz; y += py;
            xx += px * px; xz += px * pz; zz += pz * pz;
            xy += px * py; zy += pz * py;
        }

        boolean solve() {
            if (count < 3) return false;
            double cxx = xx - x * x / count, cxz = xz - x * z / count;
            double czz = zz - z * z / count;
            double cxy = xy - x * y / count, czy = zy - z * y / count;
            double determinant = cxx * czz - cxz * cxz;
            if (determinant <= 1.0e-6 * Math.max(1, cxx * czz)) return false;
            gradientX = (cxy * czz - czy * cxz) / determinant;
            gradientZ = (czy * cxx - cxy * cxz) / determinant;
            height = (y - gradientX * x - gradientZ * z) / count;
            return Double.isFinite(gradientX) && Double.isFinite(gradientZ) && Double.isFinite(height);
        }
    }

    private static void updateBody(EntityGaitState state, Entity entity) {
        state.previousBodyHeight = state.bodyHeight;
        state.previousPitch = state.pitch;
        state.previousRoll = state.roll;

        float yaw = entity instanceof LivingEntity living ? living.yBodyRot : entity.getYRot();
        double speed = Math.hypot(entity.getX() - entity.xo, entity.getZ() - entity.zo);
        // Measure the footprint, independently of step phase and foot reach failures.
        // Prediction shifts the sample window, never the frame in which tilt is applied.
        Vec3 forwardDir = rotateHorizontal(new Vec3(0, 0, -1), yaw);
        Vec3 rightDir = rotateHorizontal(new Vec3(1, 0, 0), yaw);
        Vec3 prediction = new Vec3(entity.getX() - entity.xo, 0, entity.getZ() - entity.zo);
        if (prediction.lengthSqr() > 1.25 * 1.25) prediction = prediction.normalize().scale(1.25);
        double span = 2.4 + Math.min(1.6, speed);
        double scanUp = Math.min(12, 3.5 + span * Math.hypot(state.terrainGradientX, state.terrainGradientZ));
        TerrainPlane plane = new TerrainPlane();
        for (int i = 0; i < 5; i++) {
            Vec3 offset = i == 0 ? Vec3.ZERO : i < 3
                ? forwardDir.scale(i == 1 ? span : -span)
                : rightDir.scale(i == 3 ? span : -span);
            Vec3 probe = entity.position().add(prediction).add(offset);
            BlockHitResult hit = WoodweaverSurfaceNavigation.clipLoaded(entity.level(), entity,
                probe.add(0, scanUp, 0), probe.add(0, -18, 0));
            if (hit != null && WoodweaverSurfaceNavigation.normal(hit.getDirection()).y > 0.5)
                plane.add(hit.getLocation(), entity.position());
        }
        boolean measured = plane.solve();
        if (!measured) {
            // Only actual contacts may rescue a missing terrain sample; hanging homes are not terrain.
            plane = new TerrainPlane();
            for (LegState leg : state.legs) {
                if (leg.grounded && !leg.isStepping && leg.contactNormal.y > 0.5)
                    plane.add(leg.currentPlantedPos, entity.position());
                else if (leg.targetHasContact && leg.targetContactNormal.y > 0.5)
                    plane.add(leg.targetHomePos, entity.position());
            }
            measured = plane.solve();
        }
        if (measured) {
            state.terrainGradientX = plane.gradientX;
            state.terrainGradientZ = plane.gradientZ;
            state.terrainOrigin = new Vec3(entity.getX(), entity.getY() + plane.height, entity.getZ());
            state.terrainPlaneTick = entity.tickCount;
        }
        boolean havePlane = state.terrainOrigin != null && (measured || state.terrainSupported
            || (entity.tickCount - state.terrainPlaneTick <= 4 && state.nearLanding));
        double pitch = 0, roll = 0, targetHeight = entity.getY() + STANCE_HEIGHT;
        if (havePlane) {
            double forwardSlope = state.terrainGradientX * forwardDir.x + state.terrainGradientZ * forwardDir.z;
            double rightSlope = state.terrainGradientX * rightDir.x + state.terrainGradientZ * rightDir.z;
            pitch = -Math.atan(forwardSlope);
            roll = -Math.atan2(rightSlope, Math.sqrt(1 + forwardSlope * forwardSlope));
            double supportY = state.terrainOrigin.y + state.terrainGradientX * (entity.getX() - state.terrainOrigin.x)
                + state.terrainGradientZ * (entity.getZ() - state.terrainOrigin.z);
            double crouch = Mth.clamp(Math.max(0, -forwardSlope) * span * 0.35, 0, 1.2);
            targetHeight = supportY + STANCE_HEIGHT - crouch;
        }
        double floorWeight = Mth.clamp(WoodweaverSurfaceNavigation.up(entity, 1).y, 0, 1);
        floorWeight = floorWeight * floorWeight * (3 - 2 * floorWeight);
        targetHeight = Mth.lerp(floorWeight, entity.getY() + STANCE_HEIGHT, targetHeight);
        state.bodyHeight = Mth.lerp(0.35, state.bodyHeight,
            Mth.clamp(targetHeight, entity.getY() + 0.8, entity.getY() + 4.5));
        // Fast travel needs a quicker response; tick interpolation supplies render smoothness.
        float response = (float)(1 - Math.exp(-(0.65 + Math.min(2.5, speed) * 1.6)));
        state.pitch = Mth.lerp(response, state.pitch, (float)Mth.clamp(pitch * floorWeight, -1.25, 1.25));
        state.roll = Mth.lerp(response, state.roll, (float)Mth.clamp(roll * floorWeight, -1.15, 1.15));
    }
