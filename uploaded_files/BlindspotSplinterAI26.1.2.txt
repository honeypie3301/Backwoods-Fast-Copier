	// Read-only contract for the afterimage: respect whichever controller currently owns this mob.
	public static boolean isAfterimageFlankActive(BlindspotSplinterEntity splinter, Player player) {
		if (!(splinter.level() instanceof ServerLevel level) || player == null
				|| splinter.getTarget() != player || !splinter.isAlive() || !player.isAlive()
				|| player.level() != level || player.isCreative() || player.isSpectator()
				|| splinter.getEntityData().get(BlindspotSplinterEntity.DATA_frozenByRose) != 0
				|| splinter.getEntityData().get(BlindspotSplinterEntity.DATA_mineProgress) != 0
				|| player.getMainHandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()
				|| player.getOffhandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) return false;
		net.minecraft.nbt.CompoundTag data = splinter.getPersistentData();
		long now = level.getServer().overworld().getGameTime();
		if (data.getLongOr("ahPlacePose", 0L) > now) return false;
		String owner = player.getUUID().toString();
		int tactic = data.getIntOr("htState", 0);
		if (tactic != 0) {
			long lastTick = data.getLongOr("htLastTick", 0L);
			return (tactic == 2 || tactic == 5 || tactic == 6 || tactic == 9 || tactic == 12) && data.getLongOr("htUntil", 0L) > now
					&& now >= lastTick && now - lastTick <= 2
					&& data.getLongOr("htPost", 0L) == level.getGameTime()
					&& owner.equals(data.getStringOr("htOwner", ""))
					&& level.dimension().identifier().toString().equals(data.getStringOr("htDimension", ""));
		}
		int bowState = data.getIntOr("hive_bow_state", 0);
		if (bowState != 0)
			return bowState == 3 && owner.equals(data.getStringOr("hive_bow_target", ""));
		return data.getIntOr(K_PERSONALITY, 0) == PERSONALITY_STALK_HUNT
				&& data.getIntOr(K_HM_PHASE, 0) == PHASE_STALK
				&& data.getIntOr(K_STALK_WATCH_TICKS, 0) <= 0 && data.getIntOr(K_FLANK_TICKS, 0) > 0
				&& data.getIntOr(K_FORCED_HUNT_TICKS, 0) <= 0
				&& splinter.getEntityData().get(BlindspotSplinterEntity.DATA_isEnraged) == 0;
	}

	private static void installFlankGoal(BlindspotSplinterEntity splinter) {
		if (!(splinter.level() instanceof ServerLevel) || FLANK_GOALS.contains(splinter)) return;
		for (var wrapped : new java.util.ArrayList<>(splinter.goalSelector.getAvailableGoals())) {
			if (wrapped.getGoal() instanceof net.minecraft.world.entity.ai.goal.MeleeAttackGoal original) {
				int priority = wrapped.getPriority();
				splinter.goalSelector.removeGoal(original);
				splinter.goalSelector.addGoal(priority, new FlankAwareMeleeGoal(splinter, original));
			}
		}
		FLANK_GOALS.add(splinter);
	}

	private static double customNavigationSpeed(BlindspotSplinterEntity splinter) {

	// Preserve vanilla melee rules, cooldowns and priority; only yield its MOVE/LOOK ownership.
	private static final class FlankAwareMeleeGoal extends net.minecraft.world.entity.ai.goal.Goal {
		private final BlindspotSplinterEntity splinter;
		private final net.minecraft.world.entity.ai.goal.MeleeAttackGoal original;

		FlankAwareMeleeGoal(BlindspotSplinterEntity splinter, net.minecraft.world.entity.ai.goal.MeleeAttackGoal original) {
			this.splinter = splinter;
			this.original = original;
			setFlags(original.getFlags().clone());
		}

		@Override public boolean canUse() { return !SplinterOnEntityTickUpdateProcedure.AdaptiveHive.mirrorAttackBlocked(splinter)
                && !customNavigationOwned(splinter) && original.canUse(); }
		@Override public boolean canContinueToUse() { return !SplinterOnEntityTickUpdateProcedure.AdaptiveHive.mirrorAttackBlocked(splinter)
                && !customNavigationOwned(splinter) && original.canContinueToUse(); }
		@Override public boolean isInterruptable() { return original.isInterruptable(); }
		@Override public boolean requiresUpdateEveryTick() { return original.requiresUpdateEveryTick(); }
		@Override public void start() { original.start(); }
		@Override public void tick() { if (!customNavigationOwned(splinter)) original.tick(); }
		@Override public void stop() {
			boolean preserve = customNavigationOwned(splinter);
			net.minecraft.world.level.pathfinder.Path path = preserve ? splinter.getNavigation().getPath() : null;
			original.stop();
			// MeleeAttackGoal.stop() clears navigation; restore the path selected by this tick's owner.
			if (path != null && !path.isDone()) splinter.getNavigation().moveTo(path, customNavigationSpeed(splinter));
		}
	}

	private static void lookAlongFlank(BlindspotSplinterEntity splinter, double x, double y, double z) {
		net.minecraft.world.level.pathfinder.Path path = splinter.getNavigation().getPath();
		Vec3 point = path != null && !path.isDone() ? path.getNextEntityPos(splinter) : new Vec3(x, y, z);
		if (path != null && !path.isDone() && point.subtract(splinter.position()).multiply(1, 0, 1).lengthSqr() <= 0.04
				&& path.getNextNodeIndex() + 1 < path.getNodeCount())
			point = path.getEntityPosAtNode(splinter, path.getNextNodeIndex() + 1);
		if (point.subtract(splinter.position()).multiply(1, 0, 1).lengthSqr() > 0.04)
			splinter.getLookControl().setLookAt(point.x, splinter.getEyeY(), point.z, 20.0F, 15.0F);
	}

			}
		}

		int effectiveRageThreshold = Math.max(1, RAGE_WATCH_THRESHOLD - splinter.getPersistentData().getIntOr(K_RAGE_BONUS, 0));

		boolean isWatched = false;
		if (foundPlayerDirect != null) {
			Vec3 toSplinter = splinter.getEyePosition().subtract(foundPlayerDirect.getEyePosition());
			if (toSplinter.lengthSqr() > 1.0e-8) toSplinter = toSplinter.normalize();
			double dot = foundPlayerDirect.getLookAngle().normalize().dot(toSplinter);
			isWatched = (dot > WATCH_DOT_THRESHOLD) && foundPlayerDirect.hasLineOfSight(splinter);
		}

		double distToPlayer = foundPlayerDirect != null ? splinter.position().distanceTo(foundPlayerDirect.position()) : Double.MAX_VALUE;
		if (isEnraged == 1 && distToPlayer > RAGE_ESCAPE_RANGE) {
			if (splinter.getPersistentData().getIntOr(K_FORCED_HUNT_TICKS, 0) <= 0) {
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_isEnraged, 0);
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_watchTimer, 0);
				isEnraged = 0;
			}
		}

		if (isWatched && isEnraged == 0) {
			watchTimer++;
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_watchTimer, watchTimer);
			if (watchTimer >= effectiveRageThreshold) {
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_isEnraged, 1);
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_watchTimer, 0);
				isEnraged = 1;
			}
		} else if (!isWatched && isEnraged == 0) {
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_watchTimer, 0);
		}

		boolean foundRose = false;
		if (foundPlayerDirect != null) {
			foundRose = checkHeldRose(foundPlayerDirect, splinter, world, foundPlayerDirect.getX(), foundPlayerDirect.getY(), foundPlayerDirect.getZ());
		}
		if (foundRose || frozenByRose == 1) {
			setSpeed(splinter, 0);
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_isEnraged, 0);
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_watchTimer, 0);
			splinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
			splinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
			splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
			splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
			splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			stopNav(splinter);
			return;
		}

		// Angel personality behavior: freeze while watched, otherwise direct hunt-like pursuit.
		SplinterHivemind.observe(splinter, foundPlayerDirect, splinter.getLastHurtByMob());
        if (personality == PERSONALITY_STALK_HUNT && SplinterHivemind.beginGroupHunt(splinter, foundPlayerDirect)) {
            splinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
            splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 160);
            splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
            splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
        }

		// One bounded cover move owns navigation while the archer is drawing.
		if (foundPlayerDirect != null && splinter.getTarget() == foundPlayerDirect
				&& SplinterHivemind.handleBowThreat(splinter, foundPlayerDirect,
					splinter.getEntityData().get(BlindspotSplinterEntity.DATA_mineProgress),
					isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED)) return;
		if (personality == PERSONALITY_STALK_HUNT
				&& SplinterHivemind.bowCommitActive(splinter, foundPlayerDirect)
				&& splinter.getPersistentData().getIntOr(K_HM_PHASE, 0) != PHASE_HUNT) {
			splinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
			splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 90);
		}

		if (personality == PERSONALITY_ANGEL) {
			if (isWatched && isEnraged == 0 && !SplinterHivemind.bowCommitActive(splinter, foundPlayerDirect)) {
				setSpeed(splinter, 0);
				stopNav(splinter);
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
				return;
			}

			splinter.getLookControl().setLookAt(foundPlayer, 25.0F, 20.0F);

			setSpeed(splinter, isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED);
			if (splinter instanceof Mob angelMob) {
				navMoveToEntityThrottled(splinter, angelMob, foundPlayer, 1.05, 3);
			}
		} else {
			int phase = splinter.getPersistentData().getIntOr(K_HM_PHASE, 0);
			int phaseTimer = splinter.getPersistentData().getIntOr(K_HM_PHASE_TIMER, 0);
			int silentCd = splinter.getPersistentData().getIntOr(K_HM_SILENT_CD, 0);

			if (phase == PHASE_SILENT && isEnraged == 0 && forcedHuntTicks <= 0) {
				phase = PHASE_STALK;
				phaseTimer = randomBetween(splinter, STALK_MIN_TICKS, STALK_MAX_TICKS);
				splinter.getPersistentData().putInt(K_HM_PHASE, phase);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
				splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			}

			if (silentCd > 0) {
				silentCd--;
				splinter.getPersistentData().putInt(K_HM_SILENT_CD, silentCd);
			}

			if (isEnraged == 1 || splinter.getPersistentData().getIntOr(K_FORCED_HUNT_TICKS, 0) > 0) {
				phase = PHASE_HUNT;
				phaseTimer = HUNT_TICKS + 40;
				splinter.getPersistentData().putInt(K_HM_PHASE, phase);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
				splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			} else {
				phaseTimer--;
				if (phaseTimer <= 0) {
					if (phase == PHASE_HUNT) {
						phase = PHASE_SILENT;
						phaseTimer = randomBetween(splinter, SILENT_MIN_TICKS, SILENT_MAX_TICKS);
						splinter.getPersistentData().putInt(K_HM_SILENT_CD, POST_HUNT_SILENT_CD);
						splinter.getEntityData().set(BlindspotSplinterEntity.DATA_watchTimer, 0);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					} else if (phase == PHASE_SILENT) {
						phase = PHASE_STALK;
						phaseTimer = randomBetween(splinter, STALK_MIN_TICKS, STALK_MAX_TICKS);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					} else {
						phase = PHASE_SILENT;
						phaseTimer = randomBetween(splinter, SILENT_MIN_TICKS, SILENT_MAX_TICKS);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					}
				}
				splinter.getPersistentData().putInt(K_HM_PHASE, phase);
				splinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
			}

			if (phase == PHASE_SILENT) {
				setSpeed(splinter, 0);
				stopNav(splinter);
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
				return;
			}

			if (phase != PHASE_STALK || splinter.getPersistentData().getIntOr(K_FLANK_TICKS, 0) <= 0
					|| splinter.getPersistentData().getIntOr(K_STALK_WATCH_TICKS, 0) > 0)
				splinter.getLookControl().setLookAt(foundPlayer, 25.0F, 20.0F);

			if (phase == PHASE_STALK) {
				setSpeed(splinter, STALK_SPEED);

				int watchTicks = splinter.getPersistentData().getIntOr(K_STALK_WATCH_TICKS, 0);
				if (watchTicks > 0) {
					splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, watchTicks - 1);
					stopNav(splinter);
					splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
					return;
				}

				int flankTicks = splinter.getPersistentData().getIntOr(K_FLANK_TICKS, 0);
				if (flankTicks > 0) {
					splinter.getPersistentData().putLong("blindspot_flank_post", splinter.level().getGameTime());
					splinter.getPersistentData().putInt(K_FLANK_TICKS, flankTicks - 1);
					if (splinter instanceof net.minecraft.world.entity.Mob mob2) {
						double fx = splinter.getPersistentData().getDoubleOr(K_FLANK_X, 0.0);
						double fy = splinter.getPersistentData().getDoubleOr(K_FLANK_Y, 0.0);
						double fz = splinter.getPersistentData().getDoubleOr(K_FLANK_Z, 0.0);
						navMoveToPosThrottled(splinter, mob2, fx, fy, fz, 0.9, 1);
						lookAlongFlank(splinter, fx, fy, fz);
					}
					splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
					return;
				}
				if (splinter.getRandom().nextInt(STALK_WATCH_CHANCE) == 0) {
					int flankStreak = splinter.getPersistentData().getIntOr(K_FLANK_STREAK, 0);
					boolean doFlank = !isWatched && flankStreak < 2 && splinter.getRandom().nextBoolean();
					if (doFlank) {
						splinter.getPersistentData().putInt(K_FLANK_STREAK, flankStreak + 1);
						splinter.getPersistentData().putInt(K_FLANK_TICKS, randomBetween(splinter, STALK_WATCH_MIN, STALK_WATCH_MAX));
						double dist = STALK_FOLLOW_DIST - 2.0;
						double angleOffset = (splinter.getRandom().nextDouble() * 30.0 + 60.0) * (splinter.getRandom().nextBoolean() ? 1 : -1);
						double angleRad = Math.toRadians(foundPlayer.getYRot() + angleOffset);
						double fx = foundPlayer.getX() - Math.sin(angleRad) * dist;
						double fz = foundPlayer.getZ() + Math.cos(angleRad) * dist;
						double fy = foundPlayer.getY();
						splinter.getPersistentData().putDouble(K_FLANK_X, fx);
						splinter.getPersistentData().putDouble(K_FLANK_Y, fy);
						splinter.getPersistentData().putDouble(K_FLANK_Z, fz);
						splinter.getPersistentData().putLong("blindspot_flank_post", splinter.level().getGameTime());
						navMoveToPosThrottled(splinter, splinter, fx, fy, fz, 0.9, 1);
						lookAlongFlank(splinter, fx, fy, fz);
					} else {
						splinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						splinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, randomBetween(splinter, STALK_WATCH_MIN, STALK_WATCH_MAX));
						stopNav(splinter);
					}
					splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
					return;
				}

				if (splinter instanceof Mob mob) {
					Vec3 hiveStalkPoint = SplinterHivemind.stalkPoint(mob, foundPlayer);
					if (hiveStalkPoint != null) {
						SplinterHivemind.allowPost(splinter);
						navMoveToPosThrottled(splinter, mob, hiveStalkPoint.x, hiveStalkPoint.y, hiveStalkPoint.z, 0.9, 1);
					} else if (distToPlayer > STALK_FOLLOW_DIST) {
						navMoveToEntityThrottled(splinter, mob, foundPlayer, 0.9, 1);
					} else {
						if (!mob.getNavigation().isDone() && splinter.tickCount % 10 == 0) {
							stopNav(splinter);
						}
					}
				}

				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
				return;
			}

			setSpeed(splinter, isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED);
			if (splinter instanceof Mob huntMob) {
				navMoveToEntityThrottled(splinter, huntMob, foundPlayer, 1.05, 3);
			}
		}

		Vec3 splinterEyes = splinter.getEyePosition(1f);
		Vec3 splinterView = splinter.getViewVector(1f);
		Vec3 blockCheckTarget = splinterEyes.add(splinterView.scale(MINE_RAY_DISTANCE));

		HitResult hit = world.clip(new ClipContext(splinterEyes, blockCheckTarget, ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, splinter));

		BlockPos facePos;
		BlockPos feetPos;

		if (hit.getType() == HitResult.Type.BLOCK) {
			facePos = ((BlockHitResult) hit).getBlockPos();
			feetPos = new BlockPos(facePos.getX(), Mth.floor(splinter.getY()), facePos.getZ());
		} else {
			Vec3 look = splinter.getLookAngle().normalize();
			int fx = Mth.floor(splinter.getX() + look.x);
			int fz = Mth.floor(splinter.getZ() + look.z);
			feetPos = new BlockPos(fx, Mth.floor(splinter.getY()), fz);
			facePos = new BlockPos(fx, Mth.floor(splinter.getEyeY()), fz);
		}

		if (SplinterOnEntityTickUpdateProcedure.AdaptiveHive.detourAroundCover(world, splinter, foundPlayer, feetPos, facePos)) {
                splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
                return;
            }

		boolean canMineFeet = canMine(world, feetPos, foundPlayer);
		boolean canMineFace = canMine(world, facePos, foundPlayer);

		BlockPos downPos = splinter.blockPosition().below();
		double heightDiff = foundPlayer.getY() - splinter.getY();
		double horizontalDist = splinter.position().distanceTo(foundPlayer.position());
		boolean canMineDown = (heightDiff <= -2.0 && horizontalDist <= 3.0) && canMine(world, downPos, foundPlayer);
		
		int prevProgress = splinter.getEntityData().get(BlindspotSplinterEntity.DATA_mineProgress);
		BlockPos prevTrackPos = new BlockPos(
			splinter.getEntityData().get(BlindspotSplinterEntity.DATA_mineX),
			splinter.getEntityData().get(BlindspotSplinterEntity.DATA_mineY),
			splinter.getEntityData().get(BlindspotSplinterEntity.DATA_mineZ)
		);
		BlockPos trackPos = canMineDown ? downPos : (canMineFeet ? feetPos : facePos);
		if (prevProgress > 0 && trackPos != null && !trackPos.equals(prevTrackPos)) {
			boolean lockPrev = canMine(world, prevTrackPos, foundPlayer) && prevTrackPos.closerToCenterThan(splinter.position(), 4.0) && !isPositionClaimed(world, prevTrackPos, splinter);
			if (lockPrev) {
				trackPos = prevTrackPos;
				canMineDown = trackPos.equals(downPos);
				canMineFeet = trackPos.equals(feetPos);
				canMineFace = trackPos.equals(facePos);
				if (!canMineDown && !canMineFeet && !canMineFace) {
					if (trackPos.getY() < splinter.getY()) { canMineDown = true; downPos = trackPos; }
					else { canMineFeet = true; feetPos = trackPos; }
				}
			} else {
				prevProgress = 0;
			}
		}
		if (canMineDown && isPositionClaimed(world, downPos, splinter)) canMineDown = false;
		if (canMineFeet && isPositionClaimed(world, feetPos, splinter)) canMineFeet = false;
		if (canMineFace && isPositionClaimed(world, facePos, splinter)) canMineFace = false;

		if (canMineFeet || canMineFace || canMineDown) {
			int mineProgress = prevProgress + 1;
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, mineProgress);

			if (splinter.tickCount % 6 == 0) splinter.swing(InteractionHand.MAIN_HAND);

			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineX, trackPos.getX());
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineY, trackPos.getY());
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineZ, trackPos.getZ());

			float speedRef;
			if (canMineDown) {
				speedRef = world.getBlockState(downPos).getDestroySpeed(world, downPos);
			} else if (canMineFeet) {
				speedRef = world.getBlockState(feetPos).getDestroySpeed(world, feetPos);
			} else {
				speedRef = world.getBlockState(facePos).getDestroySpeed(world, facePos);
			}

			float mineThreshold = isDegraded
					? speedRef * DEGRADED_MINE_SPEED_MULTIPLIER + DEGRADED_MINE_SPEED_BASE
					: speedRef * MINE_SPEED_MULTIPLIER + MINE_SPEED_BASE;

			SplinterOnEntityTickUpdateProcedure.AdaptiveHive.showMiningCracks(splinter, mineProgress, mineThreshold,
					canMineDown ? downPos : null, canMineFeet ? feetPos : null, canMineFace ? facePos : null);
			if (mineProgress > mineThreshold) {
				if (canMineDown) world.destroyBlock(downPos, false);
				if (canMineFeet) world.destroyBlock(feetPos, false);
				if (canMineFace) world.destroyBlock(facePos, false);
				stopNav(splinter);
				splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);
			}
		} else {
			splinter.getEntityData().set(BlindspotSplinterEntity.DATA_mineProgress, 0);

			if (!isCritical && !splinter.isSwimming() && !splinter.isInWater()) {
				if (heightDiff >= 1.4 && horizontalDist < 12) {
					BlockPos under = BlockPos.containing(splinter.getX(), splinter.getY() - 1, splinter.getZ());
					BlockPos body = BlockPos.containing(splinter.getX(), splinter.getY(), splinter.getZ());
					BlockPos bodyUp = body.above();

					if (world.getBlockState(under).isAir()) {
						if (world.setBlock(under, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState(), 3)) {
							SplinterOnEntityTickUpdateProcedure.AdaptiveHive.swingBuild(splinter);
							trackPlacedBlock(splinter, under);
						}
					}
					if (world.getBlockState(body).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) world.setBlock(body, Blocks.AIR.defaultBlockState(), 3);
					if (world.getBlockState(bodyUp).is(TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get())) world.setBlock(bodyUp, Blocks.AIR.defaultBlockState(), 3);

					world.destroyBlock(BlockPos.containing(splinter.getX(), splinter.getY() + 2, splinter.getZ()), false);
					world.destroyBlock(BlockPos.containing(splinter.getX(), splinter.getY() + 3, splinter.getZ()), false);

					if (splinter.onGround()) {
						splinter.setDeltaMovement(new Vec3(splinter.getDeltaMovement().x(), 0.4, splinter.getDeltaMovement().z()));
					}
					splinter.fallDistance = 0;
				} else if (heightDiff > -0.5 && heightDiff < 2.5) {
					Vec3 look = splinter.getLookAngle();
					BlockPos bridgePos = BlockPos.containing(splinter.getX() + look.x, splinter.getY() - 1, splinter.getZ() + look.z);
					if (!(world.getBlockFloorHeight(bridgePos) > 0)) {
						if (world.setBlock(bridgePos, TheBackwoodsModBlocks.SPLINTERED_OAK_PLANKS.get().defaultBlockState(), 3)) {
							SplinterOnEntityTickUpdateProcedure.AdaptiveHive.swingBuild(splinter);
							trackPlacedBlock(splinter, bridgePos);
						}
					}
				}
			}
		}

		// Melee reach range increase for non-players only (by 50%)
		LivingEntity target = splinter.getTarget();
		if (target != null && target.isAlive() && !(target instanceof Player)) {
			double reach = splinter.getBbWidth() + target.getBbWidth() + 0.8;
			double increasedReach = reach * 1.5;
			double distSqr = splinter.distanceToSqr(target);
			if (distSqr <= increasedReach * increasedReach) {
				int lastAttackTick = splinter.getPersistentData().getIntOr("last_melee_attack_tick", 0);
				if (splinter.tickCount - lastAttackTick >= 20) {
					splinter.doHurtTarget((net.minecraft.server.level.ServerLevel) splinter.level(), target);
					splinter.swing(InteractionHand.MAIN_HAND);
					splinter.getPersistentData().putInt("last_melee_attack_tick", splinter.tickCount);
				}
			}
		}
	}

	private static boolean isThirdPersonFrontForLocalPlayer(Player player) {
		try {
			Object minecraft = Class.forName("net.minecraft.client.Minecraft").getMethod("getInstance").invoke(null);
			Object localPlayer = minecraft.getClass().getField("player").get(minecraft);
			if (localPlayer == null || localPlayer != player) return false;
			Object options = minecraft.getClass().getField("options").get(minecraft);
			Object cameraType = options.getClass().getMethod("getCameraType").invoke(options);
			boolean isMirrored = (boolean) cameraType.getClass().getMethod("isMirrored").invoke(cameraType);
			boolean isFirstPerson = (boolean) cameraType.getClass().getMethod("isFirstPerson").invoke(cameraType);
			return isMirrored && !isFirstPerson;
		} catch (Throwable ignored) {
			return false;
		}
	}

	private static int randomBetween(BlindspotSplinterEntity entity, int min, int max) {
		if (max <= min) return min;
		return min + entity.getRandom().nextInt(max - min + 1);
	}

	private static void setSpeed(LivingEntity entity, double speed) {
		if (entity.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)) {
			entity.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(speed);
		}

	}

private static boolean isPositionClaimed(LevelAccessor world, BlockPos pos, BlindspotSplinterEntity thisEntity) {
		net.minecraft.world.phys.AABB box = new net.minecraft.world.phys.AABB(pos).inflate(3);
		java.util.List<BlindspotSplinterEntity> others = world.getEntitiesOfClass(BlindspotSplinterEntity.class, box, e -> e != thisEntity && e.isAlive());
		int ourProgress = thisEntity.getEntityData().get(BlindspotSplinterEntity.DATA_mineProgress);
		for (BlindspotSplinterEntity other : others) {
			int otherProgress = other.getEntityData().get(BlindspotSplinterEntity.DATA_mineProgress);
			if (otherProgress > 0 &&
				other.getEntityData().get(BlindspotSplinterEntity.DATA_mineX) == pos.getX() &&
				other.getEntityData().get(BlindspotSplinterEntity.DATA_mineY) == pos.getY() &&
				other.getEntityData().get(BlindspotSplinterEntity.DATA_mineZ) == pos.getZ()) {
				
				if (ourProgress == 0) return true;
				if (otherProgress > ourProgress) return true;
				if (otherProgress == ourProgress && other.getId() < thisEntity.getId()) return true;
			}
		}
		return false;
	}

	private static boolean canMine(LevelAccessor world, BlockPos pos, LivingEntity player) {
		BlockState state = world.getBlockState(pos);
		if (state.isAir()) return false;
		float speed = state.getDestroySpeed(world, pos);
		if (speed < 0 || speed >= MAX_BREAKABLE_HARDNESS) return false;
		return pos.getY() != (int) (player.getY() - 2);
	}

	private static boolean checkHeldRose(Entity holder, BlindspotSplinterEntity splinter, LevelAccessor world, double x, double y, double z) {
		if (!(holder instanceof LivingEntity living)) return false;

		ItemStack main = living.getMainHandItem();
		ItemStack off = living.getOffhandItem();

		boolean mainIsRose = main.getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem();
		boolean offIsRose = off.getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem();

		if (!mainIsRose && !offIsRose) return false;

		setSpeed(splinter, 0);
		stopNav(splinter);

		if (mainIsRose) tickRoseItem(main, world, x, y, z);
		if (offIsRose) tickRoseItem(off, world, x, y, z);

		return true;
	}

	private static void tickRoseItem(ItemStack rose, LevelAccessor world, double x, double y, double z) {
		double wiltTimer = rose.getOrDefault(DataComponents.CUSTOM_DATA, CustomData.EMPTY).copyTag().getDoubleOr("wilt_timer", 0.0) + 1.0;
		CustomData.update(DataComponents.CUSTOM_DATA, rose, tag -> tag.putDouble("wilt_timer", wiltTimer));

		if (wiltTimer > ROSE_WILT_TICKS) {
			if (world instanceof Level level) {
				if (!level.isClientSide()) {
					level.playSound(null, BlockPos.containing(x, y, z), SoundEvents.FIRE_EXTINGUISH, SoundSource.NEUTRAL, 1f, 1f);
				} else {
					level.playLocalSound(x, y, z, SoundEvents.FIRE_EXTINGUISH, SoundSource.NEUTRAL, 1f, 1f, false);
				}
			}
			rose.shrink(1);
			CustomData.update(DataComponents.CUSTOM_DATA, rose, tag -> tag.putDouble("wilt_timer", 0));
		}
	}

	private static boolean checkNearbyRoseBlocks(LevelAccessor world, BlindspotSplinterEntity splinter) {
		BlockPos base = splinter.blockPosition();
		BlockPos.MutableBlockPos check = new BlockPos.MutableBlockPos();
		for (int sx = -ROSE_SCAN_XZ; sx < ROSE_SCAN_XZ; sx++) {
			for (int sz = -ROSE_SCAN_XZ; sz < ROSE_SCAN_XZ; sz++) {
				check.set(base.getX() + sx, base.getY(), base.getZ() + sz);
				if (world instanceof Level level && !level.hasChunkAt(check)) continue;
				for (int sy = -ROSE_SCAN_Y; sy < ROSE_SCAN_Y; sy++) {
					check.set(base.getX() + sx, base.getY() + sy, base.getZ() + sz);
					BlockState roseState = world.getBlockState(check);
					if (roseState.is(TheBackwoodsModBlocks.ASH_ROSE.get())
							|| roseState.is(TheBackwoodsModBlocks.POTTED_ASH_ROSE.get())) return true;
				}
			}
		}
		return false;
	}

	private static Player findNearestPlayerInRange(LevelAccessor world, double x, double y, double z, double range) {
		AABB box = AABB.ofSize(new Vec3(x, y, z), range, range, range);
		List<Player> players = world.getEntitiesOfClass(Player.class, box, e -> !e.isCreative() && !e.isSpectator());

		Player nearest = null;
		double best = range * range;
		for (Player p : players) {
			double d = p.distanceToSqr(x, y, z);
			if (d < best) {
				best = d;
				nearest = p;
			}
		}
		return nearest;
	}

	private static void navMoveToEntityThrottled(BlindspotSplinterEntity splinter, Mob mob, Entity target, double speed, int mode) {
		Vec3 hiveApproach = SplinterHivemind.approach(mob, target);
		if (target instanceof Player && hiveApproach.distanceToSqr(target.position()) > 1.0) SplinterHivemind.allowPost(mob);
		navMoveToPosThrottled(splinter, mob, hiveApproach.x, hiveApproach.y, hiveApproach.z, speed, mode);
	}

	private static void navMoveToPosThrottled(Entity entity, Mob mob, double x, double y, double z, double speed, int throttle) {
		double dx = entity.getX() - x;
		double dy = entity.getY() - y;
		double dz = entity.getZ() - z;
		boolean closingForMelee = mob.getTarget() instanceof LivingEntity target && target.isAlive()
				&& target.position().distanceToSqr(new Vec3(x, y, z)) < 1.0
				&& !mob.isWithinMeleeAttackRange(target);
		if (!closingForMelee && dx * dx + dy * dy + dz * dz < 4.0) {
			mob.getNavigation().stop();
			return;
		}

		int currentTick = entity.tickCount;
		int lastTick = entity.getPersistentData().getIntOr(K_NAV_LAST_TICK, 0);
		int lastMode = entity.getPersistentData().getIntOr(K_NAV_MODE, 0);

		double lastTx = entity.getPersistentData().getDoubleOr(K_NAV_TX, 0.0);
		double lastTy = entity.getPersistentData().getDoubleOr(K_NAV_TY, 0.0);
		double lastTz = entity.getPersistentData().getDoubleOr(K_NAV_TZ, 0.0);

		int interval = Math.max(PATH_RECALC_INTERVAL_TICKS, throttle);
		double goalShift = (lastTx - x) * (lastTx - x) + (lastTy - y) * (lastTy - y) + (lastTz - z) * (lastTz - z);
		if (currentTick >= lastTick && currentTick - lastTick < interval && lastMode == 2
				&& goalShift < TARGET_MOVE_REPATH_DIST_SQR) return;
		entity.getPersistentData().putInt(K_NAV_LAST_TICK, currentTick);
		entity.getPersistentData().putInt(K_NAV_MODE, 2);
		entity.getPersistentData().putDouble(K_NAV_TX, x);
		entity.getPersistentData().putDouble(K_NAV_TY, y);
		entity.getPersistentData().putDouble(K_NAV_TZ, z);

		if (mob.getPersistentData().contains("hive_post_time")
				&& mob.getPersistentData().getLongOr("hive_post_time", 0L) == mob.level().getGameTime()
				&& !localFlankOwnsNavigation((BlindspotSplinterEntity) mob)) speed = 1.05;
		mob.getNavigation().moveTo(x, y, z, speed);
	}

	private static void trackPlacedBlock(Entity splinter, BlockPos pos) {
		net.minecraft.nbt.CompoundTag tag = splinter.getPersistentData();
		int size = tag.getIntOr("pb_size", 0);

		for (int i = 0; i < size; i++) {
			int x = tag.getIntOr("pb_x_" + i, 0);
			int y = tag.getIntOr("pb_y_" + i, 0);
			int z = tag.getIntOr("pb_z_" + i, 0);
			if (x == pos.getX() && y == pos.getY() && z == pos.getZ()) {
				return;
			}
		}

		if (size < 64) {
			tag.putInt("pb_x_" + size, pos.getX());
			tag.putInt("pb_y_" + size, pos.getY());
			tag.putInt("pb_z_" + size, pos.getZ());
			tag.putInt("pb_size", size + 1);
		}
	}

	private static void updatePlacedBlockCleanup(LevelAccessor world, Entity splinter, BlockState targetBlockState) {
		net.minecraft.nbt.CompoundTag tag = splinter.getPersistentData();
		int size = tag.getIntOr("pb_size", 0);
		if (size == 0) return;

		boolean standingOnAny = false;
		for (int i = 0; i < size; i++) {
			int x = tag.getIntOr("pb_x_" + i, 0);
			int y = tag.getIntOr("pb_y_" + i, 0);
			int z = tag.getIntOr("pb_z_" + i, 0);

			double dx = Math.abs(splinter.getX() - (x + 0.5));
			double dz = Math.abs(splinter.getZ() - (z + 0.5));
			double dy = splinter.getY() - y;

			if (dx <= 1.0 && dz <= 1.0 && dy >= -0.2 && dy <= 2.2) {
				standingOnAny = true;
				break;
			}
		}

		if (standingOnAny) {
			tag.putInt("pb_delay", 0);
		} else {
			int delay = tag.getIntOr("pb_delay", 0) + 1;
			tag.putInt("pb_delay", delay);

			if (delay % 2 == 0) {
				int x0 = tag.getIntOr("pb_x_0", 0);
				int y0 = tag.getIntOr("pb_y_0", 0);
				int z0 = tag.getIntOr("pb_z_0", 0);
				BlockPos pos0 = new BlockPos(x0, y0, z0);

				BlockState state = world.getBlockState(pos0);
				if (state.is(targetBlockState.getBlock())) {
					world.destroyBlock(pos0, false);
				}

				for (int i = 0; i < size - 1; i++) {
					tag.putInt("pb_x_" + i, tag.getIntOr("pb_x_" + (i + 1), 0));
					tag.putInt("pb_y_" + i, tag.getIntOr("pb_y_" + (i + 1), 0));
					tag.putInt("pb_z_" + i, tag.getIntOr("pb_z_" + (i + 1), 0));
				}
				size--;
				tag.putInt("pb_size", size);
				if (size == 0) {
					tag.putInt("pb_delay", 0);
				}
			}
		}
	}
