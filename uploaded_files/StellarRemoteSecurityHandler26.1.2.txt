package net.mcreator.thebackwoods.procedures;

import net.neoforged.neoforge.event.entity.player.PlayerInteractEvent;
import net.neoforged.neoforge.event.entity.living.LivingEntityUseItemEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.bus.api.EventPriority;

import net.minecraft.world.level.Level;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Item;
import net.minecraft.world.entity.player.Player;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.network.chat.Component;
import net.minecraft.core.registries.BuiltInRegistries;
import net.minecraft.resources.Identifier;

import java.util.Set;

@EventBusSubscriber
public class StellarRemoteSecurityHandler {
	private static final Set<String> BLOCKED_DIMENSIONS = Set.of(
		"the_sub_strata",
		"the_petrified_weald",
		"the_grain",
		"the_familiar",
		"the_still",
		"rotting",
		"backwoods",
		"loss"
	);

	public static boolean isStellarRemote(ItemStack stack) {
		if (stack == null || stack.isEmpty()) return false;
		Item item = stack.getItem();
		Identifier key = BuiltInRegistries.ITEM.getKey(item);
		if (key == null) return false;
		return (key.getNamespace().equals("shooting_star_demo") && key.getPath().contains("stellar_remote"))
			|| key.getPath().equals("stellar_remote");
	}

	public static boolean isBlockedDimension(Level level) {
		if (level == null) return false;
		Identifier dimLoc = level.dimension().identifier();
		if (dimLoc.getNamespace().equals("the_backwoods")) {
			return true;
		}
		return BLOCKED_DIMENSIONS.contains(dimLoc.getPath().toLowerCase());
	}

	@SubscribeEvent(priority = EventPriority.HIGHEST)
	public static void onPlayerRightClickItem(PlayerInteractEvent.RightClickItem event) {
		Player player = event.getEntity();
		Level level = event.getLevel();
		ItemStack stack = event.getItemStack();

		if (player != null && level != null && isStellarRemote(stack)) {
			if (isBlockedDimension(level)) {
				event.setCanceled(true);
				if (!level.isClientSide()) {
					if (player instanceof ServerPlayer serverPlayer) {
						serverPlayer.sendSystemMessage(Component.literal("§cUnknown location."), true);
					} else {
						player.sendSystemMessage(Component.literal("§cUnknown location."));
					}
				}
			}
		}
	}

	@SubscribeEvent(priority = EventPriority.HIGHEST)
	public static void onLivingUseItemStart(LivingEntityUseItemEvent.Start event) {
		if (event.getEntity() instanceof Player player) {
			Level level = player.level();
			if (isStellarRemote(event.getItem()) && isBlockedDimension(level)) {
				event.setCanceled(true);
			}
		}
	}
}
