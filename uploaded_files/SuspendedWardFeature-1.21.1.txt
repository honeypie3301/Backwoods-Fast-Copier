package net.mcreator.thebackwoods.world.feature;

/** Chunk-local, world-seeded suspended ward; every solid part joins a tower or plate. */
public class SuspendedWardFeature extends net.minecraft.world.level.levelgen.feature.Feature<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> {
    public static final boolean ENABLE_AUTO_DATAPACK_GEN = true;
    public static final boolean ENABLE_FAST_TERRAIN_GEN = true; // Single-biome Ward only; avoids generating caves just to erase them.
    public static final String DIMENSION_ID = "the_backwoods:the_crossworks";
    public static final String BIOME_ID = "the_backwoods:dustbound_galleries";
    public static final int CELL_SIZE = 48;
    public static final int TIER_SPACING = 18;
    public static final int FLOOR_SPACING = 6;

    public static final double SPAWNER_FLOOR_CHANCE = 20.0; // Percent per floor; one centered block per successful roll.
    public static final double FALSE_ROOM_FLOOR_CHANCE = 10.0; // Percent per upper room; bottom safety plate stays solid.
    public static final double FALSE_BRIDGE_CHANCE = 1.0; // Percent per complete bridge at each tier; false wins at crossings.
    public static final double ROOM_PORTAL_CHANCE = 0.001; // Percent per room: 1 in 100,000 before room-fit checks.
    public static final double GRAIN_PORTAL_CHANCE = 50.0; // Percent of portals leading to the Grain; remainder to Backwoods.

    public static final int TOP_PLATE_THICKNESS = 2;
    public static final boolean ENABLE_CEILING_PLATE = false;
    public static final int GAP_BELOW_LATTICE = 1; // ULTIMATE SETTING: Adjust the gap between top of Ward and bottom of Lattice in ONE place!

    public static int getWardRoofY(int max) {
        return CrossworksLatticeFeature.getLatticeBaseY(max) - GAP_BELOW_LATTICE;
    }

        if (roof-floor<24) return false;
        int baseX=context.origin().getX()&~15, baseZ=context.origin().getZ()&~15;
        var chunk=level.getChunk(baseX>>4,baseZ>>4,net.minecraft.world.level.chunk.status.ChunkStatus.EMPTY,false);
        if(chunk==null) return false; // Never request or create an unloaded chunk.
        byte[] plan=plan(baseX,baseZ,floor,roof,effectiveMax,level.getSeed());

            .getOptional(net.minecraft.resources.ResourceLocation.parse("the_backwoods:the_grain_portal"))
            .orElseThrow(()->new IllegalStateException("Missing the_grain_portal block")).defaultBlockState();
        var backwoodsPortal=net.minecraft.core.registries.BuiltInRegistries.BLOCK
            .getOptional(net.minecraft.resources.ResourceLocation.parse("the_backwoods:backwoods_portal"))
            .orElseThrow(()->new IllegalStateException("Missing backwoods_portal block")).defaultBlockState();
        var grainX=grainPortal.setValue(net.minecraft.world.level.block.NetherPortalBlock.AXIS,net.minecraft.core.Direction.Axis.X);
        var grainZ=grainPortal.setValue(net.minecraft.world.level.block.NetherPortalBlock.AXIS,net.minecraft.core.Direction.Axis.Z);
        var backwoodsX=backwoodsPortal.setValue(net.minecraft.world.level.block.NetherPortalBlock.AXIS,net.minecraft.core.Direction.Axis.X);
        var backwoodsZ=backwoodsPortal.setValue(net.minecraft.world.level.block.NetherPortalBlock.AXIS,net.minecraft.core.Direction.Axis.Z);
        var ladder=net.minecraft.world.level.block.Blocks.LADDER.defaultBlockState()
            .setValue(net.minecraft.world.level.block.LadderBlock.FACING,net.minecraft.core.Direction.EAST);
        var fences=new net.minecraft.world.level.block.state.BlockState[16];
        for(int mask=0;mask<16;mask++) fences[mask]=net.minecraft.world.level.block.Blocks.OAK_FENCE.defaultBlockState()
            .setValue(net.minecraft.world.level.block.FenceBlock.NORTH,(mask&1)!=0)

    public static int getTopRoomFloorY(int minY,int maxYExclusive) {
        int effectiveMax=getWardRoofY(maxYExclusive);
        int bottom=minY+BOTTOM_PLATE_OFFSET,roof=ENABLE_CEILING_PLATE?effectiveMax-TOP_PLATE_THICKNESS:effectiveMax;
        if(FLOOR_SPACING<4||roof-bottom<FLOOR_SPACING+1) throw new IllegalArgumentException("No room with two blocks of headroom");
        return bottom+Math.floorDiv(roof-bottom-FLOOR_SPACING-1,FLOOR_SPACING)*FLOOR_SPACING;
    }
    public static net.minecraft.core.BlockPos findEntryRoom(long seed,int nearX,int nearZ,int minY,int maxYExclusive) {
        int cx=Math.floorDiv(nearX,CELL_SIZE),cz=Math.floorDiv(nearZ,CELL_SIZE);
        Tower closest=null;long distance=Long.MAX_VALUE;
        for(int z=cz-4;z<=cz+4;z++) for(int x=cx-4;x<=cx+4;x++) {
            Tower t=tower(seed,x,z);if(t==null) continue;
            long dx=(long)t.x-nearX,dz=(long)t.z-nearZ,d=dx*dx+dz*dz;
            if(d<distance) {closest=t;distance=d;}
        }
        return closest==null?null:new net.minecraft.core.BlockPos(closest.x,getTopRoomFloorY(minY,maxYExclusive)+1,closest.z);
    }
    private static boolean solid(byte kind) { return kind==PLANK || kind==LOG || kind==FENCE || kind==FALSE_PLANK||kind==PILLAR||kind==31||kind==32; }
    private static int index(int x,int y,int z) { return y*PAD*PAD+z*PAD+x; }
    private static long hash(long seed,int x,int z) {
        long v=seed^((long)x*0x632BE59BD9B4E019L)^((long)z*0x9E3779B97F4A7C15L);
        v=(v^(v>>>30))*0xBF58476D1CE4E5B9L;v=(v^(v>>>27))*0x94D049BB133111EBL;return v^(v>>>31);
    }
    private record Tower(int x,int z,int halfX,int halfZ) {}
    private record Bridge(Tower a,Tower b) {}
    private record LayoutKey(long seed,int x,int z,int cellSize) {}
    private record Layout(Tower[][] towers,java.util.List<Bridge> connections) {}
    private static final java.util.Map<LayoutKey,Layout> LAYOUT_CACHE=new java.util.LinkedHashMap<>(64,0.75f,true) {
        @Override protected boolean removeEldestEntry(java.util.Map.Entry<LayoutKey,Layout> entry) {return size()>64;}
    };
    private static Layout layout(long seed,int cellX,int cellZ) {
        var key=new LayoutKey(seed,cellX,cellZ,CELL_SIZE);
        synchronized(LAYOUT_CACHE) {var cached=LAYOUT_CACHE.get(key);if(cached!=null) return cached;}
        var towers=new Tower[11][11];
        for(int z=0;z<11;z++) for(int x=0;x<11;x++) towers[z][x]=tower(seed,cellX+x-5,cellZ+z-5);
        var built=new Layout(towers,bridges(towers,cellX*CELL_SIZE,cellZ*CELL_SIZE,seed));
        synchronized(LAYOUT_CACHE) {
            var cached=LAYOUT_CACHE.get(key);if(cached!=null) return cached;
            LAYOUT_CACHE.put(key,built);return built;
        }
    }
    private static Tower candidate(long seed,int cx,int cz) {
        long h=hash(seed,cx,cz);
        return new Tower(cx*CELL_SIZE+(int)Math.floorMod(h,CELL_SIZE),
            cz*CELL_SIZE+(int)Math.floorMod(h>>>8,CELL_SIZE),
            3+2*(int)Math.floorMod(h>>>16,3L),3+2*(int)Math.floorMod(h>>>24,3L));
    }
    private static Tower tower(long seed,int cx,int cz) {
        Tower a=candidate(seed,cx,cz);long priority=hash(seed^0xA24BAED4963EE407L,cx,cz);
        // Full-cell scatter, with deterministic local rejection instead of aligned rows.
        for(int z=cz-1;z<=cz+1;z++) for(int x=cx-1;x<=cx+1;x++) {
            if(x==cx&&z==cz) continue;
            Tower b=candidate(seed,x,z);int order=Long.compareUnsigned(hash(seed^0xA24BAED4963EE407L,x,z),priority);
            if(order>0 || (order==0&&(x>cx || (x==cx&&z>cz)))) continue;
            if(Math.abs(a.x-b.x)<a.halfX+b.halfX+12 && Math.abs(a.z-b.z)<a.halfZ+b.halfZ+12) return null;
        }
        return a;
    }
    private static java.util.List<Bridge> bridges(Tower[][] towers,int baseX,int baseZ,long seed) {
        var result=new java.util.LinkedHashSet<Bridge>();
        for(int z=2;z<=8;z++) for(int x=2;x<=8;x++) {
            Tower a=towers[z][x];if(a==null) continue;
            var nearby=new java.util.ArrayList<Tower>();
            for(int dz=-2;dz<=2;dz++) for(int dx=-2;dx<=2;dx++) {
                Tower b=towers[z+dz][x+dx];if(b!=null&&b!=a) nearby.add(b);
            }
            nearby.sort(java.util.Comparator.comparingLong(b->(long)(a.x-b.x)*(a.x-b.x)+(long)(a.z-b.z)*(a.z-b.z)));
            int count=Math.min(nearby.size(),2+(int)Math.floorMod(hash(seed,a.x,a.z),2L));
            for(int i=0;i<count;i++) {
                Tower b=nearby.get(i);int margin=CELL_SIZE/4+3;
                if(Math.max(a.x,b.x)+margin<baseX-2||Math.min(a.x,b.x)-margin>baseX+CELL_SIZE+17
                        ||Math.max(a.z,b.z)+margin<baseZ-2||Math.min(a.z,b.z)-margin>baseZ+CELL_SIZE+17) continue;
                result.add(a.x<b.x || (a.x==b.x&&a.z<b.z)?new Bridge(a,b):new Bridge(b,a));
            }
        }
        return new java.util.ArrayList<>(result);
    }
    private static void route(boolean[] deck,int baseX,int baseZ,Bridge bridge,long seed,int tier) {
        Tower a=bridge.a,b=bridge.b;
        long h=hash(seed^((long)tier*0xD1B54A32D192ED03L),a.x^Integer.rotateLeft(b.x,16),a.z^Integer.rotateLeft(b.z,16));
        boolean alongX=Math.abs(a.x-b.x)>=Math.abs(a.z-b.z);
        int au=alongX?a.x:a.z,av=alongX?a.z:a.x,bu=alongX?b.x:b.z,bv=alongX?b.z:b.x;
        int ah=alongX?a.halfX:a.halfZ,bh=alongX?b.halfX:b.halfZ;
        int direction=bu>=au?1:-1,lo=au+direction*(ah+4),hi=bu-direction*(bh+4),span=Math.abs(hi-lo);
        // The dominant axis always has room because accepted towers have at least 12 blocks of clearance.
        if((hi-lo)*direction<0) {alongX=!alongX;au=alongX?a.x:a.z;av=alongX?a.z:a.x;bu=alongX?b.x:b.z;bv=alongX?b.z:b.x;
            ah=alongX?a.halfX:a.halfZ;bh=alongX?b.halfX:b.halfZ;direction=bu>=au?1:-1;
            lo=au+direction*(ah+4);hi=bu-direction*(bh+4);span=Math.abs(hi-lo);}
        int first=lo+direction*(int)Math.floorMod(h,span/2+1L);
        int last=hi-direction*(int)Math.floorMod(h>>>12,span/2+1L);
        int offset=CELL_SIZE/4,side=(av+bv)/2+(int)Math.floorMod(h>>>24,offset*2L+1)-offset;
        if((h&4)==0) {last=first;side=bv;} // Mix single elbows with three/four-turn paths.
        int[] us={au,first,first,last,last,bu},vs={av,av,side,side,bv,bv};
        for(int i=1;i<us.length;i++) {
            if(alongX) segment(deck,baseX,baseZ,us[i-1],vs[i-1],us[i],vs[i]);
            else segment(deck,baseX,baseZ,vs[i-1],us[i-1],vs[i],us[i]);
        }
    }

    private record PlanKey(int x,int z,int floor,int roof,int max,long seed) {}
    private static final java.util.Map<PlanKey,byte[]> PLAN_CACHE=new java.util.LinkedHashMap<>(16,0.75f,true){
        @Override protected boolean removeEldestEntry(java.util.Map.Entry<PlanKey,byte[]> entry){return size()>16;}
    };
    private static byte[] plan(int x,int z,int floor,int roof,int max,long seed){
        var key=new PlanKey(x,z,floor,roof,max,seed);synchronized(PLAN_CACHE){var old=PLAN_CACHE.get(key);if(old!=null)return old;}
        byte[] result=buildPlan(x,z,floor,roof,max,seed);
        addSupportPillar(result,x,z,floor,roof,max,seed);
        synchronized(PLAN_CACHE){var old=PLAN_CACHE.get(key);if(old!=null)return old;PLAN_CACHE.put(key,result);}return result;
    }
    public static byte[] plateSupportKinds(int baseX,int baseZ,int min,int max,long seed){
        int roof=getWardRoofY(max),floor=min+BOTTOM_PLATE_OFFSET;
        byte[] blocks=plan(baseX,baseZ,floor,ENABLE_CEILING_PLATE?roof-TOP_PLATE_THICKNESS:roof,roof,seed);
        byte[] result=new byte[256];for(int z=0;z<16;z++)for(int x=0;x<16;x++){byte kind=blocks[index(x+2,3,z+2)];result[z*16+x]=kind==PILLAR||kind==PILLAR_LADDER?kind:0;}
        return result;
    }
    private static void addSupportPillar(byte[] blocks,int baseX,int baseZ,int floor,int roof,int max,long seed){
        if(!CrossworksLatticeFeature.ENABLE_PLATE_SUPPORTS)return;
        int height=max-floor;if(height<3)return;
        int half=CrossworksLatticeFeature.ROOF_PLATE_2_BOUNDARY/2;
        int inner=CrossworksLatticeFeature.BOUNDARY_CLAMP_SIZE/2+CrossworksLatticeFeature.ROOF_PLATE_2_HEIGHT_OFFSET;
        if(baseX < -half+16||baseX+15>=half-16||baseZ < -half+16||baseZ+15>=half-16
                ||Math.abs(baseX)<inner+16&&Math.abs(baseZ)<inner+16)return;
        long district=hash(seed^0xA110L,Math.floorDiv(baseX,32),Math.floorDiv(baseZ,32));
        int slot=Math.floorMod(Math.floorDiv(baseX,16),2)+2*Math.floorMod(Math.floorDiv(baseZ,16),2);
        if(Math.floorMod(district,4)!=slot)return; // One candidate per 2x2 chunks, then clearance rejection.
        // Full-height occupancy projection includes every tower, bridge tier, railing and portal.
        boolean[] occupied=new boolean[PAD*PAD];
        for(int z=0;z<PAD;z++)for(int x=0;x<PAD;x++)for(int y=1;y<height;y++)
            if(blocks[index(x,y,z)]!=AIR){occupied[z*PAD+x]=true;break;}
        long h=hash(seed^0xC011D5L,baseX,baseZ);
        boolean round=Math.floorMod(h,1000)<CYLINDER_PILLAR_PERMILLE;int radius=5,clearance=3;
        for(int attempt=0;attempt<16;attempt++){
            int cx=8+(int)Math.floorMod((h>>>8)+attempt,4),cz=8+(int)Math.floorMod((h>>>16)+attempt/4,4);
            boolean clear=true;
            for(int z=cz-radius-clearance;z<=cz+radius+clearance&&clear;z++)for(int x=cx-radius-clearance;x<=cx+radius+clearance;x++)
                if(occupied[z*PAD+x]){clear=false;break;}
            if(!clear)continue;
            for(int z=cz-radius;z<=cz+radius;z++)for(int x=cx-radius;x<=cx+radius;x++){
                if(round&&(x-cx)*(x-cx)+(z-cz)*(z-cz)>radius*radius)continue;
                for(int y=1;y<height;y++)blocks[index(x,y,z)]=PILLAR;
            }
            if(Math.floorMod(h>>>24,100)<25&&CrossworksLatticeFeature.canExitPillarLadder(seed,baseX+cx-2,baseZ+cz-2)){
                for(int y=1;y<height;y++)blocks[index(cx,y,cz)]=PILLAR_LADDER;
                for(int z=cz-radius;z<cz;z++)for(int y=1;y<=2;y++)blocks[index(cx,y,z)]=AIR;
                blocks[index(cx,1,cz-radius)]=PILLAR_DOOR_LOWER;blocks[index(cx,2,cz-radius)]=PILLAR_DOOR_UPPER;
                addPillarSettlements(blocks,cx,cz,height,h,round);
            }
            return; // One thick, jittered shaft per eligible opening; no neighbor-chunk writes.
        }
    }

    private static void addPillarSettlements(byte[] blocks,int cx,int cz,int height,long seed,boolean round){
        var random=net.minecraft.util.RandomSource.create(seed);
        for(int floor=8+random.nextInt(12);floor+6<height;){
            int variant=random.nextInt(32);boolean spacious=(variant&1)==0;
            int minX=spacious?-4:variant%3==0?0:-3,maxX=spacious?4:2+random.nextInt(2),halfZ=spacious?4:2+random.nextInt(2),roomHeight=3+random.nextInt(2);
            for(int dx=minX;dx<=maxX;dx++)for(int dz=-halfZ;dz<=halfZ;dz++){
                if(round&&dx*dx+dz*dz>16||dx==-1&&dz==0)continue; // Keep the shaft's backing spine.
                for(int dy=1;dy<=roomHeight;dy++)blocks[index(cx+dx,floor+dy,cz+dz)]=AIR;
            }
            for(int y=floor;y<=floor+roomHeight+1;y++)blocks[index(cx,y,cz)]=PILLAR_LADDER;
            int furnitureZ=Math.min(halfZ,round?3:halfZ);
            int bedX=round?Math.min(maxX,3):maxX;
            blocks[index(cx+bedX,floor+1,cz-1)]=SETTLE_BED_FOOT;blocks[index(cx+bedX,floor+1,cz)]=SETTLE_BED_HEAD;
            blocks[index(cx+1,floor+1,cz-furnitureZ)]=(variant&1)==0?SETTLE_BARREL:SETTLE_CHEST;
            if((variant&2)==0)blocks[index(cx+2,floor+1,cz-furnitureZ)]=SETTLE_BOOKS;
            blocks[index(cx+2,floor+1,cz+furnitureZ)]=SETTLE_FURNACE;blocks[index(cx+1,floor+1,cz+furnitureZ)]=SETTLE_CRAFT;
            blocks[index(cx+1,floor+roomHeight,cz+1)]=SETTLE_LANTERN;
            if(minX<0){blocks[index(cx-2,floor+1,cz+1)]=SETTLE_FENCE;blocks[index(cx-2,floor+2,cz+1)]=SETTLE_TABLE;}
            if(variant>=5){ // Glazed observation slit through the shell; never an open balcony.
                for(int dx=Math.min(maxX,round?3:4)+1;dx<=5;dx++)if(!round||dx*dx+1<=25)
                    for(int dy=2;dy<=3;dy++)blocks[index(cx+dx,floor+dy,cz+1)]=SETTLE_GLASS;
            }
            if(CrossworksLatticeFeature.invertedSettlement(seed,cx,cz,floor)){
                blocks[index(cx+2,floor+1,cz+furnitureZ)]=AIR;blocks[index(cx+1,floor+1,cz+furnitureZ)]=AIR;
                blocks[index(cx+2,floor+roomHeight,cz+furnitureZ)]=28;blocks[index(cx+1,floor+roomHeight,cz+furnitureZ-1)]=27;
                blocks[index(cx+2,floor+roomHeight,cz-2)]=29;if(random.nextInt(1000)<2)blocks[index(cx+1,floor+roomHeight,cz+furnitureZ)]=30;
            }
            if(variant%8==0){
                int end=cz+7;boolean balconyFits=end<=17&&cx-2>=2&&cx+2<=17;
                if(balconyFits){boolean clear=true;
                    for(int bz=cz+6;bz<=end;bz++)for(int bx=cx-2;bx<=cx+2;bx++)for(int by=floor;by<=floor+roomHeight;by++)if(blocks[index(bx,by,bz)]!=AIR)clear=false;
                    if(clear){
                        for(int bz=cz+5;bz<=end;bz++)for(int bx=cx-2;bx<=cx+2;bx++){
                            blocks[index(bx,floor,bz)]=PLANK;
                            if(bz==end||bx==cx-2||bx==cx+2)blocks[index(bx,floor+1,bz)]=(byte)(variant%32==0?32:31);
                        }
                        for(int bz=cz+halfZ;bz<=cz+5;bz++)for(int by=1;by<=2;by++)blocks[index(cx+1,floor+by,bz)]=AIR;
                    }
                }
            }
            floor+=floor<Math.min(160,height/3)?14+random.nextInt(18):48+random.nextInt(64);
        }
    }
    private static byte[] buildPlan(int baseX,int baseZ,int floor,int roof,int max,long seed) {
        int height=max-floor;byte[] blocks=new byte[PAD*PAD*height];
        boolean[] interior=new boolean[PAD*PAD];
        for(int z=0;z<PAD;z++) for(int x=0;x<PAD;x++) {
            blocks[index(x,0,z)]=PLANK;
            for(int y=roof;y<max;y++) blocks[index(x,y-floor,z)]=PLANK;
        }
        int cellX=Math.floorDiv(baseX,CELL_SIZE),cellZ=Math.floorDiv(baseZ,CELL_SIZE);
        var layout=layout(seed,cellX,cellZ);var towers=layout.towers;
        var connections=new java.util.ArrayList<Bridge>();int margin=CELL_SIZE/4+3;
        for(Bridge b:layout.connections) if(Math.max(b.a.x,b.b.x)+margin>=baseX-2&&Math.min(b.a.x,b.b.x)-margin<=baseX+17
                &&Math.max(b.a.z,b.b.z)+margin>=baseZ-2&&Math.min(b.a.z,b.b.z)-margin<=baseZ+17) connections.add(b);
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];if(t==null) continue;
            for(int z=0;z<PAD;z++) for(int x=0;x<PAD;x++) {
                int wx=baseX+x-2,wz=baseZ+z-2,dx=wx-t.x,dz=wz-t.z;
                if(Math.abs(dx)>t.halfX || Math.abs(dz)>t.halfZ) continue;
                interior[z*PAD+x]=true;
                boolean wallX=Math.abs(dx)==t.halfX,wallZ=Math.abs(dz)==t.halfZ;
                for(int y=floor+1;y<roof;y++) {
                    int layer=y-floor;byte kind=AIR;
                    if(wallX&&wallZ) kind=LOG;
                    else if(wallX||wallZ) {
                        int along=wallX?dz+t.halfZ:dx+t.halfX;
                        boolean window=layer%FLOOR_SPACING>=2&&layer%FLOOR_SPACING<=3
                            &&Math.floorMod(along,4)>=2;
                        kind=window?GLASS:PLANK;
                    } else if(layer%FLOOR_SPACING==0&&y+FLOOR_SPACING<roof) kind=PLANK;
                    // The ladder stays against a continuous corner wall, clear of bridge doorways.
                    if(dx==-t.halfX+1&&dz==-t.halfZ+1&&y<roof-1) kind=LADDER;
                    if(y==roof-1)kind=PLANK; // Complete individual tower roofs even without a dimension-wide ceiling.
                    blocks[index(x,layer,z)]=kind;
                }
            }
        }
        boolean[] deck=new boolean[PAD*PAD],falseDeck=new boolean[PAD*PAD];
        for(int y=floor+TIER_SPACING;y+4<roof;y+=TIER_SPACING) {
            int tier=(y-floor)/TIER_SPACING;java.util.Arrays.fill(deck,false);java.util.Arrays.fill(falseDeck,false);
            for(Bridge bridge:connections) {
                route(deck,baseX,baseZ,bridge,seed,tier);
                if(roll(seed^0xC6BC279692B5CC83L^((long)tier*0x9E3779B97F4A7C15L),
                        bridge.a.x^Integer.rotateLeft(bridge.b.x,16),bridge.a.z^Integer.rotateLeft(bridge.b.z,16),FALSE_BRIDGE_CHANCE))
                    route(falseDeck,baseX,baseZ,bridge,seed,tier);
            }
            for(int z=0;z<PAD;z++) for(int x=0;x<PAD;x++) if(deck[z*PAD+x]) {
                blocks[index(x,y-floor,z)]=falseDeck[z*PAD+x]&&!interior[z*PAD+x]?FALSE_PLANK:PLANK;
                // Join doors and junctions before adding edge rails; no fences across an intersection.
                for(int dy=1;dy<=3;dy++) if(blocks[index(x,y-floor+dy,z)]!=LADDER) blocks[index(x,y-floor+dy,z)]=AIR;
            }
            for(int z=1;z<PAD-1;z++) for(int x=1;x<PAD-1;x++) {
                int k=z*PAD+x;
                if(deck[k]&&!interior[k]&&(!deck[k-1]||!deck[k+1]||!deck[k-PAD]||!deck[k+PAD]
                        ||!deck[k-PAD-1]||!deck[k-PAD+1]||!deck[k+PAD-1]||!deck[k+PAD+1]))
                    blocks[index(x,y-floor+1,z)]=FENCE;
            }

        }
        // A passing bridge may cut a doorway near another tower's ladder; retain its backing wall.
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];if(t==null) continue;
            int x=t.x-t.halfX+1-baseX+2,z=t.z-t.halfZ+1-baseZ+2;
            if(x<1||x>=PAD||z<0||z>=PAD) continue;
            for(int y=1;y<height;y++) if(blocks[index(x,y,z)]==LADDER) blocks[index(x-1,y,z)]=PLANK;
        }
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];
            if(t==null||t.x+t.halfX<baseX-2||t.x-t.halfX>baseX+17||t.z+t.halfZ<baseZ-2||t.z-t.halfZ>baseZ+17) continue;
            for(int y=floor+FLOOR_SPACING;y<roof;y+=FLOOR_SPACING) {
                if(!roll(seed^0xD1342543DE82EF95L^((long)y*0x9E3779B97F4A7C15L),t.x,t.z,FALSE_ROOM_FLOOR_CHANCE)) continue;
                int loX=Math.max(0,t.x-t.halfX+1-baseX+2),hiX=Math.min(PAD-1,t.x+t.halfX-1-baseX+2);
                int loZ=Math.max(0,t.z-t.halfZ+1-baseZ+2),hiZ=Math.min(PAD-1,t.z+t.halfZ-1-baseZ+2);
                for(int z=loZ;z<=hiZ;z++) for(int x=loX;x<=hiX;x++) {
                    int k=index(x,y-floor,z);if(blocks[k]==PLANK) blocks[k]=FALSE_PLANK;
                }
            }
            for(int y=floor;y+4<roof;y+=FLOOR_SPACING) {
                long portalSeed=seed^0x94D049BB133111EBL^((long)y*0xD1B54A32D192ED03L);
                if(roll(portalSeed,t.x,t.z,ROOM_PORTAL_CHANCE)) {
                    RoomPortal portal=roomPortal(t,portalSeed);
                    if(portal!=null) placeRoomPortal(blocks,baseX,baseZ,y-floor,portal);
                }
            }
        }
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];
            if(t==null||!roll(seed^0xDB4F0B9175AE2165L,t.x,t.z,SPAWNER_BUILDING_CHANCE)) continue;
            int x=t.x-baseX+2,z=t.z-baseZ+2;
            if(x<0||x>=PAD||z<0||z>=PAD) continue;
            for(int y=floor+1;y+2<roof;y+=FLOOR_SPACING) {
                if(!roll(seed^0xBBE0563303A4615FL^((long)y*0x9E3779B97F4A7C15L),t.x,t.z,SPAWNER_FLOOR_CHANCE)) continue;
                int k=index(x,y-floor,z);
                if(blocks[k]==AIR&&(blocks[k-PAD*PAD]==PLANK||blocks[k-PAD*PAD]==FALSE_PLANK)&&blocks[k+PAD*PAD]==AIR) blocks[k]=SPAWNER;
            }
        }
        return blocks;
    }
    private record RoomPortal(int x,int z,boolean alongX,int front,boolean grain) {}
    private static RoomPortal roomPortal(Tower t,long seed) {
        long h=hash(seed^0x9FB21C651E98DF25L,t.x,t.z);int first=(int)Math.floorMod(h,4L);
        for(int n=0;n<4;n++) {
            int choice=(first+n)&3;boolean alongX=(choice&1)==0;int side=(choice&2)==0?-1:1;
            int half=alongX?t.halfX:t.halfZ,other=alongX?t.halfZ:t.halfX,slots=2*half-4;
            for(int i=0;i<slots;i++) {
                int u=-half+1+(int)Math.floorMod((h>>>8)+i,slots);
                int x=t.x+(alongX?u:side*(other-1)),z=t.z+(alongX?side*(other-1):u),front=-side;
                int endX=x+(alongX?3:front*2),endZ=z+(alongX?front*2:3);
                int minX=Math.min(x,endX),maxX=Math.max(x,endX),minZ=Math.min(z,endZ),maxZ=Math.max(z,endZ);
                if(Math.floorDiv(minX,16)!=Math.floorDiv(maxX,16)||Math.floorDiv(minZ,16)!=Math.floorDiv(maxZ,16)) continue;
                int ladderX=t.x-t.halfX+1,ladderZ=t.z-t.halfZ+1;
                if(ladderX>=minX&&ladderX<=maxX&&ladderZ>=minZ&&ladderZ<=maxZ) continue;
                return new RoomPortal(x,z,alongX,front,roll(seed^0xABC98388FB8FAC03L,t.x,t.z,GRAIN_PORTAL_CHANCE));
            }
        }
        return null;
    }
    private static void placeRoomPortal(byte[] blocks,int baseX,int baseZ,int y,RoomPortal portal) {
        byte kind=portal.grain?(portal.alongX?GRAIN_X:GRAIN_Z):(portal.alongX?BACKWOODS_X:BACKWOODS_Z);
        for(int u=0;u<4;u++) for(int depth=0;depth<3;depth++) {
            int x=portal.x+(portal.alongX?u:depth*portal.front)-baseX+2;
            int z=portal.z+(portal.alongX?depth*portal.front:u)-baseZ+2;
            if(x<0||x>=PAD||z<0||z>=PAD) continue;
            blocks[index(x,y,z)]=PLANK; // Real oak frame base and a stable landing on either destination type.
            for(int dy=1;dy<=4;dy++) {
                if(depth==0) blocks[index(x,y+dy,z)]=(u==0||u==3||dy==4)?PLANK:kind;
                else if(dy<4) blocks[index(x,y+dy,z)]=AIR;
            }
        }
    }
    private static boolean roll(long seed,int x,int z,double chance) {
        return (hash(seed,x,z)>>>11)*0x1.0p-53<chance/100.0;
    }
    private static void segment(boolean[] deck,int baseX,int baseZ,int ax,int az,int bx,int bz) {
        int loX=Math.max(0,Math.min(ax,bx)-1-baseX+2),hiX=Math.min(PAD-1,Math.max(ax,bx)+1-baseX+2);
        int loZ=Math.max(0,Math.min(az,bz)-1-baseZ+2),hiZ=Math.min(PAD-1,Math.max(az,bz)+1-baseZ+2);
        for(int z=loZ;z<=hiZ;z++) for(int x=loX;x<=hiX;x++) deck[z*PAD+x]=true;
    }

    private static boolean fastTerrain(com.google.gson.JsonObject dimension) {
        if(!dimension.get("type").isJsonPrimitive()||!dimension.get("type").getAsString().equals(DIMENSION_ID)) return false;
        var generator=dimension.getAsJsonObject("generator");
        if(!generator.get("type").getAsString().equals("minecraft:noise")||!generator.get("settings").isJsonObject()) return false;
        var source=generator.getAsJsonObject("biome_source");String type=source.get("type").getAsString();
        if(type.equals("minecraft:fixed")) {if(!source.get("biome").getAsString().equals(BIOME_ID)) return false;}
        else if(type.equals("minecraft:multi_noise")&&source.has("biomes")) {
            for(var biome:source.getAsJsonArray("biomes")) if(!biome.getAsJsonObject().get("biome").getAsString().equals(BIOME_ID)) return false;
        } else return false; // Preserve climate sampling for other biomes and shared presets.
        var settings=generator.getAsJsonObject("settings");String before=settings.toString();
        int min=settings.getAsJsonObject("noise").get("min_y").getAsInt();
        var router=settings.getAsJsonObject("noise_router");
        var density=new com.google.gson.JsonObject();density.addProperty("type","minecraft:y_clamped_gradient");
        density.addProperty("from_y",min+4);density.addProperty("to_y",min+5);
        density.addProperty("from_value",1);density.addProperty("to_value",-1);
        for(var entry:router.entrySet()) entry.setValue(new com.google.gson.JsonPrimitive(0));
        router.add("final_density",density);
        if(router.has("initial_density_without_jaggedness")) router.add("initial_density_without_jaggedness",density.deepCopy());
        if(router.has("preliminary_surface_level")) router.addProperty("preliminary_surface_level",min+4);
        settings.addProperty("aquifers_enabled",false);settings.addProperty("ore_veins_enabled",false);
        settings.addProperty("sea_level",min-1);
        return !before.equals(settings.toString());
    }
    private static boolean migrateBiome(com.google.gson.JsonElement element) {
        boolean changed=false;
        if(element.isJsonArray()) for(var child:element.getAsJsonArray()) changed|=migrateBiome(child);
        else if(element.isJsonObject()) for(var entry:element.getAsJsonObject().entrySet()) {
            if(entry.getKey().equals("biome")&&entry.getValue().isJsonPrimitive()
                    &&entry.getValue().getAsString().equals("the_backwoods:the_dead_grain")) {
                entry.setValue(new com.google.gson.JsonPrimitive(BIOME_ID));changed=true;
            } else if(entry.getKey().equals("biome_is")&&entry.getValue().isJsonArray()) {
                var array=entry.getValue().getAsJsonArray();
                for(int i=0;i<array.size();i++) if(array.get(i).isJsonPrimitive()
                        &&array.get(i).getAsString().equals("the_backwoods:the_dead_grain")) {
                    array.set(i,new com.google.gson.JsonPrimitive(BIOME_ID));changed=true;
                }
            } else changed|=migrateBiome(entry.getValue());
        }
        return changed;
    }
    private static void removeLegacyTag(java.nio.file.Path root) throws java.io.IOException {
        var path=root.resolve("data/the_backwoods/tags/worldgen/biome/has_suspended_ward.json");
        if(!java.nio.file.Files.isRegularFile(path)) return;
        var tag=com.google.gson.JsonParser.parseString(java.nio.file.Files.readString(path)).getAsJsonObject();
        if(!tag.has("replace")||tag.get("replace").getAsBoolean()||!tag.has("values")) return;
        for(var entry:tag.getAsJsonArray("values")) {
            String id=entry.isJsonObject()?entry.getAsJsonObject().get("id").getAsString():entry.getAsString();
            if(!id.equals(BIOME_ID)&&!id.equals("the_backwoods:the_dead_grain")) return;
        }
        java.nio.file.Files.delete(path); // Retire only the previous generator's known tag, after replacing its modifier.
    }
