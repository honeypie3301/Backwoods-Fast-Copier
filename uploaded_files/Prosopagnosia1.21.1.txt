	private static Holder<MobEffect> cachedEffectHolder = null;
	private static boolean effectHolderSearched = false;

	public static boolean isBackwoodsEntity(LivingEntity entity) {
		if (entity == null) {
			return false;
		}
		try {
			ResourceLocation id = BuiltInRegistries.ENTITY_TYPE.getKey(entity.getType());
			if (id != null && "the_backwoods".equals(id.getNamespace())) {
				return true;
			}
		} catch (Throwable ignored) {
		}
		return entity.getClass().getName().startsWith("net.mcreator.thebackwoods");
	}

	public static void purgeProsopagnosia(LivingEntity entity) {
		if (entity == null) {
			return;
		}

		try {
			if (entity.hasEffect(TheBackwoodsModMobEffects.PROSOPAGNOSIA_EFFECT)) {
				entity.removeEffect(TheBackwoodsModMobEffects.PROSOPAGNOSIA_EFFECT);
			}
		} catch (Throwable ignored) {
		}

		if (cachedEffectHolder != null && entity.hasEffect(cachedEffectHolder)) {
			try {
				entity.removeEffect(cachedEffectHolder);
			} catch (Throwable ignored) {
			}
		}

		if (!entity.getActiveEffects().isEmpty()) {
			for (MobEffectInstance effect : new ArrayList<>(entity.getActiveEffects())) {
				Holder<MobEffect> holder = effect.getEffect();
				ResourceLocation id = holder.unwrapKey().map(ResourceKey::location).orElse(null);
				if (id != null && id.getNamespace().equals("the_backwoods") && id.getPath().startsWith("prosopagnosia")) {
					try {
						entity.removeEffect(holder);
					} catch (Throwable ignored) {
					}
				}
			}
		}

		try {
			if (entity.getPersistentData().contains("prosopagnosia_active")) {
				entity.getPersistentData().remove("prosopagnosia_active");
			}
			if (entity.getPersistentData().contains("prosopagnosia_hallucination")) {
				entity.getPersistentData().remove("prosopagnosia_hallucination");
			}
		} catch (Throwable ignored) {
		}
	}

	public static boolean hasProsopagnosia(LivingEntity entity) {
		if (entity == null) {
			return false;
		}

		// Mod ID entities are completely immune to Prosopagnosia: immediately purge if applied
		if (isBackwoodsEntity(entity)) {
			purgeProsopagnosia(entity);
			return false;
		}

		if (!entity.getActiveEffects().isEmpty()) {
			try {
				if (entity.hasEffect(TheBackwoodsModMobEffects.PROSOPAGNOSIA_EFFECT)) {
					return true;
				}
			} catch (Throwable ignored) {
			}

			if (!effectHolderSearched) {
				effectHolderSearched = true;
				for (Holder<MobEffect> holder : BuiltInRegistries.MOB_EFFECT.asHolderIdMap()) {
					ResourceLocation id = holder.unwrapKey().map(ResourceKey::location).orElse(null);
					if (id != null && id.getNamespace().equals("the_backwoods") && id.getPath().startsWith("prosopagnosia")) {
						cachedEffectHolder = holder;
						break;
					}
				}
			}

			if (cachedEffectHolder != null && entity.hasEffect(cachedEffectHolder)) {
				return true;
			}

			if (cachedEffectHolder == null) {
				for (MobEffectInstance effect : entity.getActiveEffects()) {
					Holder<MobEffect> holder = effect.getEffect();
					ResourceLocation id = holder.unwrapKey().map(ResourceKey::location).orElse(null);
					if (id != null && id.getNamespace().equals("the_backwoods") && id.getPath().startsWith("prosopagnosia")) {
						cachedEffectHolder = holder;
						return true;
					}
				}
			}
		}

		try {
			if (entity.getPersistentData().getBoolean("prosopagnosia_active")) {
				return true;
			}
		} catch (Throwable ignored) {
		}

		return false;
	}

	@SubscribeEvent
	public static void onEntityTick(EntityTickEvent.Pre event) {
		if (event.getEntity() == null) {
			return;
		}

		if (!(event.getEntity() instanceof LivingEntity entity) || entity instanceof Player) {
			return;
		}

		if (entity.level().isClientSide() || !entity.isAlive()) {
			return;
		}

		if (isBackwoodsEntity(entity)) {
			purgeProsopagnosia(entity);
			return;
		}

		if (!hasProsopagnosia(entity)) {
			return;
		}

		// 1. Breakdown of Pack & Protective Instincts
		// Clear aggression, targets, and pack instincts: entity cannot distinguish allies from threats
		if (entity instanceof Mob mob) {
			if (mob.getTarget() != null) {
				mob.setTarget(null);
			}
			if (mob.isAggressive()) {
				mob.setAggressive(false);
			}
			if (mob.isSprinting()) {
				mob.setSprinting(false); // Prevents sprinting/walking dust particles under feet
			}
		}

		if (entity instanceof NeutralMob neutral) {
			if (neutral.isAngry() || neutral.getPersistentAngerTarget() != null) {
				neutral.stopBeingAngry();
				neutral.setPersistentAngerTarget(null);
			}
		}

		if (entity instanceof TamableAnimal tamable) {
			if (tamable.isOrderedToSit() || tamable.isInSittingPose()) {
				tamable.setOrderedToSit(false);
				tamable.setInSittingPose(false);
			}
		}

		// 2. Scan for all living entities within paranoid sight range (16 horizontal, 8 vertical)
		AABB scanBox = entity.getBoundingBox().inflate(16.0, 8.0, 16.0);
		List<LivingEntity> nearby = entity.level().getEntitiesOfClass(
			LivingEntity.class,
			scanBox,
			other -> other != entity && other.isAlive() && !other.isSpectator()
				&& !(other instanceof Player p && (p.isCreative() || p.isSpectator()))
		);

		List<LivingEntity> seenThreats = null;
		if (!nearby.isEmpty()) {
			for (LivingEntity other : nearby) {
				// Line of sight check: flees from every entity they see (including pack mates, owners, villagers)
				if (entity.hasLineOfSight(other)) {
					if (seenThreats == null) {
						seenThreats = new ArrayList<>();
					}
					seenThreats.add(other);
				}
			}
		}

		// Read hallucination timer (1.21.1 raw int getter)
		int hallucinationTimer = 0;
		try {
			hallucinationTimer = entity.getPersistentData().getInt("prosopagnosia_hallucination");
		} catch (Throwable ignored) {
		}

		// Determine if navigation re-pathing is needed this tick
		boolean shouldUpdateNav = false;
		if (entity instanceof Mob mob) {
			// Recalculate immediately on collision, routine update every 8 ticks (staggered by entity ID),
			// or when navigation finished/idle check every 4 ticks (staggered) instead of every single tick
			shouldUpdateNav = entity.horizontalCollision
				|| ((entity.tickCount + entity.getId()) % 8 == 0)
				|| (mob.getNavigation().isDone() && ((entity.tickCount + entity.getId()) % 4 == 0));
		}

		// 3. Fleeing logic: Real Threats vs Phantom Hallucinations
		if (seenThreats != null && !seenThreats.isEmpty()) {
			// Real entities seen: clear any phantom hallucination timer because active threats take priority
			if (hallucinationTimer > 0) {
				try {
					entity.getPersistentData().putInt("prosopagnosia_hallucination", 0);
				} catch (Throwable ignored) {
				}
			}

			double entityX = entity.getX();
			double entityZ = entity.getZ();
			double repelX = 0.0;
			double repelZ = 0.0;
			LivingEntity nearestThreat = null;
			double minDistanceSq = Double.MAX_VALUE;

			for (LivingEntity threat : seenThreats) {
				double distSq = entity.distanceToSqr(threat);
				if (distSq < minDistanceSq) {
					minDistanceSq = distSq;
					nearestThreat = threat;
				}

				double dx = entityX - threat.getX();
				double dz = entityZ - threat.getZ();
				double hDistSq = dx * dx + dz * dz;
				if (hDistSq > 0.0001) {
					double hDist = Math.sqrt(hDistSq);
					double dist = Math.sqrt(distSq);
					double scale = 1.0 / (hDist * Math.max(0.5, dist));
					repelX += dx * scale;
					repelZ += dz * scale;
				}
			}

			Vec3 repelVector;
			double rLenSq = repelX * repelX + repelZ * repelZ;
			if (rLenSq < 0.0001) {
				double rx = entity.getRandom().nextDouble() - 0.5;
				double rz = entity.getRandom().nextDouble() - 0.5;
				double rLen = Math.sqrt(rx * rx + rz * rz);
				if (rLen > 0.0001) {
					repelVector = new Vec3(rx / rLen, 0.0, rz / rLen);
				} else {
					repelVector = new Vec3(1.0, 0.0, 0.0);
				}
			} else {
				double rLen = Math.sqrt(rLenSq);
				repelVector = new Vec3(repelX / rLen, 0.0, repelZ / rLen);
			}

			Vec3 fleeTarget = entity.position().add(repelVector.scale(14.0));

			if (shouldUpdateNav && entity instanceof PathfinderMob pathfinderMob && nearestThreat != null) {
				try {
					Vec3 randomAway = DefaultRandomPos.getPosAway(pathfinderMob, 16, 7, nearestThreat.position());
					if (randomAway != null) {
						fleeTarget = randomAway;
					}
				} catch (Throwable ignored) {
				}
			}

			executeFleeMovement(entity, fleeTarget, repelVector, shouldUpdateNav);

			// Erratic panicked glances while fleeing: rapidly look back at threats
			if (entity instanceof Mob mob && entity.getRandom().nextFloat() < 0.30f) {
				if (nearestThreat != null && entity.getRandom().nextBoolean()) {
					mob.getLookControl().setLookAt(nearestThreat.getX(), nearestThreat.getEyeY(), nearestThreat.getZ());
				} else {
					float jitterYaw = (entity.getRandom().nextFloat() - 0.5f) * 140.0f;
					mob.setYRot(mob.getYRot() + jitterYaw);
					mob.setYHeadRot(mob.getYRot());
				}
			}
		} else {
			// No real entities in sight: Handle Phantom Hallucinations & Paranoid Twitches
			if (hallucinationTimer > 0) {
				hallucinationTimer--;
				try {
					entity.getPersistentData().putInt("prosopagnosia_hallucination", hallucinationTimer);
				} catch (Throwable ignored) {
				}

				// Generate or maintain phantom flee vector away from an imaginary threat
				double px = Math.cos(entity.tickCount * 0.15);
				double pz = Math.sin(entity.tickCount * 0.15);
				if (entity.getRandom().nextFloat() < 0.10f) {
					px = entity.getRandom().nextDouble() - 0.5;
					pz = entity.getRandom().nextDouble() - 0.5;
				}
				double pLen = Math.sqrt(px * px + pz * pz);
				Vec3 phantomRepel = pLen > 0.0001 ? new Vec3(px / pLen, 0.0, pz / pLen) : new Vec3(1.0, 0.0, 0.0);
				Vec3 fleeTarget = entity.position().add(phantomRepel.scale(12.0));

				if (shouldUpdateNav && entity instanceof PathfinderMob pathfinderMob) {
					try {
						Vec3 randomAway = DefaultRandomPos.getPos(pathfinderMob, 14, 7);
						if (randomAway != null) {
							fleeTarget = randomAway;
						}
					} catch (Throwable ignored) {
					}
				}

				executeFleeMovement(entity, fleeTarget, phantomRepel, shouldUpdateNav);

				// Panicked erratic head jerking during hallucination
				if (entity instanceof Mob mob && entity.getRandom().nextFloat() < 0.35f) {
					float jitterYaw = (entity.getRandom().nextFloat() - 0.5f) * 180.0f;
					mob.setYRot(mob.getYRot() + jitterYaw);
					mob.setYHeadRot(mob.getYRot());
				}
			} else {
				// 2% chance per tick (~every 2.5 seconds) to hallucinate a phantom threat in the shadows
				if (entity.getRandom().nextFloat() < 0.02f) {
					int newTimer = 30 + entity.getRandom().nextInt(40);
					try {
						entity.getPersistentData().putInt("prosopagnosia_hallucination", newTimer);
					} catch (Throwable ignored) {
					}
					// Sudden startle jump
					if (entity.onGround()) {
						entity.setDeltaMovement(entity.getDeltaMovement().add(0.0, 0.25, 0.0));
					}
				} else {
					// Paranoid idle scanning: frequently look around, checking behind their back
					if (entity instanceof Mob mob && entity.getRandom().nextFloat() < 0.20f) {
						float twitchYaw = (entity.getRandom().nextFloat() - 0.5f) * 160.0f;
						mob.setYRot(mob.getYRot() + twitchYaw);
						mob.setYHeadRot(mob.getYRot());
					}
				}
			}
		}
	}

	private static void executeFleeMovement(LivingEntity entity, Vec3 fleeTarget, Vec3 escapeDir, boolean shouldUpdateNav) {
		if (entity instanceof Mob mob) {
			if (shouldUpdateNav) {
				mob.getNavigation().moveTo(fleeTarget.x, fleeTarget.y, fleeTarget.z, 1.45);
			}
			if (mob.isSprinting()) {
				mob.setSprinting(false); // Ensure no dust/walking particles spawn
			}
		} else {
			// Non-mob / aquatic / flying propulsion push
			Vec3 push = escapeDir.scale(0.035);
			entity.setDeltaMovement(entity.getDeltaMovement().add(push));
		}

		// Jump panic if obstructed by a wall or obstacle
		if (entity.horizontalCollision && entity.onGround()) {
			entity.setDeltaMovement(entity.getDeltaMovement().add(0.0, 0.38, 0.0));
		}
	}

		private static final ResourceLocation OAK_BIPED_SLIM_TEXTURE = ResourceLocation.parse("the_backwoods:textures/entities/oak_biped_slim.png");

		private static boolean isSlimPlayer(Player player) {
		if (player instanceof AbstractClientPlayer clientPlayer) {
			try {
				return clientPlayer.getSkin().model() == net.minecraft.client.resources.PlayerSkin.Model.SLIM
						|| "slim".equalsIgnoreCase(clientPlayer.getSkin().model().id());
			} catch (Throwable ignored) {
			}
		}
		return false;
	}

	@SubscribeEvent
	public static void onClientTick(ClientTickEvent.Post event) {
		Minecraft mc = Minecraft.getInstance();
		if (mc.player != null && hasProsopagnosia(mc.player)) {
			if (mc.getMusicManager() != null) {
				mc.getMusicManager().stopPlaying();
			}
		}
	}

	@SubscribeEvent
	public static void onPlaySound(PlaySoundEvent event) {
		Minecraft mc = Minecraft.getInstance();
		if (mc.player == null || !hasProsopagnosia(mc.player)) {
			return;
		}

		if (event.getSound() != null) {
			ResourceLocation loc = event.getSound().getLocation();
			if (loc != null && "the_backwoods".equals(loc.getNamespace())) {
				return;
			}

			String name = event.getName() != null ? event.getName().toLowerCase(java.util.Locale.ROOT) : "";
			if (name.startsWith("the_backwoods:") || name.contains("the_backwoods")) {
				return;
			}

			SoundSource source = event.getSound().getSource();
			if (source == SoundSource.MUSIC || source == SoundSource.AMBIENT || source == SoundSource.RECORDS) {
				event.setSound(null);
				return;
			}

			String path = loc != null ? loc.getPath().toLowerCase(java.util.Locale.ROOT) : "";
			if (name.contains("music") || name.contains("ambient") || path.contains("music") || path.contains("ambient")) {
				event.setSound(null);
			}
		}
	}

	// ---- Fog distance / sky override -------------------------------------------------
	private static final float FOG_START = 56.0f;
	private static final float FOG_END = 96.0f;

	@SubscribeEvent
	public static void onRenderFog(ViewportEvent.RenderFog event) {
		Minecraft mc = Minecraft.getInstance();
		if (mc.player == null || !hasProsopagnosia(mc.player)) {
			return;
		}
		// leave water / lava / powder snow fog alone
		if (event.getType() != FogType.NONE) {
			return;
		}
		event.setNearPlaneDistance(FOG_START);
		event.setFarPlaneDistance(FOG_END);
		event.setFogShape(FogShape.SPHERE);
		event.setCanceled(true); // required, otherwise vanilla values are used
	}

	/** Covers sky disc, sun, moon, stars (any dimension) with solid black right after the sky is drawn. */
	@SubscribeEvent
	public static void onRenderLevelStage(RenderLevelStageEvent event) {
		if (event.getStage() != RenderLevelStageEvent.Stage.AFTER_SKY) {
			return;
		}
		Minecraft mc = Minecraft.getInstance();
		if (mc.player == null || !hasProsopagnosia(mc.player)) {
			return;
		}
		Matrix4f mv = new Matrix4f(event.getModelViewMatrix());
		Matrix4fStack stack = RenderSystem.getModelViewStack();
		stack.pushMatrix();
		stack.identity();
		RenderSystem.applyModelViewMatrix();
		RenderSystem.disableDepthTest();
		RenderSystem.depthMask(false);
		RenderSystem.disableBlend();
		RenderSystem.disableCull();
		RenderSystem.setShader(GameRenderer::getPositionColorShader);
		BufferBuilder b = Tesselator.getInstance().begin(VertexFormat.Mode.QUADS, DefaultVertexFormat.POSITION_COLOR);
		float s = 50.0f;
		float[][] faces = {
			{-s, -s, -s, s, -s, -s, s, s, -s, -s, s, -s},
			{-s, -s, s, s, -s, s, s, s, s, -s, s, s},
			{-s, -s, -s, -s, -s, s, -s, s, s, -s, s, -s},
			{s, -s, -s, s, -s, s, s, s, s, s, s, -s},
			{-s, -s, -s, s, -s, -s, s, -s, s, -s, -s, s},
			{-s, s, -s, s, s, -s, s, s, s, -s, s, s}
		};
		for (float[] f : faces) {
			for (int i = 0; i < 12; i += 3) {
				b.addVertex(mv, f[i], f[i + 1], f[i + 2]).setColor(0, 0, 0, 255);
			}
		}
		BufferUploader.drawWithShader(b.buildOrThrow());
		RenderSystem.enableCull();
		RenderSystem.depthMask(true);
		RenderSystem.enableDepthTest();
		stack.popMatrix();
		RenderSystem.applyModelViewMatrix();
	}

	@SubscribeEvent
	public static void onComputeFogColor(ViewportEvent.ComputeFogColor event) {
		Minecraft mc = Minecraft.getInstance();
		if (mc.player != null && hasProsopagnosia(mc.player)) {
			event.setRed(0.0f);
			event.setGreen(0.0f);
			event.setBlue(0.0f);
		}
	}

	@SubscribeEvent
	public static void onRenderNameTag(RenderNameTagEvent event) {
		Minecraft mc = Minecraft.getInstance();
		if (mc.player != null && hasProsopagnosia(mc.player)) {
			event.setCanRender(TriState.FALSE);
			event.setContent(Component.empty());
		}
	}

	private static MultiBufferSource wrapBuffers(MultiBufferSource original, ResourceLocation texture, boolean proceduralUv) {
		RenderType splinterType = RenderType.entityCutoutNoCull(texture);
		return renderType -> {
			if (renderType != null) {
				String str = renderType.toString().toLowerCase(java.util.Locale.ROOT);
				if (str.contains("atlas") || str.contains("item") || str.contains("block") || str.contains("glint") || str.contains("gui") || str.contains("crumbling") || str.contains("translucent")) {
					return original.getBuffer(renderType);
				}
				if (str.contains("entity") || str.contains("armor") || str.contains("player")) {
					com.mojang.blaze3d.vertex.VertexConsumer base = original.getBuffer(splinterType);
					if (proceduralUv) {
						int[] count = new int[]{0};
						return (com.mojang.blaze3d.vertex.VertexConsumer) java.lang.reflect.Proxy.newProxyInstance(
							com.mojang.blaze3d.vertex.VertexConsumer.class.getClassLoader(),
							new Class<?>[]{com.mojang.blaze3d.vertex.VertexConsumer.class},
							(proxy, method, args) -> {
								String name = method.getName();
								if (name.startsWith("vertex") || name.startsWith("addVertex")) {
									count[0]++;
								} else if (name.equals("uv") || name.equals("setUv")) {
									if (args != null && args.length == 2 && args[0] instanceof Number && args[1] instanceof Number) {
										int idx = Math.max(0, count[0] - 1) & 3;
										float nu = (idx == 1 || idx == 2) ? 1.0f : 0.0f;
										float nv = (idx == 2 || idx == 3) ? 1.0f : 0.0f;
										args[0] = nu;
										args[1] = nv;
									}
								}
								Object res = method.invoke(base, args);
								return res == base ? proxy : res;
							}
						);
					}
					return base;
				}
			}
			return original.getBuffer(renderType);
		};
	}

	private static boolean isBiped(LivingEntity entity, Object renderer) {
		if (entity instanceof Player) {
			return true;
		}

		if (isBackwoodsEntity(entity)) {
			return false;
		}

		// Never replace skeleton skin or non-human shaped mobs (skeletons have bone UVs, illagers/villagers have robes & joined arms, piglins have snouts/ears)
		if (entity instanceof net.minecraft.world.entity.monster.AbstractSkeleton
				|| entity instanceof net.minecraft.world.entity.monster.piglin.AbstractPiglin
				|| entity instanceof net.minecraft.world.entity.monster.AbstractIllager
				|| entity instanceof net.minecraft.world.entity.npc.AbstractVillager
				|| entity instanceof net.minecraft.world.entity.monster.Witch
				|| entity instanceof net.minecraft.world.entity.monster.ZombieVillager) {
			return false;
		}

		// Standard human-shaped zombies (Zombie, Husk, Drowned)
		if (entity instanceof net.minecraft.world.entity.monster.Zombie) {
			return true;
		}

		// Safe model check avoiding direct PlayerModel/ZombieModel class references for maximum compatibility
		if (renderer instanceof LivingEntityRenderer<?, ?> livingRenderer) {
			Object model = livingRenderer.getModel();
			if (model != null) {
				String modelName = model.getClass().getSimpleName();
				if (modelName.contains("Player") || modelName.contains("Zombie")) {
					return true;
				}
			}
		}

		if (entity != null) {
			String name = entity.getClass().getName().toLowerCase(java.util.Locale.ROOT);
			if (name.contains("splinter")) {
				return true;
			}
		}

		return false;
	}

	@SubscribeEvent
	public static void onRenderPlayerPre(RenderPlayerEvent.Pre event) {
		if (isRenderingSplinter) {
			return;
		}
		Minecraft mc = Minecraft.getInstance();
		if (mc.player == null || !hasProsopagnosia(mc.player)) {
			return;
		}

		Player target = event.getEntity();
		if (target == null || target == mc.player) {
			return;
		}

		event.setCanceled(true);

		try {
			isRenderingSplinter = true;
			PlayerRenderer renderer = event.getRenderer();
			ResourceLocation texture = isSlimPlayer(target) ? OAK_BIPED_SLIM_TEXTURE : OAK_BIPED_TEXTURE;
			MultiBufferSource bufferSource = wrapBuffers(event.getMultiBufferSource(), texture, false);
			PoseStack poseStack = event.getPoseStack();

			renderer.render(
				(AbstractClientPlayer) target,
				target.getYRot(),
				event.getPartialTick(),
				poseStack,
				bufferSource,
				event.getPackedLight()
			);
		} finally {
			isRenderingSplinter = false;
		}
	}

	@SubscribeEvent
	public static void onRenderLivingPre(RenderLivingEvent.Pre<?, ?> event) {
		if (isRenderingSplinter) {
			return;
		}
		Minecraft mc = Minecraft.getInstance();
		if (mc.player == null || !hasProsopagnosia(mc.player)) {
			return;
		}

		LivingEntity entity = event.getEntity();
		if (entity == null || entity == mc.player || entity instanceof Player || isBackwoodsEntity(entity)) {
			return;
		}

		boolean isBipedMob = isBiped(entity, event.getRenderer());
		if (!PROCEDURAL_TEXTURING && !isBipedMob) {
			return;
		}

		event.setCanceled(true);

		try {
			isRenderingSplinter = true;
			@SuppressWarnings("rawtypes")
			LivingEntityRenderer rawRenderer = (LivingEntityRenderer) event.getRenderer();
			ResourceLocation texture = (!PROCEDURAL_TEXTURING || isBipedMob) ? OAK_BIPED_TEXTURE : OAK_PLANKS_TEXTURE;
			boolean useProceduralUv = PROCEDURAL_TEXTURING && !isBipedMob;
			MultiBufferSource bufferSource = wrapBuffers(event.getMultiBufferSource(), texture, useProceduralUv);
			PoseStack poseStack = event.getPoseStack();

			//noinspection unchecked
			rawRenderer.render(
				entity,
				entity.getYRot(),
				event.getPartialTick(),
				poseStack,
				bufferSource,
				event.getPackedLight()
			);
		} finally {
			isRenderingSplinter = false;
		}
	}
