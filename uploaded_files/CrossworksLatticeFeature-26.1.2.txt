package net.mcreator.thebackwoods.world.feature;

/**
 * 2D grid structure for the Crossworks dimension, situated above the Suspended Ward.
 * Structured into 9 larger cells (3x3 super-grid) separated by configurable chasms/gaps,
 * interconnected with bridges featuring optional fence guards (80% chance).
 * Within each sector, fine-grained lattice cells support 4 probabilistic chamber variations,
 * generating strictly bottom-to-up with a configurable ceiling plate.
 */
public class CrossworksLatticeFeature extends net.minecraft.world.level.levelgen.feature.Feature<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> {
    public static boolean ENABLE_AUTO_DATAPACK_GEN = true;
    public static String DIMENSION_ID = "the_backwoods:the_crossworks";
    public static String BIOME_ID = "the_backwoods:dustbound_galleries";
    public static final String ID = "the_backwoods:crossworks_lattice";

    // 9 Larger Cells (3x3 Super-Grid) & Gap Configuration
    public static int SECTOR_DIVISIONS = 3; // 3x3 = 9 larger cells
    public static int GAP_SIZE = 24; // Width of the chasm/gap separating the 9 larger cells
    public static int BRIDGE_WIDTH = 5; // Walkway width of connecting bridges across gaps
    public static int BRIDGE_THICKNESS = 1; // Vertical thickness of bridge decks
    public static int BRIDGE_INTERVAL = 32; // Spacing between parallel bridge spans
    public static int BRIDGE_EXTENSION = 5; // Blocks bridge extends into adjacent sectors to eliminate gaps
    public static double BRIDGE_FENCE_CHANCE = 80.0; // 80% chance of having fence guards on both sides

    // Fine Lattice Cell Dimensions & Vertical Sizing
    public static int CELL_SIZE = 5; // Spacing between beam centers in blocks
    public static int BEAM_WIDTH = 1; // Width of oak plank beams in blocks
    public static int VERTICAL_THICKNESS = 30; // 30 blocks tall, generating strictly bottom-to-up
    public static int ELEVATION_OFFSET_FROM_TOP = 20; // Distance from roof Y to lattice top

    // Probabilistic Cell Variations
    public static double SOLID_CELL_CHANCE = 2.0; // 2.0% chance for cell to be solid oak planks
    public static double HOLLOW_COVERED_CHANCE = 1.5; // 1.5% chance for cell to be covered but hollow oak
    public static double FALSE_SOLID_CHANCE = 1.0; // 1.0% chance for cell to be solid false oak planks
    public static double FALSE_HOLLOW_CHANCE = 0.6; // 0.6% chance for cell to be hollow false oak planks

    // Ceiling Plate Configuration
    public static boolean ENABLE_CEILING_PLATE = true; // Toggle for oak ceiling plate above lattice
    public static int CEILING_PLATE_GAP = 2; // Air gap between top of lattice and ceiling plate
    public static int CEILING_PLATE_THICKNESS = 2; // Thickness of the ceiling plate

    // Outer Roof Plate 2 & Adaptive Rubber-Band Stairs Configuration
    public static boolean ENABLE_ROOF_PLATE_2 = true; // Toggle for outer second roof plate
    public static int ROOF_PLATE_2_HEIGHT_OFFSET = 35; // Configurable height rise above lower plate (rubber band stair rise)
    public static int ROOF_PLATE_2_THICKNESS = 1; // Thickness of the outer roof plate
    public static int ROOF_PLATE_2_BOUNDARY = 20000; // 20km x 20km outer boundary (-10000 to +10000)

    // Configurable Boundary Clamping
    public static int BOUNDARY_CLAMP_SIZE = 1000; // Total boundary width in blocks (-500 to +500)
    public static boolean ENABLE_BOUNDARY_CLAMPING = true; // Enforces the boundary clamp

    private static final org.slf4j.Logger LOGGER = com.mojang.logging.LogUtils.getLogger();

    public CrossworksLatticeFeature() {
        this(net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration.CODEC);
    }

    public CrossworksLatticeFeature(com.mojang.serialization.Codec<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> codec) {
        super(codec);
    }

    @net.neoforged.fml.common.EventBusSubscriber(modid="the_backwoods")
    public static final class Registration {
        @net.neoforged.bus.api.SubscribeEvent
        public static void register(net.neoforged.neoforge.registries.RegisterEvent event) {
            event.register(net.minecraft.core.registries.Registries.FEATURE, helper ->
                helper.register(net.minecraft.resources.Identifier.parse(ID), new CrossworksLatticeFeature()));
            if (event.getRegistryKey().equals(net.minecraft.core.registries.Registries.FEATURE)) {
                autoGenerateDataPackJsons();
            }
        }
    }

    private record SectorCoord(int sector, boolean inGap, int gapIndex, int distToGap, boolean inExtendedGap) {}

    private static SectorCoord mapAxis(int coord, int halfBound, int boundarySize, int sectorSize, int gapSize, int extension) {
        int u = coord + halfBound;
        if (u < 0 || u >= boundarySize) return new SectorCoord(-1, false, -1, 999, false);
        int gap0Start = sectorSize;
        int gap0End = sectorSize + gapSize;
        int sec1End = gap0End + sectorSize;
        int gap1End = sec1End + gapSize;

        if (u < gap0Start) {
            int dist = gap0Start - 1 - u;
            return new SectorCoord(0, false, 0, dist, dist < extension);
        }
        if (u < gap0End) {
            return new SectorCoord(0, true, 0, 0, true);
        }
        if (u < sec1End) {
            int dist0 = u - gap0End;
            int dist1 = sec1End - 1 - u;
            int dist = Math.min(dist0, dist1);
            int idx = (dist0 <= dist1) ? 0 : 1;
            return new SectorCoord(1, false, idx, dist, dist < extension);
        }
        if (u < gap1End) {
            return new SectorCoord(1, true, 1, 0, true);
        }
        if (u < boundarySize) {
            int dist = u - gap1End;
            return new SectorCoord(2, false, 1, dist, dist < extension);
        }
        return new SectorCoord(-1, false, -1, 999, false);
    }

    private static long hash(long seed, int x, int z) {
        long v = seed ^ ((long) x * 0x632BE59BD9B4E019L) ^ ((long) z * 0x9E3779B97F4A7C15L);
        v = (v ^ (v >>> 30)) * 0xBF58476D1CE4E5B9L;
        v = (v ^ (v >>> 27)) * 0x94D049BB133111EBL;
        return v ^ (v >>> 31);
    }

    private static double rollPercentage(long seed, int x, int z) {
        return ((hash(seed, x, z) >>> 11) * 0x1.0p-53) * 100.0;
    }

    public static int TOP_MARGIN = 3; // Air padding below max build height for Roof Plate 2

    public static int getLatticeBaseY(int max) {
        int totalRoofOffset = (ENABLE_ROOF_PLATE_2 ? ROOF_PLATE_2_HEIGHT_OFFSET : 0) + CEILING_PLATE_GAP + CEILING_PLATE_THICKNESS + 1;
        int gridTopY = max - TOP_MARGIN - totalRoofOffset;
        return gridTopY - VERTICAL_THICKNESS;
    }

    @Override
    public boolean place(net.minecraft.world.level.levelgen.feature.FeaturePlaceContext<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> context) {
        var level = context.level();
        if (!level.getLevel().dimension().identifier().toString().equals(DIMENSION_ID)) return false;

        if (CELL_SIZE < 2 || BEAM_WIDTH < 1 || BEAM_WIDTH >= CELL_SIZE || VERTICAL_THICKNESS < 1) {
            throw new IllegalStateException("Invalid CrossworksLatticeFeature geometry settings");
        }

        int min = level.getMinY();
        int max = level.getMaxY() + 1; // Modern getMaxY() is inclusive;

        int gridBaseY = getLatticeBaseY(max);
        int gridTopY = gridBaseY + VERTICAL_THICKNESS;
        if (gridBaseY < min || gridTopY > max) {
            return false;
        }

        int baseX = context.origin().getX() & ~15;
        int baseZ = context.origin().getZ() & ~15;

        int halfInner = BOUNDARY_CLAMP_SIZE / 2;
        int halfOuter = (ENABLE_ROOF_PLATE_2) ? (ROOF_PLATE_2_BOUNDARY / 2) : halfInner;
        if (ENABLE_BOUNDARY_CLAMPING) {
            if (baseX + 15 < -halfOuter || baseX > halfOuter || baseZ + 15 < -halfOuter || baseZ > halfOuter) {
                return false;
            }
        }

        var chunk = level.getChunk(baseX >> 4, baseZ >> 4, net.minecraft.world.level.chunk.status.ChunkStatus.EMPTY, false);
        if (chunk == null) return false;

        var pos = new net.minecraft.core.BlockPos.MutableBlockPos();
        var plank = net.minecraft.world.level.block.Blocks.OAK_PLANKS.defaultBlockState();
        var fence = net.minecraft.world.level.block.Blocks.OAK_FENCE.defaultBlockState();

        // Outward-facing Upper Stairs (Half.BOTTOM, INNER corners) & Upside-Down Under Stairs (Half.TOP, OUTER corners)
        var baseStair = net.minecraft.world.level.block.Blocks.OAK_STAIRS.defaultBlockState();
        var stairN = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.NORTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);
        var stairS = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.SOUTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);
        var stairE = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.EAST).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);
        var stairW = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.WEST).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);

        var cornerNW = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.NORTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.INNER_LEFT);
        var cornerNE = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.NORTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.INNER_RIGHT);
        var cornerSW = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.SOUTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.INNER_RIGHT);
        var cornerSE = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.SOUTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.BOTTOM).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.INNER_LEFT);

        var underStairN = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.SOUTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);
        var underStairS = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.NORTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);
        var underStairE = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.WEST).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);
        var underStairW = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.EAST).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.STRAIGHT);

        var underCornerNW = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.SOUTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.OUTER_LEFT);
        var underCornerNE = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.SOUTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.OUTER_RIGHT);
        var underCornerSW = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.NORTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.OUTER_RIGHT);
        var underCornerSE = baseStair.setValue(net.minecraft.world.level.block.StairBlock.FACING, net.minecraft.core.Direction.NORTH).setValue(net.minecraft.world.level.block.StairBlock.HALF, net.minecraft.world.level.block.state.properties.Half.TOP).setValue(net.minecraft.world.level.block.StairBlock.SHAPE, net.minecraft.world.level.block.state.properties.StairsShape.OUTER_LEFT);

        var falsePlankBlock = net.minecraft.core.registries.BuiltInRegistries.BLOCK
            .getOptional(net.minecraft.resources.Identifier.parse("the_backwoods:false_oak_planks"))
            .orElse(net.minecraft.world.level.block.Blocks.OAK_PLANKS);
        var falsePlank = falsePlankBlock.defaultBlockState();
        if (falsePlank.hasProperty(net.minecraft.world.level.block.state.properties.BlockStateProperties.PERSISTENT)) {
            falsePlank = falsePlank.setValue(net.minecraft.world.level.block.state.properties.BlockStateProperties.PERSISTENT, true);
        }

        long seed = level.getSeed();
        int sectorSize = Math.max(16, (BOUNDARY_CLAMP_SIZE - 2 * GAP_SIZE) / 3);

        for (int lz = 0; lz < 16; lz++) {
            int wz = baseZ + lz;
            if (ENABLE_BOUNDARY_CLAMPING && (wz < -halfOuter || wz >= halfOuter)) continue;

            int distZ = (wz >= halfInner) ? (wz - halfInner) : ((wz < -halfInner) ? ((-halfInner - 1) - wz) : -1);
            int dirZ = (wz >= halfInner) ? 1 : ((wz < -halfInner) ? -1 : 0);
            SectorCoord scZ = (distZ < 0) ? mapAxis(wz, halfInner, BOUNDARY_CLAMP_SIZE, sectorSize, GAP_SIZE, BRIDGE_EXTENSION) : null;

            for (int lx = 0; lx < 16; lx++) {
                int wx = baseX + lx;
                if (ENABLE_BOUNDARY_CLAMPING && (wx < -halfOuter || wx >= halfOuter)) continue;

                int distX = (wx >= halfInner) ? (wx - halfInner) : ((wx < -halfInner) ? ((-halfInner - 1) - wx) : -1);
                int dirX = (wx >= halfInner) ? 1 : ((wx < -halfInner) ? -1 : 0);

                int outDist = Math.max(distX, distZ);

                if (outDist >= 0) {
                    // OUTER ELEVATED REGION: Rubber-Band Stairs or Lattice Roof Plate 2
                    if (ENABLE_ROOF_PLATE_2) {
                        int plate1TopY = gridTopY + CEILING_PLATE_GAP + CEILING_PLATE_THICKNESS - 1;
                        if (outDist < ROOF_PLATE_2_HEIGHT_OFFSET) {
                            // Adaptive Rubber-Band Stairs with Upside-Down Stairs underneath
                            int s = outDist;
                            int stairY = plate1TopY + 1 + s;

                            net.minecraft.world.level.block.state.BlockState stairState;
                            net.minecraft.world.level.block.state.BlockState underStairState;

                            if (distX == s && distZ == s) {
                                if (dirX == -1 && dirZ == -1) { stairState = cornerNW; underStairState = underCornerNW; }
                                else if (dirX == 1 && dirZ == -1) { stairState = cornerNE; underStairState = underCornerNE; }
                                else if (dirX == -1 && dirZ == 1) { stairState = cornerSW; underStairState = underCornerSW; }
                                else { stairState = cornerSE; underStairState = underCornerSE; }
                            } else if (distZ == s) {
                                if (dirZ == -1) { stairState = stairN; underStairState = underStairN; }
                                else { stairState = stairS; underStairState = underStairS; }
                            } else {
                                if (dirX == -1) { stairState = stairW; underStairState = underStairW; }
                                else { stairState = stairE; underStairState = underStairE; }
                            }

                            // 1. Upper Right-Side-Up Stair
                            if (stairY >= min && stairY < max) {
                                pos.set(wx, stairY, wz);
                                chunk.setBlockState(pos, stairState, 0);
                            }

                            // 2. Lower Upside-Down Stair directly underneath
                            int underY = stairY - 1;
                            if (underY >= min && underY < max) {
                                pos.set(wx, underY, wz);
                                chunk.setBlockState(pos, underStairState, 0);
                            }
                        } else if (outDist < halfOuter) {
                            // Lattice Roof Plate 2 (20km x 20km outer boundary)
                            int plate2TopY = plate1TopY + ROOF_PLATE_2_HEIGHT_OFFSET;
                            for (int dy = 0; dy < ROOF_PLATE_2_THICKNESS; dy++) {
                                int y = plate2TopY - dy;
                                if (y >= min && y < max) {
                                    pos.set(wx, y, wz);
                                    chunk.setBlockState(pos, plank, 0);
                                }
                            }
                        }
                    }
                    continue;
                }

                if (scZ == null || scZ.sector < 0) continue;
                SectorCoord scX = mapAxis(wx, halfInner, BOUNDARY_CLAMP_SIZE, sectorSize, GAP_SIZE, BRIDGE_EXTENSION);
                if (scX.sector < 0) continue;

                // 1. CEILING PLATE: Generates above the lattice if enabled
                if (ENABLE_CEILING_PLATE) {
                    int plateY = gridTopY + CEILING_PLATE_GAP;
                    for (int py = 0; py < CEILING_PLATE_THICKNESS; py++) {
                        int y = plateY + py;
                        if (y >= min && y < max) {
                            pos.set(wx, y, wz);
                            chunk.setBlockState(pos, plank, 0);
                        }
                    }
                }

                // 2. CHASM GAPS & BRIDGES: Spanning between the 9 larger cells
                boolean isCrossGap = scX.inExtendedGap && scZ.inExtendedGap;
                boolean bridgeX = false;
                boolean bridgeZ = false;

                if (!isCrossGap) {
                    int bridgeLocalZ = Math.floorMod(wz, BRIDGE_INTERVAL);
                    boolean onBridgePathX = bridgeLocalZ < BRIDGE_WIDTH;
                    int bridgeLocalX = Math.floorMod(wx, BRIDGE_INTERVAL);
                    boolean onBridgePathZ = bridgeLocalX < BRIDGE_WIDTH;

                    bridgeX = scX.inExtendedGap && !scZ.inExtendedGap && onBridgePathX;
                    bridgeZ = scZ.inExtendedGap && !scX.inExtendedGap && onBridgePathZ;

                    if (bridgeX || bridgeZ) {
                        int deckY = gridTopY - BRIDGE_THICKNESS;
                        int fenceY = gridTopY;

                        // 1. Solid Bridge Deck at gridTopY - BRIDGE_THICKNESS
                        if (deckY >= min && deckY < max) {
                            pos.set(wx, deckY, wz);
                            chunk.setBlockState(pos, plank, 0);
                        }

                        // 2. Directional Fence Railings on Outer Edges
                        if (fenceY >= min && fenceY < max) {
                            int bridgeIndexZ = Math.floorDiv(wz, BRIDGE_INTERVAL);
                            boolean hasFencesX = bridgeX && (rollPercentage(seed ^ 0xFEC839A1B704L, bridgeIndexZ, scX.gapIndex) < BRIDGE_FENCE_CHANCE);
                            int bridgeIndexX = Math.floorDiv(wx, BRIDGE_INTERVAL);
                            boolean hasFencesZ = bridgeZ && (rollPercentage(seed ^ 0xA82319FC412DL, bridgeIndexX, scZ.gapIndex) < BRIDGE_FENCE_CHANCE);

                            boolean isRailX = hasFencesX && (bridgeLocalZ == 0 || bridgeLocalZ == BRIDGE_WIDTH - 1);
                            boolean isRailZ = hasFencesZ && (bridgeLocalX == 0 || bridgeLocalX == BRIDGE_WIDTH - 1);

                            if (isRailX || isRailZ) {
                                var fenceState = fence;
                                if (isRailX) {
                                    fenceState = fenceState
                                        .setValue(net.minecraft.world.level.block.state.properties.BlockStateProperties.EAST, true)
                                        .setValue(net.minecraft.world.level.block.state.properties.BlockStateProperties.WEST, true);
                                }
                                if (isRailZ) {
                                    fenceState = fenceState
                                        .setValue(net.minecraft.world.level.block.state.properties.BlockStateProperties.NORTH, true)
                                        .setValue(net.minecraft.world.level.block.state.properties.BlockStateProperties.SOUTH, true);
                                }
                                pos.set(wx, fenceY, wz);
                                chunk.setBlockState(pos, fenceState, 0);
                            }
                        }
                    }
                }

                if (scX.inGap || scZ.inGap) {
                    // Outside sectors: only bridge exists; the rest of the chasm is open air
                    continue;
                }

                // 3. SECTOR INTERIOR LATTICE: Inside one of the 9 larger cells
                int cellX = Math.floorDiv(wx, CELL_SIZE);
                int cellZ = Math.floorDiv(wz, CELL_SIZE);
                int modX = Math.floorMod(wx, CELL_SIZE);
                int modZ = Math.floorMod(wz, CELL_SIZE);

                boolean isBeam = (modX < BEAM_WIDTH) || (modZ < BEAM_WIDTH) || (scX.distToGap < BEAM_WIDTH) || (scZ.distToGap < BEAM_WIDTH);
                boolean isPerimeterWall = (modX == BEAM_WIDTH || modX == CELL_SIZE - 1 || modZ == BEAM_WIDTH || modZ == CELL_SIZE - 1);

                // Deterministic cell roll for the 4 probabilistic variations
                double cellRoll = rollPercentage(seed ^ 0x3E8B549219AFL, cellX, cellZ);
                boolean isSolidOak = cellRoll < SOLID_CELL_CHANCE; // 2.0%
                boolean isHollowOak = !isSolidOak && (cellRoll < SOLID_CELL_CHANCE + HOLLOW_COVERED_CHANCE); // 1.5%
                boolean isSolidFalse = !isSolidOak && !isHollowOak && (cellRoll < SOLID_CELL_CHANCE + HOLLOW_COVERED_CHANCE + FALSE_SOLID_CHANCE); // 1.0%
                boolean isHollowFalse = !isSolidOak && !isHollowOak && !isSolidFalse && (cellRoll < SOLID_CELL_CHANCE + HOLLOW_COVERED_CHANCE + FALSE_SOLID_CHANCE + FALSE_HOLLOW_CHANCE); // 0.6%

                // Generate vertical column strictly bottom-to-up (dy: 0 -> VERTICAL_THICKNESS - 1)
                for (int dy = 0; dy < VERTICAL_THICKNESS; dy++) {
                    int y = gridBaseY + dy;
                    if (y < min || y >= max) continue;
                    if ((bridgeX || bridgeZ) && y >= gridTopY - BRIDGE_THICKNESS) continue; // Preserved by bridge deck

                    net.minecraft.world.level.block.state.BlockState blockToPlace = null;
                    boolean isFloorOrCeiling = (dy == 0 || dy == VERTICAL_THICKNESS - 1);

                    if (isSolidOak) {
                        blockToPlace = plank;
                    } else if (isSolidFalse) {
                        blockToPlace = falsePlank;
                    } else if (isHollowOak) {
                        if (isBeam || isFloorOrCeiling || isPerimeterWall) {
                            blockToPlace = plank;
                        }
                    } else if (isHollowFalse) {
                        if (isFloorOrCeiling) {
                            blockToPlace = falsePlank;
                        } else if (isBeam || isPerimeterWall) {
                            blockToPlace = plank;
                        }
                    } else if (isBeam) {
                        blockToPlace = plank;
                    }

                    if (blockToPlace != null) {
                        pos.set(wx, y, wz);
                        chunk.setBlockState(pos, blockToPlace, 0);
                    }
                }
            }
        }

        return true;
    }

    public static void autoGenerateDataPackJsons() {
        if (!ENABLE_AUTO_DATAPACK_GEN) return;
        try {
            var project = findDevelopmentProject(net.neoforged.fml.loading.FMLPaths.GAMEDIR.get(), java.nio.file.Path.of(""));
            if (project == null) return;
            var resources = project.resolve("src/main/resources");

            var outputs = new java.util.LinkedHashMap<String, String>();
            outputs.put("data/the_backwoods/worldgen/configured_feature/crossworks_lattice.json",
                "{\"type\":\"" + ID + "\",\"config\":{}}");
            outputs.put("data/the_backwoods/worldgen/placed_feature/crossworks_lattice_placed.json",
                "{\"feature\":\"" + ID + "\",\"placement\":[{\"type\":\"minecraft:biome\"}]}");
            outputs.put("data/the_backwoods/neoforge/biome_modifier/add_crossworks_lattice.json",
                "{\"type\":\"neoforge:add_features\",\"biomes\":\"" + BIOME_ID + "\",\"features\":\"the_backwoods:crossworks_lattice_placed\",\"step\":\"surface_structures\"}");

            for (var root : java.util.List.of(resources, project.resolve("build/resources/main"))) {
                if (!java.nio.file.Files.isDirectory(root)) continue;
                for (var output : outputs.entrySet()) {
                    write(root.resolve(output.getKey()), output.getValue() + "\n");
                }
            }
        } catch (java.io.IOException | RuntimeException failure) {
            LOGGER.error("Crossworks Lattice development data generation failed", failure);
        }
    }

    private static java.nio.file.Path findDevelopmentProject(java.nio.file.Path... starts) {
        for (var start : starts) {
            for (var current = start.toAbsolutePath().normalize(); current != null; current = current.getParent()) {
                if ((java.nio.file.Files.isRegularFile(current.resolve("build.gradle"))
                        || java.nio.file.Files.isRegularFile(current.resolve("build.gradle.kts")))
                        && java.nio.file.Files.isDirectory(current.resolve("src/main/resources"))) {
                    return current;
                }
            }
        }
        return null;
    }

    private static void write(java.nio.file.Path path, String content) throws java.io.IOException {
        if (java.nio.file.Files.isRegularFile(path) && java.nio.file.Files.readString(path).equals(content)) return;
        java.nio.file.Files.createDirectories(path.getParent());
        var temporary = java.nio.file.Files.createTempFile(path.getParent(), "crossworks-lattice-", ".tmp");
        try {
            java.nio.file.Files.writeString(temporary, content);
            try {
                java.nio.file.Files.move(temporary, path, java.nio.file.StandardCopyOption.ATOMIC_MOVE, java.nio.file.StandardCopyOption.REPLACE_EXISTING);
            } catch (java.nio.file.AtomicMoveNotSupportedException ignored) {
                java.nio.file.Files.move(temporary, path, java.nio.file.StandardCopyOption.REPLACE_EXISTING);
            }
        } finally {
            java.nio.file.Files.deleteIfExists(temporary);
        }
    }
