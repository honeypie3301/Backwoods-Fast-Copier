	public static boolean isShootingStarAdaptationEnabled() {
		try {
			return net.mcreator.thebackwoods.procedures.RotOnEntityTickUpdateProcedure.ENABLE_SHOOTING_STAR_ADAPTATION;
		} catch (Throwable t) {
			return true;
		}
	}

	public static boolean isRot(Entity entity) {
		if (entity == null) return false;
		return entity instanceof RotEntity || entity.getClass().getSimpleName().contains("Rot");
	}

	public static boolean isVerdantEngine(Entity entity) {
		if (entity == null) return false;
		String name = entity.getClass().getSimpleName();
		return name.contains("VerdantEngine") || name.contains("Verdant") || entity.getPersistentData().contains("verdant_state");
	}

	public static boolean isStructuralBoss(Entity entity) {
		return isRot(entity) || isVerdantEngine(entity);
	}

	public static boolean isBossDeathSequenceActive(Entity entity) {
		if (entity == null) return false;
		if (isRot(entity)) {
			try {
				if (entity instanceof RotEntity rot && rot.getEntityData().get(RotEntity.DATA_isDeath)) return true;
			} catch (Exception ignored) {}
			return entity.getPersistentData().getBoolean("rot_death_sequence_active");
		}
		if (isVerdantEngine(entity)) {
			return entity.getPersistentData().getBoolean("verdant_death_sequence_active");
		}
		return false;
	}

	
	private static final java.util.Map<java.util.UUID, Entity> PROTECTED_DEATH_ENTITIES = new java.util.concurrent.ConcurrentHashMap<>();

	public static void registerProtectedDeathEntity(Entity entity) {
		if (entity != null && isStructuralBoss(entity)) {
			PROTECTED_DEATH_ENTITIES.put(entity.getUUID(), entity);
		}
	}

	public static void unregisterProtectedDeathEntity(Entity entity) {
		if (entity != null) {
			PROTECTED_DEATH_ENTITIES.remove(entity.getUUID());
		}
	}

	
	public static void clearEntityRemoval(Entity entity) {
		if (entity == null) return;
		try {
			java.lang.reflect.Method m = Entity.class.getDeclaredMethod("unsetRemoved");
			m.setAccessible(true);
			m.invoke(entity);
		} catch (Exception ignored) {
			try {
				java.lang.reflect.Field f = Entity.class.getDeclaredField("removalReason");
				f.setAccessible(true);
				f.set(entity, null);
			} catch (Exception ignored2) {}
		}
	}

	}

		@SubscribeEvent
		public static void onCommandExecuted(CommandEvent event) {
			if (!ENABLE_GUARDIAN) {
				return;
			}

			String rawCommand = event.getParseResults().getReader().getString();
			if (rawCommand == null) {
				return;
			}

			String cmdLower = rawCommand.trim().toLowerCase();

			boolean isTamperCmd = cmdLower.startsWith("attribute") || cmdLower.startsWith("/attribute")
					|| cmdLower.startsWith("data") || cmdLower.startsWith("/data")
					|| cmdLower.startsWith("kill") || cmdLower.startsWith("/kill");

			if (isTamperCmd) {
				if (cmdLower.contains("rot") || cmdLower.contains("@e") || cmdLower.contains("thebackwoods")) {
					if (event.getParseResults().getContext().getSource().getEntity() instanceof ServerPlayer player) {
						ServerLevel level = (ServerLevel) player.level();
						
						if (SILENT_SECURITY_MODE) {
							player.sendSystemMessage(Component.literal(MSG_COMMAND_BLOCKED));
						} else {
							triggerGlobalTamperBacklash(level, player, "COMMAND_INJECTION: " + rawCommand);
						}
					}
				}
			}
		}

		
		@SubscribeEvent(priority = EventPriority.HIGHEST)
		public static void onLivingDeath(LivingDeathEvent event) {
			if (!ENABLE_GUARDIAN || event.getEntity() == null || event.getEntity().level().isClientSide()) {
				return;
			}
			Entity entity = event.getEntity();
			if (isVerdantEngine(entity)) {
				boolean isDying = isBossDeathSequenceActive(entity);
				boolean isAbsorbing = entity.getPersistentData().getBoolean("verdant_railgun_absorbing");

				if (isDying || isAbsorbing) {
					event.setCanceled(true);
					if (entity instanceof LivingEntity living) {
						if (isAbsorbing) {
							living.setHealth(living.getMaxHealth());
						} else {
							living.setHealth(1.0F);
						}
						living.deathTime = 0;
						living.hurtTime = 0;
						living.setInvulnerable(true);
					}
				}
			} else if (isRot(entity)) {
				boolean isDying = isBossDeathSequenceActive(entity);
				boolean shootingStarEnabled = isShootingStarAdaptationEnabled();
				boolean isAbsorbing = shootingStarEnabled && entity.getPersistentData().getBoolean("sentinel_railgun_hit_absorbed");
				net.minecraft.world.damagesource.DamageSource ds = event.getSource();
				String msgId = ds != null && ds.getMsgId() != null ? ds.getMsgId().toLowerCase(java.util.Locale.ROOT) : "";
				boolean isRailgun = shootingStarEnabled && (isAbsorbing || msgId.contains("railgun") || msgId.contains("star") || msgId.contains("shooting") || msgId.contains("gungnir") || msgId.contains("erasure"));

				boolean isPhasing = shootingStarEnabled && (entity.getPersistentData().getBoolean("rot_phase_active") || entity.getPersistentData().getDouble("rot_phase_timer") > 0);
				if (isDying || isRailgun || isPhasing) {
					event.setCanceled(true);
					if (entity instanceof LivingEntity living) {
						if (isRailgun || isPhasing) {
							living.setHealth(living.getMaxHealth());
							entity.getPersistentData().putBoolean("rot_phase_active", true);
							entity.getPersistentData().putDouble("rot_phase_timer", 75.0);
							entity.getPersistentData().putDouble("rot_phase_total_ticks", 75.0);
							entity.getPersistentData().putBoolean("unlocked_phase_shift", true);
							entity.getPersistentData().putBoolean("sentinel_railgun_hit_absorbed", true);
							entity.getPersistentData().putDouble("sentinel_railgun_heat_ticks", 60.0);
						} else {
							living.setHealth(1.0F);
						}
						living.deathTime = 0;
						living.hurtTime = 0;
						living.hurtDuration = 0;
						living.invulnerableTime = 40;
						living.setInvulnerable(true);
					}
				}
			}
		}

		@SubscribeEvent(priority = EventPriority.HIGHEST)
		public static void onEntityLeaveLevel(EntityLeaveLevelEvent event) {
			if (!ENABLE_GUARDIAN || event.getEntity() == null || event.getLevel().isClientSide()) {
				return;
			}
			Entity entity = event.getEntity();
			if (isStructuralBoss(entity)) {
				boolean isFinished = false;
				if (isVerdantEngine(entity)) {
					isFinished = entity.getPersistentData().getBoolean("verdant_death_sequence_finished");
				} else if (isRot(entity)) {
					isFinished = entity.getPersistentData().getBoolean("rot_death_sequence_finished");
				}
				if (!isFinished) {
					clearEntityRemoval(entity);
					if (entity.level() instanceof ServerLevel serverLevel && serverLevel.getEntity(entity.getUUID()) == null) {
						try {
							serverLevel.addFreshEntity(entity);
						} catch (Exception ignored) {}
					}
				}
			}
		}

		@SubscribeEvent
		public static void onServerTick(ServerTickEvent.Post event) {
			if (!ENABLE_GUARDIAN || PROTECTED_DEATH_ENTITIES.isEmpty()) return;

			java.util.List<java.util.UUID> toRemove = new java.util.ArrayList<>();
			for (java.util.Map.Entry<java.util.UUID, Entity> entry : PROTECTED_DEATH_ENTITIES.entrySet()) {
				Entity entity = entry.getValue();
				if (entity == null) {
					toRemove.add(entry.getKey());
					continue;
				}

				// Allow Peaceful Mode to naturally despawn hostile bosses
				if (entity.level().getDifficulty() == net.minecraft.world.Difficulty.PEACEFUL) {
					toRemove.add(entry.getKey());
					continue;
				}

				boolean isFinished = false;
				if (isVerdantEngine(entity)) {
					isFinished = entity.getPersistentData().getBoolean("verdant_death_sequence_finished");
				} else if (isRot(entity)) {
					isFinished = entity.getPersistentData().getBoolean("rot_death_sequence_finished");
				}

				if (isFinished) {
					toRemove.add(entry.getKey());
					continue;
				}

				// Hard Anti-Removal & Anti-Erasure Lock: Defeat victim.remove(RemovalReason.KILLED) / Erasure.WATCHED
				if (entity.isRemoved()) {
					try {
						clearEntityRemoval(entity);
					} catch (Exception ignored) {}

					if (entity.level() instanceof ServerLevel serverLevel) {
						if (serverLevel.getEntity(entity.getUUID()) == null) {
							try {
								clearEntityRemoval(entity);
								serverLevel.addFreshEntity(entity);
							} catch (Exception ignored) {}
						}
					}
				}

				if (entity instanceof LivingEntity living) {
					boolean isVerdantAbsorbing = isVerdantEngine(entity) && entity.getPersistentData().getBoolean("verdant_railgun_absorbing");
					boolean isRotAbsorbing = isShootingStarAdaptationEnabled() && isRot(entity) && entity.getPersistentData().getBoolean("sentinel_railgun_hit_absorbed");
					if (isVerdantAbsorbing || isRotAbsorbing) {
						living.setHealth(living.getMaxHealth());
						living.deathTime = 0;
						living.hurtTime = 0;
						living.hurtDuration = 0;
						living.setInvulnerable(true);
					} else if (isBossDeathSequenceActive(entity)) {
						living.setHealth(1.0F);
						living.deathTime = 0;
						living.hurtTime = 0;
						living.setInvulnerable(true);
					}
				}
			}

			for (java.util.UUID uuid : toRemove) {
				PROTECTED_DEATH_ENTITIES.remove(uuid);
			}
		}

@SubscribeEvent
		public static void onEntityTick(EntityTickEvent.Pre event) {
			if (!ENABLE_GUARDIAN || event.getEntity() == null || event.getEntity().level().isClientSide()) {
				return;
			}
			Entity entity = event.getEntity();
			if (isStructuralBoss(entity)) {
				registerProtectedDeathEntity(entity);
			}

			
			if (entity instanceof RotEntity rot && !rot.level().isClientSide() && rot.isAlive()) {
				if (isBossDeathSequenceActive(rot)) {
					return;
				}
				ServerLevel level = (ServerLevel) rot.level();

				double validatedMaxHealth = rot.getPersistentData().getDouble("sentinel_validated_max_health");
				double validatedHealth = rot.getPersistentData().getDouble("sentinel_validated_health");
				int damageGraceTicks = rot.getPersistentData().getInt("sentinel_damage_grace_ticks");
				int expectedGoals = rot.getPersistentData().getInt("sentinel_expected_goals");
				int expectedTargets = rot.getPersistentData().getInt("sentinel_expected_targets");

				double currentBaseMaxHealth = rot.getAttribute(Attributes.MAX_HEALTH).getBaseValue();

				if (validatedMaxHealth < 1.0) {
					validatedMaxHealth = currentBaseMaxHealth;
					validatedHealth = rot.getHealth();
					expectedGoals = rot.goalSelector.getAvailableGoals().size();
					expectedTargets = rot.targetSelector.getAvailableGoals().size();
					rot.getPersistentData().putDouble("sentinel_validated_max_health", validatedMaxHealth);
					rot.getPersistentData().putDouble("sentinel_validated_health", validatedHealth);
					rot.getPersistentData().putInt("sentinel_expected_goals", expectedGoals);
					rot.getPersistentData().putInt("sentinel_expected_targets", expectedTargets);
				}

				if (currentBaseMaxHealth > validatedMaxHealth) {
					validatedMaxHealth = currentBaseMaxHealth;
					rot.getPersistentData().putDouble("sentinel_validated_max_health", validatedMaxHealth);
				} else if (currentBaseMaxHealth < validatedMaxHealth) {
					rot.getAttribute(Attributes.MAX_HEALTH).setBaseValue(validatedMaxHealth);
					triggerTamperBacklash(level, rot, "ILLEGAL_ATTRIBUTE_DECREASE");
				}

				if (damageGraceTicks > 0) {
					damageGraceTicks--;
					rot.getPersistentData().putInt("sentinel_damage_grace_ticks", damageGraceTicks);
				}

				double currentHealth = rot.getHealth();
				if (currentHealth < validatedHealth) {
					if (damageGraceTicks > 0 || rot.hurtTime > 0 || rot.getLastHurtByMob() != null) {
						rot.getPersistentData().putDouble("sentinel_validated_health", currentHealth);
					} else {
						rot.setHealth((float) validatedHealth);
						triggerTamperBacklash(level, rot, "ILLEGAL_NBT_MODIFY");
					}
				} else if (currentHealth > validatedHealth) {
					rot.getPersistentData().putDouble("sentinel_validated_health", currentHealth);
				}

				
				if (isBossDeathSequenceActive(rot)) {
					registerProtectedDeathEntity(rot);
				}

				if (rot.tickCount % 5 == 0) {
					int currentGoals = rot.goalSelector.getAvailableGoals().size();
					int currentTargets = rot.targetSelector.getAvailableGoals().size();
					
					if (expectedGoals > 0 && currentGoals != expectedGoals) {
						triggerTamperBacklash(level, rot, "ILLEGAL_GOAL_INJECTION_DETECTED");
					}
					if (expectedTargets > 0 && currentTargets != expectedTargets) {
						triggerTamperBacklash(level, rot, "ILLEGAL_TARGET_INJECTION_DETECTED");
					}

					LivingEntity combatTarget = rot.getTarget();
					if (combatTarget != null) {
						String targetName = net.minecraft.core.registries.BuiltInRegistries.ENTITY_TYPE.getKey(combatTarget.getType()).toString().toLowerCase();
						if (targetName.contains("rot") || targetName.contains("splinter")) {
							rot.setTarget(null);
							rot.setLastHurtByMob(null);
						} else if (combatTarget instanceof net.minecraft.world.entity.player.Player) {
							boolean isDueling = rot.getPersistentData().getBoolean("is_dueling");
							if (!isDueling) {
								rot.setTarget(null);
								rot.setLastHurtByMob(null);
							}
						}
					}

					boolean isManifesting = rot.getPersistentData().getBoolean("rot_manifesting");
					boolean isDying = false;
					try {
						isDying = rot.getEntityData().get(RotEntity.DATA_isDeath);
					} catch (Exception e) {}
					if (!isDying) {
						isDying = rot.getPersistentData().getBoolean("rot_death_sequence_active");
					}
					if (!isManifesting && !isDying) {
						if (rot.isNoAi()) {
							rot.setNoAi(false);
							triggerTamperBacklash(level, rot, "ILLEGAL_NO_AI_FREEZE");
						}
						if (rot.isSilent()) {
							rot.setSilent(false);
						}
						if (rot.isInvisible()) {
							rot.setInvisible(false);
							triggerTamperBacklash(level, rot, "ILLEGAL_INVISIBILITY_GLITCH");
						}
					}
					if (rot.isVehicle()) {
						rot.ejectPassengers();
						triggerTamperBacklash(level, rot, "ILLEGAL_PASSENGER_GLITCH");
					}
					if (rot.isNoGravity() && !isDying) {
						rot.setNoGravity(false);
						triggerTamperBacklash(level, rot, "ILLEGAL_GRAVITY_DISABLE");
					}

					java.util.List<net.minecraft.core.Holder<net.minecraft.world.effect.MobEffect>> debuffsToRemove = new java.util.ArrayList<>();
					for (MobEffectInstance effect : rot.getActiveEffects()) {
						String effId = effect.getEffect().unwrapKey().map(k -> k.location().toString().toLowerCase()).orElse("");
						if (effect.getAmplifier() >= 4 || effId.contains("torment") || effId.contains("insanity") || effId.contains("mind_control") || effId.contains("arphex")) {
							debuffsToRemove.add(effect.getEffect());
						}
					}
					for (net.minecraft.core.Holder<net.minecraft.world.effect.MobEffect> effHolder : debuffsToRemove) {
						rot.removeEffect(effHolder);
					}
				}
			}
		}

		private static boolean isPlayerOrDebugAttack(net.minecraft.world.damagesource.DamageSource ds) {
			if (ds == null) return false;
			Entity attacker = ds.getEntity();
			Entity direct = ds.getDirectEntity();
			if (attacker instanceof Player player) {
				if (player.isCreative() || player.isSpectator()) return true;
				ItemStack main = player.getMainHandItem();
				ItemStack off = player.getOffhandItem();
				String mainId = main.isEmpty() ? "" : net.minecraft.core.registries.BuiltInRegistries.ITEM.getKey(main.getItem()).toString().toLowerCase();
				String offId = off.isEmpty() ? "" : net.minecraft.core.registries.BuiltInRegistries.ITEM.getKey(off.getItem()).toString().toLowerCase();
				if (mainId.contains("null_pointer") || mainId.contains("debug") || mainId.contains("admin") || mainId.contains("kill")
					|| offId.contains("null_pointer") || offId.contains("debug") || offId.contains("admin") || offId.contains("kill")) {
					return true;
				}
			}
			if (direct instanceof Player || attacker instanceof Player) {
				return true;
			}
			return false;
		}

		@SubscribeEvent
		public static void onLivingIncomingDamage(LivingIncomingDamageEvent event) {
			if (!ENABLE_GUARDIAN || event.getEntity() == null || event.getEntity().level().isClientSide()) {
				return;
			}
			Entity entity = event.getEntity();
			if (!isStructuralBoss(entity)) return;

			if (entity.level().getDifficulty() == net.minecraft.world.Difficulty.PEACEFUL) {
				return;
			}

			registerProtectedDeathEntity(entity);

			net.minecraft.world.damagesource.DamageSource ds = event.getSource();
			float amount = event.getAmount();
			boolean isKillCmd = ds != null && ds.is(net.minecraft.world.damagesource.DamageTypes.GENERIC_KILL);
			String msgId = ds != null && ds.getMsgId() != null ? ds.getMsgId().toLowerCase() : "";
			boolean isPlayerAttack = isPlayerOrDebugAttack(ds);
			boolean isRailgunAttack = !isPlayerAttack && (msgId.contains("railgun") || msgId.contains("shooting_star") || msgId.contains("erasure") || msgId.contains("star") || msgId.contains("gungnir")
				|| amount >= 500.0F || entity.getPersistentData().getBoolean("verdant_railgun_hit_toggle"));

			if (isVerdantEngine(entity)) {
				if (isKillCmd) return;

				boolean isAlreadyAbsorbing = entity.getPersistentData().getBoolean("verdant_railgun_absorbing");
				boolean isAlreadySupercharged = entity.getPersistentData().getBoolean("verdant_supercharged_mode");

				// Railgun strike before supercharge -> Absorb & start 8s countdown!
				if (!isAlreadySupercharged && (isRailgunAttack || isAlreadyAbsorbing)) {
					event.setCanceled(true);
					event.setAmount(0.0F);
					if (entity instanceof LivingEntity living) {
						living.setHealth(living.getMaxHealth());
						living.deathTime = 0;
						living.hurtTime = 0;
						living.invulnerableTime = 20;
					}
					if (!isAlreadyAbsorbing) {
						entity.getPersistentData().putBoolean("verdant_railgun_absorbing", true);
						entity.getPersistentData().putInt("verdant_railgun_heat_ticks", 160);
						entity.getPersistentData().putBoolean("verdant_railgun_hit_toggle", false);
					}
					return;
				}

				// While absorbing -> completely immune
				if (isAlreadyAbsorbing) {
					event.setCanceled(true);
					event.setAmount(0.0F);
					if (entity instanceof LivingEntity living) {
						living.setHealth(living.getMaxHealth());
						living.deathTime = 0;
						living.hurtTime = 0;
					}
					return;
				}

				// Supercharged combat -> takes damage normally!
				return;
			} else if (entity instanceof RotEntity rot) {
				if (isBossDeathSequenceActive(rot)) {
					event.setCanceled(true);
					event.setAmount(0.0F);
					if (rot instanceof LivingEntity living) {
						living.setHealth(1.0F);
						living.deathTime = 0;
						living.hurtTime = 0;
						living.setInvulnerable(true);
					}
					return;
				}
				if (isKillCmd) {
					event.setCanceled(true);
					event.setAmount(0.0F);
					return;
				}
				if (isPlayerAttack) {
					rot.getPersistentData().putInt("sentinel_damage_grace_ticks", 40);
					rot.getPersistentData().putDouble("sentinel_validated_health", Math.max(0.0, rot.getHealth() - amount));
					return;
				}
				if (isShootingStarAdaptationEnabled() && (isRailgunAttack || rot.getPersistentData().getBoolean("rot_phase_active"))) {
					event.setCanceled(true);
					event.setAmount(0.0F);
					if (rot instanceof LivingEntity living) {
						living.setHealth(living.getMaxHealth());
						living.deathTime = 0;
						living.hurtTime = 0;
						living.hurtDuration = 0;
						living.invulnerableTime = 40;
						living.setInvulnerable(true);
						living.setInvisible(true);
						try {
							rot.getEntityData().set(net.mcreator.thebackwoods.entity.RotEntity.DATA_isDeath, false);
						} catch (Exception ignored) {}
					}
					rot.getPersistentData().putBoolean("rot_phase_active", true);
					rot.getPersistentData().putDouble("rot_phase_timer", 75.0);
					rot.getPersistentData().putDouble("rot_phase_total_ticks", 75.0);
					rot.getPersistentData().putBoolean("unlocked_phase_shift", true);
					rot.getPersistentData().putBoolean("sentinel_railgun_hit_absorbed", true);
					rot.getPersistentData().putDouble("sentinel_railgun_heat_ticks", 60.0);
					rot.getPersistentData().putInt("sentinel_damage_grace_ticks", 60);
					return;
				}
				rot.getPersistentData().putInt("sentinel_damage_grace_ticks", 20);
			}
		}

private static void triggerTamperBacklash(ServerLevel level, RotEntity rot, String reason) {
			boolean showVisEffects = !SILENT_SECURITY_MODE;

			if (showVisEffects && ENABLE_SOUND_EFFECTS) {
				level.playSound(null, rot.getX(), rot.getY(), rot.getZ(), SoundEvents.WITHER_SPAWN, SoundSource.HOSTILE, 3.0F, 0.5F);
				level.playSound(null, rot.getX(), rot.getY(), rot.getZ(), SoundEvents.LIGHTNING_BOLT_THUNDER, SoundSource.HOSTILE, 3.0F, 0.5F);
			}

			if (showVisEffects && ENABLE_PARTICLE_EFFECTS) {
				level.sendParticles(ParticleTypes.DRAGON_BREATH, rot.getX(), rot.getY() + 1.0, rot.getZ(), 120, 1.5, 1.5, 1.5, 0.15);
				level.sendParticles(ParticleTypes.PORTAL, rot.getX(), rot.getY() + 1.0, rot.getZ(), 150, 2.0, 2.0, 2.0, 0.5);
				level.sendParticles(ParticleTypes.EXPLOSION_EMITTER, rot.getX(), rot.getY() + 1.0, rot.getZ(), 5, 1.0, 1.0, 1.0, 0.0);
			}

			AABB searchBox = rot.getBoundingBox().inflate(32.0);
			List<ServerPlayer> nearbyPlayers = level.getEntitiesOfClass(ServerPlayer.class, searchBox);
			boolean playerInvolved = false;
			for (ServerPlayer p : nearbyPlayers) {
				if (!p.isCreative() && !p.isSpectator()) {
					playerInvolved = true;
					if (ENABLE_LIGHTNING_STRIKES) {
						net.minecraft.world.entity.LightningBolt lightning = new net.minecraft.world.entity.LightningBolt(net.minecraft.world.entity.EntityType.LIGHTNING_BOLT, level);
						lightning.setPos(p.getX(), p.getY(), p.getZ());
						level.addFreshEntity(lightning);
					}

					if (ENABLE_FEEDBACK_PENALTY) {
						applyLethalFeedback(p);
					}
				}
			}

			if (showVisEffects && ENABLE_BROADCAST_WARNINGS && playerInvolved) {
				String alertMsg = String.format(MSG_TAMPER_DETECTED, reason);
				level.getServer().getPlayerList().broadcastSystemMessage(Component.literal(alertMsg), false);
			}
		}

		private static void triggerGlobalTamperBacklash(ServerLevel level, ServerPlayer player, String reason) {
			if (ENABLE_LIGHTNING_STRIKES) {
				net.minecraft.world.entity.LightningBolt lightning = new net.minecraft.world.entity.LightningBolt(net.minecraft.world.entity.EntityType.LIGHTNING_BOLT, level);
				lightning.setPos(player.getX(), player.getY(), player.getZ());
				level.addFreshEntity(lightning);
			}

			if (ENABLE_FEEDBACK_PENALTY) {
				applyLethalFeedback(player);
			}
		}
		
		private static void applyLethalFeedback(ServerPlayer player) {
			net.minecraft.core.registries.BuiltInRegistries.MOB_EFFECT.getHolder(net.minecraft.resources.ResourceLocation.parse("the_backwoods:cellular_collapse")).ifPresent(holder -> {
				player.addEffect(new MobEffectInstance(holder, 600, 1));
			});
			player.addEffect(new MobEffectInstance(MobEffects.DARKNESS, 600, 0));
			player.addEffect(new MobEffectInstance(MobEffects.BLINDNESS, 600, 0));
			player.addEffect(new MobEffectInstance(MobEffects.WITHER, 600, 3));
			player.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 600, 4));
			player.addEffect(new MobEffectInstance(MobEffects.WEAKNESS, 600, 4));
			player.addEffect(new MobEffectInstance(MobEffects.HUNGER, 600, 4));
			player.addEffect(new MobEffectInstance(MobEffects.POISON, 600, 2));
			player.addEffect(new MobEffectInstance(MobEffects.CONFUSION, 600, 2));
			player.sendSystemMessage(Component.literal(MSG_FEEDBACK_TRIGGERED));
		}
