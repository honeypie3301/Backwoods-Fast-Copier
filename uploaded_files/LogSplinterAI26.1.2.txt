	@SubscribeEvent
	public static void onEntityTickAfterGoals(EntityTickEvent.Post event) {
		if (event.getEntity() instanceof LogSplinterEntity member) {
			SplinterHivemind.followFormationPost(member, member.getEntityData().get(LogSplinterEntity.DATA_mineProgress));
		}
	}

	@SubscribeEvent
	public static void onEntityTick(EntityTickEvent.Pre event) {
		execute(event, event.getEntity().level(), event.getEntity().getX(), event.getEntity().getY(), event.getEntity().getZ(), event.getEntity());
	}

	public static void execute(LevelAccessor world, double x, double y, double z, Entity entity) {
		execute(null, world, x, y, z, entity);
	}

	private static void execute(@Nullable Event event, LevelAccessor world, double x, double y, double z, Entity entity) {
		if (!(entity instanceof LogSplinterEntity logSplinter)) return;

		updatePlacedBlockCleanup(world, logSplinter, TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get().defaultBlockState());

		if (IsRotNearbyConditionProcedure.execute(world, logSplinter)) {
			logSplinter.setTarget(null);
			logSplinter.setLastHurtByMob(null);
			setSpeed(logSplinter, ACTIVE_MOVE_SPEED);
			return;
		}

		if (logSplinter.getTarget() instanceof Player p && (p.isCreative() || p.isSpectator())) {
			logSplinter.setTarget(null);
		}
		if (logSplinter.getLastHurtByMob() instanceof Player p && (p.isCreative() || p.isSpectator())) {
			logSplinter.setLastHurtByMob(null);
		}

		if (!logSplinter.getPersistentData().getBooleanOr(K_PERSONALITY_INIT, false)) {
			int p = (logSplinter.getRandom().nextInt(100) < ANGEL_CHANCE_PERCENT) ? PERSONALITY_ANGEL : PERSONALITY_STALK_HUNT;
			logSplinter.getPersistentData().putInt(K_PERSONALITY, p);
			logSplinter.getPersistentData().putBoolean(K_PERSONALITY_INIT, true);
		}
		int personality = logSplinter.getPersistentData().getIntOr(K_PERSONALITY, PERSONALITY_STALK_HUNT);

		if (logSplinter.isPassenger()) {
			Entity vehicle = logSplinter.getVehicle();
			if (vehicle instanceof net.minecraft.world.entity.vehicle.boat.Boat || vehicle instanceof net.minecraft.world.entity.vehicle.boat.ChestBoat) {
				logSplinter.stopRiding();
				logSplinter.setDeltaMovement(logSplinter.getDeltaMovement().add(0, 0.2, 0));
			}
		}

		int age = logSplinter.getEntityData().get(LogSplinterEntity.DATA_Age);
		if (age >= AGE_LAST) {
			logSplinter.discard();
			return;
		}

		boolean isDegraded = age >= AGE_THIRD_TO_LAST;
		boolean isCritical = age >= AGE_SECOND_TO_LAST;

		Player foundPlayerDirect = findNearestPlayerInRange(world, logSplinter.getX(), logSplinter.getY(), logSplinter.getZ(), TARGET_RANGE);
		if (world instanceof Level level && level.isClientSide()) {
			return;
		}
		LivingEntity foundPlayer = foundPlayerDirect;

		LivingEntity prioritizedMob = null;
		String attackerUUIDStr = logSplinter.getPersistentData().getStringOr("splinter_attacker_uuid", "");
		if (!attackerUUIDStr.isEmpty() && world instanceof net.minecraft.server.level.ServerLevel serverLevel) {
			try {
				java.util.UUID attackerUUID = java.util.UUID.fromString(attackerUUIDStr);
				net.minecraft.world.entity.Entity foundEntity = serverLevel.getEntity(attackerUUID);
				if (foundEntity instanceof LivingEntity livingAttacker && livingAttacker.isAlive() && logSplinter.distanceToSqr(livingAttacker) < TARGET_RANGE * TARGET_RANGE) {
					boolean isWoodbound = livingAttacker.is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.Identifier.parse("the_backwoods:woodbound_entities")));
					if (!isWoodbound && !(livingAttacker instanceof Player)) {
						prioritizedMob = livingAttacker;
					}
				}
			} catch (Exception e) {}
		}

		LivingEntity attacker = logSplinter.getLastHurtByMob();
		if (attacker != null) {
			boolean isWoodbound = attacker.is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.Identifier.parse("the_backwoods:woodbound_entities")));
			if (isWoodbound) {
				if (logSplinter.getLastHurtByMob() == attacker) {
					logSplinter.setLastHurtByMob(null);
				}
				attacker = null;
			} else if (attacker.isAlive() && logSplinter.distanceToSqr(attacker) < TARGET_RANGE * TARGET_RANGE) {
				if (!(attacker instanceof Player)) {
					prioritizedMob = attacker;
					logSplinter.getPersistentData().putString("splinter_attacker_uuid", attacker.getUUID().toString());
				}
			}
		}

		boolean hadPrioritizedMob = !attackerUUIDStr.isEmpty();
		if (hadPrioritizedMob && prioritizedMob == null) {
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 0);
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
			logSplinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
			logSplinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
			logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
			logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
			logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			if (logSplinter.getLastHurtByMob() != null) {
				logSplinter.setLastHurtByMob(null);
			}
			logSplinter.getPersistentData().remove("splinter_attacker_uuid");
		}

		if (prioritizedMob != null) {
			foundPlayer = prioritizedMob;
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 1);
			if (logSplinter.getTarget() != prioritizedMob) {
				logSplinter.setTarget(prioritizedMob);
			}
			if (logSplinter.getLastHurtByMob() != prioritizedMob) {
				logSplinter.setLastHurtByMob(prioritizedMob);
			}
		} else {
			if (attacker instanceof Player p && p.isAlive() && !p.isCreative() && !p.isSpectator() && logSplinter.distanceToSqr(attacker) < TARGET_RANGE * TARGET_RANGE) {
				foundPlayer = p;
			}
			if (foundPlayer != null) {
				if (logSplinter.getTarget() != foundPlayer) {
					logSplinter.setTarget(foundPlayer);
				}
			} else {
				if (logSplinter.getTarget() != null) {
					logSplinter.setTarget(null);
				}
			}
		}

		if (foundPlayer != null) {
			boolean isWoodbound = foundPlayer.is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.Identifier.parse("the_backwoods:woodbound_entities")));
			if (isWoodbound) {
				foundPlayer = null;
			}
		}

		if (foundPlayer == null) {
			if (SplinterHivemind.search(logSplinter)) return;
			int remainingForcedHunt = logSplinter.getPersistentData().getIntOr(K_FORCED_HUNT_TICKS, 0);

			if (remainingForcedHunt <= 0) {
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 0);
				logSplinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
				logSplinter.getPersistentData().putInt(K_HM_PHASE, PHASE_SILENT);
				logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 0);
				logSplinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
				logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
				stopNav(logSplinter);
			} else {
				logSplinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, remainingForcedHunt - 1);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 1);
				logSplinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
				logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, HUNT_TICKS);
			}
			return;
		}

		int frozenByRose = logSplinter.getEntityData().get(LogSplinterEntity.DATA_frozenByRose);
		int isEnraged = logSplinter.getEntityData().get(LogSplinterEntity.DATA_isEnraged);
		int watchTimer = logSplinter.getEntityData().get(LogSplinterEntity.DATA_watchTimer);

		int forcedHuntTicks = logSplinter.getPersistentData().getIntOr(K_FORCED_HUNT_TICKS, 0);
		if (forcedHuntTicks > 0) {
			logSplinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, forcedHuntTicks - 1);
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 1);
			isEnraged = 1;
		}

		if (logSplinter.getLastHurtByMob() != null) {
			boolean isWoodbound = logSplinter.getLastHurtByMob().is(net.minecraft.tags.TagKey.create(net.minecraft.core.registries.Registries.ENTITY_TYPE, net.minecraft.resources.Identifier.parse("the_backwoods:woodbound_entities")));
			if (!isWoodbound) {
				int lastSeenHurtTime = logSplinter.getPersistentData().getIntOr(K_LAST_HURT_TIME, 0);
				int currentHurtTime = logSplinter.hurtTime;
				if (currentHurtTime > 0 && currentHurtTime != lastSeenHurtTime) {
					int bonus = logSplinter.getPersistentData().getIntOr(K_RAGE_BONUS, 0) + RAGE_THRESHOLD_HIT_BONUS;
					logSplinter.getPersistentData().putInt(K_RAGE_BONUS, bonus);
					logSplinter.getPersistentData().putInt(K_LAST_HURT_TIME, currentHurtTime);
				}
			}
		}

		int effectiveRageThreshold = Math.max(1, RAGE_WATCH_THRESHOLD - logSplinter.getPersistentData().getIntOr(K_RAGE_BONUS, 0));

		boolean isWatched = false;
		if (foundPlayerDirect != null) {
			Vec3 toLogSplinter = logSplinter.getEyePosition().subtract(foundPlayerDirect.getEyePosition());
			if (toLogSplinter.lengthSqr() > 1.0e-8) toLogSplinter = toLogSplinter.normalize();
			double dot = foundPlayerDirect.getLookAngle().normalize().dot(toLogSplinter);
			isWatched = (dot > WATCH_DOT_THRESHOLD) && foundPlayerDirect.hasLineOfSight(logSplinter);
		}

		double distToPlayer = foundPlayerDirect != null ? logSplinter.position().distanceTo(foundPlayerDirect.position()) : Double.MAX_VALUE;
		if (isEnraged == 1 && distToPlayer > RAGE_ESCAPE_RANGE) {
			if (logSplinter.getPersistentData().getIntOr(K_FORCED_HUNT_TICKS, 0) <= 0) {
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 0);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
				isEnraged = 0;
			}
		}

		if (isWatched && isEnraged == 0) {
			watchTimer++;
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, watchTimer);
			if (watchTimer >= effectiveRageThreshold) {
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 1);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
				isEnraged = 1;
			}
		} else if (!isWatched && isEnraged == 0) {
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
		}

		boolean foundRose = false;
		if (foundPlayerDirect != null) {
			foundRose = checkHeldRose(foundPlayerDirect, logSplinter, world, foundPlayerDirect.getX(), foundPlayerDirect.getY(), foundPlayerDirect.getZ());
		}
		if (!foundRose && (logSplinter.tickCount % 5 == 0)) {
			foundRose = checkNearbyRoseBlocks(world, logSplinter);
		}
		if (foundRose || frozenByRose == 1) {
			setSpeed(logSplinter, 0);
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_isEnraged, 0);
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
			logSplinter.getPersistentData().putInt(K_RAGE_BONUS, 0);
			logSplinter.getPersistentData().putInt(K_FORCED_HUNT_TICKS, 0);
			logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
			logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
			logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			stopNav(logSplinter);
			return;
		}

		SplinterHivemind.observe(logSplinter, foundPlayerDirect, logSplinter.getLastHurtByMob());
        if (personality == PERSONALITY_STALK_HUNT && SplinterHivemind.beginGroupHunt(logSplinter, foundPlayerDirect)) {
            logSplinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
            logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 160);
            logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
            logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
        }

		double heightDiff = foundPlayer.getY() - logSplinter.getY();
		double horizontalDist = logSplinter.position().distanceTo(foundPlayer.position());
		boolean isTowering = !isCritical && (heightDiff >= 1.4 && horizontalDist < 12);

		// Angel personality behavior: freeze while watched, otherwise direct hunt-like pursuit.
		// One bounded cover move owns navigation while the archer is drawing.
		if (foundPlayerDirect != null && logSplinter.getTarget() == foundPlayerDirect
				&& SplinterHivemind.handleBowThreat(logSplinter, foundPlayerDirect,
					logSplinter.getEntityData().get(LogSplinterEntity.DATA_mineProgress),
					isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED)) return;
		if (personality == PERSONALITY_STALK_HUNT
				&& SplinterHivemind.bowCommitActive(logSplinter, foundPlayerDirect)
				&& logSplinter.getPersistentData().getIntOr(K_HM_PHASE, 0) != PHASE_HUNT) {
			logSplinter.getPersistentData().putInt(K_HM_PHASE, PHASE_HUNT);
			logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, 90);
		}

		if (personality == PERSONALITY_ANGEL) {
			if (isWatched && isEnraged == 0 && !SplinterHivemind.bowCommitActive(logSplinter, foundPlayerDirect)) {
				setSpeed(logSplinter, 0);
				stopNav(logSplinter);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
				return;
			}

			if (logSplinter.tickCount % 3 == 0) {
				logSplinter.lookAt(EntityAnchorArgument.Anchor.EYES, new Vec3(foundPlayer.getX(), foundPlayer.getEyeY(), foundPlayer.getZ()));
			}

			setSpeed(logSplinter, isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED);
			if (logSplinter instanceof Mob angelMob) {
				navMoveToEntityThrottled(logSplinter, angelMob, foundPlayer, 1.0, 3);
			}
		} else {
			int phase = logSplinter.getPersistentData().getIntOr(K_HM_PHASE, 0);
			int phaseTimer = logSplinter.getPersistentData().getIntOr(K_HM_PHASE_TIMER, 0);
			int silentCd = logSplinter.getPersistentData().getIntOr(K_HM_SILENT_CD, 0);

			if (phase == PHASE_SILENT && isEnraged == 0 && forcedHuntTicks <= 0) {
				phase = PHASE_STALK;
				phaseTimer = randomBetween(logSplinter, STALK_MIN_TICKS, STALK_MAX_TICKS);
				logSplinter.getPersistentData().putInt(K_HM_PHASE, phase);
				logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
				logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			}

			if (silentCd > 0) {
				silentCd--;
				logSplinter.getPersistentData().putInt(K_HM_SILENT_CD, silentCd);
			}

			if (isEnraged == 1 || logSplinter.getPersistentData().getIntOr(K_FORCED_HUNT_TICKS, 0) > 0) {
				phase = PHASE_HUNT;
				phaseTimer = HUNT_TICKS + 40;
				logSplinter.getPersistentData().putInt(K_HM_PHASE, phase);
				logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
				logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
				logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
				logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
			} else {
				phaseTimer--;
				if (phaseTimer <= 0) {
					if (phase == PHASE_HUNT) {
						phase = PHASE_SILENT;
						phaseTimer = randomBetween(logSplinter, SILENT_MIN_TICKS, SILENT_MAX_TICKS);
						logSplinter.getPersistentData().putInt(K_HM_SILENT_CD, POST_HUNT_SILENT_CD);
						logSplinter.getEntityData().set(LogSplinterEntity.DATA_watchTimer, 0);
						logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					} else if (phase == PHASE_SILENT) {
						phase = PHASE_STALK;
						phaseTimer = randomBetween(logSplinter, STALK_MIN_TICKS, STALK_MAX_TICKS);
						logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					} else {
						phase = PHASE_SILENT;
						phaseTimer = randomBetween(logSplinter, SILENT_MIN_TICKS, SILENT_MAX_TICKS);
						logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, 0);
						logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						logSplinter.getPersistentData().putInt(K_FLANK_TICKS, 0);
					}
				}
				logSplinter.getPersistentData().putInt(K_HM_PHASE, phase);
				logSplinter.getPersistentData().putInt(K_HM_PHASE_TIMER, phaseTimer);
			}

			if (phase == PHASE_SILENT) {
				setSpeed(logSplinter, 0);
				stopNav(logSplinter);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
				return;
			}

			if (logSplinter.tickCount % 3 == 0) {
				logSplinter.lookAt(EntityAnchorArgument.Anchor.EYES, new Vec3(foundPlayer.getX(), foundPlayer.getEyeY(), foundPlayer.getZ()));
			}

			if (phase == PHASE_STALK) {
				setSpeed(logSplinter, STALK_SPEED);

				int watchTicks = logSplinter.getPersistentData().getIntOr(K_STALK_WATCH_TICKS, 0);
				if (watchTicks > 0) {
					logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, watchTicks - 1);
					stopNav(logSplinter);
					logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
					return;
				}

				int flankTicks = logSplinter.getPersistentData().getIntOr(K_FLANK_TICKS, 0);
				if (flankTicks > 0) {
					logSplinter.getPersistentData().putInt(K_FLANK_TICKS, flankTicks - 1);
					if (logSplinter instanceof net.minecraft.world.entity.Mob mob2) {
						double fx = logSplinter.getPersistentData().getDoubleOr(K_FLANK_X, 0.0);
						double fy = logSplinter.getPersistentData().getDoubleOr(K_FLANK_Y, 0.0);
						double fz = logSplinter.getPersistentData().getDoubleOr(K_FLANK_Z, 0.0);
						navMoveToPosThrottled(logSplinter, mob2, fx, fy, fz, 0.8, 1);
					}
					logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
					return;
				}
				if (logSplinter.getRandom().nextInt(STALK_WATCH_CHANCE) == 0) {
					int flankStreak = logSplinter.getPersistentData().getIntOr(K_FLANK_STREAK, 0);
					boolean doFlank = !isWatched && flankStreak < 2 && logSplinter.getRandom().nextBoolean();
					if (doFlank) {
						logSplinter.getPersistentData().putInt(K_FLANK_STREAK, flankStreak + 1);
						logSplinter.getPersistentData().putInt(K_FLANK_TICKS, randomBetween(logSplinter, STALK_WATCH_MIN, STALK_WATCH_MAX));
						double dist = STALK_FOLLOW_DIST - 2.0;
						double angleOffset = (logSplinter.getRandom().nextDouble() * 30.0 + 60.0) * (logSplinter.getRandom().nextBoolean() ? 1 : -1);
						double angleRad = Math.toRadians(foundPlayer.getYRot() + angleOffset);
						double fx = foundPlayer.getX() - Math.sin(angleRad) * dist;
						double fz = foundPlayer.getZ() + Math.cos(angleRad) * dist;
						double fy = foundPlayer.getY();
						logSplinter.getPersistentData().putDouble(K_FLANK_X, fx);
						logSplinter.getPersistentData().putDouble(K_FLANK_Y, fy);
						logSplinter.getPersistentData().putDouble(K_FLANK_Z, fz);
					} else {
						logSplinter.getPersistentData().putInt(K_FLANK_STREAK, 0);
						logSplinter.getPersistentData().putInt(K_STALK_WATCH_TICKS, randomBetween(logSplinter, STALK_WATCH_MIN, STALK_WATCH_MAX));
						stopNav(logSplinter);
					}
					logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
					return;
				}

				if (logSplinter instanceof Mob mob) {
					Vec3 hiveStalkPoint = SplinterHivemind.stalkPoint(mob, foundPlayer);
					if (hiveStalkPoint != null) {
						SplinterHivemind.allowPost(logSplinter);
						navMoveToPosThrottled(logSplinter, mob, hiveStalkPoint.x, hiveStalkPoint.y, hiveStalkPoint.z, 0.8, 1);
					} else if (distToPlayer > STALK_FOLLOW_DIST) {
						navMoveToEntityThrottled(logSplinter, mob, foundPlayer, 0.8, 1);
					} else {
						if (!mob.getNavigation().isDone() && logSplinter.tickCount % 10 == 0) {
							stopNav(logSplinter);
						}
					}
				}

				logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
				return;
			}

			setSpeed(logSplinter, isDegraded ? DEGRADED_MOVE_SPEED : ACTIVE_MOVE_SPEED);
			if (logSplinter instanceof Mob huntMob) {
				navMoveToEntityThrottled(logSplinter, huntMob, foundPlayer, 1.0, 3);
			}
		}

		Vec3 logEyes = logSplinter.getEyePosition(1f);
		Vec3 logView = logSplinter.getViewVector(1f);
		Vec3 blockCheckTarget = logEyes.add(logView.scale(MINE_RAY_DISTANCE));

		HitResult hit = world.clip(new ClipContext(logEyes, blockCheckTarget, ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, logSplinter));

		BlockPos facePos;
		BlockPos feetPos;

		if (hit.getType() == HitResult.Type.BLOCK) {
			facePos = ((BlockHitResult) hit).getBlockPos();
			feetPos = new BlockPos(facePos.getX(), Mth.floor(logSplinter.getY()), facePos.getZ());
		} else {
			Vec3 look = logSplinter.getLookAngle().normalize();
			int fx = Mth.floor(logSplinter.getX() + look.x);
			int fz = Mth.floor(logSplinter.getZ() + look.z);
			feetPos = new BlockPos(fx, Mth.floor(logSplinter.getY()), fz);
			facePos = new BlockPos(fx, Mth.floor(logSplinter.getEyeY()), fz);
		}

		boolean canMineFeet = canMine(world, feetPos, foundPlayer);
		boolean canMineFace = canMine(world, facePos, foundPlayer);

		BlockPos downPos = logSplinter.blockPosition().below();
		boolean canMineDown = (heightDiff <= -2.0 && horizontalDist <= 3.0) && canMine(world, downPos, foundPlayer);
		
		int prevProgress = logSplinter.getEntityData().get(LogSplinterEntity.DATA_mineProgress);
		BlockPos prevTrackPos = new BlockPos(
			logSplinter.getEntityData().get(LogSplinterEntity.DATA_mineX),
			logSplinter.getEntityData().get(LogSplinterEntity.DATA_mineY),
			logSplinter.getEntityData().get(LogSplinterEntity.DATA_mineZ)
		);
		BlockPos trackPos = canMineDown ? downPos : (canMineFeet ? feetPos : facePos);
		if (prevProgress > 0 && trackPos != null && !trackPos.equals(prevTrackPos)) {
			boolean lockPrev = canMine(world, prevTrackPos, foundPlayer) && prevTrackPos.closerToCenterThan(logSplinter.position(), 4.0) && !isPositionClaimed(world, prevTrackPos, logSplinter);
			if (lockPrev) {
				trackPos = prevTrackPos;
				canMineDown = trackPos.equals(downPos);
				canMineFeet = trackPos.equals(feetPos);
				canMineFace = trackPos.equals(facePos);
				if (!canMineDown && !canMineFeet && !canMineFace) {
					if (trackPos.getY() < logSplinter.getY()) { canMineDown = true; downPos = trackPos; }
					else { canMineFeet = true; feetPos = trackPos; }
				}
			} else {
				prevProgress = 0;
			}
		}
		if (canMineDown && isPositionClaimed(world, downPos, logSplinter)) canMineDown = false;
		if (canMineFeet && isPositionClaimed(world, feetPos, logSplinter)) canMineFeet = false;
		if (canMineFace && isPositionClaimed(world, facePos, logSplinter)) canMineFace = false;

		if (canMineFeet || canMineFace || canMineDown) {
			logSplinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
			int mineProgress = prevProgress + 1;
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, mineProgress);

			if (logSplinter.tickCount % 6 == 0) logSplinter.swing(InteractionHand.MAIN_HAND);

			logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineX, trackPos.getX());
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineY, trackPos.getY());
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineZ, trackPos.getZ());

			float speedRef;
			if (canMineDown) {
				speedRef = world.getBlockState(downPos).getDestroySpeed(world, downPos);
			} else if (canMineFeet) {
				speedRef = world.getBlockState(feetPos).getDestroySpeed(world, feetPos);
			} else {
				speedRef = world.getBlockState(facePos).getDestroySpeed(world, facePos);
			}

			float mineThreshold = isDegraded
					? speedRef * DEGRADED_MINE_SPEED_MULTIPLIER + DEGRADED_MINE_SPEED_BASE
					: speedRef * MINE_SPEED_MULTIPLIER + MINE_SPEED_BASE;

			if (mineProgress > mineThreshold) {
				if (canMineDown) world.destroyBlock(downPos, false);
				if (canMineFeet) world.destroyBlock(feetPos, false);
				if (canMineFace) world.destroyBlock(facePos, false);
				stopNav(logSplinter);
				logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);
			}
		} else {
			logSplinter.getEntityData().set(LogSplinterEntity.DATA_mineProgress, 0);

			if (!isCritical && !logSplinter.isSwimming() && !logSplinter.isInWater()) {
				if (heightDiff >= 1.4 && horizontalDist < 12) {
					logSplinter.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get()));
					if (logSplinter.tickCount % 6 == 0) {
						logSplinter.swing(InteractionHand.MAIN_HAND);
					}

					BlockPos under = BlockPos.containing(logSplinter.getX(), logSplinter.getY() - 1, logSplinter.getZ());
					BlockPos body = BlockPos.containing(logSplinter.getX(), logSplinter.getY(), logSplinter.getZ());
					BlockPos bodyUp = body.above();

					if (world.getBlockState(under).isAir()) {
						world.setBlock(under, TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get().defaultBlockState(), 3);
						trackPlacedBlock(logSplinter, under);
					}
					if (world.getBlockState(body).is(TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get())) {
						world.setBlock(body, Blocks.AIR.defaultBlockState(), 3);
					}
					if (world.getBlockState(bodyUp).is(TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get())) {
						world.setBlock(bodyUp, Blocks.AIR.defaultBlockState(), 3);
					}

					world.destroyBlock(BlockPos.containing(logSplinter.getX(), logSplinter.getY() + 2, logSplinter.getZ()), false);
					world.destroyBlock(BlockPos.containing(logSplinter.getX(), logSplinter.getY() + 3, logSplinter.getZ()), false);

					if (logSplinter.onGround()) {
						logSplinter.setDeltaMovement(new Vec3(logSplinter.getDeltaMovement().x(), 0.4, logSplinter.getDeltaMovement().z()));
					}
					logSplinter.fallDistance = 0;
				} else if (heightDiff > -0.5 && heightDiff < 2.5) {
					Vec3 look = logSplinter.getLookAngle();
					BlockPos bridgePos = BlockPos.containing(logSplinter.getX() + look.x, logSplinter.getY() - 1, logSplinter.getZ() + look.z);
					BlockPos farBridgePos = BlockPos.containing(logSplinter.getX() + look.x * 2.0, logSplinter.getY() - 1, logSplinter.getZ() + look.z * 2.0);
					boolean isNearGap = !(world.getBlockFloorHeight(bridgePos) > 0) || !(world.getBlockFloorHeight(farBridgePos) > 0);

					if (isNearGap) {
						logSplinter.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get()));
						if (!(world.getBlockFloorHeight(bridgePos) > 0)) {
							if (logSplinter.tickCount % 6 == 0) {
								logSplinter.swing(InteractionHand.MAIN_HAND);
							}
							world.setBlock(bridgePos, TheBackwoodsModBlocks.SPLINTERED_OAK_WOOD.get().defaultBlockState(), 3);
							trackPlacedBlock(logSplinter, bridgePos);
						}
					} else {
						logSplinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
					}
				} else {
					logSplinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
				}
			} else {
				logSplinter.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
			}
		}

		// Melee reach range increase for non-players only (by 50%)
		LivingEntity target = logSplinter.getTarget();
		if (target != null && target.isAlive() && !(target instanceof Player)) {
			double reach = logSplinter.getBbWidth() + target.getBbWidth() + 0.8;
			double increasedReach = reach * 1.5;
			double distSqr = logSplinter.distanceToSqr(target);
			if (distSqr <= increasedReach * increasedReach) {
				int lastAttackTick = logSplinter.getPersistentData().getIntOr("last_melee_attack_tick", 0);
				if (logSplinter.tickCount - lastAttackTick >= 20) {
					logSplinter.doHurtTarget((net.minecraft.server.level.ServerLevel) logSplinter.level(), target);
					logSplinter.swing(InteractionHand.MAIN_HAND);
					logSplinter.getPersistentData().putInt("last_melee_attack_tick", logSplinter.tickCount);
				}
			}
		}
	}

	private static int randomBetween(LogSplinterEntity entity, int min, int max) {
		if (max <= min) return min;
		return min + entity.getRandom().nextInt(max - min + 1);
	}

	private static void setSpeed(LivingEntity entity, double speed) {
		if (entity.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)) {
			entity.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(speed);
		}

	}

private static boolean isPositionClaimed(LevelAccessor world, BlockPos pos, LogSplinterEntity thisEntity) {
		net.minecraft.world.phys.AABB box = new net.minecraft.world.phys.AABB(pos).inflate(3);
		java.util.List<LogSplinterEntity> others = world.getEntitiesOfClass(LogSplinterEntity.class, box, e -> e != thisEntity && e.isAlive());
		int ourProgress = thisEntity.getEntityData().get(LogSplinterEntity.DATA_mineProgress);
		for (LogSplinterEntity other : others) {
			int otherProgress = other.getEntityData().get(LogSplinterEntity.DATA_mineProgress);
			if (otherProgress > 0 &&
				other.getEntityData().get(LogSplinterEntity.DATA_mineX) == pos.getX() &&
				other.getEntityData().get(LogSplinterEntity.DATA_mineY) == pos.getY() &&
				other.getEntityData().get(LogSplinterEntity.DATA_mineZ) == pos.getZ()) {
				
				if (ourProgress == 0) return true;
				if (otherProgress > ourProgress) return true;
				if (otherProgress == ourProgress && other.getId() < thisEntity.getId()) return true;
			}
		}
		return false;
	}

	private static boolean canMine(LevelAccessor world, BlockPos pos, LivingEntity player) {
		BlockState state = world.getBlockState(pos);
		if (state.isAir()) return false;
		float speed = state.getDestroySpeed(world, pos);
		if (speed < 0 || speed >= MAX_BREAKABLE_HARDNESS) return false;
		return pos.getY() != (int) (player.getY() - 2);
	}

	private static boolean checkHeldRose(Entity holder, LogSplinterEntity logSplinter, LevelAccessor world, double x, double y, double z) {
		if (!(holder instanceof LivingEntity living)) return false;

		ItemStack main = living.getMainHandItem();
		ItemStack off = living.getOffhandItem();

		boolean mainIsRose = main.getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem();
		boolean offIsRose = off.getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem();

		if (!mainIsRose && !offIsRose) return false;

		setSpeed(logSplinter, 0);
		stopNav(logSplinter);

		if (mainIsRose) tickRoseItem(main, world, x, y, z);
		if (offIsRose) tickRoseItem(off, world, x, y, z);

		return true;
	}

	private static void tickRoseItem(ItemStack rose, LevelAccessor world, double x, double y, double z) {
		double wiltTimer = rose.getOrDefault(DataComponents.CUSTOM_DATA, CustomData.EMPTY).copyTag().getDoubleOr("wilt_timer", 0.0) + 1.0;
		CustomData.update(DataComponents.CUSTOM_DATA, rose, tag -> tag.putDouble("wilt_timer", wiltTimer));

		if (wiltTimer > ROSE_WILT_TICKS) {
			if (world instanceof Level level) {
				if (!level.isClientSide()) {
					level.playSound(null, BlockPos.containing(x, y, z), SoundEvents.FIRE_EXTINGUISH, SoundSource.NEUTRAL, 1f, 1f);
				} else {
					level.playLocalSound(x, y, z, SoundEvents.FIRE_EXTINGUISH, SoundSource.NEUTRAL, 1f, 1f, false);
				}
			}
			rose.shrink(1);
			CustomData.update(DataComponents.CUSTOM_DATA, rose, tag -> tag.putDouble("wilt_timer", 0));
		}
	}

	private static boolean checkNearbyRoseBlocks(LevelAccessor world, LogSplinterEntity logSplinter) {
		BlockPos base = logSplinter.blockPosition();
		for (int sx = -ROSE_SCAN_XZ; sx < ROSE_SCAN_XZ; sx++) {
			for (int sy = -ROSE_SCAN_Y; sy < ROSE_SCAN_Y; sy++) {
				for (int sz = -ROSE_SCAN_XZ; sz < ROSE_SCAN_XZ; sz++) {
					BlockPos check = base.offset(sx, sy, sz);
					if (world.getBlockState(check).getBlock() == TheBackwoodsModBlocks.ASH_ROSE.get()) return true;
				}
			}
		}
		return false;
	}

	private static Player findNearestPlayerInRange(LevelAccessor world, double x, double y, double z, double range) {
		AABB box = AABB.ofSize(new Vec3(x, y, z), range, range, range);
		List<Player> players = world.getEntitiesOfClass(Player.class, box, e -> !e.isCreative() && !e.isSpectator());

		Player nearest = null;
		double best = range * range;
		for (Player p : players) {
			double d = p.distanceToSqr(x, y, z);
			if (d < best) {
				best = d;
				nearest = p;
			}
		}
		return nearest;
	}

	private static void stopNav(LogSplinterEntity logSplinter) {
		if (logSplinter instanceof Mob mob) {
			mob.getNavigation().stop();
		}
		logSplinter.getPersistentData().putInt(K_NAV_MODE, 0);
	}

	private static void navMoveToEntityThrottled(LogSplinterEntity logSplinter, Mob mob, Entity target, double speed, int mode) {
		Vec3 hiveApproach = SplinterHivemind.approach(mob, target);
		if (target instanceof Player && hiveApproach.distanceToSqr(target.position()) > 1.0) SplinterHivemind.allowPost(mob);
		navMoveToPosThrottled(logSplinter, mob, hiveApproach.x, hiveApproach.y, hiveApproach.z, speed, mode);
	}

	private static void navMoveToPosThrottled(Entity entity, Mob mob, double x, double y, double z, double speed, int throttle) {
		double dx = entity.getX() - x;
		double dy = entity.getY() - y;
		double dz = entity.getZ() - z;
		if (dx * dx + dy * dy + dz * dz < 4.0) {
			mob.getNavigation().stop();
			return;
		}

		int currentTick = entity.tickCount;
		int lastTick = entity.getPersistentData().getIntOr(K_NAV_LAST_TICK, 0);
		int lastMode = entity.getPersistentData().getIntOr(K_NAV_MODE, 0);

		double lastTx = entity.getPersistentData().getDoubleOr(K_NAV_TX, 0.0);
		double lastTy = entity.getPersistentData().getDoubleOr(K_NAV_TY, 0.0);
		double lastTz = entity.getPersistentData().getDoubleOr(K_NAV_TZ, 0.0);

		if (currentTick - lastTick < throttle && lastMode == 2 && Math.abs(lastTx - x) < 0.1 && Math.abs(lastTy - y) < 0.1 && Math.abs(lastTz - z) < 0.1) {
			return;
		}
		entity.getPersistentData().putInt(K_NAV_LAST_TICK, currentTick);
		entity.getPersistentData().putInt(K_NAV_MODE, 2);
		entity.getPersistentData().putDouble(K_NAV_TX, x);
		entity.getPersistentData().putDouble(K_NAV_TY, y);
		entity.getPersistentData().putDouble(K_NAV_TZ, z);

		mob.getNavigation().moveTo(x, y, z, speed);
	}

	private static double sqr(double v) {
		return v * v;

	private static void trackPlacedBlock(Entity splinter, BlockPos pos) {
		net.minecraft.nbt.CompoundTag tag = splinter.getPersistentData();
		int size = tag.getIntOr("pb_size", 0);

		for (int i = 0; i < size; i++) {
			int x = tag.getIntOr("pb_x_" + i, 0);
			int y = tag.getIntOr("pb_y_" + i, 0);
			int z = tag.getIntOr("pb_z_" + i, 0);
			if (x == pos.getX() && y == pos.getY() && z == pos.getZ()) {
				return;
			}
		}

		if (size < 64) {
			tag.putInt("pb_x_" + size, pos.getX());
			tag.putInt("pb_y_" + size, pos.getY());
			tag.putInt("pb_z_" + size, pos.getZ());
			tag.putInt("pb_size", size + 1);
		}
	}

	private static void updatePlacedBlockCleanup(LevelAccessor world, Entity splinter, BlockState targetBlockState) {
		net.minecraft.nbt.CompoundTag tag = splinter.getPersistentData();
		int size = tag.getIntOr("pb_size", 0);
		if (size == 0) return;

		boolean standingOnAny = false;
		for (int i = 0; i < size; i++) {
			int x = tag.getIntOr("pb_x_" + i, 0);
			int y = tag.getIntOr("pb_y_" + i, 0);
			int z = tag.getIntOr("pb_z_" + i, 0);

			double dx = Math.abs(splinter.getX() - (x + 0.5));
			double dz = Math.abs(splinter.getZ() - (z + 0.5));
			double dy = splinter.getY() - y;

			if (dx <= 1.0 && dz <= 1.0 && dy >= -0.2 && dy <= 2.2) {
				standingOnAny = true;
				break;
			}
		}

		if (standingOnAny) {
			tag.putInt("pb_delay", 0);
		} else {
			int delay = tag.getIntOr("pb_delay", 0) + 1;
			tag.putInt("pb_delay", delay);

			if (delay % 2 == 0) {
				int x0 = tag.getIntOr("pb_x_0", 0);
				int y0 = tag.getIntOr("pb_y_0", 0);
				int z0 = tag.getIntOr("pb_z_0", 0);
				BlockPos pos0 = new BlockPos(x0, y0, z0);

				BlockState state = world.getBlockState(pos0);
				if (state.is(targetBlockState.getBlock())) {
					world.destroyBlock(pos0, false);
				}

				for (int i = 0; i < size - 1; i++) {
					tag.putInt("pb_x_" + i, tag.getIntOr("pb_x_" + (i + 1), 0));
					tag.putInt("pb_y_" + i, tag.getIntOr("pb_y_" + (i + 1), 0));
					tag.putInt("pb_z_" + i, tag.getIntOr("pb_z_" + (i + 1), 0));
				}
				size--;
				tag.putInt("pb_size", size);
				if (size == 0) {
					tag.putInt("pb_delay", 0);
				}
			}
		}
	}

	/** Short-lived, local coordination across the four Splinter variants. Server side only. */
	private static final class SplinterHivemind {
	    private static final int ALERT_TICKS = 120;
	    private static final double ALERT_RADIUS = 28.0;
	    private static final String ID = "hive_target_uuid";
	    private static final String DIM = "hive_dimension";
	    private static final String UNTIL = "hive_until";
	    private static final String X = "hive_x";
	    private static final String Y = "hive_y";
	    private static final String Z = "hive_z";

	    private SplinterHivemind() { }

	    public static void allowPost(Mob self) {
	        self.getPersistentData().putLong("hive_post_time", self.level().getGameTime());
	    }

	    /** Reassert a flanking route after the vanilla melee goal updates navigation. */
	    public static void followFormationPost(Mob self, int mineProgress) {
	        if (!(self.level() instanceof ServerLevel) || frozen(self) || mineProgress != 0
	                || (self.getPersistentData().getLongOr("hive_post_time", -1) != self.level().getGameTime()
	                    && self.getPersistentData().getLongOr("hive_bow_post_time", 0) != self.level().getGameTime())
	                || !self.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)
	                || self.getAttribute(Attributes.MOVEMENT_SPEED).getBaseValue() <= 0.01) return;
	        if (!(self.getTarget() instanceof Player target) || !target.isAlive()
	                || target.isCreative() || target.isSpectator()) return;
	        if (self.getPersistentData().getLongOr("hive_bow_post_time", 0) == self.level().getGameTime()) {
	            if (self.getPersistentData().getIntOr("hive_bow_state", 0) == BOW_SEEK || self.getPersistentData().getIntOr("hive_bow_state", 0) == BOW_FLANK) {
	                Vec3 point = bowPoint(self);
	                self.getNavigation().moveTo(point.x, point.y, point.z, 1.0);
	            } else {
	                self.getNavigation().stop();
	            }
	            return;
	        }
	        Vec3 destination = approach(self, target);
	        if (destination.distanceToSqr(target.position()) < 1.0
	                || self.distanceToSqr(target) < 16.0) return;
	        if ((self.tickCount + self.getId()) % 3 == 0 || self.getNavigation().isDone()) {
	            self.getNavigation().moveTo(destination.x, destination.y, destination.z, 1.05);
	        }
	    }

	    private static boolean member(Entity e) {
	        return e instanceof SplinterEntity || e instanceof BlindspotSplinterEntity
	            || e instanceof LogSplinterEntity || e instanceof PetrifiedLogSplinterEntity;
	    }

	    private static boolean frozen(Mob e) {
	        if (e instanceof SplinterEntity s) return s.getEntityData().get(SplinterEntity.DATA_frozenByRose) == 1;
	        if (e instanceof BlindspotSplinterEntity s) return s.getEntityData().get(BlindspotSplinterEntity.DATA_frozenByRose) == 1;
	        if (e instanceof LogSplinterEntity s) return s.getEntityData().get(LogSplinterEntity.DATA_frozenByRose) == 1;
	        if (e instanceof PetrifiedLogSplinterEntity s) return s.getEntityData().get(PetrifiedLogSplinterEntity.DATA_frozenByRose) == 1;
	        return true;
	    }

	    private static String dimension(ServerLevel level) {
	        return level.dimension().identifier().toString();
	    }

	    private static boolean current(Mob e, Player player, ServerLevel level) {
	        return dimension(level).equals(e.getPersistentData().getStringOr(DIM, ""))
	            && player.getUUID().toString().equals(e.getPersistentData().getStringOr(ID, ""))
	            && level.getGameTime() < e.getPersistentData().getLongOr(UNTIL, 0);
	    }

	    private static void remember(Mob e, Player target, ServerLevel level) {
	        e.getPersistentData().putString(ID, target.getUUID().toString());
	        e.getPersistentData().putString(DIM, dimension(level));
	        e.getPersistentData().putLong(UNTIL, level.getGameTime() + ALERT_TICKS);
	        e.getPersistentData().putDouble(X, target.getX());
	        e.getPersistentData().putDouble(Y, target.getY());
	        e.getPersistentData().putDouble(Z, target.getZ());
	    }

	    /** A true sighting or attack shares the last known position with loaded neighbors. */
	    public static void observe(Mob source, Player visible, Entity attacker) {
	        if (!(source.level() instanceof ServerLevel level) || frozen(source)) return;
	        if ((source.tickCount + source.getId()) % 10 != 0) return;
	        Player target = null;
	        if (attacker instanceof Player p && p.isAlive() && !p.isCreative() && !p.isSpectator()
	                && source.distanceToSqr(p) <= 32 * 32 && source.hurtTime > 0) target = p;
	        else if (visible != null && visible.isAlive() && !visible.isCreative() && !visible.isSpectator()
	                && source.distanceToSqr(visible) <= 48 * 48
	                && source.hasLineOfSight(visible)) target = visible;
	        if (target == null) return;
	        if (target.getMainHandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()
	                || target.getOffhandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) return;

	        AABB area = source.getBoundingBox().inflate(ALERT_RADIUS, 12, ALERT_RADIUS);
	        List<Mob> neighbors = level.getEntitiesOfClass(Mob.class, area,
	            e -> e != source && e.isAlive() && member(e) && !frozen(e)
	                && e.distanceToSqr(source) <= ALERT_RADIUS * ALERT_RADIUS);
	        if (neighbors.isEmpty()) return;

	        boolean newAlarm = !current(source, target, level);
	        if (newAlarm) source.getPersistentData().putLong("hive_engage_until", level.getGameTime() + 160);
	        remember(source, target, level);
	        for (Mob neighbor : neighbors) {
	            // Sharing a report never sets rage, personality, melee timing, or a live target.
	            if (!current(neighbor, target, level)) {
	                newAlarm = true;
	                neighbor.getPersistentData().putLong("hive_engage_until", level.getGameTime() + 160);
	            }
	            remember(neighbor, target, level);
	        }
	        long lastCue = source.getPersistentData().getLongOr("hive_last_cue", 0);
	        if (newAlarm && (lastCue == 0 || level.getGameTime() - lastCue >= 160)) {
	            source.getPersistentData().putLong("hive_last_cue", level.getGameTime());
	            level.playSound(null, source.blockPosition(), SoundEvents.FIRE_EXTINGUISH,
	                SoundSource.HOSTILE, 1.2f, 0.55f);
	        }
	    }

	    /** Apply the group alert once; leave rage and individual attack timing alone. */
	    public static boolean beginGroupHunt(Mob self, Entity target) {
	        if (!(target instanceof Player player) || !(self.level() instanceof ServerLevel level)
	                || !current(self, player, level) || frozen(self)) return false;
	        long epoch = self.getPersistentData().getLongOr("hive_engage_until", 0);
	        if (epoch <= level.getGameTime() || self.getPersistentData().getLongOr("hive_hunt_used", 0) == epoch) return false;
	        self.getPersistentData().putLong("hive_hunt_used", epoch);
	        return true;
	    }

	    /** A lost target yields a brief, fallible search around the last reported position. */
	    public static boolean search(Mob self) {
	        if (!(self.level() instanceof ServerLevel level) || frozen(self)) return false;
	        long until = self.getPersistentData().getLongOr(UNTIL, 0);
	        if (until <= level.getGameTime() || !dimension(level).equals(self.getPersistentData().getStringOr(DIM, ""))) return false;
	        try {
	            Entity target = level.getEntity(UUID.fromString(self.getPersistentData().getStringOr(ID, "")));
	            if (!(target instanceof Player p) || !p.isAlive() || p.isCreative() || p.isSpectator()) return false;
	        } catch (IllegalArgumentException ex) {
	            return false;
	        }
	        double x = self.getPersistentData().getDoubleOr(X, 0.0);
	        double y = self.getPersistentData().getDoubleOr(Y, 0.0);
	        double z = self.getPersistentData().getDoubleOr(Z, 0.0);
	        if (self.distanceToSqr(x, y, z) > 48 * 48) return false;
	        if (self.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)) {
	            self.getAttribute(Attributes.MOVEMENT_SPEED).setBaseValue(0.23);
	        }
	        int slot = Math.floorMod(self.getUUID().hashCode(), 4);
	        double sx = x + (slot == 0 ? 4 : slot == 1 ? -4 : 0);
	        double sz = z + (slot == 2 ? 4 : slot == 3 ? -4 : 0);
	        if ((self.tickCount + self.getId()) % 12 == 0 || self.getNavigation().isDone()) {
	            self.getNavigation().moveTo(sx, y, sz, 1.0);
	        }
	        return true;
	    }


	    private static double facing(Mob self, Player player) {
	        Vec3 toward = self.getEyePosition().subtract(player.getEyePosition());
	        if (toward.lengthSqr() < 0.01) return 1.0;
	        return player.getLookAngle().normalize().dot(toward.normalize());
	    }

	    /** Strafe around a raised shield, then commit to the ordinary melee goal. */
	    private static Vec3 shieldRoute(Mob self, Player player, ServerLevel level) {
	        long now = level.getGameTime();
	        if (self.getPersistentData().getLongOr("hive_shield_strike", 0) > now) return player.position();
	        double distance = self.distanceToSqr(player);
	        if (distance < 4 || distance > 100 || facing(self, player) < 0.25
	                || !player.isBlocking() || !player.getUseItem().is(Items.SHIELD)) return null;
	        Vec3 forward = player.getLookAngle();
	        double length = Math.hypot(forward.x, forward.z);
	        if (length < 0.01) return null;
	        double side = (self.getUUID().hashCode() & 1) == 0 ? 1 : -1;
	        Vec3 goal = player.position().add(-forward.x / length * 2 + side * forward.z / length * 3,
	            0, -forward.z / length * 2 - side * forward.x / length * 3);
	        long deadline = self.getPersistentData().getLongOr("hive_shield_side_until", 0);
	        if (deadline <= now) {
	            self.getPersistentData().putLong("hive_shield_side_until", now + 32);
	            deadline = now + 32;
	        }
	        if (self.distanceToSqr(goal) < 6.25 || deadline - now <= 1) {
	            self.getPersistentData().putLong("hive_shield_strike", now + 18);
	            self.getPersistentData().putLong("hive_shield_side_until", 0);
	            return player.position();
	        }
	        return goal;
	    }

	    // Bow movement has one owner: seek cover, assess it, then take a screened exit.
	    private static final int BOW_SEEK = 1;
	    private static final int BOW_COVER = 2;
	    private static final int BOW_FLANK = 3;
	    private static final int BOW_COMMIT_TICKS = 45;
	    private static final int BOW_COOLDOWN_TICKS = 120;

	    private static boolean bowDrawn(Player player) {
	        return player != null && player.isAlive() && !player.isCreative() && !player.isSpectator()
	            && player.isUsingItem() && player.getUseItem().is(Items.BOW);
	    }

	    public static boolean bowCommitActive(Mob self, Player player) {
	        return self.level() instanceof ServerLevel level && bowDrawn(player)
	            && level.getGameTime() < self.getPersistentData().getLongOr("hive_bow_commit_until", 0)
	            && player.getUUID().toString().equals(self.getPersistentData().getStringOr("hive_bow_target", ""));
	    }

	    private static void commitBow(Mob self, Player player, long now) {
	        int previousState = self.getPersistentData().getIntOr("hive_bow_state", 0);
	        self.getPersistentData().putInt("hive_bow_state", 0);
	        self.getPersistentData().putLong("hive_bow_commit_until", now + BOW_COMMIT_TICKS);
	        self.getPersistentData().putLong("hive_bow_cooldown", now + BOW_COOLDOWN_TICKS);
	        self.getPersistentData().putString("hive_bow_target", player.getUUID().toString());
	        self.getPersistentData().putLong("hive_bow_pressure_until", 0);
	        if (previousState != BOW_FLANK && self.level() instanceof ServerLevel level
	                && bowDrawn(player) && facing(self, player) >= 0.70
	                && blockedShot(self, player, level, self.position())) {
	            Vec3 exit = pressureExit(self, player, level);
	            if (exit != null) {
	                self.getPersistentData().putDouble("hive_bow_pressure_x", exit.x);
	                self.getPersistentData().putDouble("hive_bow_pressure_y", exit.y);
	                self.getPersistentData().putDouble("hive_bow_pressure_z", exit.z);
	                self.getPersistentData().putLong("hive_bow_pressure_until", now + BOW_COMMIT_TICKS);
	            }
	        }
	    }

	    private static Vec3 bowPoint(Mob self) {
	        return new Vec3(self.getPersistentData().getDoubleOr("hive_bow_x", 0.0),
	            self.getPersistentData().getDoubleOr("hive_bow_y", 0.0), self.getPersistentData().getDoubleOr("hive_bow_z", 0.0));
	    }

	    /** Check the whole actual collision box, including headroom and solid footing. */
	    private static boolean clearStandingRoom(Mob self, ServerLevel level, BlockPos pos) {
	        Vec3 point = new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	        return !level.getBlockState(pos.below()).getCollisionShape(level, pos.below()).isEmpty()
	            && level.getFluidState(pos).isEmpty() && level.getFluidState(pos.above()).isEmpty()
	            && level.noCollision(self, self.getBoundingBox().move(point.subtract(self.position())));
	    }

	    private static boolean blockedShot(Mob self, Player player, ServerLevel level, Vec3 point) {
	        double height = self.getBbHeight();
	        double width = self.getBbWidth() * 0.35;
	        Vec3 toward = point.subtract(player.position());
	        double span = Math.hypot(toward.x, toward.z);
	        if (span < 0.01) return false;
	        double lateralX = -toward.z / span * width;
	        double lateralZ = toward.x / span * width;
	        Vec3 start = player.getEyePosition();
	        Vec3[] silhouette = {
	            point.add(0, Math.min(height - 0.12, self.getEyeHeight()), 0),
	            point.add(0, height * 0.55, 0),
	            point.add(0, 0.22, 0),
	            point.add(lateralX, height * 0.55, lateralZ),
	            point.add(-lateralX, height * 0.55, -lateralZ)
	        };
	        for (Vec3 sample : silhouette) {
	            HitResult hit = level.clip(new ClipContext(start, sample,
	                ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, self));
	            if (hit.getType() != HitResult.Type.BLOCK
	                    || hit.getLocation().distanceToSqr(sample) < 0.16) return false;
	        }
	        return true;
	    }

	    /** Scan local terrain; discard the last shelter when changing cover. */
	    private static Vec3 nearbyCover(Mob self, Player player, ServerLevel level, Vec3 avoid) {
	        Vec3 origin = self.position();
	        for (int radius = 2; radius <= 8; radius++) {
	            for (int vertical : new int[] {0, -1, 1}) {
	                for (int direction = 0; direction < 16; direction++) {
	                    double angle = direction * Math.PI / 8.0;
	                    BlockPos pos = BlockPos.containing(origin.x + Math.cos(angle) * radius,
	                        origin.y + vertical, origin.z + Math.sin(angle) * radius);
	                    Vec3 point = new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	                    if (avoid != null && point.distanceToSqr(avoid) < 9.0) continue;
	                    if (!blockedShot(self, player, level, point)
	                            || !clearStandingRoom(self, level, pos)) continue;
	                    net.minecraft.world.level.pathfinder.Path path = self.getNavigation().createPath(pos, 0);
	                    if (path != null && path.canReach()) return point;
	                }
	            }
	        }
	        return null;
	    }

	    private static Vec3 shortDodge(Mob self, Player player, ServerLevel level) {
	        Vec3 away = self.position().subtract(player.position());
	        double span = Math.hypot(away.x, away.z);
	        if (span < 0.01) return null;
	        double side = (self.getUUID().hashCode() & 1) == 0 ? 1.0 : -1.0;
	        for (int attempt = 0; attempt < 2; attempt++, side = -side) {
	            BlockPos pos = BlockPos.containing(self.getX() - away.z / span * side * 3,
	                self.getY(), self.getZ() + away.x / span * side * 3);
	            if (!clearStandingRoom(self, level, pos)) continue;
	            net.minecraft.world.level.pathfinder.Path path = self.getNavigation().createPath(pos, 0);
	            if (path != null && path.canReach())
	                return new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	        }
	        return null;
	    }

	    /** Choose a flank with a screened approach, favoring a different exit next time. */
	    private static Vec3 screenedFlank(Mob self, Player player, ServerLevel level) {
	        Vec3 toward = player.position().subtract(self.position());
	        double span = Math.hypot(toward.x, toward.z);
	        if (span < 1.0) return null;
	        Vec3 along = new Vec3(toward.x / span, 0, toward.z / span);
	        Vec3 lateral = new Vec3(-along.z, 0, along.x);
	        int lastSide = self.getPersistentData().getIntOr("hive_bow_last_side", 0);
	        int variation = self.getUUID().hashCode() ^ player.getUUID().hashCode()
	            ^ (int) (level.getGameTime() / 80);
	        int preferred = lastSide != 0 && (variation & 3) != 0
	            ? -lastSide : (variation & 4) == 0 ? 1 : -1;
	        for (int order = 0; order < 2; order++) {
	            int side = order == 0 ? preferred : -preferred;
	            for (int offset = 2; offset <= 6; offset++) {
	                for (int advance = 1; advance <= 3; advance++) {
	                    Vec3 desired = self.position().add(lateral.scale(side * offset)).add(along.scale(advance));
	                    for (int vertical : new int[] {0, -1, 1}) {
	                        BlockPos pos = BlockPos.containing(desired.x, desired.y + vertical, desired.z);
	                        Vec3 point = new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	                        if (point.distanceToSqr(player.position()) >= self.distanceToSqr(player) - 2.0
	                                || !blockedShot(self, player, level, point)
	                                || !clearStandingRoom(self, level, pos)) continue;
	                        boolean screened = true;
	                        for (double step : new double[] {0.25, 0.5, 0.75}) {
	                            if (!blockedShot(self, player, level, self.position().lerp(point, step))) {
	                                screened = false;
	                                break;
	                            }
	                        }
	                        if (!screened) continue;
	                        net.minecraft.world.level.pathfinder.Path path = self.getNavigation().createPath(pos, 0);
	                        if (path != null && path.canReach() && mostlyScreenedPath(self, player, level, path)) {
	                            self.getPersistentData().putInt("hive_bow_last_side", side);
	                            return point;
	                        }
	                    }
	                }
	            }
	        }
	        return null;
	    }

	    /** The actual navigation path matters; a straight line can hide an exposed detour. */
	    private static boolean mostlyScreenedPath(Mob self, Player player, ServerLevel level,
	            net.minecraft.world.level.pathfinder.Path path) {
	        int visible = 0;
	        int checked = 0;
	        int stride = Math.max(1, path.getNodeCount() / 7);
	        for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i += stride) {
	            net.minecraft.world.level.pathfinder.Node node = path.getNode(i);
	            Vec3 point = new Vec3(node.x + 0.5, node.y, node.z + 0.5);
	            if (!blockedShot(self, player, level, point)) visible++;
	            checked++;
	        }
	        return checked > 0 && visible <= Math.max(1, checked / 4);
	    }

	    /** If safety is impossible, leave cover around a side rather than retracing entry. */
	    private static Vec3 pressureExit(Mob self, Player player, ServerLevel level) {
	        Vec3 toward = player.position().subtract(self.position());
	        double span = Math.hypot(toward.x, toward.z);
	        if (span < 1.0) return null;
	        Vec3 along = new Vec3(toward.x / span, 0, toward.z / span);
	        Vec3 lateral = new Vec3(-along.z, 0, along.x);
	        int previous = self.getPersistentData().getIntOr("hive_bow_last_side", 0);
	        int variation = self.getUUID().hashCode() ^ player.getUUID().hashCode()
	            ^ (int) (level.getGameTime() / 80);
	        int first = previous != 0 && (variation & 3) != 0
	            ? -previous : (variation & 4) == 0 ? 1 : -1;
	        for (int order = 0; order < 2; order++) {
	            int side = order == 0 ? first : -first;
	            for (int offset = 3; offset <= 6; offset++) {
	                Vec3 goal = self.position().add(lateral.scale(side * offset)).add(along.scale(2));
	                for (int vertical : new int[] {0, -1, 1}) {
	                    BlockPos pos = BlockPos.containing(goal.x, goal.y + vertical, goal.z);
	                    if (!clearStandingRoom(self, level, pos)) continue;
	                    net.minecraft.world.level.pathfinder.Path path = self.getNavigation().createPath(pos, 0);
	                    if (path == null || !path.canReach()) continue;
	                    boolean usesSide = false;
	                    for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i++) {
	                        net.minecraft.world.level.pathfinder.Node node = path.getNode(i);
	                        Vec3 step = new Vec3(node.x + 0.5, node.y, node.z + 0.5).subtract(self.position());
	                        if (step.dot(lateral) * side > 1.0) { usesSide = true; break; }
	                    }
	                    if (!usesSide) continue;
	                    self.getPersistentData().putInt("hive_bow_last_side", side);
	                    return new Vec3(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5);
	                }
	            }
	        }
	        return null;
	    }

	    private static void bowDestination(Mob self, Vec3 point, int state, long now) {
	        self.getPersistentData().putInt("hive_bow_state", state);
	        self.getPersistentData().putDouble("hive_bow_x", point.x);
	        self.getPersistentData().putDouble("hive_bow_y", point.y);
	        self.getPersistentData().putDouble("hive_bow_z", point.z);
	        self.getPersistentData().putDouble("hive_bow_progress", self.distanceToSqr(point));
	        self.getPersistentData().putInt("hive_bow_stalls", 0);
	        self.getPersistentData().putLong("hive_bow_path_check", now + 12);
	    }

	    /** The main AI yields navigation to this state before its watched/phase rules. */
	    public static boolean handleBowThreat(Mob self, Player player, int mineProgress, double speed) {
	        if (!(self.level() instanceof ServerLevel level) || frozen(self) || mineProgress != 0
	                || player == null || self.getTarget() != player) return false;
	        long now = level.getGameTime();
	        int state = self.getPersistentData().getIntOr("hive_bow_state", 0);
	        if (!bowDrawn(player)) {
	            if (state != 0) commitBow(self, player, now);
	            return false;
	        }
	        if (player.getMainHandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()
	                || player.getOffhandItem().getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) return false;
	        if (state != 0 && !player.getUUID().toString().equals(self.getPersistentData().getStringOr("hive_bow_target", ""))) {
	            self.getPersistentData().putInt("hive_bow_state", 0);
	            state = 0;
	        }
	        if (state == 0) {
	            if (self.getPersistentData().getLongOr("hive_bow_cooldown", 0) > now || self.distanceToSqr(player) < 16
	                    || self.distanceToSqr(player) > 28 * 28 || player.getTicksUsingItem() < 4
	                    || facing(self, player) < 0.84 || !self.hasLineOfSight(player)) return false;
	            Vec3 lastCover = self.getPersistentData().getIntOr("hive_bow_has_last_cover", 0) != 0
	                ? new Vec3(self.getPersistentData().getDoubleOr("hive_bow_last_cover_x", 0.0), self.getPersistentData().getDoubleOr("hive_bow_last_cover_y", 0.0), self.getPersistentData().getDoubleOr("hive_bow_last_cover_z", 0.0)) : null;
	            Vec3 point = nearbyCover(self, player, level, lastCover);
	            if (point == null && lastCover != null) point = nearbyCover(self, player, level, null);
	            boolean hasCover = point != null;
	            if (!hasCover) point = shortDodge(self, player, level);
	            if (point == null) {
	                commitBow(self, player, now);
	                return false;
	            }
	            self.getPersistentData().putString("hive_bow_target", player.getUUID().toString());
	            self.getPersistentData().putInt("hive_bow_changes", 0);
	            self.getPersistentData().putInt("hive_bow_failed_choices", 0);
	            self.getPersistentData().putInt("hive_bow_cover", hasCover ? 1 : 0);
	            bowDestination(self, point, BOW_SEEK, now);
	            state = BOW_SEEK;
	        }

	        if (self.distanceToSqr(player) < 9) {
	            commitBow(self, player, now);
	            return false;
	        }
	        Vec3 point = bowPoint(self);
	        boolean protectedNow = blockedShot(self, player, level, self.position());
	        if (state == BOW_SEEK && protectedNow) {
	            bowDestination(self, self.position(), BOW_COVER, now);
	            self.getPersistentData().putInt("hive_bow_has_last_cover", 1);
	            self.getPersistentData().putDouble("hive_bow_last_cover_x", self.getX());
	            self.getPersistentData().putDouble("hive_bow_last_cover_y", self.getY());
	            self.getPersistentData().putDouble("hive_bow_last_cover_z", self.getZ());
	            state = BOW_COVER;
	        }
	        if (state == BOW_FLANK && !protectedNow && facing(self, player) >= 0.84
	                && self.getPersistentData().getIntOr("hive_bow_changes", 0) < 2 && now >= self.getPersistentData().getLongOr("hive_bow_next_choice", 0)) {
	            self.getPersistentData().putLong("hive_bow_next_choice", now + 12);
	            Vec3 other = nearbyCover(self, player, level, point);
	            if (other != null) {
	                self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getIntOr("hive_bow_changes", 0) + 1);
	                bowDestination(self, other, BOW_SEEK, now);
	                state = BOW_SEEK;
	                point = other;
	            }
	        }
	        if (state == BOW_SEEK || state == BOW_FLANK) {
	            if (state == BOW_FLANK && self.distanceToSqr(point) < 2.25) {
	                commitBow(self, player, now);
	                return false;
	            }
	            if (state == BOW_SEEK && self.distanceToSqr(point) < 2.25 && !protectedNow) {
	                // The spot was screened when picked but the player moved: reconsider.
	                if (self.getPersistentData().getIntOr("hive_bow_changes", 0) >= 2) {
	                    commitBow(self, player, now);
	                    return false;
	                }
	                Vec3 other = nearbyCover(self, player, level, point);
	                if (other == null) {
	                    commitBow(self, player, now);
	                    return false;
	                }
	                self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getIntOr("hive_bow_changes", 0) + 1);
	                bowDestination(self, other, BOW_SEEK, now);
	            }
	            if (now >= self.getPersistentData().getLongOr("hive_bow_path_check", 0)) {
	                double remaining = self.distanceToSqr(bowPoint(self));
	                int stalls = remaining >= self.getPersistentData().getDoubleOr("hive_bow_progress", 0.0) - 0.4 ? self.getPersistentData().getIntOr("hive_bow_stalls", 0) + 1 : 0;
	                self.getPersistentData().putInt("hive_bow_stalls", stalls);
	                self.getPersistentData().putDouble("hive_bow_progress", remaining);
	                self.getPersistentData().putLong("hive_bow_path_check", now + 12);
	                if (stalls >= 2) {
	                    if (state == BOW_SEEK && self.getPersistentData().getIntOr("hive_bow_changes", 0) < 2) {
	                        Vec3 other = nearbyCover(self, player, level, bowPoint(self));
	                        if (other != null) {
	                            self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getIntOr("hive_bow_changes", 0) + 1);
	                            bowDestination(self, other, BOW_SEEK, now);
	                        } else {
	                            commitBow(self, player, now);
	                            return false;
	                        }
	                    } else {
	                        commitBow(self, player, now);
	                        return false;
	                    }
	                }
	            }
	        } else if (state == BOW_COVER) {
	            if (facing(self, player) < 0.70) {
	                commitBow(self, player, now); // The archer looks away: exploit the opening.
	                return false;
	            }
	            if (!protectedNow) {
	                Vec3 other = self.getPersistentData().getIntOr("hive_bow_changes", 0) < 2 ? nearbyCover(self, player, level, self.position()) : null;
	                if (other == null) {
	                    commitBow(self, player, now);
	                    return false;
	                }
	                self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getIntOr("hive_bow_changes", 0) + 1);
	                bowDestination(self, other, BOW_SEEK, now);
	            } else if (now >= self.getPersistentData().getLongOr("hive_bow_next_choice", 0)) {
	                self.getPersistentData().putLong("hive_bow_next_choice", now + 12 + (self.getId() & 7));
	                Vec3 flank = screenedFlank(self, player, level);
	                if (flank != null) {
	                    bowDestination(self, flank, BOW_FLANK, now);
	                } else {
	                    int failures = self.getPersistentData().getIntOr("hive_bow_failed_choices", 0) + 1;
	                    self.getPersistentData().putInt("hive_bow_failed_choices", failures);
	                    Vec3 other = failures == 1 && self.getPersistentData().getIntOr("hive_bow_changes", 0) < 2
	                        ? nearbyCover(self, player, level, self.position()) : null;
	                    if (other != null) {
	                        self.getPersistentData().putInt("hive_bow_changes", self.getPersistentData().getIntOr("hive_bow_changes", 0) + 1);
	                        bowDestination(self, other, BOW_SEEK, now);
	                    } else if (failures >= 2) {
	                        commitBow(self, player, now); // No safe route: pressure instead of camping.
	                        return false;
	                    }
	                }
	            }
	        }
	        setSpeed(self, speed);
	        self.getPersistentData().putLong("hive_bow_post_time", now);
	        return true;
	    }

	    /** Preserve ordinary movement for lone members and for non-player targets. */
	    public static Vec3 approach(Mob self, Entity target) {
	        Vec3 direct = target.position();
	        if (!(target instanceof Player player) || !(self.level() instanceof ServerLevel level)
	                || frozen(self)) return direct;
	        Vec3 shield = shieldRoute(self, player, level);
	        if (shield != null) return shield;
	        if (bowCommitActive(self, player)) {
            if (self.getPersistentData().getLongOr("hive_bow_pressure_until", 0) > level.getGameTime()) {
                Vec3 exit = new Vec3(self.getPersistentData().getDoubleOr("hive_bow_pressure_x", 0.0), self.getPersistentData().getDoubleOr("hive_bow_pressure_y", 0.0), self.getPersistentData().getDoubleOr("hive_bow_pressure_z", 0.0));
                if (self.distanceToSqr(exit) > 2.25) return exit;
            }
            return direct;
        }
	        if (!current(self, player, level) || self.distanceToSqr(player) < 16) return direct;
	        if (self.getPersistentData().getLongOr("hive_commit_until", 0) > level.getGameTime()) return direct;
	        long huntUntil = self.getPersistentData().getLongOr("hive_engage_until", 0);
	        if (huntUntil > level.getGameTime() && huntUntil - level.getGameTime() < 110) {
	            self.getPersistentData().putLong("hive_commit_until", level.getGameTime() + 70);
	            return direct;
	        }

	        long cachedUntil = self.getPersistentData().getLongOr("hive_route_until", 0);
	        if (cachedUntil > level.getGameTime()) {
	            Vec3 cachedPoint = new Vec3(self.getPersistentData().getDoubleOr("hive_route_x", 0.0),
	                self.getPersistentData().getDoubleOr("hive_route_y", 0.0),
	                self.getPersistentData().getDoubleOr("hive_route_z", 0.0));
	            if (self.distanceToSqr(cachedPoint) < 9.0) {
	                self.getPersistentData().putLong("hive_commit_until", level.getGameTime() + 70);
	                return direct;
	            }
	            return cachedPoint;
	        }

	        List<Mob> team = new ArrayList<>();
	        team.add(self);
	        for (Mob e : level.getEntitiesOfClass(Mob.class, self.getBoundingBox().inflate(24, 12, 24),
	                other -> other != self && other.isAlive() && member(other) && !frozen(other))) {
	            if (current(e, player, level) && e.distanceToSqr(player) <= 32 * 32
	                    && e.getAttributes().hasAttribute(Attributes.MOVEMENT_SPEED)
	                    && e.getAttribute(Attributes.MOVEMENT_SPEED).getBaseValue() > 0.01) team.add(e);
	        }
	        if (team.size() < 2) return direct;

	        // Tank takes pressure when present; otherwise the closest member does.
	        Mob pursuer = team.stream().min(Comparator.comparingDouble(e ->
	            e.distanceToSqr(player) - (e instanceof PetrifiedLogSplinterEntity ? 100.0 : 0.0))).orElse(self);
	        if (self == pursuer) return direct;
	        team.remove(pursuer);
	        team.sort(Comparator.comparing(Entity::getUUID));
	        int index = team.indexOf(self);
	        if (index < 0) return direct;

	        Vec3 facing = player.getLookAngle();
	        double length = Math.hypot(facing.x, facing.z);
	        if (length < 0.01) return direct;
	        double fx = facing.x / length;
	        double fz = facing.z / length;
	        double side = index % 2 == 0 ? 1.0 : -1.0;
	        Vec3 approach;
	        if (index >= 2 && player.getDeltaMovement().horizontalDistanceSqr() > 0.003) {
	            Vec3 velocity = player.getDeltaMovement();
	            approach = direct.add(Math.max(-6, Math.min(6, velocity.x * 12)) + side * fz * 3,
	                0, Math.max(-6, Math.min(6, velocity.z * 12)) - side * fx * 3);
	        } else {
	            approach = direct.add(-fx * 3 + side * fz * 6, 0, -fz * 3 - side * fx * 6);
	        }
	        self.getPersistentData().putLong("hive_route_until", level.getGameTime() + 10);
	        self.getPersistentData().putDouble("hive_route_x", approach.x);
	        self.getPersistentData().putDouble("hive_route_y", approach.y);
	        self.getPersistentData().putDouble("hive_route_z", approach.z);
	        if (self.distanceToSqr(approach) < 9.0) {
	            self.getPersistentData().putLong("hive_commit_until", level.getGameTime() + 70);
	            return direct;
	        }
	        return approach;
	    }

	    public static Vec3 stalkPoint(Mob self, Entity target) {
	        Vec3 approach = approach(self, target);
	        return approach.distanceToSqr(target.position()) > 1 ? approach : null;
	    }
	}
