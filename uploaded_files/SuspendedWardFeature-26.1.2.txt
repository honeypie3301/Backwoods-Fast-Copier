package net.mcreator.thebackwoods.world.feature;

/** Chunk-local, world-seeded suspended ward; every solid part joins a tower or plate. */
public class SuspendedWardFeature extends net.minecraft.world.level.levelgen.feature.Feature<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> {
    public static boolean ENABLE_AUTO_DATAPACK_GEN = true;
    public static boolean ENABLE_FAST_TERRAIN_GEN = true; // Single-biome Ward only; avoids generating caves just to erase them.
    public static String DIMENSION_ID = "the_backwoods:the_crossworks";
    public static String BIOME_ID = "the_backwoods:dustbound_galleries";
    public static int CELL_SIZE = 48;
    public static int TIER_SPACING = 18;
    public static int FLOOR_SPACING = 6;
    public static double SPAWNER_BUILDING_CHANCE = 40.0; // Percent of buildings eligible for floor rolls.
    public static double SPAWNER_FLOOR_CHANCE = 20.0; // Percent per floor; one centered block per successful roll.
    public static int BOTTOM_PLATE_OFFSET = 5; // Vanilla bedrock occupies offsets 0..4; never overwritten.
    public static int TOP_PLATE_THICKNESS = 2;
    public static boolean ENABLE_CEILING_PLATE = false;
    // Dimension height is data-driven and cannot be extended after a level has loaded.
    public static boolean ENABLE_AUTO_WORLD_HEIGHT_GEN = false;
    public static int GENERATED_WORLD_HEIGHT = 768; // Multiple of 16; opt in before creating a new world.
    private static final String ID = "the_backwoods:suspended_ward";
    private static final org.slf4j.Logger LOGGER = com.mojang.logging.LogUtils.getLogger();
    private static final byte AIR=0, PLANK=1, LOG=2, GLASS=3, FENCE=4, LADDER=5, SPAWNER=6;
    private static final int PAD=20;

    public SuspendedWardFeature() { this(net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration.CODEC); }
    public SuspendedWardFeature(com.mojang.serialization.Codec<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> codec) { super(codec); }

    @net.neoforged.fml.common.EventBusSubscriber(modid="the_backwoods")
    public static final class Registration {
        @net.neoforged.bus.api.SubscribeEvent
        public static void register(net.neoforged.neoforge.registries.RegisterEvent event) {
            event.register(net.minecraft.core.registries.Registries.FEATURE, helper ->
                helper.register(net.minecraft.resources.Identifier.parse(ID), new SuspendedWardFeature()));
            if (event.getRegistryKey().equals(net.minecraft.core.registries.Registries.FEATURE)) autoGenerateDataPackJsons();
        }
    }

    @Override public boolean place(net.minecraft.world.level.levelgen.feature.FeaturePlaceContext<net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration> context) {
        var level=context.level();
        if (!level.getLevel().dimension().identifier().toString().equals(DIMENSION_ID)) return false;
        int min=level.getMinY(), max=level.getMaxY()+1; // Modern getMaxY() is inclusive.
        if (CELL_SIZE<40 || CELL_SIZE>96 || TIER_SPACING<12 || TIER_SPACING>48 || FLOOR_SPACING<4
                || TIER_SPACING%FLOOR_SPACING!=0 || BOTTOM_PLATE_OFFSET<5 || TOP_PLATE_THICKNESS<1 || TOP_PLATE_THICKNESS>8)
            throw new IllegalStateException("Invalid SuspendedWardFeature geometry settings");
        if(!Double.isFinite(SPAWNER_BUILDING_CHANCE)||SPAWNER_BUILDING_CHANCE<0||SPAWNER_BUILDING_CHANCE>100
                ||!Double.isFinite(SPAWNER_FLOOR_CHANCE)||SPAWNER_FLOOR_CHANCE<0||SPAWNER_FLOOR_CHANCE>100)
            throw new IllegalStateException("SuspendedWardFeature spawner chances must be percentages from 0 to 100");
        int floor=min+BOTTOM_PLATE_OFFSET, roof=ENABLE_CEILING_PLATE?max-TOP_PLATE_THICKNESS:max;
        if (roof-floor<24) return false;
        int baseX=context.origin().getX()&~15, baseZ=context.origin().getZ()&~15;
        var chunk=level.getChunk(baseX>>4,baseZ>>4,net.minecraft.world.level.chunk.status.ChunkStatus.EMPTY,false);
        if(chunk==null) return false; // Never request or create an unloaded chunk.
        byte[] plan=plan(baseX,baseZ,floor,roof,max,level.getSeed());
        // Only this decorator's origin chunk is accessed; padded planning never touches the world.
        var pos=new net.minecraft.core.BlockPos.MutableBlockPos();
        var plank=net.minecraft.world.level.block.Blocks.OAK_PLANKS.defaultBlockState();
        var log=net.minecraft.world.level.block.Blocks.OAK_LOG.defaultBlockState();
        var glass=net.minecraft.world.level.block.Blocks.GLASS.defaultBlockState();
        var spawner=net.minecraft.core.registries.BuiltInRegistries.BLOCK
            .getOptional(net.minecraft.resources.Identifier.parse("the_backwoods:blindspot_splinter_spawner"))
            .orElseThrow(()->new IllegalStateException("Missing blindspot_splinter_spawner block")).defaultBlockState();
        var air=net.minecraft.world.level.block.Blocks.AIR.defaultBlockState();
        var ladder=net.minecraft.world.level.block.Blocks.LADDER.defaultBlockState()
            .setValue(net.minecraft.world.level.block.LadderBlock.FACING,net.minecraft.core.Direction.EAST);
        var fences=new net.minecraft.world.level.block.state.BlockState[16];
        for(int mask=0;mask<16;mask++) fences[mask]=net.minecraft.world.level.block.Blocks.OAK_FENCE.defaultBlockState()
            .setValue(net.minecraft.world.level.block.FenceBlock.NORTH,(mask&1)!=0)
            .setValue(net.minecraft.world.level.block.FenceBlock.EAST,(mask&2)!=0)
            .setValue(net.minecraft.world.level.block.FenceBlock.SOUTH,(mask&4)!=0)
            .setValue(net.minecraft.world.level.block.FenceBlock.WEST,(mask&8)!=0);
        var sections=chunk.getSections();var initiallyEmpty=new boolean[sections.length];
        for(int i=0;i<sections.length;i++) initiallyEmpty[i]=sections[i].hasOnlyAir();
        for(int y=floor;y<max;y++) {
          int sectionIndex=chunk.getSectionIndex(y);var section=sections[sectionIndex];
          for(int z=2;z<=17;z++) for(int x=2;x<=17;x++) {
            int index=index(x,y-floor,z); byte kind=plan[index];
            if(kind==AIR&&initiallyEmpty[sectionIndex]) continue;
            var old=section.getBlockState(x-2,y&15,z-2);
            if(old.is(net.minecraft.world.level.block.Blocks.BEDROCK)) continue;
            net.minecraft.world.level.block.state.BlockState state;
            if(kind==FENCE) {
                int mask=(solid(plan[index-PAD])?1:0)|(solid(plan[index+1])?2:0)
                    |(solid(plan[index+PAD])?4:0)|(solid(plan[index-1])?8:0);
                state=fences[mask];
            } else state=switch(kind) {case PLANK->plank;case LOG->log;case GLASS->glass;case LADDER->ladder;case SPAWNER->spawner;default->air;};
            if(old!=state) {
                pos.set(baseX+x-2,y,baseZ+z-2);
                chunk.setBlockState(pos,state,0);
            }
          }
        }
        return true;
    }

    private static boolean solid(byte kind) { return kind==PLANK || kind==LOG || kind==FENCE; }
    private static int index(int x,int y,int z) { return y*PAD*PAD+z*PAD+x; }
    private static long hash(long seed,int x,int z) {
        long v=seed^((long)x*0x632BE59BD9B4E019L)^((long)z*0x9E3779B97F4A7C15L);
        v=(v^(v>>>30))*0xBF58476D1CE4E5B9L;v=(v^(v>>>27))*0x94D049BB133111EBL;return v^(v>>>31);
    }
    private record Tower(int x,int z,int halfX,int halfZ) {}
    private record Bridge(Tower a,Tower b) {}
    private record LayoutKey(long seed,int x,int z,int cellSize) {}
    private record Layout(Tower[][] towers,java.util.List<Bridge> connections) {}
    private static final java.util.Map<LayoutKey,Layout> LAYOUT_CACHE=new java.util.LinkedHashMap<>(64,0.75f,true) {
        @Override protected boolean removeEldestEntry(java.util.Map.Entry<LayoutKey,Layout> entry) {return size()>64;}
    };
    private static Layout layout(long seed,int cellX,int cellZ) {
        var key=new LayoutKey(seed,cellX,cellZ,CELL_SIZE);
        synchronized(LAYOUT_CACHE) {var cached=LAYOUT_CACHE.get(key);if(cached!=null) return cached;}
        var towers=new Tower[11][11];
        for(int z=0;z<11;z++) for(int x=0;x<11;x++) towers[z][x]=tower(seed,cellX+x-5,cellZ+z-5);
        var built=new Layout(towers,bridges(towers,cellX*CELL_SIZE,cellZ*CELL_SIZE,seed));
        synchronized(LAYOUT_CACHE) {
            var cached=LAYOUT_CACHE.get(key);if(cached!=null) return cached;
            LAYOUT_CACHE.put(key,built);return built;
        }
    }
    private static Tower candidate(long seed,int cx,int cz) {
        long h=hash(seed,cx,cz);
        return new Tower(cx*CELL_SIZE+(int)Math.floorMod(h,CELL_SIZE),
            cz*CELL_SIZE+(int)Math.floorMod(h>>>8,CELL_SIZE),
            3+2*(int)Math.floorMod(h>>>16,3L),3+2*(int)Math.floorMod(h>>>24,3L));
    }
    private static Tower tower(long seed,int cx,int cz) {
        Tower a=candidate(seed,cx,cz);long priority=hash(seed^0xA24BAED4963EE407L,cx,cz);
        // Full-cell scatter, with deterministic local rejection instead of aligned rows.
        for(int z=cz-1;z<=cz+1;z++) for(int x=cx-1;x<=cx+1;x++) {
            if(x==cx&&z==cz) continue;
            Tower b=candidate(seed,x,z);int order=Long.compareUnsigned(hash(seed^0xA24BAED4963EE407L,x,z),priority);
            if(order>0 || (order==0&&(x>cx || (x==cx&&z>cz)))) continue;
            if(Math.abs(a.x-b.x)<a.halfX+b.halfX+12 && Math.abs(a.z-b.z)<a.halfZ+b.halfZ+12) return null;
        }
        return a;
    }
    private static java.util.List<Bridge> bridges(Tower[][] towers,int baseX,int baseZ,long seed) {
        var result=new java.util.LinkedHashSet<Bridge>();
        for(int z=2;z<=8;z++) for(int x=2;x<=8;x++) {
            Tower a=towers[z][x];if(a==null) continue;
            var nearby=new java.util.ArrayList<Tower>();
            for(int dz=-2;dz<=2;dz++) for(int dx=-2;dx<=2;dx++) {
                Tower b=towers[z+dz][x+dx];if(b!=null&&b!=a) nearby.add(b);
            }
            nearby.sort(java.util.Comparator.comparingLong(b->(long)(a.x-b.x)*(a.x-b.x)+(long)(a.z-b.z)*(a.z-b.z)));
            int count=Math.min(nearby.size(),2+(int)Math.floorMod(hash(seed,a.x,a.z),2L));
            for(int i=0;i<count;i++) {
                Tower b=nearby.get(i);int margin=CELL_SIZE/4+3;
                if(Math.max(a.x,b.x)+margin<baseX-2||Math.min(a.x,b.x)-margin>baseX+CELL_SIZE+17
                        ||Math.max(a.z,b.z)+margin<baseZ-2||Math.min(a.z,b.z)-margin>baseZ+CELL_SIZE+17) continue;
                result.add(a.x<b.x || (a.x==b.x&&a.z<b.z)?new Bridge(a,b):new Bridge(b,a));
            }
        }
        return new java.util.ArrayList<>(result);
    }
    private static void route(boolean[] deck,int baseX,int baseZ,Bridge bridge,long seed,int tier) {
        Tower a=bridge.a,b=bridge.b;
        long h=hash(seed^((long)tier*0xD1B54A32D192ED03L),a.x^Integer.rotateLeft(b.x,16),a.z^Integer.rotateLeft(b.z,16));
        boolean alongX=Math.abs(a.x-b.x)>=Math.abs(a.z-b.z);
        int au=alongX?a.x:a.z,av=alongX?a.z:a.x,bu=alongX?b.x:b.z,bv=alongX?b.z:b.x;
        int ah=alongX?a.halfX:a.halfZ,bh=alongX?b.halfX:b.halfZ;
        int direction=bu>=au?1:-1,lo=au+direction*(ah+4),hi=bu-direction*(bh+4),span=Math.abs(hi-lo);
        // The dominant axis always has room because accepted towers have at least 12 blocks of clearance.
        if((hi-lo)*direction<0) {alongX=!alongX;au=alongX?a.x:a.z;av=alongX?a.z:a.x;bu=alongX?b.x:b.z;bv=alongX?b.z:b.x;
            ah=alongX?a.halfX:a.halfZ;bh=alongX?b.halfX:b.halfZ;direction=bu>=au?1:-1;
            lo=au+direction*(ah+4);hi=bu-direction*(bh+4);span=Math.abs(hi-lo);}
        int first=lo+direction*(int)Math.floorMod(h,span/2+1L);
        int last=hi-direction*(int)Math.floorMod(h>>>12,span/2+1L);
        int offset=CELL_SIZE/4,side=(av+bv)/2+(int)Math.floorMod(h>>>24,offset*2L+1)-offset;
        if((h&4)==0) {last=first;side=bv;} // Mix single elbows with three/four-turn paths.
        int[] us={au,first,first,last,last,bu},vs={av,av,side,side,bv,bv};
        for(int i=1;i<us.length;i++) {
            if(alongX) segment(deck,baseX,baseZ,us[i-1],vs[i-1],us[i],vs[i]);
            else segment(deck,baseX,baseZ,vs[i-1],us[i-1],vs[i],us[i]);
        }
    }
    private static byte[] plan(int baseX,int baseZ,int floor,int roof,int max,long seed) {
        int height=max-floor;byte[] blocks=new byte[PAD*PAD*height];
        boolean[] interior=new boolean[PAD*PAD];
        for(int z=0;z<PAD;z++) for(int x=0;x<PAD;x++) {
            blocks[index(x,0,z)]=PLANK;
            for(int y=roof;y<max;y++) blocks[index(x,y-floor,z)]=PLANK;
        }
        int cellX=Math.floorDiv(baseX,CELL_SIZE),cellZ=Math.floorDiv(baseZ,CELL_SIZE);
        var layout=layout(seed,cellX,cellZ);var towers=layout.towers;
        var connections=new java.util.ArrayList<Bridge>();int margin=CELL_SIZE/4+3;
        for(Bridge b:layout.connections) if(Math.max(b.a.x,b.b.x)+margin>=baseX-2&&Math.min(b.a.x,b.b.x)-margin<=baseX+17
                &&Math.max(b.a.z,b.b.z)+margin>=baseZ-2&&Math.min(b.a.z,b.b.z)-margin<=baseZ+17) connections.add(b);
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];if(t==null) continue;
            for(int z=0;z<PAD;z++) for(int x=0;x<PAD;x++) {
                int wx=baseX+x-2,wz=baseZ+z-2,dx=wx-t.x,dz=wz-t.z;
                if(Math.abs(dx)>t.halfX || Math.abs(dz)>t.halfZ) continue;
                interior[z*PAD+x]=true;
                boolean wallX=Math.abs(dx)==t.halfX,wallZ=Math.abs(dz)==t.halfZ;
                for(int y=floor+1;y<roof;y++) {
                    int layer=y-floor;byte kind=AIR;
                    if(wallX&&wallZ) kind=LOG;
                    else if(wallX||wallZ) {
                        int along=wallX?dz+t.halfZ:dx+t.halfX;
                        boolean window=layer%FLOOR_SPACING>=2&&layer%FLOOR_SPACING<=3
                            &&Math.floorMod(along,4)>=2;
                        kind=window?GLASS:PLANK;
                    } else if(layer%FLOOR_SPACING==0) kind=PLANK;
                    // The ladder stays against a continuous corner wall, clear of bridge doorways.
                    if(dx==-t.halfX+1&&dz==-t.halfZ+1) kind=LADDER;
                    blocks[index(x,layer,z)]=kind;
                }
            }
        }
        boolean[] deck=new boolean[PAD*PAD];
        for(int y=floor+TIER_SPACING;y+4<roof;y+=TIER_SPACING) {
            int tier=(y-floor)/TIER_SPACING;java.util.Arrays.fill(deck,false);
            for(Bridge bridge:connections) route(deck,baseX,baseZ,bridge,seed,tier);
            for(int z=0;z<PAD;z++) for(int x=0;x<PAD;x++) if(deck[z*PAD+x]) {
                blocks[index(x,y-floor,z)]=PLANK;
                // Join doors and junctions before adding edge rails; no fences across an intersection.
                for(int dy=1;dy<=3;dy++) if(blocks[index(x,y-floor+dy,z)]!=LADDER) blocks[index(x,y-floor+dy,z)]=AIR;
            }
            for(int z=1;z<PAD-1;z++) for(int x=1;x<PAD-1;x++) {
                int k=z*PAD+x;
                if(deck[k]&&!interior[k]&&(!deck[k-1]||!deck[k+1]||!deck[k-PAD]||!deck[k+PAD]
                        ||!deck[k-PAD-1]||!deck[k-PAD+1]||!deck[k+PAD-1]||!deck[k+PAD+1]))
                    blocks[index(x,y-floor+1,z)]=FENCE;
            }

        }
        // A passing bridge may cut a doorway near another tower's ladder; retain its backing wall.
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];if(t==null) continue;
            int x=t.x-t.halfX+1-baseX+2,z=t.z-t.halfZ+1-baseZ+2;
            if(x<1||x>=PAD||z<0||z>=PAD) continue;
            for(int y=1;y<height;y++) if(blocks[index(x,y,z)]==LADDER) blocks[index(x-1,y,z)]=PLANK;
        }
        for(int cz=3;cz<=7;cz++) for(int cx=3;cx<=7;cx++) {
            Tower t=towers[cz][cx];
            if(t==null||!roll(seed^0xDB4F0B9175AE2165L,t.x,t.z,SPAWNER_BUILDING_CHANCE)) continue;
            int x=t.x-baseX+2,z=t.z-baseZ+2;
            if(x<0||x>=PAD||z<0||z>=PAD) continue;
            for(int y=floor+1;y+2<roof;y+=FLOOR_SPACING) {
                if(!roll(seed^0xBBE0563303A4615FL^((long)y*0x9E3779B97F4A7C15L),t.x,t.z,SPAWNER_FLOOR_CHANCE)) continue;
                int k=index(x,y-floor,z);
                if(blocks[k]==AIR&&blocks[k-PAD*PAD]==PLANK&&blocks[k+PAD*PAD]==AIR) blocks[k]=SPAWNER;
            }
        }
        return blocks;
    }
    private static boolean roll(long seed,int x,int z,double chance) {
        return (hash(seed,x,z)>>>11)*0x1.0p-53<chance/100.0;
    }
    private static void segment(boolean[] deck,int baseX,int baseZ,int ax,int az,int bx,int bz) {
        int loX=Math.max(0,Math.min(ax,bx)-1-baseX+2),hiX=Math.min(PAD-1,Math.max(ax,bx)+1-baseX+2);
        int loZ=Math.max(0,Math.min(az,bz)-1-baseZ+2),hiZ=Math.min(PAD-1,Math.max(az,bz)+1-baseZ+2);
        for(int z=loZ;z<=hiZ;z++) for(int x=loX;x<=hiX;x++) deck[z*PAD+x]=true;
    }

    public static void autoGenerateDataPackJsons() {
        if(!ENABLE_AUTO_DATAPACK_GEN) return;
        try {
            java.nio.file.Path current=java.nio.file.Path.of("").toAbsolutePath().normalize(),project=null;
            for(int up=0;up<3&&current!=null;up++,current=current.getParent()) {
                if((java.nio.file.Files.isRegularFile(current.resolve("build.gradle"))||java.nio.file.Files.isRegularFile(current.resolve("build.gradle.kts")))
                        &&java.nio.file.Files.isDirectory(current.resolve("src/main/resources"))) {project=current;break;}
            }
            if(project==null) return; // Release builds consume packaged data; never write into a player's instance.
            var resources=project.resolve("src/main/resources");
            // Dimension data lives in data/<namespace>/dimension/<path>.json.
            String[] dimension=DIMENSION_ID.split(":",2);
            var dimensionFile=resources.resolve("data/"+dimension[0]+"/dimension/"+dimension[1]+".json");
            com.google.gson.JsonObject dimensionJson=null;
            if(java.nio.file.Files.isRegularFile(dimensionFile)) {
                dimensionJson=com.google.gson.JsonParser.parseString(java.nio.file.Files.readString(dimensionFile)).getAsJsonObject();
            }
            var outputs=new java.util.LinkedHashMap<String,String>();
            // The first 1.21.1 Crossworks export accidentally selected the Sub Strata biome.
            if(dimensionJson!=null&&migrateBiome(dimensionJson))
                outputs.put(resources.relativize(dimensionFile).toString().replace('\\','/'),dimensionJson.toString());
            outputs.put("data/the_backwoods/worldgen/configured_feature/suspended_ward.json","{\"type\":\""+ID+"\",\"config\":{}}");
            outputs.put("data/the_backwoods/worldgen/placed_feature/suspended_ward_placed.json","{\"feature\":\""+ID+"\",\"placement\":[{\"type\":\"minecraft:biome\"}]}");
            outputs.put("data/the_backwoods/neoforge/biome_modifier/add_suspended_ward.json","{\"type\":\"neoforge:add_features\",\"biomes\":\""+BIOME_ID+"\",\"features\":\"the_backwoods:suspended_ward_placed\",\"step\":\"surface_structures\"}");
            if(ENABLE_AUTO_WORLD_HEIGHT_GEN) {
                if(dimensionJson==null) throw new IllegalStateException("Create the target dimension before generating its height data");
                if(GENERATED_WORLD_HEIGHT<64||GENERATED_WORLD_HEIGHT>4064||GENERATED_WORLD_HEIGHT%16!=0)
                    throw new IllegalStateException("World height must be a supported multiple of 16");
                if(!dimensionJson.get("type").getAsString().equals(DIMENSION_ID))
                    throw new IllegalStateException("Refusing to change a shared dimension type");
                var typeId=dimensionJson.get("type").getAsString().split(":",2);
                var typePath=resources.resolve("data/"+typeId[0]+"/dimension_type/"+typeId[1]+".json");
                var type=com.google.gson.JsonParser.parseString(java.nio.file.Files.readString(typePath)).getAsJsonObject();
                int minY=type.get("min_y").getAsInt();
                if(minY+GENERATED_WORLD_HEIGHT>2032) throw new IllegalStateException("Requested height exceeds Minecraft's upper bound");
                type.addProperty("height",GENERATED_WORLD_HEIGHT);type.addProperty("logical_height",GENERATED_WORLD_HEIGHT);
                var settings=dimensionJson.getAsJsonObject("generator").get("settings");
                if(!settings.isJsonObject()) throw new IllegalStateException("Height generation requires inline noise settings");
                settings.getAsJsonObject().getAsJsonObject("noise").addProperty("height",GENERATED_WORLD_HEIGHT);
                outputs.put(resources.relativize(typePath).toString().replace('\\','/'),type.toString());
                outputs.put(resources.relativize(dimensionFile).toString().replace('\\','/'),dimensionJson.toString());
            }
            if(ENABLE_FAST_TERRAIN_GEN&&dimensionJson!=null&&fastTerrain(dimensionJson)) {
                var backup=project.resolve(".suspended-ward-backups").resolve(resources.relativize(dimensionFile));
                if(!java.nio.file.Files.exists(backup)) {
                    java.nio.file.Files.createDirectories(backup.getParent());java.nio.file.Files.copy(dimensionFile,backup);
                }
                outputs.put(resources.relativize(dimensionFile).toString().replace(java.io.File.separatorChar,'/'),dimensionJson.toString());
            }
            for(var root:java.util.List.of(resources,project.resolve("build/resources/main"))) {
                if(!java.nio.file.Files.isDirectory(root)) continue;
                for(var output:outputs.entrySet()) write(root.resolve(output.getKey()),output.getValue()+"\n");
                removeLegacyTag(root);
            }
        } catch(java.io.IOException|RuntimeException failure) {LOGGER.error("Suspended Ward development data generation failed",failure);}
    }
    private static boolean fastTerrain(com.google.gson.JsonObject dimension) {
        if(!dimension.get("type").isJsonPrimitive()||!dimension.get("type").getAsString().equals(DIMENSION_ID)) return false;
        var generator=dimension.getAsJsonObject("generator");
        if(!generator.get("type").getAsString().equals("minecraft:noise")||!generator.get("settings").isJsonObject()) return false;
        var source=generator.getAsJsonObject("biome_source");String type=source.get("type").getAsString();
        if(type.equals("minecraft:fixed")) {if(!source.get("biome").getAsString().equals(BIOME_ID)) return false;}
        else if(type.equals("minecraft:multi_noise")&&source.has("biomes")) {
            for(var biome:source.getAsJsonArray("biomes")) if(!biome.getAsJsonObject().get("biome").getAsString().equals(BIOME_ID)) return false;
        } else return false; // Preserve climate sampling for other biomes and shared presets.
        var settings=generator.getAsJsonObject("settings");String before=settings.toString();
