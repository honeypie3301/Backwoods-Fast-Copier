    @SubscribeEvent
    public static void registerRenderers(EntityRenderersEvent.RegisterRenderers event) {
        EntityType<?> rawType = BuiltInRegistries.ENTITY_TYPE.get(Identifier.parse("the_backwoods:woodweaver"))
            .map(net.minecraft.core.Holder::value)
            .orElse(null);
        if (rawType != null) {
            @SuppressWarnings("unchecked")
            EntityType<WoodweaverEntity> entityType = (EntityType<WoodweaverEntity>) rawType;
            event.registerEntityRenderer(entityType, CustomWoodweaverRenderer::new);
        }
    }

            super(context, new AnimatedModel(context.bakeLayer(ModelWoodweaver.LAYER_LOCATION)), 3.0f);

        @Override
        public LivingEntityRenderState createRenderState() {
            return new WeaverRenderState();
        }

        @Override
        public void extractRenderState(WoodweaverEntity entity, LivingEntityRenderState state, float partialTicks) {
            super.extractRenderState(entity, state, partialTicks);
            if (state instanceof WeaverRenderState weaverState) weaverState.entity = entity;
            if (this.model instanceof AnimatedModel) {
                ((AnimatedModel) this.model).setEntity(entity);
            }
        }

        @Override
        public Identifier getTextureLocation(LivingEntityRenderState state) {
            return entityTexture;
        }

            private KeyframeAnimation safeBake(AnimationDefinition source) {
                var lookup = root.createPartLookup();
                var channels = new java.util.HashMap<>(source.boneAnimations());
                channels.keySet().removeIf(name -> lookup.apply(name) == null);
                if (source == WoodweaverAnimation.Orb) {
                    channels.remove("head"); // Continuous ambient orb loop must not force a permanent downward head pitch
                }
                return new AnimationDefinition(source.lengthInSeconds(), source.looping(), channels).bake(root);
            }
