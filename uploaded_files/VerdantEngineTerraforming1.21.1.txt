		if (level == null) return false;

	}

	private static final java.util.Map<ServerLevel, Set<ChunkPos>> PENDING_BIOME_CHUNKS_BY_LEVEL =
		java.util.Collections.synchronizedMap(new java.util.WeakHashMap<>());
	private static final java.util.Map<ServerLevel, Long> LAST_BIOME_RESEND_TICK_BY_LEVEL =
		java.util.Collections.synchronizedMap(new java.util.WeakHashMap<>());
	private static final long BIOME_RESEND_INTERVAL_TICKS = 5L;
	private static final int BIOME_RESEND_BATCH_SIZE = 48;

	private static Set<ChunkPos> getPendingBiomeChunks(ServerLevel level) {
		synchronized (PENDING_BIOME_CHUNKS_BY_LEVEL) {
			return PENDING_BIOME_CHUNKS_BY_LEVEL.computeIfAbsent(level, ignored -> new HashSet<>());
		}
	}

	/**
	 * Maximum radius for biome terraforming: 16 chunks radius = 256.0 blocks.
	 */
	public static double MAX_BIOME_RADIUS = 432.0;
	/**
	 * Configurable infection ratio threshold required in the 3x3 chunks to unlock biome conversion (default: 0.65 / 65%).
	 * Can be changed globally via INFECTION_RATIO_THRESHOLD or overridden per entity via NBT "verdant_biome_infection_ratio_threshold".
	 */
	public static double INFECTION_RATIO_THRESHOLD = 0.65;

	public static double getInfectionRatioThreshold(CompoundTag nbt) {
		if (nbt != null && nbt.contains("verdant_biome_infection_ratio_threshold")) {
			return nbt.getDouble("verdant_biome_infection_ratio_threshold");
		}
		return INFECTION_RATIO_THRESHOLD;
	}

	private static final int SINGLE_ENGINE_BIOME_QUART_BUDGET = 48;
	private static final int MULTI_ENGINE_BIOME_QUART_BUDGET = 64;

	/**
	 * Smooth, gradual biome terraforming to the 'wood_plains' biome.
	 * Activates only when the 3x3 chunks around the World Engine reach the required infection ratio.
	 * Once activated, gradually expands the biome radius outwards from the center in smooth quart waves up to 16 chunks.
	 */
	public static void processBiomeTerraforming(ServerLevel level, CompoundTag nbt, double cx, double gy, double cz, double curInfectionRadius) {
		if (level == null || nbt == null) return;
		if (!isDimensionAllowedForTerraforming(level)) return;

		Holder<Biome> plainsBiome = getWoodPlainsHolder(level);
		Holder<Biome> deepBiome = getDeepBackwoodsHolder(level);
		Holder<Biome> thicketBiome = getTheThicketHolder(level);
		if (!isRegisteredBiomeHolder(level, plainsBiome)) plainsBiome = null;
		if (!isRegisteredBiomeHolder(level, deepBiome)) deepBiome = null;
		if (!isRegisteredBiomeHolder(level, thicketBiome)) thicketBiome = null;
		if (plainsBiome == null && deepBiome == null && thicketBiome == null) return;
		Set<ChunkPos> pendingBiomeChunks = getPendingBiomeChunks(level);

		int centerChunkX = Mth.floor(cx) >> 4;
		int centerChunkZ = Mth.floor(cz) >> 4;

		// Freeze-Ray Gate: If the beam core is hitting an unbroken indestructible barrier at the epicenter,
		// biome conversion is completely halted until that barrier is shattered or penetrated.
		double verdantLockedY = nbt.contains("verdant_locked_y") ? nbt.getDouble("verdant_locked_y") : (gy + 60.0);
		if (verdantLockedY <= 0) verdantLockedY = gy + 60.0;
		if (isEpicenterBlockedByIndestructibleBarrier(level, cx, gy, cz, verdantLockedY)) {
			return;
		}

		// Step 1: Verify 3x3 Chunk Infection Requirement
		boolean unlocked = nbt.getBoolean("verdant_biome_unlocked");
		if (!unlocked) {
			int checkTimer = nbt.getInt("verdant_biome_check_timer") + 1;
			nbt.putInt("verdant_biome_check_timer", checkTimer);

			if (curInfectionRadius >= 24.0 && checkTimer % 20 == 0) {
				double infectionRatio = calculate3x3InfectionRatio(level, cx, gy, cz, centerChunkX, centerChunkZ);
				double requiredRatio = getInfectionRatioThreshold(nbt);
				if (infectionRatio >= requiredRatio) {
					unlocked = true;
					nbt.putBoolean("verdant_biome_unlocked", true);
					nbt.putDouble("verdant_biome_radius", 2.0);
				}
			}
		}

		if (!unlocked) return;

		// Step 2: Smooth, gradual radial biome expansion with dynamic catchup (capped at 16 chunks / 256 blocks)
		double effectiveMaxR = Math.min(MAX_BIOME_RADIUS, curInfectionRadius);
		double biomeR = nbt.contains("verdant_biome_radius") ? nbt.getDouble("verdant_biome_radius") : 2.0;
		if (biomeR < effectiveMaxR) {
			double diff = effectiveMaxR - biomeR;
			double speed = Math.min(diff, Math.max(0.20, diff * 0.08));
			biomeR += speed;
			biomeR = Math.min(effectiveMaxR, biomeR);
			nbt.putDouble("verdant_biome_radius", biomeR);
		} else if (biomeR > effectiveMaxR) {
			biomeR = effectiveMaxR;
			nbt.putDouble("verdant_biome_radius", biomeR);
		}

		biomeR = Math.min(MAX_BIOME_RADIUS, biomeR);

		// Step 3: Optimized Radial Biome Quart Stepper (Bounded budget to guarantee high TPS)
		int minQuartX = QuartPos.fromBlock(Mth.floor(cx - biomeR));
		int maxQuartX = QuartPos.fromBlock(Mth.floor(cx + biomeR));
		int minQuartZ = QuartPos.fromBlock(Mth.floor(cz - biomeR));
		int maxQuartZ = QuartPos.fromBlock(Mth.floor(cz + biomeR));

		int spanX = Math.max(1, maxQuartX - minQuartX + 1);
		int spanZ = Math.max(1, maxQuartZ - minQuartZ + 1);
		int totalQuartsInBox = spanX * spanZ;

		double biomeRSqr = biomeR * biomeR;
		int scanIndex = nbt.getInt("verdant_biome_scan_idx");
		int activeTerraformers = Math.max(1, nbt.contains("verdant_active_terraformer_count") ? nbt.getInt("verdant_active_terraformer_count") : 1);
		int quartBudget;
		if (activeTerraformers <= 1) {
			quartBudget = SINGLE_ENGINE_BIOME_QUART_BUDGET;
			nbt.remove("verdant_biome_work_debt");
		} else {
			int queuedBiomeWork = Math.min(MULTI_ENGINE_BIOME_QUART_BUDGET * 16,
				(nbt.contains("verdant_biome_work_debt") ? nbt.getInt("verdant_biome_work_debt") : 0) + SINGLE_ENGINE_BIOME_QUART_BUDGET);
			int fairShare = Math.max(1, (MULTI_ENGINE_BIOME_QUART_BUDGET + activeTerraformers - 1) / activeTerraformers);
			quartBudget = Math.min(queuedBiomeWork, fairShare);
			int remainingBiomeWork = queuedBiomeWork - quartBudget;
			if (remainingBiomeWork > 0) nbt.putInt("verdant_biome_work_debt", remainingBiomeWork);
			else nbt.remove("verdant_biome_work_debt");
		}

		for (int step = 0; step < quartBudget; step++) {
			int currentIdx = (scanIndex + step) % totalQuartsInBox;
			int qx = minQuartX + (currentIdx % spanX);
			int qz = minQuartZ + (currentIdx / spanX);

			int bx = QuartPos.toBlock(qx) + 2;
			int bz = QuartPos.toBlock(qz) + 2;
			double dx = bx - cx;
			double dz = bz - cz;
			double distSqr = dx * dx + dz * dz;

			if (distSqr <= biomeRSqr) {
				int chunkX = bx >> 4;
				int chunkZ = bz >> 4;
				if (!level.hasChunk(chunkX, chunkZ)) continue;

				LevelChunk chunk = level.getChunk(chunkX, chunkZ);
				if (chunk == null) continue;

				boolean chunkChanged = applyQuartColumnBiome(level, chunk, qx, qz, bx, bz, verdantLockedY, deepBiome, thicketBiome, plainsBiome);
				if (chunkChanged) {
					pendingBiomeChunks.add(chunk.getPos());
					chunk.setUnsaved(true);
				}
			}
		}
		nbt.putInt("verdant_biome_scan_idx", (scanIndex + quartBudget) % totalQuartsInBox);

		// Step 4: Share one batched biome resend across all active Verdants in this dimension.
		if (!pendingBiomeChunks.isEmpty()) {
			long gameTime = level.getGameTime();
			Long lastResendTick = LAST_BIOME_RESEND_TICK_BY_LEVEL.get(level);
			if (lastResendTick == null || gameTime - lastResendTick >= BIOME_RESEND_INTERVAL_TICKS
					|| pendingBiomeChunks.size() >= BIOME_RESEND_BATCH_SIZE) {
				List<ChunkAccess> chunkList = new ArrayList<>();
				for (ChunkPos cp : pendingBiomeChunks) {
					if (level.hasChunk(cp.x, cp.z)) {
						LevelChunk c = level.getChunk(cp.x, cp.z);
						if (c != null) chunkList.add(c);
					}
				}
				if (!chunkList.isEmpty()) {
					level.getChunkSource().chunkMap.resendBiomesForChunks(chunkList);
				}
				pendingBiomeChunks.clear();
				LAST_BIOME_RESEND_TICK_BY_LEVEL.put(level, gameTime);
			}
		}
	}

	private static boolean isEpicenterBlockedByIndestructibleBarrier(ServerLevel level, double cx, double gy, double cz, double verdantY) {
		int bx = Mth.floor(cx);
		int bz = Mth.floor(cz);
		// Only check for barriers obstructing the space between the Verdant and the active ground target
		int bottomY = (int) Math.max(level.getMinBuildHeight() + 2.0, gy - 2.0);
		int scanY = (int) Math.min(level.getMaxBuildHeight() - 1.0, verdantY);
		BlockPos.MutableBlockPos pos = new BlockPos.MutableBlockPos(bx, scanY, bz);

		while (pos.getY() >= bottomY) {
			BlockState st = level.getBlockState(pos);
			if (st.getBlock() == Blocks.BEDROCK || st.getBlock() == Blocks.BARRIER || st.getBlock() == Blocks.REINFORCED_DEEPSLATE) {
				return true; // Still unbroken barrier directly intercepting the beam before reaching ground
			}
			pos.move(Direction.DOWN);
		}
		return false;
	}

	private static final java.util.Map<Block, Boolean> INFECTED_BLOCK_CACHE = new java.util.concurrent.ConcurrentHashMap<>();

	private static boolean applyQuartColumnBiome(ServerLevel level, LevelChunk chunk, int qx, int qz, int worldBx, int worldBz, double verdantY,
			Holder<Biome> deepBiome, Holder<Biome> thicketBiome, Holder<Biome> plainsBiome) {
		boolean changed = false;
		int sectionCount = chunk.getSections().length;
		int localBx = QuartPos.toBlock(qx) & 15;
		int localBz = QuartPos.toBlock(qz) & 15;
		int localQx = qx & 3;
		int localQz = qz & 3;
		BlockPos.MutableBlockPos pos = new BlockPos.MutableBlockPos();

		// Freeze-Ray Indestructible Barrier Check:
		int barrierStopY = (int) level.getMinBuildHeight();

		for (int sIndex = 0; sIndex < sectionCount; sIndex++) {
			LevelChunkSection section = chunk.getSection(sIndex);
			if (section == null || section.hasOnlyAir()) continue;

			int secBottomY = chunk.getSectionYFromSectionIndex(sIndex) << 4;
			int secTopY = secBottomY + 15;

			if (secTopY < barrierStopY) {
				continue;
			}

			if (!isSectionOrColumnInfected(chunk, localBx, localBz, secBottomY, barrierStopY, pos)) {
				continue;
			}

			try {
				PalettedContainer<Holder<Biome>> container = (PalettedContainer<Holder<Biome>>) section.getBiomes();
				for (int qy = 0; qy < 4; qy++) {
					int quartWorldY = secBottomY + (qy << 2);
					if (quartWorldY < barrierStopY) continue;

					Holder<Biome> current = container.get(localQx, qy, localQz);
					Holder<Biome> target = determineTargetBiome(level, current, chunk, worldBx, worldBz, deepBiome, thicketBiome, plainsBiome);
					if (current != target && target != null) {
						container.set(localQx, qy, localQz, target);
						changed = true;
					}
				}
			} catch (Exception ignored) {
			}
		}
		return changed;
	}

	private static boolean isSectionOrColumnInfected(LevelChunk chunk, int localBx, int localBz, int secBottomY, int barrierStopY, BlockPos.MutableBlockPos pos) {
		int worldMinBx = chunk.getPos().getMinBlockX() + localBx;
		int worldMinBz = chunk.getPos().getMinBlockZ() + localBz;

		int checkMinY = Math.max((int) chunk.getLevel().getMinBuildHeight(), secBottomY - 16);
		int checkMaxY = secBottomY + 15;

		for (int ox = 0; ox <= 2; ox += 2) {
			for (int oz = 0; oz <= 2; oz += 2) {
				int worldBx = worldMinBx + ox;
				int worldBz = worldMinBz + oz;

				for (int y = checkMaxY; y >= checkMinY; y--) {
					if (y < barrierStopY) break;
					pos.set(worldBx, y, worldBz);
					BlockState st = chunk.getBlockState(pos);
					if (isInfectedBlock(st)) {
						return true;
					}
					if (y < secBottomY && (st.getBlock() == Blocks.BEDROCK || st.getBlock() == Blocks.BARRIER || st.getBlock() == Blocks.REINFORCED_DEEPSLATE)) {
						continue;
					}
				}
			}
		}

		return false;
	}

	private static int findIndestructibleBarrierBelow(ServerLevel level, int bx, int bz, double startY) {
		int minY = (int) level.getMinBuildHeight();
		if (!level.dimensionType().hasCeiling()) {
			return minY;
		}
		int scanY = (int) Math.min(level.getMaxBuildHeight() - 1.0, startY);
		BlockPos.MutableBlockPos pos = new BlockPos.MutableBlockPos(bx, scanY, bz);

		while (pos.getY() > minY + 5) {
			BlockState st = level.getBlockState(pos);
			if (st.getBlock() == Blocks.BEDROCK || st.getBlock() == Blocks.BARRIER || st.getBlock() == Blocks.REINFORCED_DEEPSLATE) {
				return pos.getY();
			}
			pos.move(Direction.DOWN);
		}
		return minY;
	}

	private static double calculate3x3InfectionRatio(ServerLevel level, double cx, double gy, double cz, int centerChunkX, int centerChunkZ) {
		int totalSamples = 0;
		int infectedSamples = 0;
		BlockPos.MutableBlockPos pos = new BlockPos.MutableBlockPos();
		double scanTop = Math.min(level.getMaxBuildHeight() - 5.0, gy + 85.0);

		int startX = (centerChunkX - 1) << 4;
		int startZ = (centerChunkZ - 1) << 4;

		for (int x = startX + 3; x < startX + 48; x += 6) {
			for (int z = startZ + 3; z < startZ + 48; z += 6) {
				pos.set(x, (int) gy, z);
				if (!level.hasChunkAt(pos)) continue;

				totalSamples++;
				int groundY = findSurfaceY(level, x, z, scanTop, gy);
				pos.set(x, groundY, z);
				BlockState state = level.getBlockState(pos);

				if (isInfectedBlock(state)) {
					infectedSamples++;
				}
			}
		}

		return totalSamples > 0 ? ((double) infectedSamples / totalSamples) : 0.0;
	}

	private static boolean isInfectedBlock(BlockState state) {
		if (state == null || state.isAir()) return false;
		Block b = state.getBlock();
		return INFECTED_BLOCK_CACHE.computeIfAbsent(b, block -> {
			if (block == Blocks.OAK_PLANKS || block == Blocks.OAK_LOG || block == Blocks.OAK_WOOD ||
				block == Blocks.SPRUCE_PLANKS || block == Blocks.SPRUCE_LOG || block == Blocks.SPRUCE_WOOD ||
				block == Blocks.PETRIFIED_OAK_SLAB) return true;
			ResourceLocation loc = BuiltInRegistries.BLOCK.getKey(block);
			if (loc == null) return false;
			String id = loc.toString().toLowerCase();
			return id.contains("petrified") || id.contains("lignum_caro") || id.contains("splintered") ||
				id.contains("oak_planks") || id.contains("spruce_planks") || id.contains("wood_plains") || id.contains("false_oak");
		});
	}

	private static int findSurfaceY(ServerLevel level, int x, int z, double scanTop, double gy) {
		try {
			int height = level.getHeight(Heightmap.Types.WORLD_SURFACE, x, z);
			boolean hasCeiling = level.dimensionType().hasCeiling();
			int startY = (int) Math.min(scanTop, height);
			if (hasCeiling && height > gy + 35.0) {
				startY = (int) Math.min(height - 1, gy + 30.0);
			}
			BlockPos.MutableBlockPos pos = new BlockPos.MutableBlockPos();
			int bottom = (int) level.getMinBuildHeight() + 1;
			for (int y = startY; y >= bottom; y--) {
				pos.set(x, y, z);
				BlockState st = level.getBlockState(pos);
				if (!st.isAir()) {
					return y;
				}
			}
			return startY;
		} catch (Exception ignored) {
			BlockPos.MutableBlockPos pos = new BlockPos.MutableBlockPos();
			int top = (int) scanTop;
			int bottom = (int) level.getMinBuildHeight() + 1;
			for (int y = top; y >= bottom; y--) {
				pos.set(x, y, z);
				BlockState st = level.getBlockState(pos);
				if (!st.isAir() && st.getBlock() != Blocks.BEDROCK) {
					return y;
				}
			}
			return (int) scanTop;
		}
	}

	public static Holder<Biome> getWoodPlainsHolder(ServerLevel level) {
		return findBiomeHolder(level, WOOD_PLAINS_KEY, "wood_plains");
	}

	public static Holder<Biome> getDeepBackwoodsHolder(ServerLevel level) {
		return findBiomeHolder(level, DEEP_BACKWOODS_KEY, "deep_backwoods");
	}

	public static Holder<Biome> getTheThicketHolder(ServerLevel level) {
		return findBiomeHolder(level, THE_THICKET_KEY, "the_thicket");
	}

	private static boolean isRegisteredBiomeHolder(ServerLevel level, Holder<Biome> biomeHolder) {
		if (level == null || biomeHolder == null || !biomeHolder.isBound()) return false;
		try {
			ResourceKey<Biome> biomeKey = biomeHolder.unwrapKey().orElse(null);
			if (biomeKey == null) return false;
			return level.registryAccess().registryOrThrow(Registries.BIOME).getHolder(biomeKey)
				.map(currentHolder -> currentHolder == biomeHolder || currentHolder.equals(biomeHolder)).orElse(false);
		} catch (Exception ignored) {
			return false;
		}
	}

	private static Holder<Biome> findBiomeHolder(ServerLevel level, ResourceKey<Biome> primaryKey, String targetPath) {
		try {
			var reg = level.registryAccess().registryOrThrow(Registries.BIOME);
			var opt = reg.getHolder(primaryKey);
			if (opt.isPresent()) {
				return opt.get();
			}
			for (var entry : reg.entrySet()) {
				ResourceLocation loc = entry.getKey().location();
				if ("the_backwoods".equals(loc.getNamespace()) && targetPath.equals(loc.getPath())) {
					return reg.getHolder(entry.getKey()).orElse(null);
				}
			}
		} catch (Exception ignored) {}
		return null;
	}

	public static boolean isBackwoodsBiome(ServerLevel level, Holder<Biome> biome) {
		if (biome == null) return false;
		Holder<Biome> wp = getWoodPlainsHolder(level);
		if (wp != null && biome.equals(wp)) return true;
		Holder<Biome> db = getDeepBackwoodsHolder(level);
		if (db != null && biome.equals(db)) return true;
		Holder<Biome> tt = getTheThicketHolder(level);
		if (tt != null && biome.equals(tt)) return true;

		ResourceLocation loc = biome.unwrapKey().map(ResourceKey::location).orElse(null);
		if (loc != null) {
			String ns = loc.getNamespace().toLowerCase();
			String path = loc.getPath().toLowerCase();
			if ("the_backwoods".equals(ns)) {
				if ("wood_plains".equals(path) || "deep_backwoods".equals(path) || "the_thicket".equals(path)) {
					return true;
				}
			}
		}
		return false;
	}

	public static Holder<Biome> determineTargetBiome(ServerLevel level, Holder<Biome> currentBiome) {
		return determineTargetBiome(level, currentBiome, null, 0, 0);
	}

	public static Holder<Biome> determineTargetBiome(ServerLevel level, Holder<Biome> currentBiome, LevelChunk chunk, int worldBx, int worldBz) {
		return determineTargetBiome(level, currentBiome, chunk, worldBx, worldBz,
			getDeepBackwoodsHolder(level), getTheThicketHolder(level), getWoodPlainsHolder(level));
	}

	private static Holder<Biome> determineTargetBiome(ServerLevel level, Holder<Biome> currentBiome, LevelChunk chunk, int worldBx, int worldBz,
			Holder<Biome> deep, Holder<Biome> thicket, Holder<Biome> plains) {
		if (currentBiome == null) return plains;

		// Preserve established deep_backwoods, the_thicket, and wood_plains biomes
		if (deep != null && currentBiome.equals(deep)) return currentBiome;
		if (thicket != null && currentBiome.equals(thicket)) return currentBiome;
		if (plains != null && currentBiome.equals(plains)) return currentBiome;

		ResourceLocation loc = currentBiome.unwrapKey().map(ResourceKey::location).orElse(null);

		String path = loc != null ? loc.getPath().toLowerCase() : "";
		String dimId = level != null ? level.dimension().location().toString().toLowerCase() : "";
		if (dimId.contains("nether")) {
			if (path.contains("warped_forest")) {
				if (deep != null) return deep;
			} else if (path.contains("crimson_forest")) {
				if (thicket != null) return thicket;
			} else {
				if (plains != null) return plains;
			}
		} else if (dimId.contains("end")) {
			if (plains != null) return plains;
		}
		Biome biomeVal = currentBiome.value();
		boolean isColdOrSnow = biomeVal != null && (biomeVal.getBaseTemperature() <= 0.35F);
		boolean isBeachOrWater = path.contains("beach") || path.contains("shore") || path.contains("river") || path.contains("ocean");

		// 1. Deep Backwoods: Taiga, Spruce, Jungle, Mangrove, Swamp, Cold/Snowy/Alpine (beaches/rivers & bamboo excluded)
		if (!isBeachOrWater && (currentBiome.is(net.minecraft.tags.BiomeTags.IS_TAIGA) || currentBiome.is(net.minecraft.tags.BiomeTags.IS_JUNGLE) ||
			isColdOrSnow ||
			path.contains("spruce") || path.contains("taiga") || path.contains("jungle") ||
			path.contains("dark_forest") || path.contains("mangrove") || path.contains("swamp") || path.contains("cold") ||
			path.contains("snow") || path.contains("frozen") || path.contains("grove") || path.contains("pine") ||
			path.contains("alpine") || path.contains("boreal") || path.contains("tundra") || path.contains("rainforest") ||
			path.contains("ice") || path.contains("peak") || path.contains("slopes") || path.contains("deep_backwoods"))) {
			if (deep != null) return deep;
		}

		// 2. The Thicket: Temperate Forest, Birch, Wooded hills, Orchards, Groves (beaches/rivers excluded)
		if (!isBeachOrWater && (currentBiome.is(net.minecraft.tags.BiomeTags.IS_FOREST) ||
			path.contains("forest") || path.contains("birch") || path.contains("wood") || path.contains("cherry") ||
			path.contains("orchard") || path.contains("grove") || path.contains("the_thicket") || path.contains("canopy"))) {
			if (thicket != null) return thicket;
		}

		// 3. Physical block inspection for column terrain/trees
		if (chunk != null) {
			int localX = worldBx & 15;
			int localZ = worldBz & 15;
			BlockPos.MutableBlockPos p = new BlockPos.MutableBlockPos();
			int topY = chunk.getHeight(Heightmap.Types.WORLD_SURFACE, localX, localZ);
			int minY = Math.max((int) level.getMinBuildHeight(), topY - 30);
			for (int y = topY; y >= minY; y--) {
				p.set(worldBx, y, worldBz);
				BlockState st = chunk.getBlockState(p);
				if (st.isAir()) continue;
				Block b = st.getBlock();
				if (b == Blocks.SPRUCE_LOG || b == Blocks.SPRUCE_WOOD || b == Blocks.SPRUCE_LEAVES ||
					b == Blocks.SPRUCE_PLANKS || b == Blocks.PODZOL || b == Blocks.SNOW_BLOCK ||
					b == Blocks.POWDER_SNOW || b == Blocks.DARK_OAK_LOG || b == Blocks.DARK_OAK_WOOD ||
					b == Blocks.JUNGLE_LOG || b == Blocks.JUNGLE_WOOD || b == Blocks.MANGROVE_LOG) {
					if (deep != null) return deep;
				}
				if (b == Blocks.BIRCH_LOG || b == Blocks.BIRCH_WOOD || b == Blocks.BIRCH_LEAVES ||
					b == Blocks.CHERRY_LOG || b == Blocks.CHERRY_WOOD || b == Blocks.OAK_LEAVES) {
					if (thicket != null) return thicket;
				}
			}
		}

		// 4. Wood Plains: Plains, Savanna, Desert, Badlands, Oceans, Rivers, Nether, End, and universal default
		return plains != null ? plains : currentBiome;
	}
}
