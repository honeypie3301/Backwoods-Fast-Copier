    @SubscribeEvent
    public static void onPlayerTick(PlayerTickEvent.Post event) {
        Player player = event.getEntity();
        if (player == null) return;

        Level level = player.level();
        if (level.isClientSide()) return;

        ResourceKey<Level> dim = level.dimension();
        if (!FIRE_VALID_DIMENSIONS.contains(dim)) return;

        // 1. Fire Defense (Valid in: The Grain, Backwoods, The Sub Strata, The Petrified Weald)
        tickFireDefense(level, player, dim);

        // 2, 3 & 4. Lava Defense, Lava Suppression & TNT Defense (Valid in: The Grain, Backwoods, The Sub Strata)
        if (LAVA_AND_TNT_VALID_DIMENSIONS.contains(dim)) {
            tickLavaDefense(level, player);
            tickLavaSuppression(level, player, dim);
            tickTntDefense(level, player);
        }
    }

    private static void tickFireDefense(Level level, Player player, ResourceKey<Level> dim) {
        // Process active domino purge queue every tick
        processDominoPurge(level, player);

        boolean isStrongDim = dim == THE_GRAIN || dim == THE_SUB_STRATA || dim == THE_CROSSWORKS;
        int interval = isStrongDim ? 4 : FIRE_CHECK_INTERVAL_TICKS;
        if (player.tickCount % interval != 0) return;

        BlockPos center = player.blockPosition();

        // 1. SILENT CLEANUP (Beyond close range, up to full SCAN_RADIUS_XZ)
        silentPurgeSliced(level, center, player.tickCount);

        // 2. ACTIVE THREAT RADIUS CHECK
        int cd = player.getPersistentData().getIntOr(NBT_FIRE_COOLDOWN_KEY, 0);
        if (cd > 0) {
            player.getPersistentData().putInt(NBT_FIRE_COOLDOWN_KEY, Math.max(0, cd - interval));
            return;
        }

        int threshold = isStrongDim ? 15 : BIG_FIRE_THRESHOLD;
        int fireCount = countFireInThreatRadius(level, center);
        if (fireCount < threshold) return;

        // Perform the domino threat purge
        startDominoPurge(level, player, center);

        // Set the cooldown
        int cooldown = isStrongDim ? 40 : FIRE_PURGE_COOLDOWN_TICKS;
        player.getPersistentData().putInt(NBT_FIRE_COOLDOWN_KEY, cooldown);
    }

    private static long chunkKey(int cx, int cz) {
        return ((long) cx & 0xFFFFFFFFL) | (((long) cz & 0xFFFFFFFFL) << 32);
    }

    private static void silentPurgeSliced(Level level, BlockPos center, int tickCount) {
        if (!(level instanceof net.minecraft.server.level.ServerLevel serverLevel)) return;
        net.minecraft.server.level.ServerChunkCache cache = serverLevel.getChunkSource();
        Map<Long, net.minecraft.world.level.chunk.LevelChunk> loadedChunks = new HashMap<>();

        boolean isStrongDim = level.dimension() == THE_GRAIN || level.dimension() == THE_SUB_STRATA || level.dimension() == THE_CROSSWORKS;
        int totalSlices = isStrongDim ? 3 : 12;
        int interval = isStrongDim ? 4 : FIRE_CHECK_INTERVAL_TICKS;
        int currentSlice = (int) ((tickCount / interval) % totalSlices);

        int range = FIRE_SCAN_RADIUS_XZ;
        int sliceWidth = (range * 2 + 1 + totalSlices - 1) / totalSlices;
        int startDx = -range + currentSlice * sliceWidth;
        int endDx = Math.min(range, startDx + sliceWidth - 1);

        BlockPos.MutableBlockPos p = new BlockPos.MutableBlockPos();
        int cx = center.getX();
        int cy = center.getY();
        int cz = center.getZ();

        int minY = cy - FIRE_SCAN_RADIUS_Y;
        int maxY = cy + FIRE_SCAN_RADIUS_Y;

        for (int dx = startDx; dx <= endDx; dx++) {
            int absX = cx + dx;
            int cxChunk = absX >> 4;
            int lastChunkZ = -999;
            net.minecraft.world.level.chunk.LevelChunk currentChunk = null;

            for (int dz = -FIRE_SCAN_RADIUS_XZ; dz <= FIRE_SCAN_RADIUS_XZ; dz++) {
                int absZ = cz + dz;
                int czChunk = absZ >> 4;

                if (currentChunk == null || czChunk != lastChunkZ) {
                    lastChunkZ = czChunk;
                    long chunkKey = chunkKey(cxChunk, czChunk);
                    currentChunk = loadedChunks.get(chunkKey);
                    if (currentChunk == null) {
                        currentChunk = cache.getChunkNow(cxChunk, czChunk);
                        if (currentChunk != null) loadedChunks.put(chunkKey, currentChunk);
                    }
                }

                if (currentChunk == null) continue;

                int localMinY = Math.max(level.getMinY(), minY);
                int localMaxY = Math.min(level.getMaxY() - 1, maxY);

                int minSection = (localMinY - level.getMinY()) >> 4;
                int maxSection = (localMaxY - level.getMinY()) >> 4;

                net.minecraft.world.level.chunk.LevelChunkSection[] sections = currentChunk.getSections();

                for (int secIndex = minSection; secIndex <= maxSection; secIndex++) {
                    if (secIndex < 0 || secIndex >= sections.length) continue;
                    net.minecraft.world.level.chunk.LevelChunkSection section = sections[secIndex];
                    if (section == null || section.hasOnlyAir()) continue;
                    if (!section.getStates().maybeHas(DimensionDefenseProcedure::isFire)) continue;

                    int secMinY = level.getMinY() + (secIndex << 4);
                    int startY = Math.max(localMinY, secMinY);
                    int endY = Math.min(localMaxY, secMinY + 15);

                    for (int absY = startY; absY <= endY; absY++) {
                        int dy = absY - cy;
                        if (Math.abs(dx) <= FIRE_THREAT_RADIUS_XZ && Math.abs(dy) <= FIRE_THREAT_RADIUS_Y && Math.abs(dz) <= FIRE_THREAT_RADIUS_XZ) {
                            continue;
                        }
                        BlockState state = section.getBlockState(absX & 15, absY & 15, absZ & 15);
                        if (isFire(state)) {
                            p.set(absX, absY, absZ);
                            level.setBlock(p, Blocks.AIR.defaultBlockState(), 3);
                            if (level.getRandom().nextFloat() < 0.05f) {
                                BuiltInRegistries.SOUND_EVENT.getOptional(Identifier.parse("block.fire.extinguish")).ifPresent(extinguish -> {
                                    level.playSound(null, p, extinguish, SoundSource.BLOCKS, 0.2f, 1.2f + level.getRandom().nextFloat() * 0.4f);
                                });
                            }
                        }
                    }
                }
            }
        }
    }

    private static void startDominoPurge(Level level, Player player, BlockPos center) {
        if (!(level instanceof net.minecraft.server.level.ServerLevel serverLevel)) return;
        net.minecraft.server.level.ServerChunkCache cache = serverLevel.getChunkSource();
        Map<Long, net.minecraft.world.level.chunk.LevelChunk> loadedChunks = new HashMap<>();

        List<BlockPos> firePositions = new ArrayList<>();
        List<BlockPos> plankPositions = new ArrayList<>();
        int cx = center.getX();
        int cy = center.getY();
        int cz = center.getZ();

        int minY = cy - FIRE_THREAT_RADIUS_Y;
        int maxY = cy + FIRE_THREAT_RADIUS_Y + 1;

        for (int dx = -FIRE_THREAT_RADIUS_XZ; dx <= FIRE_THREAT_RADIUS_XZ + 1; dx++) {
            int absX = cx + dx;
            int cxChunk = absX >> 4;
            int lastChunkZ = -999;
            net.minecraft.world.level.chunk.LevelChunk currentChunk = null;

            for (int dz = -FIRE_THREAT_RADIUS_XZ; dz <= FIRE_THREAT_RADIUS_XZ + 1; dz++) {
                int absZ = cz + dz;
                int czChunk = absZ >> 4;

                if (currentChunk == null || czChunk != lastChunkZ) {
                    lastChunkZ = czChunk;
                    long chunkKey = chunkKey(cxChunk, czChunk);
                    currentChunk = loadedChunks.get(chunkKey);
                    if (currentChunk == null) {
                        currentChunk = cache.getChunkNow(cxChunk, czChunk);
                        if (currentChunk != null) loadedChunks.put(chunkKey, currentChunk);
                    }
                }

                if (currentChunk == null) continue;

                int localMinY = Math.max(level.getMinY(), minY);
                int localMaxY = Math.min(level.getMaxY() - 1, maxY);

                int minSection = (localMinY - level.getMinY()) >> 4;
                int maxSection = (localMaxY - level.getMinY()) >> 4;

                net.minecraft.world.level.chunk.LevelChunkSection[] sections = currentChunk.getSections();

                for (int secIndex = minSection; secIndex <= maxSection; secIndex++) {
                    if (secIndex < 0 || secIndex >= sections.length) continue;
                    net.minecraft.world.level.chunk.LevelChunkSection section = sections[secIndex];
                    if (section == null || section.hasOnlyAir()) continue;
                    if (!section.getStates().maybeHas(state -> isFire(state) || isSmolderingPlank(state))) continue;

                    int secMinY = level.getMinY() + (secIndex << 4);
                    int startY = Math.max(localMinY, secMinY);
                    int endY = Math.min(localMaxY, secMinY + 15);

                    for (int absY = startY; absY <= endY; absY++) {
                        BlockState state = section.getBlockState(absX & 15, absY & 15, absZ & 15);
                        if (state.isAir()) continue;
                        if (isFire(state)) {
                            firePositions.add(new BlockPos(absX, absY, absZ));
                        } else if (isSmolderingPlank(state)) {
                            plankPositions.add(new BlockPos(absX, absY, absZ));
                        }
                    }
                }
            }
        }

        if (firePositions.isEmpty() && plankPositions.isEmpty()) return;

        firePositions.sort((p1, p2) -> Double.compare(p1.distSqr(center), p2.distSqr(center)));

        plankPositions = sortAsInfectionSpread(plankPositions, center);

        if (firePositions.size() > 1000) {
            firePositions = firePositions.subList(0, 1000);
        }

        ListTag fireList = new ListTag();
        for (BlockPos pos : firePositions) {
            fireList.add(StringTag.valueOf(pos.getX() + "," + pos.getY() + "," + pos.getZ()));
        }

        if (!fireList.isEmpty()) {
            player.getPersistentData().put("grain_domino_queue", fireList);
        }

        if (!plankPositions.isEmpty()) {
            plankPositions = sortAsInfectionSpread(plankPositions, center);

            ListTag plankList = new ListTag();
            for (BlockPos pos : plankPositions) {
                plankList.add(StringTag.valueOf(pos.getX() + "," + pos.getY() + "," + pos.getZ()));
            }

            if (!plankList.isEmpty()) {
                player.getPersistentData().put("grain_plank_queue", plankList);
                player.getPersistentData().putInt("grain_plank_total", plankList.size());
            }
        }

        if (!firePositions.isEmpty() || !plankPositions.isEmpty()) {
            if (!firePositions.isEmpty()) {
                BuiltInRegistries.SOUND_EVENT.getOptional(Identifier.parse("block.fire.extinguish")).ifPresent(extinguish -> {
                    level.playSound(null, center, extinguish, SoundSource.BLOCKS, 1.3f, 0.85f);
                });
            }
        }
    }

    private static boolean isBurntBlock(BlockState state) {
        if (state == null || state.isAir()) return false;
        Block block = state.getBlock();
        Identifier reg = BuiltInRegistries.BLOCK.getKey(block);
        if (reg != null) {
            String namespace = reg.getNamespace();
            String path = reg.getPath().toLowerCase();
            if (namespace.equals("burnt") && (path.equals("wood_fire") || path.matches("wood_fire_[2-9]"))) {
                return false;
            }
            if (path.contains("brick") || path.contains("stone") || path.contains("tile") || path.contains("glass") || path.contains("metal") || path.contains("ore")) {
                return false;
            }
            if (namespace.equals("burnt") || namespace.equals("burnt_basic") || namespace.equals("firesdelight") || namespace.equals("nether_delight")) {
                return true;
            }
            if (path.contains("burnt") || path.contains("smoldering") || path.contains("sooty") || path.contains("charred")) {
                return true;
            }
        }
        return false;
    }

    private static BlockState getHealthyEquivalent(BlockState burntState) {
        if (burntState == null) {
            return TheBackwoodsModBlocks.LIGNUM_CARO.get().defaultBlockState();
        }
        Block block = burntState.getBlock();
        Identifier reg = BuiltInRegistries.BLOCK.getKey(block);
        if (reg != null) {

                boolean isLog = path.contains("log") || path.contains("stem") || path.contains("wood") || path.contains("hyphae") || path.contains("bark");
                Block healthyBlock = path.contains("spruce") ? (isLog ? Blocks.SPRUCE_LOG : Blocks.SPRUCE_PLANKS) :
                                     path.contains("birch") ? (isLog ? Blocks.BIRCH_LOG : Blocks.BIRCH_PLANKS) :
                                     path.contains("jungle") ? (isLog ? Blocks.JUNGLE_LOG : Blocks.JUNGLE_PLANKS) :
                                     path.contains("acacia") ? (isLog ? Blocks.ACACIA_LOG : Blocks.ACACIA_PLANKS) :
                                     path.contains("dark_oak") ? (isLog ? Blocks.DARK_OAK_LOG : Blocks.DARK_OAK_PLANKS) :

    private static List<BlockPos> sortAsInfectionSpread(List<BlockPos> positions, BlockPos center) {
        if (positions.isEmpty()) return positions;
        Set<BlockPos> remaining = new HashSet<>(positions);
        List<BlockPos> ordered = new ArrayList<>();

        BlockPos seed = positions.get(0);
        double minDist = seed.distSqr(center);
        for (BlockPos p : positions) {
            double d = p.distSqr(center);
            if (d < minDist) {
                minDist = d;
                seed = p;
            }
        }

        ArrayDeque<BlockPos> frontier = new ArrayDeque<>();
        frontier.add(seed);
        remaining.remove(seed);
        ordered.add(seed);

        while (!frontier.isEmpty() && !remaining.isEmpty()) {
            BlockPos current = frontier.removeFirst();
            for (Direction d : Direction.values()) {
                BlockPos n = current.relative(d);
                if (!remaining.remove(n)) continue;
                ordered.add(n);
                frontier.add(n);
            }
            if (frontier.isEmpty() && !remaining.isEmpty()) {
                BlockPos nextSeed = remaining.iterator().next();
                double minNextDist = nextSeed.distSqr(current);
                for (BlockPos r : remaining) {
                    double d = r.distSqr(current);
                    if (d < minNextDist) {
                        minNextDist = d;
                        nextSeed = r;
                    }
                }
                remaining.remove(nextSeed);
                ordered.add(nextSeed);
                frontier.add(nextSeed);
            }
        }

        for (BlockPos r : remaining) {
            ordered.add(r);
        }
        return ordered;
    }

    private static void processDominoPurge(Level level, Player player) {
        CompoundTag data = player.getPersistentData();
        boolean hasFire = data.contains("grain_domino_queue");
        boolean hasPlanks = data.contains("grain_plank_queue");

        if (!hasFire && !hasPlanks) return;

        if (hasFire) {
            ListTag list = data.getListOrEmpty("grain_domino_queue");
            if (list.isEmpty()) {
                data.remove("grain_domino_queue");
            } else {
                int extinguishCount = Math.min(list.size(), 12);
                List<BlockPos> toExtinguish = new ArrayList<>();

                for (int i = 0; i < extinguishCount; i++) {
                    String s = list.getString(0).orElse("");
                    list.remove(0);
                    BlockPos p = parsePos(s);
                    if (p != null) {
                        toExtinguish.add(p);
                    }
                }

                if (list.isEmpty()) {
                    data.remove("grain_domino_queue");
                } else {
                    data.put("grain_domino_queue", list);
                }

                if (level instanceof net.minecraft.server.level.ServerLevel serverLevel) {
                    for (BlockPos pos : toExtinguish) {
                        if (isFire(level.getBlockState(pos))) {
                            level.setBlock(pos, Blocks.AIR.defaultBlockState(), 3);
                            serverLevel.sendParticles(ParticleTypes.SMOKE, pos.getX() + 0.5, pos.getY() + 0.5, pos.getZ() + 0.5, 2, 0.15, 0.15, 0.15, 0.05);
                            if (serverLevel.getRandom().nextFloat() < 0.2f) {
                                BuiltInRegistries.SOUND_EVENT.getOptional(Identifier.parse("block.fire.extinguish")).ifPresent(extinguish -> {
                                    level.playSound(null, pos, extinguish, SoundSource.BLOCKS, 0.25f, 1.2f + serverLevel.getRandom().nextFloat() * 0.4f);
                                });
                            }
                        }
                    }
                }
            }
        }

        if (hasPlanks) {
            int plankDelay = data.getIntOr("grain_plank_delay", 0);
            if (plankDelay > 0) {
                data.putInt("grain_plank_delay", plankDelay - 1);
            } else {
                data.putInt("grain_plank_delay", 14);
                ListTag list = data.getListOrEmpty("grain_plank_queue");
                if (list.isEmpty()) {
                    data.remove("grain_plank_queue");
                    data.remove("grain_plank_delay");
                } else {
                    int processCount = Math.min(list.size(), 1);
                    List<BlockPos> toRestore = new ArrayList<>();

                    for (int i = 0; i < processCount; i++) {
                        String s = list.getString(0).orElse("");
                        list.remove(0);
                        BlockPos p = parsePos(s);
                        if (p != null) {
                            toRestore.add(p);
                        }
                    }

                    if (list.isEmpty()) {
                        data.remove("grain_plank_queue");
                        data.remove("grain_plank_delay");
                        data.remove("grain_plank_total");
                    } else {
                        data.put("grain_plank_queue", list);
                    }

                    if (level instanceof net.minecraft.server.level.ServerLevel serverLevel) {
                        int totalInitial = data.getIntOr("grain_plank_total", 50);
                        double conversionChance;
                        if (totalInitial <= 100) {
                            conversionChance = 0.05 + (double) totalInitial * (0.45 / 100.0);
                        } else {
                            conversionChance = 0.50 + Math.min(0.48, (double) (totalInitial - 100) * (0.48 / 100.0));
                        }
                        conversionChance = Math.max(0.02, Math.min(0.98, conversionChance + (serverLevel.getRandom().nextDouble() * 0.1 - 0.05)));

                        for (BlockPos pos : toRestore) {
                            BlockState currentState = level.getBlockState(pos);
                            if (isSmolderingPlank(currentState)) {
                                if (serverLevel.getRandom().nextDouble() < conversionChance) {
                                    BlockState restoredState = getRestoredPlankState(currentState, serverLevel.getRandom());
                                    level.setBlock(pos, restoredState, 3);
                                    serverLevel.sendParticles(ParticleTypes.HAPPY_VILLAGER, pos.getX() + 0.5, pos.getY() + 0.5, pos.getZ() + 0.5, 2, 0.2, 0.2, 0.2, 0.02);
                                    if (serverLevel.getRandom().nextFloat() < 0.15f) {
                                        BuiltInRegistries.SOUND_EVENT.getOptional(Identifier.parse("block.cherry_wood.place")).ifPresent(placeSound -> {
                                            level.playSound(null, pos, placeSound, SoundSource.BLOCKS, 0.4f, 0.8f + serverLevel.getRandom().nextFloat() * 0.3f);
                                        });
                                    }
                                } else {
                                    serverLevel.sendParticles(ParticleTypes.SMOKE, pos.getX() + 0.5, pos.getY() + 0.5, pos.getZ() + 0.5, 1, 0.1, 0.1, 0.1, 0.01);
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    private static boolean isSmolderingPlank(BlockState state) {
        if (state == null || state.isAir()) return false;
        Block block = state.getBlock();
        Boolean cached = SMOLDERING_PLANK_CACHE.get(block);
        if (cached != null) return cached;

        boolean result = false;
        Identifier reg = BuiltInRegistries.BLOCK.getKey(block);
        if (reg != null) {
            String path = reg.getPath().toLowerCase();
            if (path.contains("smoldering") && path.contains("plank")) {
                result = true;
            }
        }
        SMOLDERING_PLANK_CACHE.put(block, result);
        return result;
    }

    private static BlockState getRestoredPlankState(BlockState burntState, net.minecraft.util.RandomSource random) {
        Block targetBlock = random.nextBoolean() ? TheBackwoodsModBlocks.LIGNUM_CARO.get() : Blocks.OAK_PLANKS;
        return copyBlockStateProperties(burntState, targetBlock.defaultBlockState());
    }

    private static BlockState copyBlockStateProperties(BlockState from, BlockState to) {
        BlockState result = to;
        for (Property<?> property : from.getProperties()) {
            if (result.hasProperty(property)) {
                result = copyPropertyHelper(from, result, property);
            }
        }
        return result;
    }

    @SuppressWarnings("unchecked")
    private static <T extends Comparable<T>> BlockState copyPropertyHelper(BlockState from, BlockState to, Property<T> property) {
        return to.setValue(property, from.getValue(property));
    }

    private static BlockPos parsePos(String s) {
        if (s == null || s.isEmpty()) return null;
        try {
            String[] parts = s.split(",");
            if (parts.length == 3) {
                return new BlockPos(
                    Integer.parseInt(parts[0]),
                    Integer.parseInt(parts[1]),
                    Integer.parseInt(parts[2])
                );
            }
        } catch (Exception e) {}
        return null;
    }

    private static boolean isFire(BlockState state) {
        if (state == null || state.isAir()) return false;
        Block block = state.getBlock();
        Boolean cached = FIRE_CACHE.get(block);
        if (cached != null) return cached;

        boolean result = false;
        if (block instanceof BaseFireBlock || block == Blocks.FIRE || block == Blocks.SOUL_FIRE) {
            result = true;
        } else {
            Identifier reg = BuiltInRegistries.BLOCK.getKey(block);
            if (reg != null) {
                String namespace = reg.getNamespace();
                String path = reg.getPath().toLowerCase();
                if (namespace.equals("burnt") && (path.equals("wood_fire") || path.matches("wood_fire_[2-9]"))) {
                    result = true;
                }
            }
        }
        FIRE_CACHE.put(block, result);
        return result;
    }

    private static void tickLavaDefense(Level level, Player player) {
        if (player.tickCount % LAVA_CHECK_INTERVAL_TICKS != 0) return;

        int cd = player.getPersistentData().getIntOr(NBT_LAVA_COOLDOWN_KEY, 0);
        if (cd > 0) {
            player.getPersistentData().putInt(NBT_LAVA_COOLDOWN_KEY, Math.max(0, cd - LAVA_CHECK_INTERVAL_TICKS));
            return;
        }

        BlockPos center = player.blockPosition();

        int lavaCount = countLavaInRadius(level, center);
        if (lavaCount < BIG_LAVA_THRESHOLD) return;

        BuiltInRegistries.SOUND_EVENT.getOptional(Identifier.parse("minecraft:ambient.cave")).ifPresent(cave -> {
            level.playSound(null, center, cave, SoundSource.AMBIENT, 1.3f, 0.85f);
        });

        purgeLavaInRadius(level, center);

        player.getPersistentData().putInt(NBT_LAVA_COOLDOWN_KEY, LAVA_PURGE_COOLDOWN_TICKS);
    }

    private static int countLavaInRadius(Level level, BlockPos center) {
        if (!(level instanceof net.minecraft.server.level.ServerLevel serverLevel)) return 0;
        net.minecraft.server.level.ServerChunkCache cache = serverLevel.getChunkSource();
        Map<Long, net.minecraft.world.level.chunk.LevelChunk> loadedChunks = new HashMap<>();

        int count = 0;
        int cx = center.getX();
        int cy = center.getY();
        int cz = center.getZ();

        int minY = cy - LAVA_SCAN_RADIUS_Y;
        int maxY = cy + LAVA_SCAN_RADIUS_Y;

        for (int dx = -LAVA_SCAN_RADIUS_XZ; dx <= LAVA_SCAN_RADIUS_XZ; dx += 2) {
            int absX = cx + dx;
            int cxChunk = absX >> 4;
            int lastChunkZ = -999;
            net.minecraft.world.level.chunk.LevelChunk currentChunk = null;

            for (int dz = -LAVA_SCAN_RADIUS_XZ; dz <= LAVA_SCAN_RADIUS_XZ; dz += 2) {
                int absZ = cz + dz;
                int czChunk = absZ >> 4;

                if (currentChunk == null || czChunk != lastChunkZ) {
                    lastChunkZ = czChunk;
                    long chunkKey = chunkKey(cxChunk, czChunk);
                    currentChunk = loadedChunks.get(chunkKey);
                    if (currentChunk == null) {
                        currentChunk = cache.getChunkNow(cxChunk, czChunk);
                        if (currentChunk != null) loadedChunks.put(chunkKey, currentChunk);
                    }
                }

                if (currentChunk == null) continue;

                int localMinY = Math.max(level.getMinY(), minY);
                int localMaxY = Math.min(level.getMaxY() - 1, maxY);

                int minSection = (localMinY - level.getMinY()) >> 4;
                int maxSection = (localMaxY - level.getMinY()) >> 4;

                net.minecraft.world.level.chunk.LevelChunkSection[] sections = currentChunk.getSections();

                for (int secIndex = minSection; secIndex <= maxSection; secIndex++) {
                    if (secIndex < 0 || secIndex >= sections.length) continue;
                    net.minecraft.world.level.chunk.LevelChunkSection section = sections[secIndex];
                    if (section == null || section.hasOnlyAir()) continue;
                    if (!section.getStates().maybeHas(DimensionDefenseProcedure::isLava)) continue;

                    int secMinY = level.getMinY() + (secIndex << 4);
                    int startY = Math.max(localMinY, secMinY);
                    int endY = Math.min(localMaxY, secMinY + 15);

                    for (int dy = -LAVA_SCAN_RADIUS_Y; dy <= LAVA_SCAN_RADIUS_Y; dy += 2) {
                        int absY = cy + dy;
                        if (absY >= startY && absY <= endY) {
                            BlockState state = section.getBlockState(absX & 15, absY & 15, absZ & 15);
                            if (isLava(state)) count += 8;
                        }
                    }
                }
            }
        }
        return count;
    }

    private static boolean isLava(BlockState state) {
        if (state == null || state.isAir()) return false;
        if (state.is(Blocks.LAVA)) return true;
        net.minecraft.world.level.material.FluidState fluid = state.getFluidState();
        return !fluid.isEmpty() && (fluid.is(Fluids.LAVA) || fluid.is(Fluids.FLOWING_LAVA));
    }

    // =========================================================================
    // LAVA SUPPRESSION LOGIC
    // =========================================================================
    private static void tickLavaSuppression(Level level, Player player, ResourceKey<Level> dim) {
        if (!(level instanceof net.minecraft.server.level.ServerLevel serverLevel)) return;
        if (player.tickCount % SUPPRESSION_CHECK_INTERVAL_TICKS != 0) return;

        if (!isTargetDimension(dim)) return;

        // Trigger scan only when the player is deep underground where underground pools exist (-30 and below)
        if (player.getY() > -30) return;

        cleanupOldPlayerPlacedLava(level);

        net.minecraft.server.level.ServerChunkCache cache = serverLevel.getChunkSource();
        Map<Long, net.minecraft.world.level.chunk.LevelChunk> loadedChunks = new HashMap<>();
        BlockPos center = player.blockPosition();

        // Slicing to distribute performance budget over multiple ticks
        int totalSlices = 12;
        int currentSlice = (int) ((player.tickCount / SUPPRESSION_CHECK_INTERVAL_TICKS) % totalSlices);

        int range = SUPPRESSION_SCAN_RADIUS_XZ;
        int sliceWidth = (range * 2 + 1 + totalSlices - 1) / totalSlices;
        int startDx = -range + currentSlice * sliceWidth;
        int endDx = Math.min(range, startDx + sliceWidth - 1);

        // OPTIMIZATION: Reusable pos pointers prevent billions of temporary object allocations
        BlockPos.MutableBlockPos mutablePos = new BlockPos.MutableBlockPos();
        BlockPos.MutableBlockPos neighborPos = new BlockPos.MutableBlockPos();

        int checked = 0;
        int removed = 0;

        for (int dx = startDx; dx <= endDx; dx++) {
            int absX = center.getX() + dx;
            int cxChunk = absX >> 4;
            int lastChunkZ = -999;
            net.minecraft.world.level.chunk.LevelChunk currentChunk = null;

            for (int dz = -SUPPRESSION_SCAN_RADIUS_XZ; dz <= SUPPRESSION_SCAN_RADIUS_XZ; dz++) {
                int absZ = center.getZ() + dz;
                int czChunk = absZ >> 4;

                if (currentChunk == null || czChunk != lastChunkZ) {
                    lastChunkZ = czChunk;
                    long chunkKey = chunkKey(cxChunk, czChunk);
                    currentChunk = loadedChunks.get(chunkKey);
                    if (currentChunk == null) {
                        currentChunk = cache.getChunkNow(cxChunk, czChunk);
                        if (currentChunk != null) loadedChunks.put(chunkKey, currentChunk);
                    }
                }

                if (currentChunk == null) continue;

                int minBuildY = level.getMinY();
                if (-64 < minBuildY) continue;
                int secIndex = (-64 - minBuildY) >> 4;
                net.minecraft.world.level.chunk.LevelChunkSection[] sections = currentChunk.getSections();
                if (secIndex < 0 || secIndex >= sections.length) continue;
                net.minecraft.world.level.chunk.LevelChunkSection section = sections[secIndex];
                if (section == null || section.hasOnlyAir()) continue;
                if (!section.getStates().maybeHas(state -> isLava(state) || state.getBlock() instanceof BaseFireBlock || state.is(Blocks.FIRE) || state.is(Blocks.SOUL_FIRE))) continue;

                for (int absY = -64; absY <= -49; absY++) {
                    checked++;
                    if (checked > MAX_CHECKS_PER_PLAYER_SCAN) return;

                    BlockState state = section.getBlockState(absX & 15, absY & 15, absZ & 15);
                    if (state.isAir()) continue;

                    if (absY <= Y_THRESHOLD) {
                        // --- LAVA PLACEMENT ZONE (-64 to -54) ---
                        if (state.is(Blocks.LAVA) || state.getFluidState().is(Fluids.LAVA)) {
                            mutablePos.set(absX, absY, absZ);
                            // Skip any coordinates protected by player placement actions
                            if (PLAYER_PLACED_LAVA.containsKey(mutablePos.asLong())) continue;

                            level.setBlock(mutablePos, Blocks.OAK_PLANKS.defaultBlockState(), 2);
                            removed++;

                            // Extinguish adjacent fire blocks immediately around the newly converted wood
                            for (net.minecraft.core.Direction dir : net.minecraft.core.Direction.values()) {
                                if (dir == net.minecraft.core.Direction.DOWN) continue;
                                neighborPos.set(absX + dir.getStepX(), absY + dir.getStepY(), absZ + dir.getStepZ());
                                if (level.hasChunkAt(neighborPos)) {
                                    BlockState neighborState = level.getBlockState(neighborPos);
                                    if (neighborState.getBlock() instanceof net.minecraft.world.level.block.BaseFireBlock
                                        || neighborState.is(Blocks.FIRE)
                                        || neighborState.is(Blocks.SOUL_FIRE)) {
                                        level.setBlock(neighborPos, Blocks.AIR.defaultBlockState(), 3);
                                    }
                                }
                            }

                            if (removed >= MAX_REMOVALS_PER_PLAYER_SCAN) {
                                return;
                            }
                        }
                    } else {
                        // --- THE ACTIVE FIRE CEILING ZONE (-53 to -49) ---
                        // Extinguish any fire blocks in the air layers sitting directly on top of the suppression threshold
                        if (state.getBlock() instanceof net.minecraft.world.level.block.BaseFireBlock
                            || state.is(Blocks.FIRE)
                            || state.is(Blocks.SOUL_FIRE)) {
                            mutablePos.set(absX, absY, absZ);
                            level.setBlock(mutablePos, Blocks.AIR.defaultBlockState(), 3);
                        }
                    }
                }
            }
        }
    }

    private static void cleanupOldPlayerPlacedLava(Level level) {
        if (level.getGameTime() % PROTECTION_CLEANUP_INTERVAL_TICKS != 0) return;

        long now = level.getGameTime();
        PLAYER_PLACED_LAVA.entrySet().removeIf(entry -> now - entry.getValue() > PLAYER_LAVA_PROTECTION_TICKS);
    }

    }

    // =========================================================================
    // TNT DEFENSE LOGIC (Matched 1:1 with 1.21.1)
    // =========================================================================
    private static void tickTntDefense(Level level, Player player) {
        if (player.tickCount % TNT_CHECK_INTERVAL_TICKS != 0) return;

        int cd = player.getPersistentData().getIntOr(NBT_TNT_COOLDOWN_KEY, 0);
        if (cd > 0) {
            player.getPersistentData().putInt(NBT_TNT_COOLDOWN_KEY, Math.max(0, cd - TNT_CHECK_INTERVAL_TICKS));
            return;
        }

        BlockPos center = player.blockPosition();

        int tntCount = countPrimedTntInRadius(level, center);
        if (tntCount < BIG_TNT_THRESHOLD) return;

        player.addEffect(new MobEffectInstance(MobEffects.BLINDNESS, 60, 5, false, false));
        player.addEffect(new MobEffectInstance(MobEffects.DARKNESS, 80, 5, false, false));

        BuiltInRegistries.SOUND_EVENT.getOptional(Identifier.parse("minecraft:ambient.cave")).ifPresent(cave ->
            level.playSound(null, center, cave, SoundSource.AMBIENT, 1.3f, 0.85f)
        );

        BlockPos capturedCenter = center;
        Level capturedLevel = level;
        net.mcreator.thebackwoods.TheBackwoodsMod.queueServerWork(40, () -> {
            purgePrimedTntInRadius(capturedLevel, capturedCenter);
        });

        player.getPersistentData().putInt(NBT_TNT_COOLDOWN_KEY, TNT_PURGE_COOLDOWN_TICKS);
    }

    private static boolean isExplosiveEntity(Entity entity) {
        if (!entity.isAlive()) return false;
        if (entity instanceof PrimedTnt) return true;

        Identifier registryName = BuiltInRegistries.ENTITY_TYPE.getKey(entity.getType());
        if (registryName != null) {
            String idStr = registryName.toString();
            return idStr.equals("alexscaves:nuclear_bomb");
        }
        return false;
    }

    private static int countPrimedTntInRadius(Level level, BlockPos center) {
        AABB box = AABB.ofSize(
                new Vec3(center.getX() + 0.5, center.getY() + 0.5, center.getZ() + 0.5),
                TNT_SCAN_RADIUS_XZ * 2.0,
                TNT_SCAN_RADIUS_Y * 2.0,
                TNT_SCAN_RADIUS_XZ * 2.0
        );
        return level.getEntitiesOfClass(Entity.class, box, DimensionDefenseProcedure::isExplosiveEntity).size();
    }

    private static void purgePrimedTntInRadius(Level level, BlockPos center) {
        AABB box = AABB.ofSize(
                new Vec3(center.getX() + 0.5, center.getY() + 0.5, center.getZ() + 0.5),
                TNT_SCAN_RADIUS_XZ * 2.0,
                TNT_SCAN_RADIUS_Y * 2.0,
                TNT_SCAN_RADIUS_XZ * 2.0
        );

        List<Entity> tnts = level.getEntitiesOfClass(Entity.class, box, DimensionDefenseProcedure::isExplosiveEntity);
        for (Entity tnt : tnts) {
            Identifier registryName = BuiltInRegistries.ENTITY_TYPE.getKey(tnt.getType());
            if (registryName != null && registryName.toString().equals("alexscaves:nuclear_bomb")) {
                tnt.teleportTo(tnt.getX(), 600.0, tnt.getZ());
                tnt.setDeltaMovement(new Vec3(0, -0.2, 0));
                tnt.hurtMarked = true;
            } else {
                tnt.discard();
            }
        }
    }
