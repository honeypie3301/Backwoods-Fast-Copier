	public static boolean ENABLE_GLITCH = true;

	// Thread-safe triggers for client integration
	public static boolean clientIsNearby = false;
	public static double clientMinDistance = 24.0;
	public static double clientActiveIntensity = 0.0;
	private static int clientLastTick = -1;
	private static LevelData.RespawnData originalSpawn = null;

	private static double getEntityMultiplier(String className) {
		if (className.contains("LignumVermis") || className.contains("AshWeaver")) {
			return 0.15; // A LOT weaker
		}
		if (className.contains("Splinter")) {
			return 0.45; // Weaker
		}
		if (className.contains("FractusPrime") || className.contains("PrimeFractus") || className.contains("LignumGigas") || className.contains("Rot")) {
			return 1.8; // Stronger
		}
		return 1.0; // Normal for LignumPalus, StiltStalker, StiltWalker, regular Fractus, Hollow
	}

	@SubscribeEvent
	public static void onEntityTick(EntityTickEvent.Post event) {
		if (event.getEntity() != null && event.getEntity() instanceof Player player) {
			if (player.level().isClientSide()) {
				tickClientSideGlitch(player);
			}
		}
	}

	private static void tickClientSideGlitch(Player player) {
		net.minecraft.client.Minecraft mc = net.minecraft.client.Minecraft.getInstance();
		if (mc.level == null)
			return;

		if (!ENABLE_GLITCH || player.isCreative() || player.isSpectator()) {
			if (originalSpawn != null) {
				mc.level.setRespawnData(originalSpawn);
				originalSpawn = null;
			}
			clientIsNearby = false;
			clientActiveIntensity = 0.0;
			return;
		}

		int tickCount = player.tickCount;
		if (tickCount != clientLastTick) {
			clientLastTick = tickCount;

			// Scan creepy entities only once every 5 ticks to guarantee absolute zero footprint
			if (tickCount % 5 == 0) {
				clientIsNearby = false;
				clientMinDistance = 24.0;
				clientActiveIntensity = 0.0;
				List<Entity> surrounding = mc.level.getEntities(null, player.getBoundingBox().inflate(24));
				for (Entity e : surrounding) {
					String name = e.getClass().getName();

						double d = e.distanceTo(player);
						if (d < 24.0) {
							double mult = getEntityMultiplier(name);
							double intensity = mult * (1.0 - (d / 24.0));
							if (intensity > clientActiveIntensity) {
								clientActiveIntensity = intensity;
								clientMinDistance = d;
							}
							clientIsNearby = true;
						}
					}
				}
			}

			// Dynamic client-side compass needle glitching without changing NBT or converting to Lodestone Compass!

				// Initialize and store the real spawn position of the world once

					originalSpawn = mc.level.getRespawnData();
				}

				double ratio = Math.max(0.0, Math.min(1.0, clientActiveIntensity));
				// Scale the angle velocity and position jitter relative to player proximity

				double distance = 5.0 + (1.0 - ratio) * 200.0;
				double spawnX = player.getX() + Math.cos(angle) * distance;
				double spawnY = player.getY();
				double spawnZ = player.getZ() + Math.sin(angle) * distance;

				mc.level.setRespawnData(LevelData.RespawnData.of(mc.level.dimension(), new BlockPos((int) spawnX, (int) spawnY, (int) spawnZ), 0.0f, 0.0f));
			} else {
				// Re-align the world spawn position cleanly when creepy entities retreat
				if (originalSpawn != null) {
					mc.level.setRespawnData(originalSpawn);
					originalSpawn = null;
				}
			}
		}
	}

@EventBusSubscriber(value = Dist.CLIENT)
class ClientCompassAndF3GlitchSubscriber {

	private static boolean isBlank(String str) {
		if (str == null) return true;
		int len = str.length();
		if (len == 0) return true;
		for (int i = 0; i < len; i++) {
			if (!Character.isWhitespace(str.charAt(i))) {
				return false;
			}
		}
		return true;
	}

	private static boolean containsIgnoreCase(String src, String search) {
		if (src == null || search == null) return false;
		final int length = search.length();
		if (length == 0) return true;
		for (int i = src.length() - length; i >= 0; i--) {
			if (src.regionMatches(true, i, search, 0, length)) {
				return true;
			}
		}
		return false;
	}

	private static String fastStripColor(String input) {
		if (input == null) return "";
		int len = input.length();
		if (len == 0) return "";
		int firstSection = input.indexOf('\u00A7');
		if (firstSection == -1) {
			return input;
		}
		StringBuilder sb = new StringBuilder(len);
		for (int i = 0; i < len; i++) {
			char c = input.charAt(i);
			if (c == '\u00A7' && i + 1 < len) {
				i++; // Skip the § and the following format code
			} else {
				sb.append(c);
			}
		}
		return sb.toString();
	}

    private static boolean installedDebugEntries;

    @SubscribeEvent
    public static void onClientTick(ClientTickEvent.Post event) {
        if (installedDebugEntries) return;
        Map<Identifier, DebugScreenEntry> entries = DebugScreenEntries.allEntries();
        if (entries.isEmpty()) return;
        try {
            entries.replaceAll((id, original) -> original instanceof GlitchedEntry ? original : new GlitchedEntry(original, id));
            installedDebugEntries = true;
        } catch (UnsupportedOperationException ignored) {
            // The registry may be read-only in some loader builds; leave the vanilla entries intact.
        }
    }

    private static class GlitchedEntry implements DebugScreenEntry {
        private final DebugScreenEntry original;
        private final Identifier id;

        private GlitchedEntry(DebugScreenEntry original, Identifier id) {
            this.original = original;
            this.id = id;
        }

        @Override
        public DebugEntryCategory category() {
            return original.category();
        }

        @Override
        public boolean isAllowed(boolean reducedDebugInfo) {
            return original.isAllowed(reducedDebugInfo);
        }

        @Override
        public void display(DebugScreenDisplayer displayer, Level level, LevelChunk clientChunk, LevelChunk serverChunk) {
            boolean right = id.equals(DebugScreenEntries.SYSTEM_SPECS)
                    || id.equals(DebugScreenEntries.MEMORY)
                    || id.equals(DebugScreenEntries.DETAILED_MEMORY);
            original.display(new DebugScreenDisplayer() {
                @Override
                public void addLine(String line) {
                    displayer.addLine(scramble(line, right));
                }

                @Override
                public void addPriorityLine(String line) {
                    displayer.addPriorityLine(scramble(line, right));
                }

                @Override
                public void addToGroup(Identifier group, String line) {
                    displayer.addToGroup(group, scramble(line, right));
                }

                @Override
                public void addToGroup(Identifier group, Collection<String> lines) {
                    java.util.List<String> changed = new java.util.ArrayList<>(lines.size());
                    for (String line : lines) changed.add(scramble(line, right));
                    displayer.addToGroup(group, changed);
                }
            }, level, clientChunk, serverChunk);
        }
    }

    private static String scramble(String line, boolean right) {
        if (!CompassAndF3GlitchProcedure.ENABLE_GLITCH || isBlank(line) || line.startsWith("§f§k")) return line;
        net.minecraft.client.Minecraft mc = net.minecraft.client.Minecraft.getInstance();
        if (mc.player == null || mc.level == null || mc.player.isCreative() || mc.player.isSpectator()
                || !CompassAndF3GlitchProcedure.clientIsNearby || CompassAndF3GlitchProcedure.clientActiveIntensity <= 0.005) return line;

        double ratio = Math.max(0.0, Math.min(1.0, CompassAndF3GlitchProcedure.clientActiveIntensity));
        boolean shouldScramble;
        if (ratio >= 0.8) {
            shouldScramble = true;
        } else if (right) {
            shouldScramble = ratio >= 0.4
                    ? line.contains("CPU:") || line.contains("Display:") || line.contains("Server:") || line.contains("Client:")
                        || line.contains("Mem:") || containsIgnoreCase(line, "nvidia") || containsIgnoreCase(line, "intel") || containsIgnoreCase(line, "amd")
                    : line.contains("Server:") || line.contains("Client:");
        } else {
            shouldScramble = ratio >= 0.4
                    ? line.startsWith("XYZ:") || line.startsWith("Block:") || line.startsWith("Chunk:") || line.startsWith("Facing:")
                        || line.startsWith("Biome:") || line.startsWith("E:") || containsIgnoreCase(line, "fps")
                    : line.startsWith("Facing:") || line.startsWith("Biome:") || line.startsWith("E:");
        }
        return shouldScramble && Math.random() < ratio ? "§f§k" + fastStripColor(line) : line;
    }
