	private static void execute(@Nullable Event event, LevelAccessor world, double x, double y, double z, Entity entity) {
		// AshWeaverEntity.baseTick() also calls this procedure. The post-tick event is the
		// sole authority so vanilla goals cannot overwrite this state machine afterward.
		if (event == null || !(world instanceof ServerLevel level) || !(entity instanceof AshWeaverEntity ashWeaver)) return;
		if (!ashWeaver.isAlive()) {
			cancelTransientActions(ashWeaver, level.getGameTime());
			return;
		}

		CompoundTag data = ashWeaver.getPersistentData();
		long gameTime = level.getGameTime();
		if (dataLong(data, KEY_LAST_TICK, Long.MIN_VALUE) == gameTime) return;
		long previousTick = dataLong(data, KEY_LAST_TICK, gameTime);
		if (gameTime - previousTick > 1L) cancelTransientActions(ashWeaver, gameTime);
		data.putLong(KEY_LAST_TICK, gameTime);
		if (!dataBoolean(data, "ashWeaverInitialized", false))
			AshWeaverOnInitialEntitySpawnProcedure.initialize(ashWeaver);
		rememberNearDeath(level, ashWeaver, data, gameTime);

		int roseCooldown = ashWeaver.getEntityData().get(AshWeaverEntity.DATA_roseCooldown);
		if (roseCooldown > 0) ashWeaver.getEntityData().set(AshWeaverEntity.DATA_roseCooldown, roseCooldown - 1);

		Player assignedPlayer = resolveAssignedPlayer(level, ashWeaver);
		if (assignedPlayer == null) assignedPlayer = tryAssignPlayer(level, ashWeaver, x, y, z);
		if (lifecycleOwnsTick(level, ashWeaver, assignedPlayer, data, gameTime)) return;
		AshWeaverChatCommandProcedure.processPendingReply(ashWeaver);
		updateHomeSurvey(level, ashWeaver, data, gameTime);
		if (dataBoolean(data, "ashWeaverHomeStay", false) && homeHere(level, data)
				&& (assignedPlayer == null || !assignedPlayer.isAlive() || assignedPlayer.isSpectator() || assignedPlayer.level() != level)) {
			ensurePlayerPace(ashWeaver);
			if (!ashWeaver.isInWater() && handleFearAndHiding(level, ashWeaver, null, data, gameTime)) return;
			handleHomeStay(level, ashWeaver, null, data, gameTime);
			return;
		}
		if (assignedPlayer == null) {
			resetVisibleAction(ashWeaver);
			if (dataString(data, "ashWeaverRelationshipOwner", "").isEmpty()
					&& ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer).isEmpty()
					&& !dataBoolean(data, "ashWeaverHomeKnown", false) && ashWeaver.tickCount % 100 == 0
					&& level.getNearestPlayer(x, y, z, 128.0, false) == null) ashWeaver.discard();
			return;
		}

		// Keep loyalty when the owner changes dimensions; never silently bind to another player.
		if (assignedPlayer.level() != level) {
			setState(data, STATE_REGROUP, gameTime, gameTime + 40L);
			ashWeaver.getNavigation().stop();
			resetVisibleAction(ashWeaver);
			return;
		}
		updateRelationship(ashWeaver, assignedPlayer, data, gameTime);
		ensurePlayerPace(ashWeaver);
		if (ashWeaver.isInWater()) {
			clearHeldRose(ashWeaver);
			setSneaking(ashWeaver, false);
			ashWeaver.setSprinting(false);
			clearPlantingIntent(data);
			data.remove(KEY_PARKOUR_JUMPED);
			if (dataBoolean(data, KEY_CHAT_STAY, false) || dataBoolean(data, "ashWeaverHomeStay", false))
				ashWeaver.getNavigation().stop();
			else if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) == 0)
				ashWeaver.getNavigation().moveTo(assignedPlayer, PLAYER_NAV_SPEED);
			// Float/swim behavior retains vertical control; never apply ground jumps in water.
			return;
		}

		if (handleFearAndHiding(level, ashWeaver, assignedPlayer, data, gameTime)) return;

		// Avoidance owns movement before follow, recovery, planting and door handlers run.
		if (dataBoolean(data, "ashWeaverAvoidOwner", false)) {
			handleRelationshipAvoidance(level, ashWeaver, assignedPlayer, data, gameTime);
			return;
		}

		LivingEntity attacker = findOwnerAttacker(ashWeaver, assignedPlayer);
		if (dataBoolean(data, "ashWeaverHomeStay", false) && homeHere(level, data)
				&& ashWeaver.distanceToSqr(assignedPlayer) > 12.0 * 12.0)
			attacker = null;
		Entity threat = updateThreatAwareness(level, ashWeaver, data, gameTime);
		boolean immediateDanger = threat != null && ashWeaver.distanceToSqr(threat) <= IMMEDIATE_DANGER_RANGE_SQ;
		if (dataBoolean(data, "ashWeaverHomeStay", false) && homeHere(level, data) && !immediateDanger
				&& (attacker == null || ashWeaver.distanceToSqr(assignedPlayer) > 12.0 * 12.0)) {
			handleHomeStay(level, ashWeaver, assignedPlayer, data, gameTime);
			return;
		}
		int state = dataInt(data, KEY_STATE, STATE_IDLE);
		if (!data.getBoolean(KEY_CHAT_STAY) && state != STATE_SEEK_PLANT_SITE && state != STATE_PLANT && state != STATE_WAIT
				&& state != STATE_REGROUP && (attacker == null || attacker instanceof Mob mob
					&& isHostileSplinter(mob) && mob.getTarget() == assignedPlayer)
				&& tryStartPlanting(level, ashWeaver, assignedPlayer, data, gameTime))
			state = STATE_SEEK_PLANT_SITE;
		boolean planting = state == STATE_SEEK_PLANT_SITE || state == STATE_PLANT;
		boolean directThreat = threat instanceof Mob mob && mob.getTarget() == ashWeaver;

		if (attacker != null && !(planting && attacker instanceof Mob mob
				&& isHostileSplinter(mob) && mob.getTarget() == assignedPlayer)
				&& (state != STATE_DEFEND || !attacker.getUUID().toString().equals(dataString(data, KEY_DEFEND_UUID, "")))) {
			clearPlantingIntent(data);
			data.putString(KEY_DEFEND_UUID, attacker.getUUID().toString());
			setState(data, STATE_DEFEND, gameTime, gameTime + 120L);
			state = STATE_DEFEND;
		} else if (immediateDanger && (!planting || directThreat) && state != STATE_ALERT && state != STATE_DEFEND) {
			clearPlantingIntent(data);
			setState(data, STATE_ALERT, gameTime, gameTime + 30L);
			state = STATE_ALERT;
		} else if (!immediateDanger && state != STATE_DEFEND && state != STATE_ALERT
				&& gameTime >= dataLong(data, KEY_NEXT_DODGE, 0L) && isOwnerAimingAtWeaver(assignedPlayer, ashWeaver)) {
			if (startDodge(level, ashWeaver, assignedPlayer, data, gameTime)) {
				clearPlantingIntent(data);
				state = STATE_DODGE;
			}
		} else if (!immediateDanger && threat != null && !planting && state != STATE_DEFEND && state != STATE_ALERT
				&& state != STATE_DODGE && state != STATE_WARN && gameTime >= dataLong(data, KEY_NEXT_WARNING, 0L)
				&& shouldWarnPlayer(ashWeaver, assignedPlayer, threat)) {
			clearPlantingIntent(data);
			setState(data, STATE_WARN, gameTime, gameTime + 25L);
			data.putLong(KEY_NEXT_WARNING, gameTime + 180L);
			state = STATE_WARN;
		}

		if (!dataBoolean(data, KEY_CHAT_STAY, false)
				&& !(dataBoolean(data, "ashWeaverHomeStay", false) && homeHere(level, data)))
			updateProgressAndRecovery(level, ashWeaver, assignedPlayer, data, state, gameTime);
		state = dataInt(data, KEY_STATE, state);
		if (ashWeaver.distanceToSqr(assignedPlayer) > 10.0 * 10.0
				&& (state == STATE_SEEK_PLANT_SITE || state == STATE_PLANT || state == STATE_WAIT)) {
			clearPlantingIntent(data);
			clearHeldRose(ashWeaver);
			setSneaking(ashWeaver, false);
			setState(data, STATE_FOLLOW, gameTime, 0L);
			state = STATE_FOLLOW;
		}
		if (ashWeaver.distanceToSqr(assignedPlayer) > 20.0 * 20.0 && state == STATE_DEFEND) {
			data.remove(KEY_DEFEND_UUID);
			setState(data, STATE_FOLLOW, gameTime, 0L);
			state = STATE_FOLLOW;
		}

		if (data.getBoolean(KEY_CHAT_STAY) && state != STATE_DEFEND && state != STATE_DODGE
				&& state != STATE_WARN && state != STATE_ALERT) {
			clearPlantingIntent(data);
			clearHeldRose(ashWeaver);
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			setSneaking(ashWeaver, false);
			data.remove(KEY_PARKOUR_JUMPED);
			setStateIfChanged(data, STATE_IDLE, gameTime, 0L);
			state = STATE_IDLE;
		} else switch (state) {
			case STATE_DEFEND -> handleDefend(level, ashWeaver, assignedPlayer, data, gameTime);
			case STATE_DODGE -> handleDodge(ashWeaver, assignedPlayer, data, gameTime);
			case STATE_WARN -> handleWarn(level, ashWeaver, assignedPlayer, threat, data, gameTime);
			case STATE_ALERT -> handleAlert(level, ashWeaver, assignedPlayer, threat, data, gameTime);
			case STATE_SEEK_PLANT_SITE -> handleSeekPlantSite(level, ashWeaver, assignedPlayer, data, gameTime);
			case STATE_PLANT -> handlePlant(level, ashWeaver, assignedPlayer, data, gameTime);
			case STATE_WAIT -> handleWait(ashWeaver, assignedPlayer, data, gameTime);
			case STATE_REGROUP -> handleRegroup(level, ashWeaver, assignedPlayer, data, gameTime);
			default -> handleSocialFollowAndPlanting(level, ashWeaver, assignedPlayer, threat, data, gameTime);
		}

		handleDoorInteraction(level, ashWeaver, data, state, gameTime);
		maybeDropEmergencyRose(level, ashWeaver, assignedPlayer, data, gameTime, state);
		continueParkour(level, ashWeaver, data, gameTime);
		applySneakEdgeBrake(level, ashWeaver);
		enforceGroundSpeed(ashWeaver);
		separateFromOtherWeavers(level, ashWeaver, gameTime);
		AshWeaverChatCommandProcedure.updateConversationLook(ashWeaver);
	}

	private static net.minecraft.resources.ResourceLocation bondKey(String owner) {

	}

	public static boolean isCanonicalCompanion(AshWeaverEntity ashWeaver, String owner) {
		if (!(ashWeaver.level() instanceof ServerLevel level) || owner == null || owner.isEmpty()) return false;
		try {
			return ashWeaver.getUUID().toString().equals(dataString(bondRecord(level, owner), "Weaver", ""));
		} catch (IllegalArgumentException ignored) { return false; }
	}

	private static boolean registerBond(ServerLevel level, AshWeaverEntity ashWeaver, String owner, long now) {
		CompoundTag record = bondRecord(level, owner);
		String claimed = dataString(record, "Weaver", "");
		if (!claimed.isEmpty() && !claimed.equals(ashWeaver.getUUID().toString())) return false;
		if (claimed.isEmpty() && hasEarlierClaim(level, ashWeaver, owner)) return false;
		ashWeaver.setPersistenceRequired();
		if (claimed.isEmpty() || now >= dataLong(ashWeaver.getPersistentData(), "ashWeaverBondCheckpoint", 0L)) {
			record = record.copy();
			record.putString("Weaver", ashWeaver.getUUID().toString());
			record.putString("Dimension", level.dimension().toString());
			record.putLong("Pos", ashWeaver.blockPosition().asLong());
			saveBond(level, owner, record);
			ashWeaver.getPersistentData().putLong("ashWeaverBondCheckpoint", now + 200L);
		}
		return true;
	}

	private static void cancelTransientActions(AshWeaverEntity ashWeaver, long now) {
		CompoundTag data = ashWeaver.getPersistentData();
		clearPlantingIntent(data);
		clearHeldRose(ashWeaver);
		ashWeaver.setTarget(null);
		ashWeaver.getNavigation().stop();
		ashWeaver.setSprinting(false);
		setSneaking(ashWeaver, false);
		for (String key : new String[] {KEY_DEFEND_UUID, KEY_THREAT_UUID, KEY_CRIT_UNTIL, KEY_PARKOUR_JUMPED,
				KEY_PARKOUR_X, KEY_PARKOUR_Y, KEY_PARKOUR_Z, KEY_BOUNDARY_UNTIL, KEY_DETOUR_UNTIL,
				KEY_DOOR_PENDING, KEY_DOOR_POS, KEY_DOOR_CLOSE_AT, KEY_DISTRACTION_UUID, KEY_DISTRACTION_UNTIL,
				KEY_MIRROR_UNTIL, KEY_EYE_CONTACT_UNTIL, KEY_AIM_WATCH_UNTIL, KEY_BOLD_UNTIL, KEY_HESITATE_UNTIL,
				"ashWeaverHomePendingAction", "ashWeaverHomePendingUntil", "ashWeaverHomePendingDimension",
				"ashWeaverHomePendingPos"}) data.remove(key);
		data.putInt(KEY_STUCK_TICKS, 0);
		setState(data, STATE_IDLE, now, 0L);
		AshWeaverChatCommandProcedure.cancelConversation(ashWeaver);
	}

				&& dataInt(ashWeaver.getPersistentData(), KEY_STATE, STATE_IDLE) != STATE_HIDE)

	}


	@SubscribeEvent
	public static void onOwnerTravel(net.neoforged.neoforge.event.entity.EntityTravelToDimensionEvent event) {
		if (!(event.getEntity() instanceof Player owner) || !(owner.level() instanceof ServerLevel level)) return;
		BlockPos feet = owner.blockPosition();
		BlockPos entrance = null;
		for (int dx = -1; dx <= 1 && entrance == null; dx++)
			for (int dz = -1; dz <= 1 && entrance == null; dz++) for (int dy = -1; dy <= 1; dy++) {
				BlockPos pos = feet.offset(dx, dy, dz);
				if (level.hasChunkAt(pos) && level.getBlockState(pos).getBlock() instanceof net.minecraft.world.level.block.Portal
						&& owner.getBoundingBox().inflate(0.1).intersects(new AABB(pos))) { entrance = pos; break; }
			}
		if (entrance == null) return; // Commands, respawns and unknown teleporters are not assumed to be portals.
		for (AshWeaverEntity weaver : level.getEntitiesOfClass(AshWeaverEntity.class, owner.getBoundingBox().inflate(16.0),
				candidate -> owner.getUUID().toString().equals(candidate.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer)))) {
			if (!weaver.hasLineOfSight(owner) || !isCanonicalCompanion(weaver, owner.getUUID().toString())) continue;
			CompoundTag data = weaver.getPersistentData();
			data.putLong("ashWeaverPortalPos", entrance.asLong());
			data.putString("ashWeaverPortalBlock", BuiltInRegistries.BLOCK.getKey(level.getBlockState(entrance).getBlock()).toString());
			data.putString("ashWeaverPortalFrom", level.dimension().toString());
			data.putLong("ashWeaverPortalSeenAt", level.getGameTime());
		}
	}


	private static void holdWitnessedPortalRoute(ServerLevel level, AshWeaverEntity weaver, BlockPos portal) {
		// One short-lived vanilla lease per crossing, only for the two already-loaded route chunks.
		// Never refresh this from the entity tick or turn it into a permanent chunk loader.
		for (BlockPos pos : new BlockPos[] {weaver.blockPosition(), portal}) {
			if (!level.hasChunkAt(pos)) continue;
			level.getChunkSource().addRegionTicket(net.minecraft.server.level.TicketType.PORTAL,
					new net.minecraft.world.level.ChunkPos(pos), 2, pos);
		}
	}


	@SubscribeEvent
	public static void onOwnerDimensionChange(net.neoforged.neoforge.event.entity.player.PlayerEvent.PlayerChangedDimensionEvent event) {
		Player owner = event.getEntity();
		if (!(owner.level() instanceof ServerLevel current)) return;
		for (ServerLevel source : current.getServer().getAllLevels()) for (Entity candidate : source.getAllEntities()) {
			if (!(candidate instanceof AshWeaverEntity weaver)
					|| !owner.getUUID().toString().equals(weaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))) continue;
			CompoundTag data = weaver.getPersistentData();
			long now = source.getGameTime();
			boolean witnessed = event.getFrom().toString().equals(dataString(data, "ashWeaverPortalFrom", ""))
					&& now >= dataLong(data, "ashWeaverPortalCooldown", 0L)
					&& now - dataLong(data, "ashWeaverPortalSeenAt", Long.MIN_VALUE / 2L) <= 60L;
			cancelTransientActions(weaver, now);
			if (witnessed && !dataBoolean(data, KEY_CHAT_STAY, false) && !dataBoolean(data, "ashWeaverHomeStay", false)) {
				data.putString("ashWeaverPortalTo", event.getTo().toString());
				data.putLong("ashWeaverPortalUntil", now + 600L);
				data.putLong("ashWeaverPortalReady", now + 8L + weaver.getRandom().nextInt(13));
				data.putInt("ashWeaverPortalAttempts", 0);
				holdWitnessedPortalRoute(source, weaver, BlockPos.of(dataLong(data, "ashWeaverPortalPos", 0L)));
			} else data.remove("ashWeaverPortalUntil");
		}
	}

	private static void resetOwnerActions(Player owner) {
		if (!(owner.level() instanceof ServerLevel level)) return;
		for (ServerLevel dimension : level.getServer().getAllLevels()) for (Entity candidate : dimension.getAllEntities())
			if (candidate instanceof AshWeaverEntity weaver
					&& owner.getUUID().toString().equals(weaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))) {
				cancelTransientActions(weaver, dimension.getGameTime());
				weaver.getPersistentData().remove("ashWeaverPortalUntil");
				weaver.getPersistentData().remove("ashWeaverPortalSeenAt");
			}
	}

	@SubscribeEvent
	public static void onCompanionRemoved(net.neoforged.neoforge.event.entity.EntityLeaveLevelEvent event) {
		if (!(event.getLevel() instanceof ServerLevel level) || !(event.getEntity() instanceof AshWeaverEntity weaver)) return;
		Entity.RemovalReason reason = weaver.getRemovalReason();
		// Unload and dimension transfer preserve the claim; actual removal releases only this UUID.
		if (reason != Entity.RemovalReason.KILLED && reason != Entity.RemovalReason.DISCARDED) return;
		String owner = weaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer);
		if (!isCanonicalCompanion(weaver, owner)) return;
		CompoundTag record = bondRecord(level, owner).copy();
		for (String key : new String[] {"Weaver", "Dimension", "Pos"}) record.remove(key);
		saveBond(level, owner, record); // Rose quotas deliberately survive deaths and replacement companions.
		HOME_SURVEYS.remove(weaver);
		AshWeaverChatCommandProcedure.cancelConversation(weaver);
	}

	private static void observeOwnerPortal(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long now) {
		if (now < dataLong(data, "ashWeaverPortalCooldown", 0L)
				|| Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 5) != 0
				|| ashWeaver.distanceToSqr(player) > 16.0 * 16.0 || !ashWeaver.hasLineOfSight(player)) return;
		BlockPos feet = player.blockPosition();
		for (int dx = -1; dx <= 1; dx++) for (int dz = -1; dz <= 1; dz++) for (int dy = 0; dy <= 1; dy++) {
			BlockPos portal = feet.offset(dx, dy, dz);
			if (!level.hasChunkAt(portal) || !(level.getBlockState(portal).getBlock() instanceof net.minecraft.world.level.block.Portal)
					|| !player.getBoundingBox().inflate(0.1).intersects(new AABB(portal))) continue;
			data.putLong("ashWeaverPortalPos", portal.asLong());
			data.putString("ashWeaverPortalBlock", BuiltInRegistries.BLOCK.getKey(level.getBlockState(portal).getBlock()).toString());
			data.putString("ashWeaverPortalFrom", level.dimension().toString());
			data.putLong("ashWeaverPortalSeenAt", now);
			return;
		}
	}

	private static boolean portalPathSafe(ServerLevel level, Path path, BlockPos portal) {
		if (path == null || !path.canReach() || path.getNodeCount() > 80) return false;
		for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i++) {
			var node = path.getNode(i);
			BlockPos feet = new BlockPos(node.x, node.y, node.z);
			if (!level.hasChunkAt(feet) || !level.getWorldBorder().isWithinBounds(feet)) return false;
			boolean gate = feet.distSqr(portal) <= 4.0
					&& level.getBlockState(feet).getBlock() instanceof net.minecraft.world.level.block.Portal;
			if (!gate && !isSafeStandingPosition(level, feet)
					&& !(level.getBlockState(feet).getBlock() instanceof DoorBlock)) return false;
			BlockPos below = feet.below();
			BlockState floor = level.getBlockState(below);
			if (!floor.isFaceSturdy(level, below, Direction.UP) || isHazardousFloor(floor)
					|| !floor.getFluidState().isEmpty()) return false;
			if (i > path.getNextNodeIndex() && Math.abs(node.y - path.getNode(i - 1).y) > 1) return false;
		}
		return true;
	}

	private static void handlePortalFollow(ServerLevel level, AshWeaverEntity ashWeaver, Player owner, CompoundTag data, long now) {
		ashWeaver.setTarget(null);
		clearHeldRose(ashWeaver);
		if (dataBoolean(data, KEY_CHAT_STAY, false) || dataBoolean(data, "ashWeaverHomeStay", false)
				|| dataBoolean(data, "ashWeaverAvoidOwner", false) || ashWeaver.isPassenger()
				|| now >= dataLong(data, "ashWeaverPortalUntil", 0L)
				|| !level.dimension().toString().equals(dataString(data, "ashWeaverPortalFrom", ""))
				|| !owner.level().dimension().toString().equals(dataString(data, "ashWeaverPortalTo", ""))) {
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			setStateIfChanged(data, STATE_TRAVEL, now, 0L);
			return;
		}
		BlockPos portal = BlockPos.of(dataLong(data, "ashWeaverPortalPos", 0L));
		if (!level.hasChunkAt(portal) || ashWeaver.distanceToSqr(Vec3.atCenterOf(portal)) > 32.0 * 32.0) {
			data.remove("ashWeaverPortalUntil"); ashWeaver.getNavigation().stop(); return;
		}
		var block = level.getBlockState(portal).getBlock();
		if (!(block instanceof net.minecraft.world.level.block.Portal gate)
				|| !BuiltInRegistries.BLOCK.getKey(block).toString().equals(dataString(data, "ashWeaverPortalBlock", ""))) {
			data.remove("ashWeaverPortalUntil"); ashWeaver.getNavigation().stop(); return;
		}
		setStateIfChanged(data, STATE_TRAVEL, now, 0L);
		ashWeaver.getLookControl().setLookAt(portal.getX() + 0.5, portal.getY() + 0.8, portal.getZ() + 0.5, 15.0F, 12.0F);
		if (now < dataLong(data, "ashWeaverPortalReady", 0L)) {
			ashWeaver.getNavigation().stop(); ashWeaver.setSprinting(false); return;
		}
		if (ashWeaver.getBoundingBox().intersects(new AABB(portal))) {
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			if (ashWeaver.canUsePortal(false) && !ashWeaver.isOnPortalCooldown()) ashWeaver.setAsInsidePortal(gate, portal);
			return; // The portal itself decides the destination; never synthesize a cross-dimension teleport.
		}
		if (now >= dataLong(data, "ashWeaverPortalRepath", 0L)) {
			data.putLong("ashWeaverPortalRepath", now + 40L);
			Path path = fearCreatePath(level, ashWeaver, portal);
			if (portalPathSafe(level, path, portal)) ashWeaver.getNavigation().moveTo(path, PLAYER_NAV_SPEED);
			else {
				int attempts = dataInt(data, "ashWeaverPortalAttempts", 0) + 1;
				data.putInt("ashWeaverPortalAttempts", attempts);
				ashWeaver.getNavigation().stop();
				if (attempts >= 3) data.remove("ashWeaverPortalUntil");
			}
		}
		setRunning(ashWeaver, !ashWeaver.getNavigation().isDone() && ashWeaver.distanceToSqr(Vec3.atCenterOf(portal)) > 8.0 * 8.0);
		handleDoorInteraction(level, ashWeaver, data, STATE_TRAVEL, now);
		applySneakEdgeBrake(level, ashWeaver);
		enforceGroundSpeed(ashWeaver);
	}

	private static boolean lifecycleOwnsTick(ServerLevel level, AshWeaverEntity ashWeaver, @Nullable Player owner,
			CompoundTag data, long now) {
		String dimension = level.dimension().toString();
		String previous = dataString(data, "ashWeaverLastDimension", dimension);
		if (!dimension.equals(previous)) {
			cancelTransientActions(ashWeaver, now);
			data.remove("ashWeaverPortalUntil");
			data.putLong("ashWeaverPortalCooldown", now + 200L);
			data.remove("ashWeaverBondCheckpoint");
			if (!homeHere(level, data)) data.remove("ashWeaverHomeStay");
		}
		data.putString("ashWeaverLastDimension", dimension);
		if (dataBoolean(data, "ashWeaverDuplicateClaim", false)) {
			if (dataInt(data, KEY_STATE, STATE_IDLE) != STATE_TRAVEL) cancelTransientActions(ashWeaver, now);
			setStateIfChanged(data, STATE_TRAVEL, now, 0L);
			ashWeaver.getNavigation().stop();
			return true;
		}
		if (owner == null || !owner.isAlive() || owner.isSpectator()) {
			if (!dataBoolean(data, "ashWeaverOwnerUnavailable", false)) {
				cancelTransientActions(ashWeaver, now);
				data.putBoolean("ashWeaverOwnerUnavailable", true);
			}
			if (!ashWeaver.isInWater() && handleFearAndHiding(level, ashWeaver, null, data, now)) return true;
			if (owner == null && ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer).isEmpty()) return false;
			boolean waiting = !dataBoolean(data, "ashWeaverHomeStay", false) || !homeHere(level, data);
			if (waiting) {
				ashWeaver.getNavigation().stop();
				ashWeaver.setSprinting(false);
				setStateIfChanged(data, STATE_TRAVEL, now, 0L);
			}
			return waiting;
		}
		data.remove("ashWeaverOwnerUnavailable");
		if (owner.level() != level) {
			if (dataBoolean(data, "ashWeaverHomeStay", false) && homeHere(level, data)) return false;
			if (!ashWeaver.isInWater() && handleFearAndHiding(level, ashWeaver, null, data, now)) return true;
			handlePortalFollow(level, ashWeaver, owner, data, now); return true;
		}
		if (data.contains("ashWeaverPortalUntil")) {
			cancelTransientActions(ashWeaver, now);
			data.remove("ashWeaverPortalUntil");
		}
		observeOwnerPortal(level, ashWeaver, owner, data, now);
		if (dataInt(data, KEY_STATE, STATE_IDLE) == STATE_TRAVEL)
			setState(data, STATE_FOLLOW, now, 0L);
		if (ashWeaver.isPassenger()) {
			if (!dataBoolean(data, "ashWeaverRiding", false)) cancelTransientActions(ashWeaver, now);
			data.putBoolean("ashWeaverRiding", true);
			setStateIfChanged(data, STATE_TRAVEL, now, 0L);
			ashWeaver.getLookControl().setLookAt(owner, 12.0F, 10.0F);
			return true; // Vehicle control belongs to its rider, not navigation/parkour.
		}
		data.remove("ashWeaverRiding");
		return false;
	}

	private static boolean homeHere(ServerLevel level, CompoundTag data) {
		return dataBoolean(data, "ashWeaverHomeKnown", false)
				&& level.dimension().toString().equals(dataString(data, "ashWeaverHomeDimension", ""));
	}

	@Nullable
	private static BlockHitResult homeRay(ServerLevel level, AshWeaverEntity ashWeaver, Vec3 from, Vec3 to) {
		int minX = ((int) Math.floor(Math.min(from.x, to.x))) >> 4;
		int maxX = ((int) Math.floor(Math.max(from.x, to.x))) >> 4;
		int minZ = ((int) Math.floor(Math.min(from.z, to.z))) >> 4;
		int maxZ = ((int) Math.floor(Math.max(from.z, to.z))) >> 4;
		if (maxX - minX > 4 || maxZ - minZ > 4) return null;
		for (int cx = minX; cx <= maxX; cx++) for (int cz = minZ; cz <= maxZ; cz++)
			if (!level.hasChunkAt(new BlockPos(cx << 4, (int) from.y, cz << 4))) return null;
		return level.clip(new ClipContext(from, to, ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, ashWeaver));
	}

	// 0 = unobserved, 1 = sheltered interior (including caves), 2 = exposed/partly roofed.
	private static int homeKind(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos feet) {
		Vec3 eye = Vec3.atBottomCenterOf(feet).add(0.0, 1.5, 0.0);
		BlockHitResult roof = homeRay(level, ashWeaver, eye, eye.add(0.0, 24.0, 0.0));
		if (roof == null) return 0;
		if (roof.getType() != HitResult.Type.BLOCK) return level.canSeeSky(feet.above()) ? 2 : 0;
		int walls = 0;
		for (int side = 0; side < 8; side++) {
			double angle = side * Math.PI / 4.0;
			BlockHitResult wall = homeRay(level, ashWeaver, eye,
					eye.add(Math.cos(angle) * 24.0, 0.0, Math.sin(angle) * 24.0));
			if (wall == null) return 0; // Never load chunks to decide whether a shelter is real.
			if (wall.getType() == HitResult.Type.BLOCK) walls++;
		}
		return walls >= 5 ? 1 : 2;
	}

	// Command-only footprint check: a roof above a single pillar is not a usable room.
	private static boolean homeInteriorSafe(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos seed) {
		if (!isSafeStandingPosition(level, seed) || homeKind(level, ashWeaver, seed) != 1) return false;
		int shelteredFloor = 0;
		for (Direction side : Direction.Plane.HORIZONTAL) {
			BlockPos next = homeFloor(level, seed.relative(side));
			if (next == null) continue;
			Vec3 eye = Vec3.atBottomCenterOf(next).add(0.0, 1.5, 0.0);
			BlockHitResult access = homeRay(level, ashWeaver,
					Vec3.atBottomCenterOf(seed).add(0.0, 1.5, 0.0), eye);
			if (access == null || access.getType() == HitResult.Type.BLOCK) continue;
			BlockHitResult roof = homeRay(level, ashWeaver, eye, eye.add(0.0, 24.0, 0.0));
			if (roof != null && roof.getType() == HitResult.Type.BLOCK) shelteredFloor++;
		}
		return shelteredFloor >= 3;
	}

	// Stable room types: 1 shelter/cave, 3 campsite, 4 watchtower, 5 open base.
	// Physical exposure (homeKind == 2) is never itself evidence of a home.
	private static int homeBaseKind(int facilities, boolean tower) {
		if (tower && facilities != 0) return 4;
		if (Integer.bitCount(facilities) < 3) return 0;
		if ((facilities & 3) != 0 && (facilities & 12) != 0) return 3;
		return (facilities & 28) == 28 ? 5 : 0;
	}

	private static boolean homeFootprintSafe(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos seed) {
		if (!isSafeStandingPosition(level, seed)) return false;
		Vec3 center = Vec3.atBottomCenterOf(seed).add(0.0, 1.5, 0.0);
		for (Direction side : Direction.Plane.HORIZONTAL) {
			BlockPos next = homeFloor(level, seed.relative(side));
			if (next == null) return false;
			BlockHitResult access = homeRay(level, ashWeaver, center,
					Vec3.atBottomCenterOf(next).add(0.0, 1.5, 0.0));
			if (access == null || access.getType() == HitResult.Type.BLOCK) return false;
		}
		return true;
	}

	private static int homeOutdoorKind(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos seed) {
		if (!homeEvidenceLoaded(level, seed) || !homeFootprintSafe(level, ashWeaver, seed)) return 0;
		int facilities = 0;
		Vec3 eye = Vec3.atBottomCenterOf(seed).add(0.0, 1.5, 0.0);
		// At most 484 already-loaded blocks, only on commands or staggered maintenance.
		for (int dx = -5; dx <= 5; dx++) for (int dz = -5; dz <= 5; dz++)
			for (int dy = -1; dy <= 2; dy++) {
				BlockPos pos = seed.offset(dx, dy, dz);
				if (!level.hasChunkAt(pos) || !level.getWorldBorder().isWithinBounds(pos)) continue;
				BlockState state = level.getBlockState(pos);
				var block = state.getBlock();
				int feature = block instanceof net.minecraft.world.level.block.BedBlock ? 1
						: block instanceof net.minecraft.world.level.block.CampfireBlock ? 2
						: block == Blocks.CHEST || block == Blocks.TRAPPED_CHEST || block == Blocks.BARREL
								|| block == Blocks.ENDER_CHEST ? 4
						: block == Blocks.CRAFTING_TABLE || block == Blocks.FURNACE || block == Blocks.SMOKER
								|| block == Blocks.BLAST_FURNACE || block == Blocks.SMITHING_TABLE
								|| block == Blocks.STONECUTTER || block == Blocks.LOOM || block == Blocks.ANVIL ? 8
						: block == Blocks.TORCH || block == Blocks.SOUL_TORCH || block == Blocks.LANTERN
								|| block == Blocks.SOUL_LANTERN ? 16 : 0;
				if (feature == 0 || (facilities & feature) != 0) continue;
				BlockHitResult view = homeRay(level, ashWeaver, eye, Vec3.atCenterOf(pos));
				if (view != null && (view.getType() != HitResult.Type.BLOCK || view.getBlockPos().equals(pos)))
					facilities |= feature; // Duplicate objects never count as different facilities.
			}
		if (facilities == 0) return 0;
		int guarded = 0;
		int elevated = 0;
		for (Direction side : Direction.Plane.HORIZONTAL) {
			for (int distance = 2; distance <= 3; distance++) {
				BlockPos edge = seed.relative(side, distance);
				if (!level.hasChunkAt(edge)) continue;
				BlockState rail = level.getBlockState(edge);
				if (rail.is(net.minecraft.tags.BlockTags.FENCES) || rail.is(net.minecraft.tags.BlockTags.WALLS)) {
					guarded++; break;
				}
			}
			Vec3 outside = Vec3.atBottomCenterOf(seed.relative(side, 6));
			BlockHitResult ground = homeRay(level, ashWeaver, outside, outside.add(0.0, -32.0, 0.0));
			if (ground != null && ground.getType() == HitResult.Type.BLOCK
					&& seed.getY() - ground.getLocation().y >= 5.0) elevated++;
		}
		return homeBaseKind(facilities, guarded >= 2 && elevated >= 3);
	}

	private static boolean homeEvidenceLoaded(ServerLevel level, BlockPos seed) {
		for (int cx = (seed.getX() - 6) >> 4; cx <= (seed.getX() + 6) >> 4; cx++)
			for (int cz = (seed.getZ() - 6) >> 4; cz <= (seed.getZ() + 6) >> 4; cz++)
				if (!level.hasChunkAt(new BlockPos(cx << 4, seed.getY(), cz << 4))) return false;
		return true;
	}

	private static boolean homeAreaContains(CompoundTag data, int room, BlockPos feet) {
		if (room < 0 || room >= Math.min(8, dataInt(data, "ashWeaverHomeRooms", 0))) return false;
		int kind = dataInt(data, "ashWeaverHomeRoom" + room + "Kind", 0);
		BlockPos seed = BlockPos.of(dataLong(data, "ashWeaverHomeRoom" + room + "Seed", 0L));
		int dy = Math.abs(seed.getY() - feet.getY());
		double distance = seed.distSqr(feet);
		if (kind == 1) return distance <= 48.0 * 48.0 && dy <= 24;
		if (dataInt(data, "ashWeaverHomeRoom" + room + "Confidence", 0) <= 0) return false;
		return (kind == 3 || kind == 5) ? distance <= 6.0 * 6.0 && dy <= 2
				: kind == 4 && distance <= 4.0 * 4.0 && dy <= 1;
	}

	private static int homeMappedKind(CompoundTag data, int room, BlockPos feet, int physicalKind) {
		if (physicalKind == 0 || !homeAreaContains(data, room, feet)) return 0;
		int saved = dataInt(data, "ashWeaverHomeRoom" + room + "Kind", 0);
		return saved == 1 ? physicalKind == 1 ? 1 : 0 : saved;
	}

	private static int homeRememberedRoom(CompoundTag data, BlockPos feet) {
		// Explicit outdoor zones take priority over an indoor room's broad search bounds.
		for (int i = 0; i < Math.min(8, dataInt(data, "ashWeaverHomeRooms", 0)); i++)
			if (dataInt(data, "ashWeaverHomeRoom" + i + "Kind", 0) >= 3 && homeAreaContains(data, i, feet)) return i;
		for (int i = 0; i < Math.min(8, dataInt(data, "ashWeaverHomeRooms", 0)); i++)
			if (homeAreaContains(data, i, feet)) return i;
		return -1;
	}

	private static int homeEvidenceConfidence(int confidence, boolean supported) {
		return Math.max(0, Math.min(100, confidence + (supported ? 10 : -10)));
	}

	private static String homeDescription(int kind) {
		return kind == 3 ? "our camp" : kind == 4 ? "our lookout" : kind == 5 ? "our open base" : "home";
	}

	private static boolean confirmHomeChange(CompoundTag data, String action, BlockPos seed,
			ServerLevel level, long now) {
		boolean confirmed = action.equals(dataString(data, "ashWeaverHomePendingAction", ""))
				&& level.dimension().toString().equals(dataString(data, "ashWeaverHomePendingDimension", ""))
				&& now < dataLong(data, "ashWeaverHomePendingUntil", 0L)
				&& seed.distSqr(BlockPos.of(dataLong(data, "ashWeaverHomePendingPos", seed.asLong()))) <= 4.0;
		if (confirmed) {
			for (String key : new String[] {"Action", "Dimension", "Until", "Pos"})
				data.remove("ashWeaverHomePending" + key);
			return true;
		}
		data.putString("ashWeaverHomePendingAction", action);
		data.putString("ashWeaverHomePendingDimension", level.dimension().toString());
		data.putLong("ashWeaverHomePendingUntil", now + 200L);
		data.putLong("ashWeaverHomePendingPos", seed.asLong());
		return false;
	}

	private static void clearHomeMemory(AshWeaverEntity ashWeaver, CompoundTag data) {
		long revision = dataLong(data, "ashWeaverHomeRevision", 0L) + 1L;
		for (String key : new String[] {"Known", "Dimension", "Base", "Anchor", "Stay", "Rooms", "Points",
				"Portals", "PulseX", "PulseY", "PulseZ", "NextRest", "RestTarget", "Repath", "Familiarity", "FamiliarAt"})
			data.remove("ashWeaverHome" + key);
		for (int i = 0; i < 8; i++) for (String suffix : new String[] {"Seed", "Kind", "Confidence"})
			data.remove("ashWeaverHomeRoom" + i + suffix);
		for (int i = 0; i < 64; i++) for (String suffix : new String[] {"Pos", "Room", "Kind", "Valid"})
			data.remove("ashWeaverHomePoint" + i + suffix);
		for (int i = 0; i < 16; i++) for (String suffix : new String[] {"Door", "Beyond", "From", "To"})
			data.remove("ashWeaverHomePortal" + i + suffix);
		data.putLong("ashWeaverHomeRevision", revision);
		HOME_SURVEYS.remove(ashWeaver);
	}

	private static boolean homeVisible(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos feet) {
		Vec3 point = Vec3.atBottomCenterOf(feet).add(0.0, 1.3, 0.0);
		if (ashWeaver.getEyePosition().distanceToSqr(point) > 24.0 * 24.0) return false;
		BlockHitResult ray = homeRay(level, ashWeaver, ashWeaver.getEyePosition(), point);
		return ray != null && ray.getType() != HitResult.Type.BLOCK;
	}

	@Nullable
	private static BlockPos homeFloor(ServerLevel level, BlockPos near) {
		for (int dy : new int[] {0, 1, -1}) {
			BlockPos feet = near.offset(0, dy, 0);
			if (isSafeStandingPosition(level, feet)) return feet;
		}
		return null;
	}

	private static CompoundTag homeSnapshot(CompoundTag data, long now) {
		CompoundTag snapshot = new CompoundTag();
		for (String key : data.getAllKeys()) {
			if (key.matches("ashWeaverHome(Known|Dimension|Base|Anchor|Rooms|Points|Portals|Familiarity|Room[0-7].*|Point[0-9]+.*|Portal[0-9]+.*)")) {
				var value = data.get(key);
				if (value != null) snapshot.put(key, value.copy());
			}
		}
		snapshot.putLong("SavedAt", now);
		return snapshot;
	}

	private static boolean stashCurrentHome(CompoundTag data, long now) {
		if (!dataBoolean(data, "ashWeaverHomeKnown", false)) return true;
		int slot = -1;
		long oldest = Long.MAX_VALUE;
		BlockPos current = BlockPos.of(dataLong(data, "ashWeaverHomeBase", 0L));
		for (int i = 0; i < 4; i++) {
			CompoundTag saved = savedHome(data, i);
			if (!dataBoolean(saved, "ashWeaverHomeKnown", false)) { slot = i; break; }
			if (dataString(saved, "ashWeaverHomeDimension", "").equals(dataString(data, "ashWeaverHomeDimension", ""))
					&& current.distSqr(BlockPos.of(dataLong(saved, "ashWeaverHomeBase", 0L))) <= 9.0) { slot = i; break; }
			// Temporary outdoor bases can expire; a permanent shelter is never silently evicted.
			long age = dataLong(saved, "SavedAt", 0L);
			if (dataInt(saved, "ashWeaverHomeRoom0Kind", 1) >= 3 && age < oldest) { oldest = age; slot = i; }
		}
		if (slot < 0) return false;
		data.put("ashWeaverSavedHome" + slot, homeSnapshot(data, now));
		return true;
	}

	private static String restoreNearbyHome(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long now) {
		CompoundTag selected = null;
		double distance = 8.0 * 8.0;
		for (int i = 0; i < 4; i++) {
			CompoundTag saved = savedHome(data, i);
			if (!homeHere(level, saved)) continue;
			BlockPos base = BlockPos.of(dataLong(saved, "ashWeaverHomeBase", 0L));
			double nearby = player.distanceToSqr(Vec3.atBottomCenterOf(base));
			if (nearby < distance) { selected = saved.copy(); distance = nearby; }
		}
		if (selected == null) return "Show me a remembered home nearby.";
		BlockPos seed = BlockPos.of(dataLong(selected, "ashWeaverHomeBase", 0L));
		int remembered = dataInt(selected, "ashWeaverHomeRoom0Kind", 1);
		boolean valid = remembered == 1 ? homeInteriorSafe(level, ashWeaver, seed)
				: homeOutdoorKind(level, ashWeaver, seed) == remembered;
		if (!valid || !homeVisible(level, ashWeaver, seed)
				|| !homePathSafe(level, fearCreatePath(level, ashWeaver, seed))) return "That place isn't safe or reachable now.";
		if (!confirmHomeChange(data, "restore", seed, level, now)) return "Say 'restore home' again to confirm.";
		if (!stashCurrentHome(data, now)) return "I can't remember another permanent home.";
		clearHomeMemory(ashWeaver, data);
		for (String key : selected.getAllKeys()) {
			var value = selected.get(key);
			if (key.startsWith("ashWeaverHome") && value != null) data.put(key, value.copy());
		}
		return "I remember this home again.";
	}


	public static String homeCommand(AshWeaverEntity ashWeaver, Player player, String action) {
		if (ashWeaver == null || player == null || !ashWeaver.isAlive() || !player.isAlive()
				|| player.isSpectator() || !(ashWeaver.level() instanceof ServerLevel level)) return "";
		if (!"set".equals(action) && !"extend".equals(action) && !"forget".equals(action)
				&& !"stay".equals(action) && !"status".equals(action) && !"restore".equals(action)) return "";
		CompoundTag data = ashWeaver.getPersistentData();
		if (!player.getUUID().toString().equals(ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))) return "";
		if (!isCanonicalCompanion(ashWeaver, player.getUUID().toString())) return "";
		if (player.level() != level || ashWeaver.distanceToSqr(player) > ("status".equals(action) ? 1024.0 : 64.0))
			return "Come closer.";
		if (!"status".equals(action) && !ashWeaver.hasLineOfSight(player)) return "I can't see you.";
		long now = level.getGameTime();
		if (now < dataLong(data, "ashWeaverHomeCommandAt", 0L)) return "Give me a moment.";
		data.putLong("ashWeaverHomeCommandAt", now + 40L);
		if (!"status".equals(action) && (dataBoolean(data, "ashWeaverAvoidOwner", false)
				|| relationship(ashWeaver) <= -25.0)) return "Keep your distance.";
		if ("forget".equals(action)) {
			if (!dataBoolean(data, "ashWeaverHomeKnown", false)) return "No home yet.";
			if (!confirmHomeChange(data, action, player.blockPosition(), level, now))
				return "Say 'forget home' again to confirm.";
			clearHomeMemory(ashWeaver, data);
			for (int i = 0; i < 4; i++) data.remove("ashWeaverSavedHome" + i);
			return "Home forgotten.";
		}
		if ("restore".equals(action)) {
			int state = dataInt(data, KEY_STATE, STATE_IDLE);
			if (state == STATE_HIDE || state == STATE_DEFEND || state == STATE_ALERT || state == STATE_TRAVEL)
				return "Let's get safe first.";
			return restoreNearbyHome(level, ashWeaver, player, data, now);
		}
		if ("stay".equals(action)) {
			if (!homeHere(level, data)) return "Show me home first.";
			BlockPos anchor = BlockPos.of(dataLong(data, "ashWeaverHomeAnchor", dataLong(data, "ashWeaverHomeBase", 0L)));
			if (ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(anchor)) > 48.0 * 48.0
					|| !isSafeStandingPosition(level, anchor)
					|| homeMappedKind(data, homeRememberedRoom(data, anchor), anchor,
							homeKind(level, ashWeaver, anchor)) == 0) return "Home isn't safe now.";
			Path path = fearCreatePath(level, ashWeaver, anchor);
			if (!homePathSafe(level, path)) return "I can't reach home.";
			data.putBoolean("ashWeaverHomeStay", true);
			if (dataInt(data, KEY_STATE, STATE_IDLE) != STATE_HIDE) {
				cancelTransientActions(ashWeaver, now);
				setState(data, STATE_HOME, level.getGameTime(), 0L);
				clearPlantingIntent(data);
			}
			data.remove(KEY_CHAT_STAY);
			data.remove("ashWeaverHomeRestTarget");
			return "Staying home.";
		}
		if ("status".equals(action)) {
			if (!dataBoolean(data, "ashWeaverHomeKnown", false)) return "No home yet.";
			if (!homeHere(level, data)) return "Home is elsewhere.";
			int rememberedKind = dataInt(data, "ashWeaverHomeRoom0Kind", 1);
			if (rememberedKind >= 3 && dataInt(data, "ashWeaverHomeRoom0Confidence", 0) < 30)
				return "I'm not sure " + homeDescription(rememberedKind) + " is still usable.";
			return rememberedKind == 1 ? "I remember home." : "I remember " + homeDescription(rememberedKind) + ". It's exposed.";
		}
		if (player.level() != level || ashWeaver.distanceToSqr(player) > 8.0 * 8.0) return "Come closer.";
		BlockPos seed = homeFloor(level, player.blockPosition());
		if (seed == null) return "Show me a safe place.";
		int commandState = dataInt(data, KEY_STATE, STATE_IDLE);
		if (commandState == STATE_HIDE || commandState == STATE_DEFEND || commandState == STATE_ALERT || commandState == STATE_TRAVEL)
			return "Let's get safe first.";
		if (!player.onGround() || !ashWeaver.onGround() || !homeVisible(level, ashWeaver, seed))
			return "Show me from nearby safe ground.";
		if (ashWeaver.getTarget() != null || ashWeaver.getLastHurtByMobTimestamp() > 0
				&& ashWeaver.tickCount - ashWeaver.getLastHurtByMobTimestamp() < 100
				|| player.getLastHurtByMobTimestamp() > 0
				&& player.tickCount - player.getLastHurtByMobTimestamp() < 100)
			return "Let's get safe first.";
		int kind = homeInteriorSafe(level, ashWeaver, seed) ? 1 : homeOutdoorKind(level, ashWeaver, seed);
		if (kind == 0) return "Show me shelter or a usable, equipped base.";
		if (!homePathSafe(level, fearCreatePath(level, ashWeaver, seed))) return "I can't reach that place.";
		if ("extend".equals(action)) {
			if (!homeHere(level, data)) return "Show me home first.";
			BlockPos base = BlockPos.of(dataLong(data, "ashWeaverHomeBase", 0L));
			if (base.distSqr(seed) > 48.0 * 48.0 || Math.abs(base.getY() - seed.getY()) > 24) return "Too far from home.";
			BlockPos anchor = BlockPos.of(dataLong(data, "ashWeaverHomeAnchor", base.asLong()));
			Path connection = fearCreatePath(level, ashWeaver, anchor);
			if (!homePathSafe(level, connection)) return "That room isn't connected to home.";
			// Nearby but separate buildings must not be joined through an outdoor route.
			for (int i = connection.getNextNodeIndex(); i < connection.getNodeCount(); i++) {
				var node = connection.getNode(i);
				Vec3 eye = Vec3.atBottomCenterOf(new BlockPos(node.x, node.y, node.z)).add(0.0, 1.5, 0.0);
				BlockHitResult roof = homeRay(level, ashWeaver, eye, eye.add(0.0, 24.0, 0.0));
				BlockPos feet = new BlockPos(node.x, node.y, node.z);
				boolean insideNewArea = kind >= 3 && seed.distSqr(feet) <= (kind == 4 ? 16.0 : 36.0)
						&& Math.abs(seed.getY() - feet.getY()) <= (kind == 4 ? 1 : 2);
				if ((roof == null || roof.getType() != HitResult.Type.BLOCK)
						&& !insideNewArea && homeRememberedRoom(data, feet) < 0)
					return "Show me a connected route within our base.";
			}
			int room = addHomeRoom(data, seed, kind);
			if (room < 0) return "I can't remember more rooms.";
		} else {
			if (!confirmHomeChange(data, action, seed, level, now))
				return "Say 'this is home' again here to confirm.";
			if (!stashCurrentHome(data, now)) return "I can't remember another permanent home.";
			// Commit only after every check; failed surveys never erase the previous home.
			clearHomeMemory(ashWeaver, data);
			data.putBoolean("ashWeaverHomeKnown", true);
			data.putString("ashWeaverHomeDimension", level.dimension().toString());
			data.putLong("ashWeaverHomeBase", seed.asLong());
			data.putLong("ashWeaverHomeAnchor", seed.asLong());
			addHomeRoom(data, seed, kind);
		}
		data.putLong("ashWeaverHomeRevision", dataLong(data, "ashWeaverHomeRevision", 0L) + 1L);
		Vec3 eye = ashWeaver.getEyePosition();
		data.putDouble("ashWeaverHomePulseX", eye.x);
		data.putDouble("ashWeaverHomePulseY", eye.y);
		data.putDouble("ashWeaverHomePulseZ", eye.z);
		return "extend".equals(action)
				? "I'll remember this area too." : kind == 1 ? "I'll remember this place."
				: "I'll remember " + homeDescription(kind) + ". It's exposed.";
	}

	private static int addHomeRoom(CompoundTag data, BlockPos seed, int kind) {
		int count = Math.min(8, dataInt(data, "ashWeaverHomeRooms", 0));
		for (int i = 0; i < count; i++) {
			if (BlockPos.of(dataLong(data, "ashWeaverHomeRoom" + i + "Seed", 0L)).distSqr(seed) < 3.0 * 3.0) {
				if (kind != 0) {
					data.putInt("ashWeaverHomeRoom" + i + "Kind", kind);
					data.putInt("ashWeaverHomeRoom" + i + "Confidence", 80);
				}
				return i;
			}
		}
		if (count >= 8) return -1;
		data.putLong("ashWeaverHomeRoom" + count + "Seed", seed.asLong());
		data.putInt("ashWeaverHomeRoom" + count + "Kind", kind);
		data.putInt("ashWeaverHomeRoom" + count + "Confidence", 80);
		data.putInt("ashWeaverHomeRooms", count + 1);
		return count;
	}

	private static void rememberHomePortal(CompoundTag data, BlockPos door, BlockPos beyond, int from) {
		int count = Math.min(16, dataInt(data, "ashWeaverHomePortals", 0));
		for (int i = 0; i < count; i++)
			if (dataLong(data, "ashWeaverHomePortal" + i + "Door", Long.MIN_VALUE) == door.asLong()) return;
		if (count >= 16) return;
		String key = "ashWeaverHomePortal" + count;
		data.putLong(key + "Door", door.asLong());
		data.putLong(key + "Beyond", beyond.asLong());
		data.putInt(key + "From", from);
		data.putInt(key + "To", -1);
		data.putInt("ashWeaverHomePortals", count + 1);
	}

	private static void rememberHomePoint(CompoundTag data, BlockPos feet, int room, int kind) {
		int count = Math.min(64, dataInt(data, "ashWeaverHomePoints", 0));
		int empty = -1;
		int start = Math.max(0, Math.min(7, room)) * 8;
		for (int i = start; i < start + 8; i++) {
			String key = "ashWeaverHomePoint" + i;
			if (!dataBoolean(data, key + "Valid", false)) { empty = i; continue; }
			if (dataInt(data, key + "Room", -1) == room
					&& BlockPos.of(dataLong(data, key + "Pos", 0L)).distSqr(feet) < 3.0 * 3.0) return;
		}
		int slot = empty;
		if (slot < 0) return;
		String key = "ashWeaverHomePoint" + slot;
		data.putLong(key + "Pos", feet.asLong());
		data.putInt(key + "Room", room);
		data.putInt(key + "Kind", kind);
		data.putBoolean(key + "Valid", true);
		if (slot >= count) data.putInt("ashWeaverHomePoints", slot + 1);
	}

	private static void updateHomeSurvey(ServerLevel level, AshWeaverEntity ashWeaver, CompoundTag data, long now) {
		if (!homeHere(level, data)) { HOME_SURVEYS.remove(ashWeaver); return; }
		BlockPos base = BlockPos.of(dataLong(data, "ashWeaverHomeBase", 0L));
		if (ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(base)) > 64.0 * 64.0
				|| Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 5) != 0) return;
		long revision = dataLong(data, "ashWeaverHomeRevision", 0L);
		HomeSurvey survey = HOME_SURVEYS.get(ashWeaver);
		if (survey == null || survey.revision != revision || now - survey.started > 6000L) {
			survey = new HomeSurvey(revision, now);
			for (int i = 0; i < Math.min(8, dataInt(data, "ashWeaverHomeRooms", 0)); i++)
				survey.offer(BlockPos.of(dataLong(data, "ashWeaverHomeRoom" + i + "Seed", 0L)), i);
			for (int i = 0; i < Math.min(16, dataInt(data, "ashWeaverHomePortals", 0)); i++)
				data.putInt("ashWeaverHomePortal" + i + "To", -1);
			HOME_SURVEYS.put(ashWeaver, survey);
		}
		if (now >= dataLong(data, "ashWeaverHomeFamiliarAt", 0L)
				&& ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(base)) <= 12.0 * 12.0) {
			data.putLong("ashWeaverHomeFamiliarAt", now + 1200L);
			data.putInt("ashWeaverHomeFamiliarity", Math.min(100, dataInt(data, "ashWeaverHomeFamiliarity", 0) + 1));
		}
		if (now >= survey.evidenceAt) {
			survey.evidenceAt = now + 100L;
			int rooms = Math.min(8, dataInt(data, "ashWeaverHomeRooms", 0));
			if (rooms > 0) {
				int room = survey.evidenceCursor++ % rooms;
				String key = "ashWeaverHomeRoom" + room;
				int savedKind = dataInt(data, key + "Kind", 0);
				BlockPos seed = BlockPos.of(dataLong(data, key + "Seed", 0L));
				if (savedKind >= 3 && homeVisible(level, ashWeaver, seed)
						&& homeEvidenceLoaded(level, seed)) {
					int observed = homeOutdoorKind(level, ashWeaver, seed);
					int before = dataInt(data, key + "Confidence", 80);
					int after = homeEvidenceConfidence(before, observed == savedKind);
					data.putInt(key + "Confidence", after);
					if (before <= 0 && after > 0)
						data.putLong("ashWeaverHomeRevision", revision + 1L); // Resurvey recovered areas, not stale points.
				}
			}
		}
		if (survey.pulse < 24) {
			Vec3 eye = new Vec3(dataDouble(data, "ashWeaverHomePulseX", ashWeaver.getX()),
					dataDouble(data, "ashWeaverHomePulseY", ashWeaver.getEyeY()),
					dataDouble(data, "ashWeaverHomePulseZ", ashWeaver.getZ()));
			for (int i = 0; i < 4 && survey.pulse < 24; i++, survey.pulse++) {
				double angle = (survey.pulse % 8) * Math.PI / 4.0;
				double slope = survey.pulse / 8 - 1;
				BlockHitResult hit = homeRay(level, ashWeaver, eye,
						eye.add(Math.cos(angle) * 12.0, slope * 8.0, Math.sin(angle) * 12.0));
				if (hit != null && hit.getType() == HitResult.Type.BLOCK && survey.surfaces.size() < 64)
					survey.surfaces.add(hit.getBlockPos().asLong());
			}
			return;
		}
		long observer = ashWeaver.blockPosition().asLong();
		if (survey.observer != observer) { survey.observer = observer; survey.idleScanAt = 0L; }
		if (survey.frontier.isEmpty() && now < survey.idleScanAt) return;
		if (survey.frontier.isEmpty()) survey.idleScanAt = now + 100L
				- Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20);
		for (int budget = 0; budget < 2; budget++) {
			HomeCell cell = !survey.unseen.isEmpty() && (budget == 1 || survey.frontier.isEmpty())
					? survey.unseen.pollFirst() : survey.frontier.pollFirst();
			if (cell == null) break;
			BlockPos feet = cell.feet;
			if (base.distSqr(feet) > 48.0 * 48.0 || Math.abs(base.getY() - feet.getY()) > 24) continue;
			if (!homeVisible(level, ashWeaver, feet)) { survey.unseen.addLast(cell); continue; }
			if (!isSafeStandingPosition(level, feet)) continue;
			int kind = homeMappedKind(data, cell.room, feet, homeKind(level, ashWeaver, feet));
			if (kind >= 3 && !homeFootprintSafe(level, ashWeaver, feet)) continue;
			int roomKind = dataInt(data, "ashWeaverHomeRoom" + cell.room + "Kind", 0);
			if (roomKind == 0 && kind != 0) {
				roomKind = kind;
				data.putInt("ashWeaverHomeRoom" + cell.room + "Kind", kind);
			}
			BlockPos seed = BlockPos.of(dataLong(data, "ashWeaverHomeRoom" + cell.room + "Seed", feet.asLong()));
			// Uncertain archways and shelter changes are frontiers, never automatic outdoor extensions.
			if (kind == 0 || kind != roomKind) {
				if (kind == 0) survey.unseen.addLast(cell);
				for (Direction side : Direction.Plane.HORIZONTAL) {
					Integer from = survey.floors.get(feet.relative(side).asLong());
					if (from != null) { rememberHomePortal(data, feet, feet, from); break; }
				}
				continue;
			}
			survey.floors.put(feet.asLong(), cell.room);
			survey.accepted++;
			rememberHomePoint(data, feet, cell.room, kind);
			for (Direction side : Direction.Plane.HORIZONTAL) {
				BlockPos next = feet.relative(side);
				if (!level.hasChunkAt(next)) continue;
				if (level.getBlockState(next).getBlock() instanceof DoorBlock
						|| level.getBlockState(next.above()).getBlock() instanceof DoorBlock) {
					rememberHomePortal(data, next, next.relative(side), cell.room);
					continue; // OPEN doors remain explicit room boundaries.
				}
				BlockPos floor = homeFloor(level, next);
				if (floor != null) survey.offer(floor, cell.room);
			}
		}
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) == 0) {
			int count = Math.min(16, dataInt(data, "ashWeaverHomePortals", 0));
			if (count > 0) {
				int index = survey.portalCursor++ % count;
				String key = "ashWeaverHomePortal" + index;
				BlockPos beyond = homeFloor(level, BlockPos.of(dataLong(data, key + "Beyond", 0L)));
				int beyondKind = beyond != null && dataInt(data, key + "To", -1) == -1
						&& base.distSqr(beyond) <= 48.0 * 48.0 && homeVisible(level, ashWeaver, beyond)
						? homeKind(level, ashWeaver, beyond) : 0;
				if (beyondKind == 2) {
					Integer explicit = survey.floors.get(beyond.asLong());
					data.putInt(key + "To", explicit == null ? -2 : explicit);
				}
				if (beyondKind == 1) {
					Integer known = survey.floors.get(beyond.asLong());
					int room = known == null ? addHomeRoom(data, beyond, 1) : known;
					if (room >= 0) { data.putInt(key + "To", room); survey.offer(beyond, room); }
				}
			}
			int points = Math.min(64, dataInt(data, "ashWeaverHomePoints", 0));
			if (points > 0) {
				String key = "ashWeaverHomePoint" + (survey.pointCursor++ % points);
				BlockPos point = BlockPos.of(dataLong(data, key + "Pos", 0L));
				if (dataBoolean(data, key + "Valid", false) && level.hasChunkAt(point)
						&& (!isSafeStandingPosition(level, point)
								|| homeMappedKind(data, dataInt(data, key + "Room", -1), point,
										homeKind(level, ashWeaver, point)) == 0)) data.putBoolean(key + "Valid", false);
			}
			if (dataInt(data, "ashWeaverHomeRoom0Kind", 1) == 1
					&& !survey.floors.isEmpty() && survey.centeredAt != survey.accepted) {
				survey.centeredAt = survey.accepted;
				Vec3 center = Vec3.ZERO;
				int total = 0;
				for (var entry : survey.floors.entrySet()) {
					if (dataInt(data, "ashWeaverHomeRoom" + entry.getValue() + "Kind", 0) == 1) {
						center = center.add(Vec3.atBottomCenterOf(BlockPos.of(entry.getKey()))); total++;
					}
				}
				if (total > 0) {
					center = center.scale(1.0 / total);
					double best = Double.MAX_VALUE;
					long anchor = dataLong(data, "ashWeaverHomeBase", 0L);
					for (var entry : survey.floors.entrySet()) {
						if (dataInt(data, "ashWeaverHomeRoom" + entry.getValue() + "Kind", 0) != 1) continue;
						BlockPos floor = BlockPos.of(entry.getKey());
						double score = Vec3.atBottomCenterOf(floor).distanceToSqr(center);
						if (score < best && isSafeStandingPosition(level, floor)) { best = score; anchor = floor.asLong(); }
					}
					data.putLong("ashWeaverHomeAnchor", anchor);
				}
			}
		}
	}

	private static boolean homePathSafe(ServerLevel level, Path path) {
		if (path == null || !path.canReach() || path.getNodeCount() > 80) return false;
		for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i++) {
			var node = path.getNode(i);
			BlockPos feet = new BlockPos(node.x, node.y, node.z);
			if (!isSafeStandingPosition(level, feet)
					&& !(level.hasChunkAt(feet) && level.getBlockState(feet).getBlock() instanceof DoorBlock)) return false;
			if (i > path.getNextNodeIndex() && Math.abs(node.y - path.getNode(i - 1).y) > 1) return false;
		}
		return true;
	}

	private static void handleHomeStay(ServerLevel level, AshWeaverEntity ashWeaver,
			@Nullable Player player, CompoundTag data, long now) {
		ensurePlayerPace(ashWeaver);
		if (ashWeaver.isInWater() || !ashWeaver.onGround()) return;
		clearHeldRose(ashWeaver);
		clearPlantingIntent(data);
		ashWeaver.setTarget(null);
		setStateIfChanged(data, STATE_HOME, now, 0L);
		BlockPos target = BlockPos.of(dataLong(data, "ashWeaverHomeRestTarget",
				dataLong(data, "ashWeaverHomeAnchor", dataLong(data, "ashWeaverHomeBase", 0L))));
		LivingEntity danger = ashWeaver.getLastHurtByMob();
		boolean threatened = danger != null && danger.isAlive() && danger.level() == level
				&& ashWeaver.tickCount - ashWeaver.getLastHurtByMobTimestamp() < 100;
		boolean invalidRestTarget = false;
		if (now >= dataLong(data, "ashWeaverHomeNextRest", 0L)) {
			data.putLong("ashWeaverHomeNextRest", now + 200L + ashWeaver.getRandom().nextInt(201));
			int count = Math.min(64, dataInt(data, "ashWeaverHomePoints", 0));
			double best = -Double.MAX_VALUE;
			double restVariation = 3.0 - Mth.clamp(dataInt(data, "ashWeaverHomeFamiliarity", 0), 0, 100) * 0.02;
			for (int i = 0; i < count; i++) {
				String key = "ashWeaverHomePoint" + i;
				if (!dataBoolean(data, key + "Valid", false)) continue;
				BlockPos point = BlockPos.of(dataLong(data, key + "Pos", 0L));
				if (!homeAreaContains(data, dataInt(data, key + "Room", -1), point)
						|| !isSafeStandingPosition(level, point)
						|| ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(point)) > 48.0 * 48.0) continue;
				double score = -ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(point)) * 0.03
						+ (dataInt(data, key + "Kind", 0) == 1 ? 10.0 : 0.0) + ashWeaver.getRandom().nextDouble() * restVariation;
				if (threatened) score += danger.distanceToSqr(Vec3.atBottomCenterOf(point)) * 0.1;
				if (score > best) { best = score; target = point; }
			}
			int kind = isSafeStandingPosition(level, target) ? homeMappedKind(data,
					homeRememberedRoom(data, target), target, homeKind(level, ashWeaver, target)) : 0;
			if (kind == 0) {
				data.remove("ashWeaverHomeRestTarget");
				invalidRestTarget = true;
			}
			else data.putLong("ashWeaverHomeRestTarget", target.asLong());
		}
		int targetRoom = homeRememberedRoom(data, target);
		if (invalidRestTarget || !isSafeStandingPosition(level, target) || targetRoom < 0
				|| Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) == 0
						&& (homeMappedKind(data, targetRoom, target, homeKind(level, ashWeaver, target)) == 0
								|| dataInt(data, "ashWeaverHomeRoom" + targetRoom + "Kind", 0) >= 3
										&& !homeFootprintSafe(level, ashWeaver, target))) {
			data.remove("ashWeaverHomeRestTarget");
			data.remove("ashWeaverHomeStay");
			setStateIfChanged(data, STATE_IDLE, now, 0L);
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			setSneaking(ashWeaver, false);
			return;
		}
		setSneaking(ashWeaver, isNearLedge(level, ashWeaver) || threatened);
		if (ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(target)) > 1.0) {
			if (now >= dataLong(data, "ashWeaverHomeRepath", 0L)) {
				data.putLong("ashWeaverHomeRepath", now + 40L);
				Path path = fearCreatePath(level, ashWeaver, target);
				if (homePathSafe(level, path)) ashWeaver.getNavigation().moveTo(path, PLAYER_NAV_SPEED);
				else { ashWeaver.getNavigation().stop(); data.remove("ashWeaverHomeRestTarget"); }
			}
			setRunning(ashWeaver, threatened && !ashWeaver.isShiftKeyDown());
		} else {
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			if (player != null && player.level() == level && ashWeaver.distanceToSqr(player) < 8.0 * 8.0)
				updateSocialGaze(ashWeaver, player, data, now);
			else if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 60) == 0) {
				int count = Math.min(16, dataInt(data, "ashWeaverHomePortals", 0));
				if (count > 0) {
					BlockPos door = BlockPos.of(dataLong(data, "ashWeaverHomePortal" + ashWeaver.getRandom().nextInt(count) + "Door", 0L));
					ashWeaver.getLookControl().setLookAt(door.getX() + 0.5, door.getY() + 1.0, door.getZ() + 0.5, 12.0F, 10.0F);
				}
			}
		}
		handleDoorInteraction(level, ashWeaver, data, STATE_HOME, now);
		applySneakEdgeBrake(level, ashWeaver);
		enforceGroundSpeed(ashWeaver);
		AshWeaverChatCommandProcedure.updateConversationLook(ashWeaver);
	}

	// Event bridge: shares the same low-health latch as the normal post-tick check.
	public static void captureCombatFear(ServerLevel level, AshWeaverEntity ashWeaver) {
		if (ashWeaver.level() == level && ashWeaver.isAlive())
			rememberNearDeath(level, ashWeaver, ashWeaver.getPersistentData(), level.getGameTime());
	}

	// Four bounded memories: exact incident position plus a soft 12-to-32-block fear area.
	private static void rememberNearDeath(ServerLevel level, AshWeaverEntity ashWeaver, CompoundTag data, long now) {
		boolean low = ashWeaver.getHealth() <= ashWeaver.getMaxHealth() * 0.25F;
		if (ashWeaver.getHealth() > ashWeaver.getMaxHealth() * 0.35F) data.remove("ashWeaverFearLowHealth");
		if (!low || ashWeaver.hurtTime <= 0 || dataBoolean(data, "ashWeaverFearLowHealth", false)) return;
		data.putBoolean("ashWeaverFearLowHealth", true);
		var damage = ashWeaver.getLastDamageSource();
		Entity attacker = damage == null ? null : damage.getEntity();
		String type = attacker == null ? "" : BuiltInRegistries.ENTITY_TYPE.getKey(attacker.getType()).toString();
		String dimension = level.dimension().toString();
		int slot = 0;
		double weakest = Double.MAX_VALUE;
		for (int i = 0; i < 4; i++) {
			String key = "ashWeaverFear" + i;
			double severity = dataDouble(data, key + "Strength", 0.0);
			if (dimension.equals(dataString(data, key + "Dimension", ""))
					&& type.equals(dataString(data, key + "Type", ""))
					&& ashWeaver.position().distanceToSqr(fearPosition(data, key)) <= 16.0 * 16.0) {
				slot = i;
				break;
			}
			double priority = severity - Math.min(0.5, Math.max(0L,
					now - dataLong(data, key + "Recorded", now)) / 240000.0);
			if (priority < weakest) { weakest = priority; slot = i; }
		}
		String key = "ashWeaverFear" + slot;
		boolean merged = dimension.equals(dataString(data, key + "Dimension", ""))
				&& type.equals(dataString(data, key + "Type", ""))
				&& ashWeaver.position().distanceToSqr(fearPosition(data, key)) <= 16.0 * 16.0;
		data.putString(key + "Dimension", dimension);
		data.putString(key + "Type", type);
		data.putDouble(key + "X", ashWeaver.getX());
		data.putDouble(key + "Y", ashWeaver.getY());
		data.putDouble(key + "Z", ashWeaver.getZ());
		data.putDouble(key + "Strength", Math.min(1.0,
				(merged ? dataDouble(data, key + "Strength", 0.0) : 0.0) + 0.75));
		data.putLong(key + "Recorded", now);
		data.putLong(key + "VisitAt", now + 1200L);
	}

	private static Vec3 fearPosition(CompoundTag data, String key) {
		return new Vec3(dataDouble(data, key + "X", 0.0), dataDouble(data, key + "Y", 0.0),
				dataDouble(data, key + "Z", 0.0));
	}

	private static double rememberedFear(ServerLevel level, AshWeaverEntity ashWeaver, CompoundTag data, Entity threat) {
		String type = BuiltInRegistries.ENTITY_TYPE.getKey(threat.getType()).toString();
		double strongest = 0.0;
		for (int i = 0; i < 4; i++) {
			String key = "ashWeaverFear" + i;
			if (!level.dimension().toString().equals(dataString(data, key + "Dimension", ""))
					|| !type.equals(dataString(data, key + "Type", ""))) continue;
			double distance = ashWeaver.position().distanceTo(fearPosition(data, key));
			double area = Math.max(0.0, Math.min(1.0, (32.0 - distance) / 20.0));
			strongest = Math.max(strongest, dataDouble(data, key + "Strength", 0.0) * (0.55 + 0.45 * area));
		}
		return strongest;
	}

	@Nullable
	private static LivingEntity hidingThreat(ServerLevel level, AshWeaverEntity ashWeaver, @Nullable Player player, CompoundTag data) {
		boolean ownerHurtMe = player != null && ashWeaver.getLastHurtByMob() == player
				&& ashWeaver.tickCount - ashWeaver.getLastHurtByMobTimestamp() < 200
				&& ashWeaver.getHealth() <= ashWeaver.getMaxHealth() * 0.4F;
		if (player != null && ashWeaver.distanceToSqr(player) < 16.0 * 16.0 && (ownerHurtMe
				|| dataBoolean(data, "ashWeaverAvoidOwner", false)
					&& (isOwnerAimingAtWeaver(player, ashWeaver) || ashWeaver.distanceToSqr(player) < 6.0 * 6.0))) return player;
		LivingEntity best = null;
		double bestScore = 0.0;
		for (LivingEntity candidate : level.getEntitiesOfClass(LivingEntity.class,
				ashWeaver.getBoundingBox().inflate(20.0),
				entity -> entity.isAlive() && entity != ashWeaver && entity != player
						&& !(entity instanceof AshWeaverEntity))) {
			double distance = ashWeaver.distanceToSqr(candidate);
			if (distance > 20.0 * 20.0) continue;
			boolean pursuing = candidate instanceof Mob mob && (mob.getTarget() == ashWeaver
					|| player != null && mob.getTarget() == player);
			boolean recentAttacker = candidate == ashWeaver.getLastHurtByMob()
					&& ashWeaver.tickCount - ashWeaver.getLastHurtByMobTimestamp() < 200;
			double memory = rememberedFear(level, ashWeaver, data, candidate);
			boolean injured = ashWeaver.getHealth() <= ashWeaver.getMaxHealth() * 0.4F;
			if (!(injured && (pursuing || recentAttacker)) && !(memory >= 0.35 && (pursuing || recentAttacker
					|| distance < 14.0 * 14.0 && ashWeaver.hasLineOfSight(candidate)))) continue;
			double score = (pursuing ? 3.0 : 0.0) + (recentAttacker ? 2.0 : 0.0) + memory
					+ 1.0 / (1.0 + Math.sqrt(distance));
			if (score > bestScore) { best = candidate; bestScore = score; }
		}
		return best;
	}

	private static boolean handleFearAndHiding(ServerLevel level, AshWeaverEntity ashWeaver,
			@Nullable Player player, CompoundTag data, long now) {
		if (dataInt(data, KEY_STATE, STATE_IDLE) == STATE_HIDE) {
			LivingEntity threat = null;
			try {
				Entity cached = level.getEntity(UUID.fromString(dataString(data, "ashWeaverHideThreat", "")));
				if (cached instanceof LivingEntity living && living.isAlive()) threat = living;
			} catch (IllegalArgumentException ignored) { }
			if (!level.dimension().toString().equals(dataString(data, "ashWeaverHideDimension", ""))) {
				finishHiding(ashWeaver, data, now);
				return false;
			}
			return handleHiding(level, ashWeaver, player, threat, data, now);
		}
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) == 0 || ashWeaver.hurtTime == 10) {
			LivingEntity threat = hidingThreat(level, ashWeaver, player, data);
			boolean urgent = threat != null && ashWeaver.getHealth() <= ashWeaver.getMaxHealth() * 0.4F
					&& (ashWeaver.hurtTime > 0 || threat instanceof Mob mob && mob.getTarget() == ashWeaver);
			boolean ready = now >= dataLong(data, "ashWeaverHideRetry", 0L) || urgent;
			if (threat != null && ready && !urgent && threat != player
					&& ashWeaver.getHealth() > ashWeaver.getMaxHealth() * 0.5F
					&& ashWeaver.getRandom().nextDouble() > rememberedFear(level, ashWeaver, data, threat)) {
				// A healthy Weaver may choose courage; keep that decision stable for ten seconds.
				data.putLong("ashWeaverHideRetry", now + 200L);
				ready = false;
			}
			if (threat != null && ready && ashWeaver.onGround()) {
				clearPlantingIntent(data);
				clearHeldRose(ashWeaver);
				ashWeaver.setTarget(null);
				ashWeaver.getNavigation().stop();
				for (String key : new String[] {KEY_DOOR_PENDING, KEY_DOOR_CLOSE_AT, KEY_PARKOUR_JUMPED,
						KEY_BOUNDARY_POS, KEY_BOUNDARY_UNTIL, KEY_DETOUR_POS, "ashWeaverHideCover",
						"ashWeaverHidePeek", "ashWeaverHideLostAt"}) data.remove(key);
				data.putString("ashWeaverHideThreat", threat.getUUID().toString());
				data.putString("ashWeaverHideDimension", level.dimension().toString());
				data.putLong("ashWeaverHideOrigin", ashWeaver.blockPosition().asLong());
				data.putInt("ashWeaverHideScan", 0);
				data.putInt("ashWeaverHidePhase", 0);
				data.putLong("ashWeaverHideNextPeek", now + 60L);
				setState(data, STATE_HIDE, now, 0L);
				return handleHiding(level, ashWeaver, player, threat, data, now);
			}
			if (threat == null && !dataBoolean(data, "ashWeaverAvoidOwner", false)
					&& ashWeaver.hurtTime == 0 && (player == null || player.hurtTime == 0)
					&& (ashWeaver.getLastHurtByMobTimestamp() <= 0
						|| ashWeaver.tickCount - ashWeaver.getLastHurtByMobTimestamp() > 300)) {
				for (int i = 0; i < 4; i++) {
					String key = "ashWeaverFear" + i;
					double strength = dataDouble(data, key + "Strength", 0.0);
					if (strength < 0.1 || !level.dimension().toString().equals(dataString(data, key + "Dimension", ""))
							|| ashWeaver.position().distanceToSqr(fearPosition(data, key)) > 32.0 * 32.0
							|| now < dataLong(data, key + "VisitAt", 0L)) continue;
					data.putLong(key + "VisitAt", now + 1200L);
					if (ashWeaver.getHealth() > ashWeaver.getMaxHealth() * 0.5F)
						data.putDouble(key + "Strength", Math.max(0.0, strength - 0.08));
					if (player != null && ashWeaver.distanceToSqr(player) < 12.0 * 12.0 && ashWeaver.onGround()
							&& dataInt(data, KEY_STATE, STATE_IDLE) <= STATE_FOLLOW
							&& ashWeaver.position().distanceToSqr(fearPosition(data, key)) < 12.0 * 12.0) {
						data.putLong("ashWeaverFearPauseUntil", now + 20L);
						data.putInt("ashWeaverFearPauseSlot", i);
					}
				}
			}
		}
		if (player != null && now < dataLong(data, "ashWeaverFearPauseUntil", 0L)
				&& dataInt(data, KEY_STATE, STATE_IDLE) <= STATE_FOLLOW && ashWeaver.hurtTime == 0
				&& ashWeaver.distanceToSqr(player) < 12.0 * 12.0) {
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			setSneaking(ashWeaver, isNearLedge(level, ashWeaver));
			Vec3 point = fearPosition(data, "ashWeaverFear" + dataInt(data, "ashWeaverFearPauseSlot", 0));
			ashWeaver.getLookControl().setLookAt(point.x, point.y + 1.0, point.z, 12.0F, 10.0F);
			applySneakEdgeBrake(level, ashWeaver);
			return true;
		}
		return false;
	}

	// Ray queries never enter an unloaded chunk, including chunks between endpoints.
	private static boolean fearRayBlocked(ServerLevel level, AshWeaverEntity ashWeaver, Vec3 from, Vec3 to) {
		int minX = ((int) Math.floor(Math.min(from.x, to.x))) >> 4;
		int maxX = ((int) Math.floor(Math.max(from.x, to.x))) >> 4;
		int minZ = ((int) Math.floor(Math.min(from.z, to.z))) >> 4;
		int maxZ = ((int) Math.floor(Math.max(from.z, to.z))) >> 4;
		if (maxX - minX > 4 || maxZ - minZ > 4) return false;
		for (int cx = minX; cx <= maxX; cx++)
			for (int cz = minZ; cz <= maxZ; cz++)
				if (!level.hasChunkAt(new BlockPos(cx << 4, (int) from.y, cz << 4))) return false;
		BlockHitResult hit = level.clip(new ClipContext(from, to, ClipContext.Block.COLLIDER,
				ClipContext.Fluid.NONE, ashWeaver));
		return hit.getType() == HitResult.Type.BLOCK && hit.getLocation().distanceToSqr(to) > 0.04;
	}

	private static boolean fearCovered(ServerLevel level, AshWeaverEntity ashWeaver, LivingEntity threat, Vec3 feet) {
		Vec3 eye = threat.getEyePosition();
		Vec3 direction = feet.subtract(eye);
		double length = Math.max(0.01, Math.sqrt(direction.horizontalDistanceSqr()));
		Vec3 side = new Vec3(-direction.z / length * 0.25, 0.0, direction.x / length * 0.25);
		return fearRayBlocked(level, ashWeaver, eye, feet.add(0.0, 1.4, 0.0))
				&& fearRayBlocked(level, ashWeaver, eye, feet.add(side).add(0.0, 0.8, 0.0))
				&& fearRayBlocked(level, ashWeaver, eye, feet.subtract(side).add(0.0, 0.8, 0.0));
	}

	private static boolean threatWatching(LivingEntity threat, Vec3 feet) {
		Vec3 toward = feet.add(0.0, 1.3, 0.0).subtract(threat.getEyePosition());
		return toward.lengthSqr() < 0.01 || threat.getLookAngle().dot(toward.normalize()) > 0.75;
	}


	@Nullable
	private static Path fearCreatePath(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos target) {
		// Vanilla pathfinding also inspects a region around the mob, beyond its destination.
		int range = (int) Math.ceil(ashWeaver.getAttributeValue(Attributes.FOLLOW_RANGE)) + 8;
		if (range > 64) return null;
		int minX = (ashWeaver.getBlockX() - range) >> 4;
		int maxX = (ashWeaver.getBlockX() + range) >> 4;
		int minZ = (ashWeaver.getBlockZ() - range) >> 4;
		int maxZ = (ashWeaver.getBlockZ() + range) >> 4;
		for (int cx = minX; cx <= maxX; cx++)
			for (int cz = minZ; cz <= maxZ; cz++)
				if (!level.hasChunkAt(new BlockPos(cx << 4, ashWeaver.getBlockY(), cz << 4))) return null;
		return ashWeaver.getNavigation().createPath(target, 0);
	}

	private static boolean fearPathSafe(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			LivingEntity threat, Path path) {
		if (path == null || !path.canReach() || path.getNodeCount() > 24) return false;
		int previousY = ashWeaver.getBlockY();
		double ownerClearance = player == null ? 0.0 : Math.min(4.0 * 4.0, Math.max(0.0, ashWeaver.distanceToSqr(player) - 1.0));
		for (int i = path.getNextNodeIndex(); i < path.getNodeCount(); i++) {
			var node = path.getNode(i);
			BlockPos feet = new BlockPos(node.x, node.y, node.z);
			if (Math.abs(node.y - previousY) > 1 || !isSafeStandingPosition(level, feet)) return false;
			if (player != null && threat != player && threat instanceof Mob mob && mob.getTarget() == ashWeaver
					&& Vec3.atBottomCenterOf(feet).distanceToSqr(player.position()) < ownerClearance) return false;
			previousY = node.y;
		}
		return true;
	}

	private static boolean coverHasExit(ServerLevel level, BlockPos feet, LivingEntity threat) {
		for (Direction side : Direction.Plane.HORIZONTAL) {
			BlockPos next = feet.relative(side);
			if (isSafeStandingPosition(level, next)
					&& threat.position().distanceToSqr(Vec3.atBottomCenterOf(next))
						>= threat.position().distanceToSqr(Vec3.atBottomCenterOf(feet)) - 1.0) return true;
		}
		return false;
	}


	private static boolean tryRememberedHomeCover(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			LivingEntity threat, CompoundTag data, long now) {
		if (!homeHere(level, data)) return false;
		BlockPos best = null;
		double nearest = 10.0 * 10.0;
		for (int i = 0; i < Math.min(64, dataInt(data, "ashWeaverHomePoints", 0)); i++) {
			String key = "ashWeaverHomePoint" + i;
			if (!dataBoolean(data, key + "Valid", false)) continue;
			BlockPos point = BlockPos.of(dataLong(data, key + "Pos", 0L));
			Vec3 feet = Vec3.atBottomCenterOf(point);
			double distance = ashWeaver.distanceToSqr(feet);
			if (distance < nearest && feet.distanceToSqr(threat.position()) > 5.0 * 5.0
					&& (player == null || feet.distanceToSqr(player.position()) <= 24.0 * 24.0)
					&& isSafeStandingPosition(level, point)) {
				best = point; nearest = distance;
			}
		}
		if (best == null || !fearCovered(level, ashWeaver, threat, Vec3.atBottomCenterOf(best))
				|| !coverHasExit(level, best, threat)) return false;
		Path path = fearCreatePath(level, ashWeaver, best);
		if (!fearPathSafe(level, ashWeaver, player, threat, path)) return false;
		data.putLong("ashWeaverHideCover", best.asLong());
		data.putLong("ashWeaverHideRepath", now + 20L);
		ashWeaver.getNavigation().moveTo(path, ashWeaver.isSprinting() ? PLAYER_NAV_SPEED : PLAYER_NAV_SPEED * 0.3);
		return true;
	}

	private static void searchFearCover(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			LivingEntity threat, CompoundTag data, long now) {
		BlockPos origin = BlockPos.of(dataLong(data, "ashWeaverHideOrigin", ashWeaver.blockPosition().asLong()));
		int scan = dataInt(data, "ashWeaverHideScan", 0);
		if (scan == 0 && tryRememberedHomeCover(level, ashWeaver, player, threat, data, now)) return;
		// Eight terrain candidates and at most two path searches per scheduled batch.
		int paths = 0;
		for (int count = 0; count < 8 && scan < 144; count++, scan++) {
			int direction = scan % 12;
			int height = (scan / 12) % 3;
			double radius = 3.0 + (scan / 36) * 2.0;
			double angle = direction * Math.PI / 6.0;
			BlockPos feet = BlockPos.containing(origin.getX() + 0.5 + Math.cos(angle) * radius,
					origin.getY() + (height == 0 ? 0 : height == 1 ? -1 : 1),
					origin.getZ() + 0.5 + Math.sin(angle) * radius);
			Vec3 point = Vec3.atBottomCenterOf(feet);
			if (!isSafeStandingPosition(level, feet) || player != null && point.distanceToSqr(player.position()) > 24.0 * 24.0
					|| point.distanceToSqr(threat.position()) < 5.0 * 5.0
					|| !coverHasExit(level, feet, threat) || !fearCovered(level, ashWeaver, threat, point)) continue;
			Path path = fearCreatePath(level, ashWeaver, feet);
			paths++;
			if (fearPathSafe(level, ashWeaver, player, threat, path)) {
				data.putLong("ashWeaverHideCover", feet.asLong());
				data.putLong("ashWeaverHideRepath", now + 20L);
				ashWeaver.getNavigation().moveTo(path, ashWeaver.isSprinting() ? PLAYER_NAV_SPEED : PLAYER_NAV_SPEED * 0.3);
				return;
			}
			if (paths >= 2) { scan++; break; }
		}
		data.putInt("ashWeaverHideScan", scan);
		if (scan >= 144) {
			// No cover exists here: fall back to ordinary escape rather than crouching forever.
			finishHiding(ashWeaver, data, now);
			data.putLong("ashWeaverHideRetry", now + 160L);
		}
	}


	private static void retreatWhileSeekingCover(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			LivingEntity threat, CompoundTag data, long now) {
		if (now < dataLong(data, "ashWeaverHideRetreatAt", 0L)) return;
		data.putLong("ashWeaverHideRetreatAt", now + 20L);
		Vec3 away = ashWeaver.position().subtract(threat.position()).multiply(1.0, 0.0, 1.0);
		if (away.lengthSqr() < 0.01) away = new Vec3(1.0, 0.0, 0.0);
		away = away.normalize();
		// One path search after cheap terrain tests; never retreat through the player's position.
		for (int i = 0; i < 4; i++) {
			Vec3 offset = away.yRot((float) ((i % 2 == 0 ? 1 : -1) * (i / 2) * Math.PI / 4.0)).scale(3.0);
			BlockPos feet = BlockPos.containing(ashWeaver.position().add(offset));
			Vec3 point = Vec3.atBottomCenterOf(feet);
			if (!isSafeStandingPosition(level, feet) || player != null && point.distanceToSqr(player.position()) > 24.0 * 24.0
					|| point.distanceToSqr(threat.position()) <= ashWeaver.distanceToSqr(threat) + 1.0) continue;
			Path path = fearCreatePath(level, ashWeaver, feet);
			if (fearPathSafe(level, ashWeaver, player, threat, path))
				ashWeaver.getNavigation().moveTo(path, PLAYER_NAV_SPEED);
			return;
		}
	}

	private static void finishHiding(AshWeaverEntity ashWeaver, CompoundTag data, long now) {
		for (String key : new String[] {"ashWeaverHideCover", "ashWeaverHidePeek", "ashWeaverHideLostAt",
				"ashWeaverHideThreat", "ashWeaverHideDimension"}) data.remove(key);
		ashWeaver.getNavigation().stop();
		setSneaking(ashWeaver, false);
		ashWeaver.setSprinting(false);
		data.putLong("ashWeaverHideRetry", now + 80L);
		setState(data, STATE_FOLLOW, now, 0L);
	}

	private static boolean handleHiding(ServerLevel level, AshWeaverEntity ashWeaver, @Nullable Player player,
			@Nullable LivingEntity threat, CompoundTag data, long now) {
		if (threat == null || threat.level() != level || ashWeaver.distanceToSqr(threat) > 32.0 * 32.0) {
			if (!data.contains("ashWeaverHideLostAt")) data.putLong("ashWeaverHideLostAt", now);
			if (now - dataLong(data, "ashWeaverHideLostAt", now) >= 100L) {
				finishHiding(ashWeaver, data, now);
				return false;
			}
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			setSneaking(ashWeaver, true);
			applySneakEdgeBrake(level, ashWeaver);
			return true;
		}
		ashWeaver.setTarget(null);
		clearHeldRose(ashWeaver);
		if (ashWeaver.hurtTime == 10 && ashWeaver.getLastDamageSource() != null
				&& ashWeaver.getLastDamageSource().getEntity() instanceof LivingEntity newThreat
				&& newThreat != threat && newThreat != ashWeaver && newThreat.isAlive() && newThreat.level() == level) {
			// Only an actual new attacker may take priority over the remembered hiding threat.
			threat = newThreat;
			data.putString("ashWeaverHideThreat", threat.getUUID().toString());
			data.remove("ashWeaverHideCover");
			data.remove("ashWeaverHidePeek");
			data.putLong("ashWeaverHideOrigin", ashWeaver.blockPosition().asLong());
			data.putInt("ashWeaverHideScan", 0);
			data.putInt("ashWeaverHidePhase", 0);
			data.putLong(KEY_STATE_STARTED, now);
			ashWeaver.getNavigation().stop();
		}
		boolean check = Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 5) == 0;
		boolean close = ashWeaver.distanceToSqr(threat) < 8.0 * 8.0;
		if (check) {
			boolean visible = !fearCovered(level, ashWeaver, threat, ashWeaver.position());
			boolean pursuing = threat instanceof Mob mob && mob.getTarget() == ashWeaver;
			if (visible || pursuing || close || ashWeaver.hurtTime > 0) data.remove("ashWeaverHideLostAt");
			else if (!data.contains("ashWeaverHideLostAt")) data.putLong("ashWeaverHideLostAt", now);
		}
		// Avoid an immediate exit when the cooldown key is absent.
		if (data.contains("ashWeaverHideLostAt") && now - dataLong(data, "ashWeaverHideLostAt", now) >= 100L) {
			finishHiding(ashWeaver, data, now);
			return false;
		}
		if (!ashWeaver.onGround()) return true;
		boolean urgentEscape = ashWeaver.distanceToSqr(threat) < 5.0 * 5.0
				&& ashWeaver.getHealth() <= ashWeaver.getMaxHealth() * 0.4F
				&& (!data.contains("ashWeaverHideCover") || ashWeaver.distanceToSqr(Vec3.atBottomCenterOf(
						BlockPos.of(dataLong(data, "ashWeaverHideCover", 0L)))) > 3.0 * 3.0);
		setSneaking(ashWeaver, !urgentEscape);
		setRunning(ashWeaver, urgentEscape);
		ashWeaver.getLookControl().setLookAt(threat, 18.0F, 12.0F);
		if (!data.contains("ashWeaverHideCover")) {
			if (check) searchFearCover(level, ashWeaver, player, threat, data, now);
			if (dataInt(data, KEY_STATE, STATE_IDLE) != STATE_HIDE) return false;
			if (!data.contains("ashWeaverHideCover") && check) {
				retreatWhileSeekingCover(level, ashWeaver, player, threat, data, now);
				setRunning(ashWeaver, urgentEscape);
			}
			applySneakEdgeBrake(level, ashWeaver);
			enforceGroundSpeed(ashWeaver);
			return true;
		}
		BlockPos cover = BlockPos.of(dataLong(data, "ashWeaverHideCover", 0L));
		Vec3 coverPoint = Vec3.atBottomCenterOf(cover);
		int phase = dataInt(data, "ashWeaverHidePhase", 0);
		if (check && (!isSafeStandingPosition(level, cover)
				|| !fearCovered(level, ashWeaver, threat, coverPoint) || threat.distanceToSqr(coverPoint) < 4.0 * 4.0
				|| player != null && ashWeaver.distanceToSqr(player) > 28.0 * 28.0
				|| now - dataLong(data, KEY_STATE_STARTED, now) > 240L && phase == 0)) {
			data.remove("ashWeaverHideCover");
			data.remove("ashWeaverHidePeek");
			data.putLong("ashWeaverHideOrigin", ashWeaver.blockPosition().asLong());
			data.putInt("ashWeaverHideScan", 0);
			data.putInt("ashWeaverHidePhase", 0);
			data.putLong(KEY_STATE_STARTED, now);
			ashWeaver.getNavigation().stop();
			return true;
		}
		if (phase == 2) {

			if (close || ashWeaver.hurtTime > 0 || threatWatching(threat, ashWeaver.position())
					|| Math.sqrt(ashWeaver.distanceToSqr(threat)) + 1.0 < dataDouble(data, "ashWeaverHidePeekDistance", 0.0)
					|| now >= dataLong(data, "ashWeaverHidePeekUntil", 0L)) {
				data.putInt("ashWeaverHidePhase", 3);
				Path retreat = fearCreatePath(level, ashWeaver, cover);
				if (fearPathSafe(level, ashWeaver, player, threat, retreat) && retreat.getNodeCount() <= 3)
					ashWeaver.getNavigation().moveTo(retreat, PLAYER_NAV_SPEED * 0.3);
				else ashWeaver.getNavigation().stop();
			}
		} else if (ashWeaver.distanceToSqr(coverPoint) > 0.45 * 0.45) {
			if (check && now >= dataLong(data, "ashWeaverHideRepath", 0L)) {
				data.putLong("ashWeaverHideRepath", now + 20L);
				Path path = fearCreatePath(level, ashWeaver, cover);
				if (fearPathSafe(level, ashWeaver, player, threat, path))
					ashWeaver.getNavigation().moveTo(path, ashWeaver.isSprinting() ? PLAYER_NAV_SPEED : PLAYER_NAV_SPEED * 0.3);
				else {
					data.remove("ashWeaverHideCover");
					data.putInt("ashWeaverHidePhase", 0);
					data.putLong("ashWeaverHideOrigin", ashWeaver.blockPosition().asLong());
					data.putInt("ashWeaverHideScan", 0);
					data.putLong(KEY_STATE_STARTED, now);
					ashWeaver.getNavigation().stop();
				}
			}
		} else {
			ashWeaver.getNavigation().stop();
			data.putInt("ashWeaverHidePhase", 1);
			if (phase != 1) data.putLong("ashWeaverHideNextPeek", now + 40L + ashWeaver.getRandom().nextInt(41));
			if (check && !close && ashWeaver.getHealth() > ashWeaver.getMaxHealth() * 0.35F
					&& now >= dataLong(data, "ashWeaverHideNextPeek", now)) {
				data.putLong("ashWeaverHideNextPeek", now + 60L + ashWeaver.getRandom().nextInt(61));
				Vec3 toward = threat.position().subtract(coverPoint);
				Vec3 side = new Vec3(-toward.z, 0.0, toward.x).normalize();
				for (int sign : new int[] {-1, 1}) {
					BlockPos peek = BlockPos.containing(coverPoint.add(side.scale(sign * 1.0)));
					Vec3 peekPoint = Vec3.atBottomCenterOf(peek);
					if (peek.equals(cover) || !isSafeStandingPosition(level, peek)
							|| threatWatching(threat, peekPoint) || fearCovered(level, ashWeaver, threat, peekPoint)) continue;
					Path path = fearCreatePath(level, ashWeaver, peek);
					if (!fearPathSafe(level, ashWeaver, player, threat, path) || path.getNodeCount() > 3) continue;
					data.putLong("ashWeaverHidePeek", peek.asLong());
					data.putLong("ashWeaverHidePeekUntil", now + 28L);
					data.putDouble("ashWeaverHidePeekDistance", Math.sqrt(ashWeaver.distanceToSqr(threat)));
					data.putInt("ashWeaverHidePhase", 2);
					ashWeaver.getNavigation().moveTo(path, PLAYER_NAV_SPEED * 0.3);
					break;
				}
			}
		}
		applySneakEdgeBrake(level, ashWeaver);
		enforceGroundSpeed(ashWeaver);
		return true;
	}

	// Scores belong to the assigned player; offline time earns no trust.
	public static double relationship(AshWeaverEntity ashWeaver) {
		CompoundTag data = ashWeaver.getPersistentData();
		String owner = ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer);
		if (!owner.equals(dataString(data, "ashWeaverRelationshipOwner", ""))) {
			data.putString("ashWeaverRelationshipOwner", owner);
			data.putDouble("ashWeaverRelationship", 0.0);
			data.putDouble("ashWeaverFamiliarity", 0.0);
			for (String key : new String[] {"ashWeaverAvoidOwner", "ashWeaverSharedTicks",
					"ashWeaverRelationshipChatAt", "ashWeaverRelationshipHurtAt",
					"ashWeaverRelationshipHelpAt", "ashWeaverRelationshipHelpNext",
					"ashWeaverRelationshipHarmUntil", "ashWeaverLastOwnerHarm", "ashWeaverOwnerHarmRun",
					"ashWeaverOwnerHarmSerial", "ashWeaverApologyFor", "ashWeaverApologyWindow",
					"ashWeaverApologyAttempts", "ashWeaverApologyReplyAt", "ashWeaverApologyRefundWindow",
					"ashWeaverApologiesAccepted"}) data.remove(key);
		}
		return dataDouble(data, "ashWeaverRelationship", 0.0);
	}

	public static void noteConversation(AshWeaverEntity ashWeaver, long gameTime) {
		double bond = relationship(ashWeaver);
		CompoundTag data = ashWeaver.getPersistentData();
		if (gameTime < dataLong(data, "ashWeaverRelationshipChatAt", 0L)) return;
		data.putLong("ashWeaverRelationshipChatAt", gameTime + 1200L);
		if (gameTime >= dataLong(data, "ashWeaverRelationshipHarmUntil", 0L))
			data.putDouble("ashWeaverRelationship", Math.min(100.0, bond + 1.0));
		data.putDouble("ashWeaverFamiliarity", Math.min(100.0,
				dataDouble(data, "ashWeaverFamiliarity", 0.0) + 0.5));
	}

	public static double speechObfuscation(AshWeaverEntity ashWeaver) {
		double bond = relationship(ashWeaver);
		double familiarity = dataDouble(ashWeaver.getPersistentData(), "ashWeaverFamiliarity", 0.0);
		return Math.min(1.0, Math.max(0.0, 0.9 * (1.0 - familiarity / 100.0)
				+ Math.max(0.0, -bond) * 0.007));
	}


	// Owner-caused harm is recorded once even if an apology arrives before the next normal tick.
	private static double noteOwnerHarm(AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		double bond = relationship(ashWeaver);
		int hurtAt = ashWeaver.getLastHurtByMobTimestamp();
		var damage = ashWeaver.getLastDamageSource();
		if (hurtAt >= 0 && ashWeaver.tickCount >= hurtAt && ashWeaver.tickCount - hurtAt <= 20
				&& hurtAt != dataInt(data, "ashWeaverRelationshipHurtAt", -1)
				&& damage != null && damage.getEntity() == player) {
			long previous = dataLong(data, "ashWeaverLastOwnerHarm", Long.MIN_VALUE);
			int repeated = previous != Long.MIN_VALUE && gameTime >= previous && gameTime - previous <= 1200L
					? dataInt(data, "ashWeaverOwnerHarmRun", 0) : 0;
			long serial = dataLong(data, "ashWeaverOwnerHarmSerial", 0L);
			data.putLong("ashWeaverOwnerHarmSerial", serial >= 1_000_000_000L || serial < 0L ? 1L : serial + 1L);
			data.putLong("ashWeaverLastOwnerHarm", gameTime);
			data.putInt("ashWeaverOwnerHarmRun", Math.min(20, Math.max(0, repeated) + 1));
			data.putInt("ashWeaverRelationshipHurtAt", hurtAt);
			bond = Math.max(-100.0, bond - 18.0);
			data.putDouble("ashWeaverRelationship", bond);
			data.putLong("ashWeaverRelationshipHarmUntil", gameTime + 2400L);
			if (bond <= -25.0) data.putBoolean("ashWeaverAvoidOwner", true);
		}
		return bond;
	}

	public static String apologize(AshWeaverEntity ashWeaver, Player player) {
		if (ashWeaver == null || player == null || !ashWeaver.isAlive() || !player.isAlive()
				|| player.isSpectator() || !(ashWeaver.level() instanceof ServerLevel level)
				|| player.level() != level
				|| !player.getUUID().toString().equals(ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))
				|| !isCanonicalCompanion(ashWeaver, player.getUUID().toString())) return "";
		CompoundTag data = ashWeaver.getPersistentData();
		long now = level.getGameTime();
		double bond = noteOwnerHarm(ashWeaver, player, data, now);
		long window = dataLong(data, "ashWeaverApologyWindow", Long.MIN_VALUE);
		if (window == Long.MIN_VALUE || now < window || now - window >= 600L) {
			data.putLong("ashWeaverApologyWindow", now); data.putInt("ashWeaverApologyAttempts", 0);
		}
		int attempts = Math.min(32, Math.max(0, dataInt(data, "ashWeaverApologyAttempts", 0)) + 1);
		data.putInt("ashWeaverApologyAttempts", attempts);
		long next = dataLong(data, "ashWeaverApologyReplyAt", 0L);
		if (now < next && next - now <= 200L) return ""; // Do not postpone an already queued reply.
		data.putLong("ashWeaverApologyReplyAt", now + 200L);
		if (attempts > 4) return "Give me time.";
		if (ashWeaver.distanceToSqr(player) > 12.0 * 12.0 || !ashWeaver.hasLineOfSight(player))
			return "Come closer.";
		long serial = dataLong(data, "ashWeaverOwnerHarmSerial", 0L);
		if (serial <= 0L) return "There's nothing to forgive.";
		if (serial == dataLong(data, "ashWeaverApologyFor", -1L)) return "I heard you.";
		long harmed = dataLong(data, "ashWeaverLastOwnerHarm", Long.MIN_VALUE);
		if (harmed == Long.MIN_VALUE || now < harmed || now - harmed > 900L) return "I need time.";
		if (dataInt(data, "ashWeaverOwnerHarmRun", 0) > 1) return "Words aren't enough.";
		int state = dataInt(data, KEY_STATE, STATE_IDLE);
		if (ashWeaver.getHealth() <= ashWeaver.getMaxHealth() * 0.35F || ashWeaver.hurtTime > 0
				|| player.hurtTime > 0 || ashWeaver.getTarget() != null
				|| state == STATE_HIDE || state == STATE_DEFEND || state == STATE_DODGE || state == STATE_ALERT
				|| isOwnerAimingAtWeaver(player, ashWeaver)) return "Let's get safe first.";
		long refundWindow = dataLong(data, "ashWeaverApologyRefundWindow", Long.MIN_VALUE);
		if (refundWindow == Long.MIN_VALUE || now < refundWindow || now - refundWindow >= 6000L) {
			data.putLong("ashWeaverApologyRefundWindow", now); data.putInt("ashWeaverApologiesAccepted", 0);
		}
		int accepted = Math.max(0, Math.min(3, dataInt(data, "ashWeaverApologiesAccepted", 0)));
		data.putLong("ashWeaverApologyFor", serial); // This injury can never earn a second refund, even after reload.
		if (accepted >= 3) return "Show me you mean it.";
		data.putInt("ashWeaverApologiesAccepted", accepted + 1);
		bond = Math.min(100.0, bond + (8.0 / (1 << accepted))); // 8, 4, 2: always less than the 18 lost per hit.
		data.putDouble("ashWeaverRelationship", bond);
		data.putLong("ashWeaverRelationshipHarmUntil",
				Math.min(dataLong(data, "ashWeaverRelationshipHarmUntil", now + 600L), now + 600L));
		if (bond >= -10.0) data.remove("ashWeaverAvoidOwner");
		// No familiarity, conversation reward, healing or fear-memory deletion is granted for apologizing.
		return accepted == 0 ? "Be careful." : "I need you to be careful.";
	}

	private static void updateRelationship(AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		double bond = noteOwnerHarm(ashWeaver, player, data, gameTime);
		int hurtAt = ashWeaver.getLastHurtByMobTimestamp();
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) != 0) return;
		LivingEntity aggressor = ashWeaver.getLastHurtByMob();
		int helpAt = player.getLastHurtMobTimestamp();
		if (aggressor != null && aggressor != player && player.getLastHurtMob() == aggressor
				&& ashWeaver.tickCount - hurtAt <= 200 && player.tickCount - helpAt <= 40
				&& helpAt != dataInt(data, "ashWeaverRelationshipHelpAt", -1)
				&& ashWeaver.distanceToSqr(player) <= 16.0 * 16.0) {
			data.putInt("ashWeaverRelationshipHelpAt", helpAt);
			if (gameTime >= dataLong(data, "ashWeaverRelationshipHelpNext", 0L)) {
				bond = Math.min(100.0, bond + 3.0);
				data.putLong("ashWeaverRelationshipHelpNext", gameTime + 600L);
			}
		}
		boolean calm = ashWeaver.distanceToSqr(player) <= 12.0 * 12.0
				&& player.isAlive() && !player.isSpectator() && ashWeaver.hasLineOfSight(player)
				&& ashWeaver.hurtTime == 0 && player.hurtTime == 0
				&& (hurtAt <= 0 || ashWeaver.tickCount - hurtAt > 200)
				&& !isOwnerAimingAtWeaver(player, ashWeaver)
				&& gameTime >= dataLong(data, "ashWeaverRelationshipHarmUntil", 0L);
		if (calm) {
			int shared = dataInt(data, "ashWeaverSharedTicks", 0) + 20;
			if (shared >= 1200) {
				shared -= 1200;
				bond = Math.min(100.0, bond + 0.5);
				data.putDouble("ashWeaverFamiliarity", Math.min(100.0,
						dataDouble(data, "ashWeaverFamiliarity", 0.0) + 2.0));
			}
			data.putInt("ashWeaverSharedTicks", shared);
		}
		data.putDouble("ashWeaverRelationship", bond);
		// Separate entry/exit thresholds prevent movement flickering.
		if (bond <= -25.0) data.putBoolean("ashWeaverAvoidOwner", true);
		else if (bond >= -10.0) data.remove("ashWeaverAvoidOwner");
	}

	private static void handleRelationshipAvoidance(ServerLevel level, AshWeaverEntity ashWeaver,
			Player player, CompoundTag data, long gameTime) {
		clearPlantingIntent(data);
		clearHeldRose(ashWeaver);
		ashWeaver.setTarget(null);
		setStateIfChanged(data, STATE_IDLE, gameTime, 0L);
		data.remove(KEY_PARKOUR_JUMPED);
		double distance = ashWeaver.distanceToSqr(player);
		double comfort = 6.0 + Math.max(0.0, -relationship(ashWeaver)) * 0.06;
		setRunning(ashWeaver, distance < 3.0 * 3.0);
		setSneaking(ashWeaver, isNearLedge(level, ashWeaver));
		ashWeaver.getLookControl().setLookAt(player, 16.0F, 12.0F);
		if (!ashWeaver.onGround()) return; // Finish airborne motion before choosing a retreat.
		if (distance >= comfort * comfort) {
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
		} else if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) == 0) {
			Vec3 away = ashWeaver.position().subtract(player.position()).multiply(1.0, 0.0, 1.0);
			if (away.lengthSqr() < 0.01) away = new Vec3(1.0, 0.0, 0.0);
			away = away.normalize();
			boolean found = false;
			for (int i = 0; i < 8 && !found; i++) {
				Vec3 offset = away.yRot((float) ((i % 2 == 0 ? 1 : -1) * ((i + 1) / 2) * Math.PI / 8.0)).scale(4.0);
				BlockPos feet = BlockPos.containing(ashWeaver.position().add(offset));
				for (int dy = 0; dy >= -1; dy--) {
					BlockPos candidate = feet.offset(0, dy, 0);
					if (!isSafeStandingPosition(level, candidate)
							|| player.distanceToSqr(Vec3.atBottomCenterOf(candidate)) <= distance + 1.0) continue;
					Path path = ashWeaver.getNavigation().createPath(candidate, 0);
					if (path != null && path.canReach()) {
						ashWeaver.getNavigation().moveTo(path, PLAYER_NAV_SPEED);
						found = true;
						break;
					}
				}
			}
			if (!found) ashWeaver.getNavigation().stop();
		}
		applySneakEdgeBrake(level, ashWeaver);
		enforceGroundSpeed(ashWeaver);
	}

	private static void ensurePlayerPace(AshWeaverEntity ashWeaver) {
		AttributeInstance movement = ashWeaver.getAttribute(Attributes.MOVEMENT_SPEED);
		if (movement != null && Math.abs(movement.getBaseValue() - PLAYER_WALK_ATTRIBUTE) > 0.0001)
			movement.setBaseValue(PLAYER_WALK_ATTRIBUTE);
	}

	private static void setRunning(AshWeaverEntity ashWeaver, boolean needed) {
		ashWeaver.setSprinting(needed && !ashWeaver.getNavigation().isDone()
				&& !ashWeaver.isShiftKeyDown() && !ashWeaver.isInWater());
	}

	private static void enforceGroundSpeed(AshWeaverEntity ashWeaver) {
		if (!ashWeaver.onGround() || ashWeaver.isInWater() || ashWeaver.hurtTime > 0) return;
		Vec3 velocity = ashWeaver.getDeltaMovement();
		double cap = ashWeaver.isSprinting() ? 0.281 : 0.216;
		double horizontal = velocity.horizontalDistanceSqr();
		if (horizontal > cap * cap) {
			double scale = cap / Math.sqrt(horizontal);
			ashWeaver.setDeltaMovement(velocity.x * scale, velocity.y, velocity.z * scale);
		}
	}

	private static void maybeDropEmergencyRose(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			CompoundTag data, long gameTime, int state) {
		if (state == STATE_PLANT || state == STATE_SEEK_PLANT_SITE || state == STATE_DODGE
				|| player.isCreative() || player.isSpectator() || !ashWeaver.onGround()
				|| ashWeaver.distanceToSqr(player) > 10.0 * 10.0
				|| gameTime < dataLong(data, KEY_NEXT_ROSE_GIFT, 0L)
				|| player.getHealth() > player.getMaxHealth() * 0.5F) return;
		int hurtTime = player.getLastHurtByMobTimestamp();
		if (hurtTime <= 0 || player.tickCount - hurtTime > 70) return;
		LivingEntity attacker = player.getLastHurtByMob();
		if (!(attacker instanceof Mob mob) || !mob.isAlive() || mob.level() != level
				|| mob instanceof AshWeaverEntity || mob instanceof net.minecraft.world.entity.TamableAnimal pet && pet.isTame()
				|| mob.distanceToSqr(player) > 16.0 * 16.0 || mob.getTarget() != player
				|| !isCanonicalCompanion(ashWeaver, player.getUUID().toString())
				|| !roseBudgetAvailable(level, player, gameTime)) return;
		AABB nearby = AABB.ofSize(player.position(), 6.0, 4.0, 6.0);
		if (!level.getEntitiesOfClass(ItemEntity.class, nearby,
				item -> item.isAlive() && item.getItem().is(TheBackwoodsModBlocks.ASH_ROSE.get().asItem())).isEmpty()) return;
		Vec3 toward = player.position().subtract(ashWeaver.position());
		if (toward.horizontalDistanceSqr() < 0.01) toward = new Vec3(0.0, 0.0, 1.0);
		ItemEntity rose = new ItemEntity(level, ashWeaver.getX(), ashWeaver.getEyeY() - 0.4,
				ashWeaver.getZ(), new ItemStack(TheBackwoodsModBlocks.ASH_ROSE.get()));
		Vec3 horizontal = new Vec3(toward.x, 0.0, toward.z).normalize().scale(0.19);
		rose.setDeltaMovement(horizontal.x, 0.18, horizontal.z);
		rose.setTarget(player.getUUID());
		ashWeaver.getLookControl().setLookAt(player, 36.0F, 25.0F);
		swingMainHand(level, ashWeaver);
		if (!level.addFreshEntity(rose)) return;
		spendRoseBudget(level, player, gameTime);
		data.putLong(KEY_NEXT_ROSE_GIFT, gameTime + 600L);
	}

	private static boolean isRangedReady(Player player) {
		ItemStack main = player.getMainHandItem();
		ItemStack offhand = player.getOffhandItem();
		return isRangedReady(player, main) || isRangedReady(player, offhand);
	}

	private static boolean isRangedReady(Player player, ItemStack stack) {
		if (stack.isEmpty()) return false;
		if (stack.is(Items.BOW) || stack.is(Items.TRIDENT)) return player.isUsingItem();
		if (stack.is(Items.CROSSBOW)) return player.isUsingItem() || CrossbowItem.isCharged(stack);
		if (player.isUsingItem() && player.getUseItem().getItem() == stack.getItem()) {
			String animation = stack.getUseAnimation().name();
			if (animation.equals("BOW") || animation.equals("CROSSBOW") || animation.equals("SPEAR")) return true;
		}
		// Generic fallback for gun mods without a common aiming API.
		String path = BuiltInRegistries.ITEM.getKey(stack.getItem()).getPath();
		return path.contains("gun") || path.contains("rifle") || path.contains("pistol")
				|| path.contains("musket") || path.contains("blaster") || path.contains("shotgun");
	}

	private static boolean isOwnerAimingAtWeaver(Player player, AshWeaverEntity ashWeaver) {
		if (!isRangedReady(player) || player.distanceToSqr(ashWeaver) > 20.0 * 20.0
				|| !player.hasLineOfSight(ashWeaver)) return false;
		Vec3 ray = player.getLookAngle().normalize();
		Vec3 toward = ashWeaver.getEyePosition().subtract(player.getEyePosition());
		double forward = ray.dot(toward);
		double lateralSq = toward.lengthSqr() - forward * forward;
		return forward > 0.0 && forward / Math.sqrt(toward.lengthSqr()) > 0.95
				&& lateralSq < 0.75 * 0.75;
	}

	private static boolean startDodge(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		Vec3 look = player.getLookAngle();
		Vec3 side = new Vec3(-look.z, 0.0, look.x);
		if (side.horizontalDistanceSqr() < 0.001) return false;
		side = side.normalize();
		int preferred = (ashWeaver.getUUID().hashCode() & 1) == 0 ? 1 : -1;
		for (int sign : new int[] {preferred, -preferred}) {
			Vec3 point = ashWeaver.position().add(side.scale(sign * 2.6));
			for (int dy = 1; dy >= -1; dy--) {
				BlockPos feet = BlockPos.containing(point.x, ashWeaver.getY() + dy, point.z);
				if (!isSafeStandingPosition(level, feet)) continue;
				AABB moved = ashWeaver.getBoundingBox().move(feet.getX() + 0.5 - ashWeaver.getX(),
						feet.getY() - ashWeaver.getY(), feet.getZ() + 0.5 - ashWeaver.getZ());
				if (!level.noCollision(ashWeaver, moved)) continue;
				data.putDouble(KEY_DODGE_X, feet.getX() + 0.5);
				data.putDouble(KEY_DODGE_Y, feet.getY());
				data.putDouble(KEY_DODGE_Z, feet.getZ() + 0.5);
				data.putLong(KEY_NEXT_DODGE, gameTime + 65L);
				setState(data, STATE_DODGE, gameTime, gameTime + 24L);
				return true;
			}
		}
		data.putLong(KEY_NEXT_DODGE, gameTime + 30L);
		return false;
	}

	private static void handleDodge(AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		if (gameTime >= dataLong(data, KEY_STATE_UNTIL, 0L)
				|| ashWeaver.distanceToSqr(dataDouble(data, KEY_DODGE_X, ashWeaver.getX()),
						dataDouble(data, KEY_DODGE_Y, ashWeaver.getY()), dataDouble(data, KEY_DODGE_Z, ashWeaver.getZ())) < 1.2) {
			setState(data, STATE_FOLLOW, gameTime, 0L);
			return;
		}
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
		ashWeaver.setSprinting(false);
		ashWeaver.getLookControl().setLookAt(player, 26.0F, 20.0F);
		if (ashWeaver.getNavigation().isDone() || (ashWeaver.tickCount + ashWeaver.getId()) % 6 == 0)
			ashWeaver.getNavigation().moveTo(dataDouble(data, KEY_DODGE_X, ashWeaver.getX()),
					dataDouble(data, KEY_DODGE_Y, ashWeaver.getY()), dataDouble(data, KEY_DODGE_Z, ashWeaver.getZ()), PLAYER_NAV_SPEED);
	}

	private static boolean shouldWarnPlayer(AshWeaverEntity ashWeaver, Player player, Entity threat) {
		// "Unaware" is approximated by facing away and not having been hit recently.
		if (player.distanceToSqr(threat) > 12.0 * 12.0 || player.distanceToSqr(threat) < 3.0 * 3.0
				|| ashWeaver.distanceToSqr(player) > 16.0 * 16.0 || !ashWeaver.hasLineOfSight(threat)
				|| !(threat instanceof LivingEntity living) || !living.hasLineOfSight(player)) return false;
		Vec3 toward = threat.position().subtract(player.position());
		toward = new Vec3(toward.x, 0.0, toward.z);
		Vec3 facing = player.getLookAngle();
		facing = new Vec3(facing.x, 0.0, facing.z);
		return toward.lengthSqr() > 0.01 && facing.lengthSqr() > 0.01
				&& facing.normalize().dot(toward.normalize()) < 0.2
				&& (player.getLastHurtByMobTimestamp() <= 0
						|| player.tickCount - player.getLastHurtByMobTimestamp() > 60);
	}

	private static void handleWarn(ServerLevel level, AshWeaverEntity ashWeaver, Player player, @Nullable Entity threat, CompoundTag data, long gameTime) {
		if (threat == null || !threat.isAlive() || gameTime >= dataLong(data, KEY_STATE_UNTIL, 0L)) {
			setState(data, STATE_FOLLOW, gameTime, 0L);
			return;
		}
		ashWeaver.getNavigation().stop();
		ashWeaver.setSprinting(false);
		setSneaking(ashWeaver, false);
		clearHeldRose(ashWeaver);
		// Keep attention on the threat while signaling; the Look Around goal must not steal this cue.
		ashWeaver.getLookControl().setLookAt(threat, 32.0F, 25.0F);
		long age = gameTime - dataLong(data, KEY_STATE_STARTED, gameTime);
		if (age == 7 || age == 13 || age == 19) swingMainHand(level, ashWeaver);
	}

	@Nullable
	private static LivingEntity findOwnerAttacker(AshWeaverEntity ashWeaver, Player player) {
		LivingEntity attacker = player.getLastHurtByMob();
		int hurtTime = player.getLastHurtByMobTimestamp();
		if (attacker instanceof Player || attacker instanceof net.minecraft.world.entity.TamableAnimal pet && pet.isTame()) return null;
		if (attacker == null || hurtTime <= 0 || player.tickCount - hurtTime > 100 || !attacker.isAlive()
				|| attacker == player || attacker instanceof AshWeaverEntity || attacker.level() != player.level()
				|| player.distanceToSqr(attacker) > 24.0 * 24.0) return null;
		return attacker;
	}

	private static void handleDefend(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		Entity target;
		try {
			target = level.getEntity(UUID.fromString(dataString(data, KEY_DEFEND_UUID, "")));
		} catch (IllegalArgumentException ignored) {
			target = null;
		}
		if (!(target instanceof LivingEntity victim) || !victim.isAlive() || victim == player
				|| victim instanceof Player || victim instanceof net.minecraft.world.entity.TamableAnimal pet && pet.isTame()
				|| gameTime >= dataLong(data, KEY_STATE_UNTIL, 0L) || player.distanceToSqr(victim) > 24.0 * 24.0) {
			data.remove(KEY_DEFEND_UUID);
			setState(data, STATE_FOLLOW, gameTime, 0L);
			return;
		}
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
		ashWeaver.getLookControl().setLookAt(victim, 45.0F, 35.0F);
		double distanceSq = ashWeaver.distanceToSqr(victim);
		if (distanceSq <= 2.8 * 2.8 && ashWeaver.onGround() && gameTime >= dataLong(data, KEY_NEXT_CRIT, 0L)
				&& !isNearLedge(level, ashWeaver) && ashWeaver.getRandom().nextFloat() < 0.18F) {
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			ashWeaver.getJumpControl().jump();
			data.putLong(KEY_CRIT_UNTIL, gameTime + 14L);
			data.putLong(KEY_NEXT_CRIT, gameTime + 110L);
			return;
		}
		boolean critPending = gameTime < dataLong(data, KEY_CRIT_UNTIL, 0L);
		boolean chase = distanceSq > 4.2 * 4.2;
		setRunning(ashWeaver, chase && distanceSq > 8.0 * 8.0);
		if (chase) {
			if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 5) == 0 || ashWeaver.getNavigation().isDone()) {
				ashWeaver.getNavigation().moveTo(victim, PLAYER_NAV_SPEED);
			}
			tryJump(level, ashWeaver, victim, data, gameTime);
		} else {
			ashWeaver.getNavigation().stop();
			boolean fallingCrit = !ashWeaver.onGround() && ashWeaver.fallDistance > 0.1F
					&& ashWeaver.getDeltaMovement().y < -0.03 && !ashWeaver.isSprinting() && !ashWeaver.isInWater();
			if (gameTime >= dataLong(data, KEY_NEXT_ATTACK, 0L)
					&& distanceSq <= 2.6 * 2.6 && (!critPending || fallingCrit)
					&& ashWeaver.getSensing().hasLineOfSight(victim)) {
				AttributeInstance damage = ashWeaver.getAttribute(Attributes.ATTACK_DAMAGE);
				if (damage != null && damage.getBaseValue() < 4.0) damage.setBaseValue(4.0);
				double baseDamage = damage == null ? 4.0 : damage.getBaseValue();
				if (fallingCrit && damage != null) damage.setBaseValue(baseDamage * 1.5);
				swingMainHand(level, ashWeaver);
				boolean landed = ashWeaver.doHurtTarget(victim);
				if (damage != null && fallingCrit) damage.setBaseValue(baseDamage);
				if (landed && fallingCrit)
					level.sendParticles(ParticleTypes.CRIT, victim.getX(), victim.getEyeY(), victim.getZ(),
							12, 0.3, 0.3, 0.3, 0.08);
				data.remove(KEY_CRIT_UNTIL);
				data.putLong(KEY_NEXT_ATTACK, gameTime + 15L);
			}
		}
	}

	private static Player resolveAssignedPlayer(ServerLevel level, AshWeaverEntity ashWeaver) {
		String assigned = ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer);
		if (assigned == null || assigned.isEmpty()) return null;
		try {
			boolean canonical = registerBond(level, ashWeaver, assigned, level.getGameTime());
			ashWeaver.getPersistentData().putBoolean("ashWeaverDuplicateClaim", !canonical);
			return canonical ? level.getServer().getPlayerList().getPlayer(UUID.fromString(assigned)) : null;
		} catch (IllegalArgumentException ignored) {
			ashWeaver.getEntityData().set(AshWeaverEntity.DATA_assignedPlayer, "");
			ashWeaver.getPersistentData().remove("ashWeaverDuplicateClaim");
			return null;
		}
	}

	private static Player tryAssignPlayer(ServerLevel level, AshWeaverEntity ashWeaver, double x, double y, double z) {
		String assigned = ashWeaver.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer);
		if (assigned != null && !assigned.isEmpty()) return null;
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 40) != 0) return null;
		Player nearest = null;
		double bestDistanceSq = PLAYER_RANGE * PLAYER_RANGE;
		for (Player candidate : level.players()) {
			if (!candidate.isAlive() || candidate.isSpectator() || candidate.isCreative()) continue;
			double distanceSq = ashWeaver.distanceToSqr(candidate);
			if (distanceSq >= bestDistanceSq || isPlayerTaken(level, ashWeaver, candidate.getUUID().toString())) continue;
			nearest = candidate;
			bestDistanceSq = distanceSq;
		}
		if (nearest == null) return null;
		ashWeaver.getEntityData().set(AshWeaverEntity.DATA_assignedPlayer, nearest.getUUID().toString());
		if (!registerBond(level, ashWeaver, nearest.getUUID().toString(), level.getGameTime())) {
			ashWeaver.getEntityData().set(AshWeaverEntity.DATA_assignedPlayer, "");
			return null;
		}
		// Use the persistent personality chosen by initialization, including migrated existing Weavers.
		ashWeaver.getPersistentData().putLong(KEY_GREETING_UNTIL,
				dataBoolean(ashWeaver.getPersistentData(), "ashWeaverGreetingStyle", false) ? 0L : -1L);
		return nearest;
	}

	private static boolean isPlayerTaken(ServerLevel level, AshWeaverEntity ashWeaver, String playerUuid) {
		String claimed = dataString(bondRecord(level, playerUuid), "Weaver", "");
		if (!claimed.isEmpty() && !claimed.equals(ashWeaver.getUUID().toString())) return true;
		for (ServerLevel dimension : level.getServer().getAllLevels()) {
			for (Entity candidate : dimension.getAllEntities()) {
				if (candidate != ashWeaver && candidate instanceof AshWeaverEntity other && other.isAlive()
						&& playerUuid.equals(other.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))) return true;
			}
		}
		return false;
	}

	private static boolean hasEarlierClaim(ServerLevel level, AshWeaverEntity ashWeaver, String playerUuid) {
		for (ServerLevel dimension : level.getServer().getAllLevels()) {
			for (Entity candidate : dimension.getAllEntities()) {
				if (candidate instanceof AshWeaverEntity other && other != ashWeaver && other.isAlive()
						&& other.getUUID().compareTo(ashWeaver.getUUID()) < 0
						&& playerUuid.equals(other.getEntityData().get(AshWeaverEntity.DATA_assignedPlayer))) return true;
			}
		}
		return false;
	}

	private static Entity updateThreatAwareness(ServerLevel level, AshWeaverEntity ashWeaver, CompoundTag data, long gameTime) {
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 10) == 0) {
			Entity nearestThreat = findNearestThreat(level, ashWeaver, THREAT_RANGE);
			if (nearestThreat == null) data.remove(KEY_THREAT_UUID);
			else data.putString(KEY_THREAT_UUID, nearestThreat.getUUID().toString());
			return nearestThreat;
		}
		String cachedUuid = dataString(data, KEY_THREAT_UUID, "");
		if (cachedUuid.isEmpty()) return null;
		try {
			Entity cached = level.getEntity(UUID.fromString(cachedUuid));
			return cached != null && cached.isAlive() && ashWeaver.distanceToSqr(cached) <= THREAT_RANGE * THREAT_RANGE ? cached : null;
		} catch (IllegalArgumentException ignored) {
			data.remove(KEY_THREAT_UUID);
			return null;
		}
	}

	private static Entity findNearestThreat(ServerLevel level, AshWeaverEntity ashWeaver, double range) {
		AABB bounds = AABB.ofSize(ashWeaver.position(), range * 2.0, range * 2.0, range * 2.0);
		List<LivingEntity> threats = level.getEntitiesOfClass(LivingEntity.class, bounds, candidate -> candidate.isAlive()
				&& (candidate instanceof SplinterEntity || candidate instanceof LogSplinterEntity || candidate instanceof HollowEntity));
		Entity nearest = null;
		double nearestDistanceSq = range * range;
		for (LivingEntity candidate : threats) {
			double distanceSq = ashWeaver.distanceToSqr(candidate);
			if (distanceSq < nearestDistanceSq) {
				nearestDistanceSq = distanceSq;
				nearest = candidate;
			}
		}
		return nearest;
	}

	private static void handleSocialFollowAndPlanting(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			Entity threat, CompoundTag data, long gameTime) {
		double playerDistanceSq = ashWeaver.distanceToSqr(player);
		if (playerDistanceSq < 1.8 * 1.8 && isNearLedge(level, ashWeaver)) {
			pauseAtBoundary(ashWeaver, player, ashWeaver.blockPosition(), true);
			return; // Do not crowd the owner while approaching a drop.
		}
		Vec3 formationTarget = getFormationTarget(ashWeaver, player);
		formationTarget = avoidPlayerWork(level, ashWeaver, player, formationTarget, data, gameTime);
		boolean workDetour = data.contains(KEY_WORK_POS) && gameTime < dataLong(data, KEY_WORK_UNTIL, 0L)
				&& ashWeaver.distanceToSqr(formationTarget) > 1.5 * 1.5;
		boolean shouldFollow = playerDistanceSq > FOLLOW_START_DISTANCE_SQ || workDetour
				|| data.contains(KEY_BOUNDARY_POS) || data.contains(KEY_DETOUR_POS)
				|| data.contains(KEY_PARKOUR_JUMPED)
				|| gameTime < dataLong(data, KEY_BOUNDARY_UNTIL, 0L);
		if (shouldFollow) {
			setStateIfChanged(data, STATE_FOLLOW, gameTime, 0L);
			if (handleBoundaryFollow(level, ashWeaver, player, formationTarget, data, gameTime)) {
				clearHeldRose(ashWeaver);
				return;
			}
			if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 3) == 0 || ashWeaver.getNavigation().isDone()) {
				ashWeaver.getNavigation().moveTo(formationTarget.x, formationTarget.y, formationTarget.z, PLAYER_NAV_SPEED);
			}
			boolean ownerPullingAway = player.isSprinting()
					&& player.position().subtract(ashWeaver.position()).dot(player.getDeltaMovement()) > 0.45;
			boolean catchUp = playerDistanceSq > 8.0 * 8.0
					|| ownerPullingAway && playerDistanceSq > 4.0 * 4.0;
			if (!catchUp && !player.isSprinting() && playerDistanceSq > 5.0 * 5.0
					&& gameTime >= dataLong(data, KEY_NEXT_PACE_BURST, 0L)) {
				if (ashWeaver.getRandom().nextFloat() < 0.12F)
					data.putLong(KEY_PACE_BURST_UNTIL, gameTime + 8L + ashWeaver.getRandom().nextInt(7));
				data.putLong(KEY_NEXT_PACE_BURST, gameTime + 120L + ashWeaver.getRandom().nextInt(100));
			}
			setRunning(ashWeaver, catchUp || gameTime < dataLong(data, KEY_PACE_BURST_UNTIL, 0L));
			tryJump(level, ashWeaver, player, data, gameTime);
			ashWeaver.getLookControl().setLookAt(formationTarget.x, player.getEyeY(), formationTarget.z, 24.0F, 18.0F);
			if (playerDistanceSq < 12.0 * 12.0) updateSocialGaze(ashWeaver, player, data, gameTime);
		} else {
			setStateIfChanged(data, STATE_IDLE, gameTime, 0L);
			ashWeaver.getNavigation().stop();
			ashWeaver.setSprinting(false);
			updateSocialGaze(ashWeaver, player, data, gameTime);
		}

		updateHumanSneak(level, ashWeaver, player, data, gameTime);
		if (ashWeaver.isShiftKeyDown()) ashWeaver.setSprinting(false);
		clearHeldRose(ashWeaver);
	}

	private static boolean tryStartPlanting(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			CompoundTag data, long gameTime) {
		if (!roseBudgetAvailable(level, player, gameTime)
				|| !net.neoforged.neoforge.event.EventHooks.canEntityGrief(level, ashWeaver)
				|| ashWeaver.distanceToSqr(player) > 8.0 * 8.0
				|| ashWeaver.getEntityData().get(AshWeaverEntity.DATA_roseCooldown) > 0
				|| Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 10) != 0
				|| ashWeaver.getRandom().nextDouble() >= PLACE_CHANCE_PER_CHECK) return false;
		Mob plantThreat = findPlantThreat(level, player);
		if (plantThreat == null) return false;
		int nearbyRoseCount = countNearbyAshRoses(level, player.blockPosition(), ROSE_COUNT_RADIUS_XZ, ROSE_COUNT_RADIUS_Y);
		ashWeaver.getEntityData().set(AshWeaverEntity.DATA_roseCount, nearbyRoseCount);
		if (nearbyRoseCount >= MAX_NEARBY_ROSES) return false;
		BlockPos plantTarget = findPlantSite(level, ashWeaver, player, plantThreat);
		if (plantTarget != null) {
			data.putLong(KEY_TARGET_POS, plantTarget.asLong());
			data.putString(KEY_PLANT_THREAT_UUID, plantThreat.getUUID().toString());
			data.putBoolean(KEY_PLANT_COMMITTED, false);
			setState(data, STATE_SEEK_PLANT_SITE, gameTime, gameTime + 200L);
			equipRose(ashWeaver);
			return true;
		}
		return false;
	}

	private static Mob findPlantThreat(ServerLevel level, Player player) {
		AABB area = AABB.ofSize(player.position(), 32.0, 16.0, 32.0);
		Mob nearest = null;
		double nearestDistanceSq = 16.0 * 16.0;
		for (Mob candidate : level.getEntitiesOfClass(Mob.class, area,
				mob -> mob.isAlive() && isHostileSplinter(mob) && mob.getTarget() == player)) {
			double distanceSq = candidate.distanceToSqr(player);
			if (distanceSq < nearestDistanceSq) {
				nearest = candidate;
				nearestDistanceSq = distanceSq;
			}
		}
		return nearest;
	}

	private static boolean isHostileSplinter(Entity entity) {
		return entity instanceof SplinterEntity || entity instanceof LogSplinterEntity
				|| entity instanceof BlindspotSplinterEntity || entity instanceof PetrifiedLogSplinterEntity;
	}

	private static boolean isPlantThreatStillValid(ServerLevel level, Player player, CompoundTag data) {
		try {
			Entity candidate = level.getEntity(UUID.fromString(dataString(data, KEY_PLANT_THREAT_UUID, "")));
			return candidate instanceof Mob mob && mob.isAlive() && isHostileSplinter(mob)
					&& mob.getTarget() == player && player.isAlive() && !player.isSpectator()
					&& mob.distanceToSqr(player) <= 16.0 * 16.0;
		} catch (IllegalArgumentException ignored) {
			return false;
		}
	}

	private static void handleAlert(ServerLevel level, AshWeaverEntity ashWeaver, Player player, @Nullable Entity threat,
			CompoundTag data, long gameTime) {
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
		setRunning(ashWeaver, threat != null && ashWeaver.distanceToSqr(threat) < 4.0 * 4.0);
		if (threat == null || !threat.isAlive() || ashWeaver.distanceToSqr(threat) > IMMEDIATE_DANGER_RANGE_SQ * 1.35) {
			if (gameTime >= dataLong(data, KEY_STATE_UNTIL, gameTime)) setState(data, STATE_FOLLOW, gameTime, 0L);
			return;
		}
		ashWeaver.getLookControl().setLookAt(threat, 45.0F, 35.0F);
		// Borrow the Rot's facing-preserving back-off idea, but only briefly and on safe ground.
		if (player.getLastHurtMob() == threat && player.tickCount - player.getLastHurtMobTimestamp() < 35
				&& gameTime >= dataLong(data, KEY_NEXT_HESITATION, 0L)) {
			float impulse = ashWeaver.getRandom().nextFloat();
			if (impulse < 0.2F) data.putLong(KEY_HESITATE_UNTIL, gameTime + 10L + ashWeaver.getRandom().nextInt(8));
			else if (impulse < 0.27F && ashWeaver.getHealth() > ashWeaver.getMaxHealth() * 0.65F
					&& ashWeaver.distanceToSqr(threat) > 4.0 * 4.0)
				data.putLong(KEY_BOLD_UNTIL, gameTime + 8L);
			data.putLong(KEY_NEXT_HESITATION, gameTime + 180L);
		}
		if (gameTime < dataLong(data, KEY_BOLD_UNTIL, 0L) && ashWeaver.distanceToSqr(threat) > 3.5 * 3.5) {
			ashWeaver.getNavigation().moveTo(threat, PLAYER_NAV_SPEED);
			return;
		}
		if (gameTime < dataLong(data, KEY_HESITATE_UNTIL, 0L) && ashWeaver.distanceToSqr(threat) > 3.0 * 3.0) {
			Vec3 back = ashWeaver.position().subtract(threat.position());
			back = new Vec3(back.x, 0.0, back.z);
			if (back.horizontalDistanceSqr() > 0.01) {
				back = back.normalize();
				BlockPos step = BlockPos.containing(ashWeaver.getX() + back.x, ashWeaver.getY(), ashWeaver.getZ() + back.z);
				if (isSafeStandingPosition(level, step)) {
					ashWeaver.getNavigation().stop();
					ashWeaver.setSprinting(false);
					Vec3 motion = ashWeaver.getDeltaMovement();
					ashWeaver.setDeltaMovement(back.x * 0.08, motion.y, back.z * 0.08);
					return;
				}
			}
		}
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 6) == 0 || ashWeaver.getNavigation().isDone()) {
			Vec3 away = ashWeaver.position().subtract(threat.position());
			if (away.horizontalDistanceSqr() < 0.001) away = new Vec3(1.0, 0.0, 0.0);
			away = new Vec3(away.x, 0.0, away.z).normalize();
			Vec3 towardPlayer = player.position().subtract(ashWeaver.position());
			if (towardPlayer.horizontalDistanceSqr() > 0.001) towardPlayer = new Vec3(towardPlayer.x, 0.0, towardPlayer.z).normalize();
			Vec3 escape = ashWeaver.position().add(away.scale(7.0)).add(towardPlayer.scale(2.0));
			ashWeaver.getNavigation().moveTo(escape.x, escape.y, escape.z, PLAYER_NAV_SPEED);
		}
		data.putLong(KEY_STATE_UNTIL, gameTime + 20L);
	}

	private static void handleSeekPlantSite(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		BlockPos target = getTargetPos(data);
		if (target == null || !isPlantThreatStillValid(level, player, data) || !isPlantBaseValid(level, target)
				|| gameTime > dataLong(data, KEY_STATE_UNTIL, gameTime)) {
			abortPlanting(ashWeaver, data, gameTime);
			return;
		}
		equipRose(ashWeaver);
		ashWeaver.setSprinting(false);
		ashWeaver.getLookControl().setLookAt(target.getX() + 0.5, target.getY() + 0.2, target.getZ() + 0.5, 30.0F, 30.0F);
		double distanceSq = ashWeaver.distanceToSqr(target.getX() + 0.5, target.getY(), target.getZ() + 0.5);
		setSneaking(ashWeaver, distanceSq <= 3.5 * 3.5 && isNearLedge(level, ashWeaver));
		if (distanceSq <= 2.15 * 2.15) {
			ashWeaver.getNavigation().stop();
			data.putBoolean(KEY_PLANT_SNEAK, isNearLedge(level, ashWeaver)
					|| target.getY() < ashWeaver.getY() - 0.5);
			data.remove(KEY_PLANT_SWING_AT);
			setState(data, STATE_PLANT, gameTime, gameTime + 40L);
			return;
		}
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 8) == 0 || ashWeaver.getNavigation().isDone()) {
			ashWeaver.getNavigation().moveTo(target.getX() + 0.5, target.getY(), target.getZ() + 0.5, PLAYER_NAV_SPEED);
		}
	}

	private static void handlePlant(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		BlockPos target = getTargetPos(data);
		if (target == null || !isPlantThreatStillValid(level, player, data)
				|| !dataBoolean(data, KEY_PLANT_COMMITTED, false) && !roseBudgetAvailable(level, player, gameTime)) {
			abortPlanting(ashWeaver, data, gameTime);
			return;
		}
		ashWeaver.getNavigation().stop();
		ashWeaver.setSprinting(false);
		equipRose(ashWeaver);
		setSneaking(ashWeaver, dataBoolean(data, KEY_PLANT_SNEAK, false));
		Vec3 aim = new Vec3(target.getX() + 0.5, target.getY() + 0.35, target.getZ() + 0.5);
		ashWeaver.getLookControl().setLookAt(aim.x, aim.y, aim.z, 45.0F, 45.0F);
		Vec3 aimDelta = aim.subtract(ashWeaver.getEyePosition());
		float desiredYaw = (float) (Math.atan2(aimDelta.z, aimDelta.x) * 180.0 / Math.PI) - 90.0F;
		float bodyYaw = Mth.approachDegrees(ashWeaver.getYRot(), desiredYaw, 18.0F);
		ashWeaver.setYRot(bodyYaw);
		ashWeaver.setYBodyRot(bodyYaw);
		long actionAge = gameTime - dataLong(data, KEY_STATE_STARTED, gameTime);
		long swingAt = dataLong(data, KEY_PLANT_SWING_AT, 0L);
		boolean facingTarget = aimDelta.lengthSqr() > 0.01
				&& Vec3.directionFromRotation(ashWeaver.getXRot(), ashWeaver.getYHeadRot()).dot(aimDelta.normalize()) > 0.86;
		if (swingAt == 0L && actionAge >= 4L && facingTarget) {
			swingMainHand(level, ashWeaver);
			data.putLong(KEY_PLANT_SWING_AT, gameTime);
			swingAt = gameTime;
		}
		if (swingAt > 0L && gameTime >= swingAt + 2L && !dataBoolean(data, KEY_PLANT_COMMITTED, false)) {
			if (!commitRosePlacement(level, ashWeaver, player, target, gameTime)) {
				abortPlanting(ashWeaver, data, gameTime);
				return;
			}
			level.playSound(null, target, SoundEvents.GRASS_PLACE, SoundSource.BLOCKS, 0.65F, 1.05F + ashWeaver.getRandom().nextFloat() * 0.12F);
			data.putBoolean(KEY_PLANT_COMMITTED, true);
			ashWeaver.getEntityData().set(AshWeaverEntity.DATA_roseCooldown, PLACE_COOLDOWN_TICKS);
			ashWeaver.getEntityData().set(AshWeaverEntity.DATA_roseCount,
					ashWeaver.getEntityData().get(AshWeaverEntity.DATA_roseCount) + 1);
		}
		if (actionAge >= 40L && !dataBoolean(data, KEY_PLANT_COMMITTED, false)) {
			abortPlanting(ashWeaver, data, gameTime);
			return;
		}
		if (dataBoolean(data, KEY_PLANT_COMMITTED, false) && gameTime >= swingAt + 10L) {
			clearHeldRose(ashWeaver);
			setSneaking(ashWeaver, false);
			setState(data, STATE_WAIT, gameTime, gameTime + 24L);
		}
	}

	private static void handleWait(AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		ashWeaver.getNavigation().stop();
		ashWeaver.setSprinting(false);
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
		BlockPos target = getTargetPos(data);
		if (target != null && gameTime + 8L < dataLong(data, KEY_STATE_UNTIL, gameTime)) {
			ashWeaver.getLookControl().setLookAt(target.getX() + 0.5, target.getY() + 0.4, target.getZ() + 0.5, 18.0F, 18.0F);
		} else {
			ashWeaver.getLookControl().setLookAt(player, 16.0F, 14.0F);
		}
		if (gameTime >= dataLong(data, KEY_STATE_UNTIL, gameTime)) {
			clearPlantingIntent(data);
			setState(data, STATE_IDLE, gameTime, 0L);
		}
	}

	private static void handleRegroup(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
		ashWeaver.setSprinting(false);
		if (player.level() != level) return;
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 10) == 0 && trySafeTeleportNearPlayer(level, ashWeaver, player)) {
			data.putInt(KEY_STUCK_TICKS, 0);
			setState(data, STATE_FOLLOW, gameTime, 0L);
		} else if (gameTime >= dataLong(data, KEY_STATE_UNTIL, gameTime)) {
			Vec3 target = getFormationTarget(ashWeaver, player);
			ashWeaver.getNavigation().moveTo(target.x, target.y, target.z, PLAYER_NAV_SPEED);
			data.putLong(KEY_STATE_UNTIL, gameTime + 40L);
		}
	}

	private static void updateProgressAndRecovery(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			CompoundTag data, int state, long gameTime) {
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) != 0) return;
		double previousX = dataDouble(data, KEY_LAST_X, ashWeaver.getX());
		double previousZ = dataDouble(data, KEY_LAST_Z, ashWeaver.getZ());
		double dx = ashWeaver.getX() - previousX;
		double dz = ashWeaver.getZ() - previousZ;
		boolean expectedToMove = state == STATE_FOLLOW || state == STATE_ALERT || state == STATE_SEEK_PLANT_SITE || state == STATE_REGROUP;
		int stuckTicks = dataInt(data, KEY_STUCK_TICKS, 0);
		if (expectedToMove && dx * dx + dz * dz < 0.15 * 0.15) stuckTicks += 20;
		else stuckTicks = 0;
		data.putDouble(KEY_LAST_X, ashWeaver.getX());
		data.putDouble(KEY_LAST_Z, ashWeaver.getZ());
		data.putInt(KEY_STUCK_TICKS, stuckTicks);
		if (ashWeaver.distanceToSqr(player) > TELEPORT_DISTANCE_SQ || stuckTicks >= 60) {
			clearPlantingIntent(data);
			setState(data, STATE_REGROUP, gameTime, gameTime + 40L);
		}
	}

	private static Vec3 avoidPlayerWork(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			Vec3 formationTarget, CompoundTag data, long gameTime) {
		if (ashWeaver.distanceToSqr(player) > 8.0 * 8.0 || player.isSprinting()) {
			data.remove(KEY_WORK_POS);
			return formationTarget;
		}
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 10) == 0) {
			ItemStack held = player.getMainHandItem();
			String itemPath = BuiltInRegistries.ITEM.getKey(held.getItem()).getPath();
			boolean workingItem = held.getItem() instanceof BlockItem || itemPath.contains("pickaxe")
					|| itemPath.contains("shovel") || itemPath.contains("drill") || itemPath.endsWith("_axe");
			if (workingItem) {
				Vec3 eye = player.getEyePosition();
				BlockHitResult hit = level.clip(new ClipContext(eye, eye.add(player.getLookAngle().scale(5.0)),
						ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, player));
				if (hit.getType() == HitResult.Type.BLOCK
						&& hit.getBlockPos().distToCenterSqr(ashWeaver.getX(), ashWeaver.getY(), ashWeaver.getZ()) < 2.8 * 2.8
						&& ashWeaver.getRandom().nextFloat() >= 0.12F) {
					data.putLong(KEY_WORK_POS, hit.getBlockPos().asLong());
					data.putLong(KEY_WORK_UNTIL, gameTime + 25L);
				}
			}
		}
		if (!data.contains(KEY_WORK_POS) || gameTime >= dataLong(data, KEY_WORK_UNTIL, 0L)) {
			data.remove(KEY_WORK_POS);
			return formationTarget;
		}
		Vec3 work = Vec3.atCenterOf(BlockPos.of(dataLong(data, KEY_WORK_POS, 0L)));
		if (formationTarget.distanceToSqr(work) > 2.8 * 2.8) {
			data.remove(KEY_WORK_POS);
			return formationTarget;
		}
		Vec3 away = new Vec3(formationTarget.x - work.x, 0.0, formationTarget.z - work.z);
		if (away.horizontalDistanceSqr() < 0.1) {
			Vec3 look = player.getLookAngle();
			away = new Vec3(-look.z, 0.0, look.x);
		}
		return formationTarget.add(away.normalize().scale(2.0));
	}

	private static boolean handleBoundaryFollow(ServerLevel level, AshWeaverEntity ashWeaver, Player player,
			Vec3 target, CompoundTag data, long gameTime) {
		long jumped = dataLong(data, KEY_PARKOUR_JUMPED, 0L);
		if (jumped > 0L && gameTime <= jumped + 14L) {
			setSneaking(ashWeaver, false);
			return true; // Airborne steering has sole control until landing or timeout.
		}
		if (!ashWeaver.onGround() || ashWeaver.isInWater()) return false;
		if (gameTime < dataLong(data, KEY_DETOUR_UNTIL, 0L) && data.contains(KEY_DETOUR_POS)) {
			BlockPos detour = BlockPos.of(dataLong(data, KEY_DETOUR_POS, 0L));
			if (ashWeaver.distanceToSqr(Vec3.atCenterOf(detour)) > 1.3 * 1.3
					&& isSafeStandingPosition(level, detour)) {
				setSneaking(ashWeaver, false);
				ashWeaver.setSprinting(false);
				if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 8) == 0)
					ashWeaver.getNavigation().moveTo(detour.getX() + 0.5, detour.getY(), detour.getZ() + 0.5, PLAYER_NAV_SPEED);
				ashWeaver.getLookControl().setLookAt(detour.getX() + 0.5, detour.getY() + 1.0, detour.getZ() + 0.5, 18.0F, 15.0F);
				return true;
			}
		}
		data.remove(KEY_DETOUR_POS);
		data.remove(KEY_DETOUR_UNTIL);

		Path path = ashWeaver.getNavigation().getPath();
		Vec3 waypoint = path != null && !path.isDone() ? Vec3.atCenterOf(path.getNextNodePos()) : target;
		if (data.contains(KEY_BOUNDARY_POS)) waypoint = Vec3.atCenterOf(BlockPos.of(dataLong(data, KEY_BOUNDARY_POS, 0L)));
		Vec3 forward = new Vec3(waypoint.x - ashWeaver.getX(), 0.0, waypoint.z - ashWeaver.getZ());
		if (forward.horizontalDistanceSqr() < 0.25) forward = new Vec3(target.x - ashWeaver.getX(), 0.0, target.z - ashWeaver.getZ());
		if (forward.horizontalDistanceSqr() < 0.25) return false;
		forward = forward.normalize();
		BlockPos ahead = BlockPos.containing(ashWeaver.getX() + forward.x * 1.1,
				ashWeaver.getY(), ashWeaver.getZ() + forward.z * 1.1);
		if (!level.hasChunkAt(ahead)) {
			pauseAtBoundary(ashWeaver, player, ahead, true);
			return true;
		}
		double nearDrop = terrainDrop(level, ashWeaver, ahead);
		BlockPos farther = BlockPos.containing(ashWeaver.getX() + forward.x * 2.0,
				ashWeaver.getY(), ashWeaver.getZ() + forward.z * 2.0);
		double farDrop = nearDrop > 0.01 ? terrainDrop(level, ashWeaver, farther) : 0.0;
		boolean walkableDescent = nearDrop >= 0.0 && nearDrop <= 2.05
				&& farDrop >= Math.max(0.0, nearDrop - 0.55)
				&& farDrop <= Math.min(3.05, nearDrop + 1.05);
		boolean gap = nearDrop > 0.01 && !walkableDescent;
		if (gap) {
			if (!data.contains(KEY_BOUNDARY_POS) || dataLong(data, KEY_BOUNDARY_POS, 0L) != ahead.asLong()) {
				data.putLong(KEY_BOUNDARY_POS, ahead.asLong());
				data.putLong(KEY_BOUNDARY_UNTIL, gameTime + 12L);
			}
			if (gameTime < dataLong(data, KEY_BOUNDARY_UNTIL, 0L)) {
				pauseAtBoundary(ashWeaver, player, ahead, nearDrop > 3.05);
				return true;
			}
			BlockPos landing = findConservativeLanding(level, ashWeaver, player, forward, ahead);
			if (landing != null && gameTime >= dataLong(data, KEY_NEXT_JUMP, 0L)) {
				data.putDouble(KEY_PARKOUR_X, landing.getX() + 0.5);
				data.putDouble(KEY_PARKOUR_Y, landing.getY());
				data.putDouble(KEY_PARKOUR_Z, landing.getZ() + 0.5);
				data.putLong(KEY_PARKOUR_JUMPED, gameTime);
				data.putLong(KEY_NEXT_JUMP, gameTime + 24L);
				data.remove(KEY_BOUNDARY_POS);
				data.remove(KEY_BOUNDARY_UNTIL);
				ashWeaver.getNavigation().stop();
				setSneaking(ashWeaver, false);
				ashWeaver.setSprinting(true);
				ashWeaver.getJumpControl().jump();
				return true;
			}
			BlockPos detour = findBoundaryDetour(level, ashWeaver, forward);
			if (detour != null) {
				data.putLong(KEY_DETOUR_POS, detour.asLong());
				data.putLong(KEY_DETOUR_UNTIL, gameTime + 36L);
				data.remove(KEY_BOUNDARY_POS);
				data.remove(KEY_BOUNDARY_UNTIL);
				setSneaking(ashWeaver, false);
				ashWeaver.setSprinting(false);
				ashWeaver.getNavigation().moveTo(detour.getX() + 0.5, detour.getY(), detour.getZ() + 0.5, PLAYER_NAV_SPEED);
				return true;
			}
			data.putLong(KEY_BOUNDARY_UNTIL, gameTime + 20L);
			pauseAtBoundary(ashWeaver, player, ahead, nearDrop > 3.05);
			return true;
		}
		data.remove(KEY_BOUNDARY_POS);
		if (gameTime < dataLong(data, KEY_BOUNDARY_UNTIL, 0L)) {
			pauseAtBoundary(ashWeaver, player, ahead, false);
			return true;
		}
		data.remove(KEY_BOUNDARY_UNTIL);
		if (gameTime >= dataLong(data, KEY_BOUNDARY_NEXT, 0L) && isNarrowPassage(level, ashWeaver, forward, ahead)
				&& player.position().subtract(ashWeaver.position()).dot(forward) > 1.5) {
			data.putLong(KEY_BOUNDARY_NEXT, gameTime + 90L);
			data.putLong(KEY_BOUNDARY_UNTIL, gameTime + 12L);
		}
		if (gameTime < dataLong(data, KEY_BOUNDARY_UNTIL, 0L)) {
			pauseAtBoundary(ashWeaver, player, ahead, false);
			return true;
		}
		return false;
	}

	private static void pauseAtBoundary(AshWeaverEntity ashWeaver, Player player, BlockPos ahead, boolean dangerousDrop) {
		ashWeaver.getNavigation().stop();
		ashWeaver.setSprinting(false);
		setSneaking(ashWeaver, dangerousDrop);
		ashWeaver.getLookControl().setLookAt(player.distanceToSqr(ashWeaver) < 8.0 * 8.0
				? player.getX() : ahead.getX() + 0.5,
				player.distanceToSqr(ashWeaver) < 8.0 * 8.0 ? player.getEyeY() : ahead.getY() - 1.0,
				player.distanceToSqr(ashWeaver) < 8.0 * 8.0 ? player.getZ() : ahead.getZ() + 0.5,
				14.0F, 12.0F);
		Vec3 motion = ashWeaver.getDeltaMovement();
		ashWeaver.setDeltaMovement(0.0, motion.y, 0.0);
	}

	private static boolean isNarrowPassage(ServerLevel level, AshWeaverEntity ashWeaver, Vec3 forward, BlockPos ahead) {
		Vec3 side = new Vec3(-forward.z * 0.9, 0.0, forward.x * 0.9);
		for (int sign : new int[] {-1, 1}) {
			BlockPos edge = BlockPos.containing(ahead.getX() + 0.5 + side.x * sign,
					ashWeaver.getY(), ahead.getZ() + 0.5 + side.z * sign);
			if (terrainDrop(level, ashWeaver, edge) > 2.05) return true;
		}
		return false;
	}

	@Nullable
	private static BlockPos findConservativeLanding(ServerLevel level, AshWeaverEntity ashWeaver,
			Player player, Vec3 forward, BlockPos gap) {
		double depth = terrainDrop(level, ashWeaver, gap);
		if (depth < 0.0 || depth > 2.05) return null; // Never gamble over a deep drop.
		if (!player.onGround() || player.getY() < ashWeaver.getY() - 1.5
				|| player.position().subtract(ashWeaver.position()).dot(forward) < 2.5) return null;
		for (int step = 1; step <= 4; step++) {
			double distance = step * 0.5;
			BlockPos through = BlockPos.containing(ashWeaver.getX() + forward.x * distance,
					ashWeaver.getY(), ashWeaver.getZ() + forward.z * distance);
			if (!level.hasChunkAt(through) || !level.getBlockState(through).isAir()
					|| !level.getBlockState(through.above()).isAir() || !level.getBlockState(through.above(2)).isAir()) return null;
		}
		for (int dy = 0; dy >= -1; dy--) {
			BlockPos landing = BlockPos.containing(ashWeaver.getX() + forward.x * 2.0,
					ashWeaver.getY() + dy, ashWeaver.getZ() + forward.z * 2.0);
			if (!isSafeStandingPosition(level, landing)) continue;
			AABB atLanding = ashWeaver.getBoundingBox().move(landing.getX() + 0.5 - ashWeaver.getX(),
					landing.getY() - ashWeaver.getY(), landing.getZ() + 0.5 - ashWeaver.getZ());
			if (level.noCollision(ashWeaver, atLanding)) return landing;
		}
		return null;
	}

	@Nullable
	private static BlockPos findBoundaryDetour(ServerLevel level, AshWeaverEntity ashWeaver, Vec3 forward) {
		Vec3 side = new Vec3(-forward.z, 0.0, forward.x);
		for (int distance = 2; distance <= 4; distance++) {
			for (int sign : new int[] {-1, 1}) {
				BlockPos candidate = BlockPos.containing(ashWeaver.getX() + side.x * sign * distance,
						ashWeaver.getY(), ashWeaver.getZ() + side.z * sign * distance);
				if (!isSafeStandingPosition(level, candidate)) continue;
				BlockPos midway = BlockPos.containing(ashWeaver.getX() + side.x * sign,
						ashWeaver.getY(), ashWeaver.getZ() + side.z * sign);
				if (isSafeStandingPosition(level, midway)) return candidate;
			}
		}
		return null;
	}

	private static void tryJump(ServerLevel level, AshWeaverEntity ashWeaver, LivingEntity target, CompoundTag data, long gameTime) {
		if (!ashWeaver.onGround() || ashWeaver.isInWater() || ashWeaver.isShiftKeyDown() || target.isPassenger()
				|| gameTime < dataLong(data, KEY_NEXT_JUMP, 0L)) return;
		Vec3 toward = target.position().subtract(ashWeaver.position());
		toward = new Vec3(toward.x, 0.0, toward.z);
		if (toward.lengthSqr() < 1.0) return;
		toward = toward.normalize();
		BlockPos next = BlockPos.containing(ashWeaver.getX() + toward.x * 1.3, ashWeaver.getY(), ashWeaver.getZ() + toward.z * 1.3);
		if (!level.hasChunkAt(next)) return;
		boolean stepUp = ashWeaver.horizontalCollision && isSafeStandingPosition(level, next.above());
		boolean gap = level.getBlockState(next.below()).isAir();
		if (gap) return; // The follow boundary handler owns all gap decisions.
		boolean mimicJump = target instanceof Player player && ashWeaver.isSprinting()
				&& !player.onGround() && player.getDeltaMovement().y > 0.08
				&& ashWeaver.distanceToSqr(player) < 8.0 * 8.0
				&& isSafeStandingPosition(level, next) && ashWeaver.getRandom().nextFloat() < 0.55F;
		boolean followPlayer = target instanceof Player;
		boolean clearSprintRunway = !gap && !isNearLedge(level, ashWeaver)
				&& hasClearSprintRunway(level, ashWeaver, toward, 4);
		boolean sprintLeap = followPlayer && ashWeaver.isSprinting()
				&& ashWeaver.distanceToSqr(target) > 5.0 * 5.0
				&& clearSprintRunway
				&& ashWeaver.getDeltaMovement().horizontalDistanceSqr() > 0.035
				&& ashWeaver.getDeltaMovement().dot(toward) > 0.15
				&& !ashWeaver.horizontalCollision;
		boolean needsJump = stepUp || mimicJump || sprintLeap;
		if (needsJump) {
			// Vanilla sprint jumping adds its impulse along body yaw, not head/gaze yaw.
			if (sprintLeap) {
				float jumpYaw = (float) (Math.atan2(-toward.x, toward.z) * 180.0 / Math.PI);
				ashWeaver.setYRot(jumpYaw);
				ashWeaver.setYBodyRot(jumpYaw);
			}
			ashWeaver.getJumpControl().jump();
			// Retry on the first eligible grounded tick after the normal jump arc.
			data.putLong(KEY_NEXT_JUMP, gameTime + (sprintLeap ? 10L : 18L));
		}
	}

	private static boolean hasClearSprintRunway(ServerLevel level, AshWeaverEntity ashWeaver, Vec3 toward, int blocks) {
		for (int step = 1; step <= blocks; step++) {
			BlockPos floor = BlockPos.containing(ashWeaver.getX() + toward.x * step, ashWeaver.getY(),
					ashWeaver.getZ() + toward.z * step);
			if (!level.hasChunkAt(floor) || !isSafeStandingPosition(level, floor)) return false;
			AABB clearance = ashWeaver.getBoundingBox().move(toward.x * step, 0.0, toward.z * step)
					.expandTowards(0.0, 1.3, 0.0);
			if (!level.noCollision(ashWeaver, clearance)) return false;
		}
		return true;
	}

	private static void continueParkour(ServerLevel level, AshWeaverEntity ashWeaver, CompoundTag data, long gameTime) {
		long jumped = dataLong(data, KEY_PARKOUR_JUMPED, 0L);
		if (jumped == 0L) return;
		if (gameTime > jumped + 14L || (gameTime > jumped + 2L && ashWeaver.onGround())) {
			data.remove(KEY_PARKOUR_JUMPED);
			return;
		}
		BlockPos landing = BlockPos.containing(dataDouble(data, KEY_PARKOUR_X, ashWeaver.getX()),
				dataDouble(data, KEY_PARKOUR_Y, ashWeaver.getY()), dataDouble(data, KEY_PARKOUR_Z, ashWeaver.getZ()));
		if (!isSafeStandingPosition(level, landing)) {
			data.remove(KEY_PARKOUR_JUMPED);
			return;
		}
		Vec3 delta = new Vec3(dataDouble(data, KEY_PARKOUR_X, ashWeaver.getX()) - ashWeaver.getX(), 0.0,
				dataDouble(data, KEY_PARKOUR_Z, ashWeaver.getZ()) - ashWeaver.getZ());
		if (delta.horizontalDistanceSqr() < 0.25) return;
		Vec3 forward = delta.normalize().scale(0.27);
		Vec3 motion = ashWeaver.getDeltaMovement();
		ashWeaver.setDeltaMovement(forward.x, motion.y, forward.z);
		ashWeaver.setSprinting(true);
	}

	private static Vec3 getFormationTarget(AshWeaverEntity ashWeaver, Player player) {
		Vec3 look = player.getLookAngle();
		Vec3 horizontalLook = new Vec3(look.x, 0.0, look.z);
		if (horizontalLook.lengthSqr() < 0.001) horizontalLook = new Vec3(0.0, 0.0, 1.0);
		horizontalLook = horizontalLook.normalize();
		double sideSign = (ashWeaver.getUUID().hashCode() & 1) == 0 ? 1.0 : -1.0;
		Vec3 side = new Vec3(-horizontalLook.z, 0.0, horizontalLook.x).scale(1.35 * sideSign);
		// Small, slow spacing errors read as character without compromising long-range pursuit.
		double spacing = 2.75 + Math.max(0.0, -relationship(ashWeaver)) * 0.06
				+ 0.8 * Math.sin((ashWeaver.tickCount + (ashWeaver.getUUID().hashCode() & 255)) * 0.012);
		return player.position().subtract(horizontalLook.scale(spacing)).add(side);
	}

	private static void updateHumanSneak(ServerLevel level, AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		boolean closeAndStill = ashWeaver.distanceToSqr(player) <= 5.0 * 5.0
				&& ashWeaver.getNavigation().isDone() && ashWeaver.getSensing().hasLineOfSight(player);
		boolean nearbyCrouch = closeAndStill && player.isShiftKeyDown();
		long mirrorAt = dataLong(data, KEY_MIRROR_AT, 0L);
		if (nearbyCrouch) {
			if (mirrorAt == 0L) data.putLong(KEY_MIRROR_AT, gameTime + 12L + ashWeaver.getRandom().nextInt(12));
			else if (mirrorAt > 0L && gameTime >= mirrorAt) {
				if (ashWeaver.getRandom().nextFloat() < 0.35F)
					data.putLong(KEY_MIRROR_UNTIL, gameTime + 20L + ashWeaver.getRandom().nextInt(15));
				data.putLong(KEY_MIRROR_AT, -1L); // One response per player crouch, not a repeating loop.
			}
		} else {
			data.remove(KEY_MIRROR_AT);
		}

		long greetingUntil = dataLong(data, KEY_GREETING_UNTIL, -1L);
		if (greetingUntil == 0L && closeAndStill) {
			greetingUntil = gameTime + 42L;
			data.putLong(KEY_GREETING_UNTIL, greetingUntil);
		}
		if (greetingUntil > gameTime && !closeAndStill) {
			greetingUntil = -1L;
			data.putLong(KEY_GREETING_UNTIL, -1L);
		}
		boolean greeting = closeAndStill && greetingUntil > gameTime
				&& greetingUntil - gameTime > 28L;
		setSneaking(ashWeaver, greeting
				|| closeAndStill && gameTime < dataLong(data, KEY_MIRROR_UNTIL, 0L));
	}

	private static void updateSocialGaze(AshWeaverEntity ashWeaver, Player player, CompoundTag data, long gameTime) {
		if (!ashWeaver.getSensing().hasLineOfSight(player)) return;
		if (isRangedReady(player) && ashWeaver.distanceToSqr(player) <= 14.0 * 14.0
				&& !isOwnerAimingAtWeaver(player, ashWeaver)
				&& (gameTime < dataLong(data, KEY_AIM_WATCH_UNTIL, 0L)
						|| (Math.floorMod(gameTime, 30L) == Math.floorMod(ashWeaver.getId(), 30)
								&& ashWeaver.getRandom().nextFloat() < 0.35F))) {
			data.putLong(KEY_AIM_WATCH_UNTIL, gameTime + 20L);
			Vec3 watchedPoint = player.getEyePosition().add(player.getLookAngle().scale(12.0));
			ashWeaver.getLookControl().setLookAt(watchedPoint.x, watchedPoint.y, watchedPoint.z, 16.0F, 14.0F);
			return;
		}
		Vec3 toWeaver = ashWeaver.getEyePosition().subtract(player.getEyePosition());
		if (gameTime >= dataLong(data, KEY_NEXT_EYE_CONTACT, 0L)
				&& toWeaver.lengthSqr() > 0.1 && toWeaver.lengthSqr() < 9.0 * 9.0
				&& player.getLookAngle().dot(toWeaver.normalize()) > 0.94
				&& player.hasLineOfSight(ashWeaver)) {
			data.putLong(KEY_EYE_CONTACT_UNTIL, gameTime + 14L);
			data.putLong(KEY_NEXT_EYE_CONTACT, gameTime + 150L);
		}
		long attentionUntil = dataLong(data, KEY_EYE_CONTACT_UNTIL, 0L);
		if (gameTime < attentionUntil) {
			ashWeaver.getLookControl().setLookAt(player, 16.0F, 12.0F);
			return;
		}
		if (gameTime < attentionUntil + 14L) {
			Vec3 sideways = new Vec3(-player.getLookAngle().z, 0.0, player.getLookAngle().x);
			Vec3 glance = player.getEyePosition().add(sideways.scale(3.0));
			ashWeaver.getLookControl().setLookAt(glance.x, glance.y, glance.z, 12.0F, 10.0F);
			return;
		}
		if (gameTime < dataLong(data, KEY_DISTRACTION_UNTIL, 0L)) {
			try {
				Entity distraction = ((ServerLevel) ashWeaver.level()).getEntity(UUID.fromString(dataString(data, KEY_DISTRACTION_UUID, "")));
				if (distraction != null && distraction.isAlive() && ashWeaver.distanceToSqr(distraction) < 10.0 * 10.0) {
					ashWeaver.getLookControl().setLookAt(distraction, 12.0F, 10.0F);
					return;
				}
			} catch (IllegalArgumentException ignored) {
			}
		} else if (gameTime >= dataLong(data, KEY_NEXT_GAZE, 0L) && ashWeaver.getRandom().nextFloat() < 0.28F) {
			AABB nearby = ashWeaver.getBoundingBox().inflate(7.0, 3.0, 7.0);
			List<Entity> interesting = ashWeaver.level().getEntities(ashWeaver, nearby,
					candidate -> candidate.isAlive() && candidate != player
							&& (candidate instanceof ItemEntity || candidate instanceof LivingEntity && !(candidate instanceof AshWeaverEntity)));
			if (!interesting.isEmpty()) {
				Entity distraction = interesting.get(ashWeaver.getRandom().nextInt(interesting.size()));
				data.putString(KEY_DISTRACTION_UUID, distraction.getUUID().toString());
				data.putLong(KEY_DISTRACTION_UNTIL, gameTime + 12L + ashWeaver.getRandom().nextInt(14));
				data.putLong(KEY_NEXT_GAZE, gameTime + 100L + ashWeaver.getRandom().nextInt(100));
				ashWeaver.getLookControl().setLookAt(distraction, 12.0F, 10.0F);
				return;
			}
		}
		long nextGaze = dataLong(data, KEY_NEXT_GAZE, 0L);
		if (gameTime >= nextGaze) {
			data.putLong(KEY_GAZE_UNTIL, gameTime + 18L + ashWeaver.getRandom().nextInt(28));
			data.putLong(KEY_NEXT_GAZE, gameTime + 65L + ashWeaver.getRandom().nextInt(80));
			data.putInt(KEY_GAZE_MODE, ashWeaver.getRandom().nextFloat() < 0.72F ? 0 : 1);
		}
		if (gameTime <= dataLong(data, KEY_GAZE_UNTIL, 0L)) {
			if (dataInt(data, KEY_GAZE_MODE, 0) == 0) {
				ashWeaver.getLookControl().setLookAt(player, 14.0F, 12.0F);
			} else {
				Vec3 inspectPoint = player.getEyePosition().add(player.getLookAngle().scale(4.0));
				ashWeaver.getLookControl().setLookAt(inspectPoint.x, inspectPoint.y, inspectPoint.z, 12.0F, 10.0F);
			}
		}
	}

	private static BlockPos findPlantSite(ServerLevel level, AshWeaverEntity ashWeaver, Player player, Entity threat) {
		BlockPos best = null;
		double bestScore = Double.POSITIVE_INFINITY;
		Vec3 threatDirection = threat.position().subtract(player.position());
		if (threatDirection.horizontalDistanceSqr() < 0.001) threatDirection = new Vec3(1.0, 0.0, 0.0);
		double threatAngle = Math.atan2(threatDirection.z, threatDirection.x);
		BlockPos.MutableBlockPos mutable = new BlockPos.MutableBlockPos();
		for (int attempt = 0; attempt < 28; attempt++) {
			double angle = attempt < 9 ? threatAngle + (ashWeaver.getRandom().nextDouble() - 0.5) * 1.15
					: ashWeaver.getRandom().nextDouble() * Math.PI * 2.0;
			double radius = 2.25 + ashWeaver.getRandom().nextDouble() * 3.75;
			int candidateX = (int) Math.floor(player.getX() + Math.cos(angle) * radius);
			int candidateZ = (int) Math.floor(player.getZ() + Math.sin(angle) * radius);
			for (int verticalOffset = 3; verticalOffset >= -4; verticalOffset--) {
				mutable.set(candidateX, player.getBlockY() + verticalOffset, candidateZ);
				if (!isPlantSiteValid(level, mutable)) continue;
				double playerDistanceSq = mutable.distToCenterSqr(player.getX(), player.getY(), player.getZ());
				if (playerDistanceSq < 1.75 * 1.75) continue;
				double score = ashWeaver.distanceToSqr(mutable.getX() + 0.5, mutable.getY(), mutable.getZ() + 0.5)
						+ Math.abs(Math.sqrt(playerDistanceSq) - 3.5) * 3.0;
				if (score < bestScore) {
					bestScore = score;
					best = mutable.immutable();
				}
				break;
			}
		}
		return best;
	}


	private static boolean roseBudgetAvailable(ServerLevel level, Player player, long now) {
		CompoundTag record = bondRecord(level, player.getUUID().toString());
		long start = dataLong(record, "RoseWindow", now);
		int used = now - start >= 24000L ? 0 : dataInt(record, "RosesUsed", 0);
		return used < 8 && now >= dataLong(record, "NextRose", 0L);
	}

	private static void spendRoseBudget(ServerLevel level, Player player, long now) {
		String owner = player.getUUID().toString();
		CompoundTag record = bondRecord(level, owner).copy();
		long start = dataLong(record, "RoseWindow", now);
		int used = dataInt(record, "RosesUsed", 0);
		if (!record.contains("RoseWindow") || now - start >= 24000L) {
			record.putLong("RoseWindow", now);
			used = 0;
		}
		record.putInt("RosesUsed", used + 1);
		record.putLong("NextRose", now + 1200L);
		saveBond(level, owner, record);
	}

	private static boolean rosePlacementReachable(ServerLevel level, AshWeaverEntity ashWeaver, Player owner, BlockPos target) {
		if (!ashWeaver.isAlive() || !isCanonicalCompanion(ashWeaver, owner.getUUID().toString())
				|| owner.level() != level || !owner.isAlive() || owner.isSpectator()
				|| ashWeaver.distanceToSqr(Vec3.atCenterOf(target)) > 3.0 * 3.0
				|| owner.distanceToSqr(Vec3.atCenterOf(target)) > 8.0 * 8.0) return false;
		BlockHitResult sight = homeRay(level, ashWeaver, ashWeaver.getEyePosition(),
				Vec3.atBottomCenterOf(target).add(0.0, 0.35, 0.0));
		if (sight == null || sight.getType() == HitResult.Type.BLOCK) return false;
		return level.getEntitiesOfClass(LivingEntity.class, new AABB(target),
				living -> living.isAlive()).isEmpty();
	}

	private static boolean commitRosePlacement(ServerLevel level, AshWeaverEntity ashWeaver, Player owner, BlockPos target, long now) {
		if (!isPlantSiteValid(level, target) || !rosePlacementReachable(level, ashWeaver, owner, target)
				|| !roseBudgetAvailable(level, owner, now)
				|| countNearbyAshRoses(level, owner.blockPosition(), ROSE_COUNT_RADIUS_XZ, ROSE_COUNT_RADIUS_Y) >= MAX_NEARBY_ROSES
				|| !net.neoforged.neoforge.event.EventHooks.canEntityGrief(level, ashWeaver)) return false;
		BlockState rose = TheBackwoodsModBlocks.ASH_ROSE.get().defaultBlockState();
		var snapshot = net.neoforged.neoforge.common.util.BlockSnapshot.create(level.dimension(), level, target);
		if (!level.setBlock(target, rose, 2)) return false;
		boolean denied = net.neoforged.neoforge.event.EventHooks.onBlockPlace(ashWeaver, snapshot, Direction.UP);
		if (denied || !level.getBlockState(target).is(rose.getBlock()) || !rose.canSurvive(level, target)) {
			if (level.getBlockState(target).is(rose.getBlock())) snapshot.restore(3);
			return false;
		}
		level.updateNeighborsAt(target, rose.getBlock());
		if (!level.getBlockState(target).is(rose.getBlock())) return false;
		spendRoseBudget(level, owner, now);
		return true;
	}

	private static boolean isRoseBlock(BlockState state) {
		return state.is(TheBackwoodsModBlocks.ASH_ROSE.get()) || state.is(TheBackwoodsModBlocks.POTTED_ASH_ROSE.get());
	}

	private static boolean isPlantSiteValid(ServerLevel level, BlockPos target) {
		return isPlantBaseValid(level, target) && !hasNearbyAshRose(level, target, MIN_ROSE_SPACING);
	}

	private static boolean isPlantBaseValid(ServerLevel level, BlockPos target) {
		if (level.isOutsideBuildHeight(target) || level.isOutsideBuildHeight(target.above())
				|| !level.hasChunkAt(target) || !level.getWorldBorder().isWithinBounds(target)) return false;
		BlockState targetState = level.getBlockState(target);
		BlockState aboveState = level.getBlockState(target.above());
		BlockPos below = target.below();
		BlockState floor = level.getBlockState(below);
		if (!targetState.isAir() || !aboveState.isAir()) return false;
		if (!targetState.getFluidState().isEmpty() || !aboveState.getFluidState().isEmpty()) return false;
		return floor.isFaceSturdy(level, below, Direction.UP) && floor.getFluidState().isEmpty() && !isHazardousFloor(floor)
				&& TheBackwoodsModBlocks.ASH_ROSE.get().defaultBlockState().canSurvive(level, target);
	}

	private static boolean isHazardousFloor(BlockState state) {
		return state.is(Blocks.MAGMA_BLOCK) || state.is(Blocks.CACTUS) || state.is(Blocks.CAMPFIRE)
				|| state.is(Blocks.SOUL_CAMPFIRE) || state.is(Blocks.FIRE) || state.is(Blocks.SOUL_FIRE)
				|| state.is(Blocks.LAVA) || state.is(Blocks.POWDER_SNOW);
	}

	private static boolean trySafeTeleportNearPlayer(ServerLevel level, AshWeaverEntity ashWeaver, Player player) {
		long now = level.getGameTime();
		CompoundTag data = ashWeaver.getPersistentData();
		if (player.level() != level || !player.isAlive() || player.isSpectator() || player.isPassenger()
				|| !player.onGround() || !ashWeaver.onGround() || ashWeaver.isPassenger()
				|| !ashWeaver.hasLineOfSight(player) || ashWeaver.distanceToSqr(player) < 24.0 * 24.0
				|| dataBoolean(data, KEY_CHAT_STAY, false) || dataBoolean(data, "ashWeaverHomeStay", false)
				|| now < dataLong(data, "ashWeaverRecoveryTeleportAt", 0L)) return false;
		data.putLong("ashWeaverRecoveryTeleportAt", now + 1200L);
		Vec3 behind = player.getLookAngle().scale(-1.0);
		for (int attempt = 0; attempt < 20; attempt++) {
			double angle = Math.atan2(behind.z, behind.x) + (ashWeaver.getRandom().nextDouble() - 0.5) * Math.PI * 1.5;
			double radius = 3.0 + ashWeaver.getRandom().nextDouble() * 5.0;
			int candidateX = (int) Math.floor(player.getX() + Math.cos(angle) * radius);
			int candidateZ = (int) Math.floor(player.getZ() + Math.sin(angle) * radius);
			for (int verticalOffset = 3; verticalOffset >= -4; verticalOffset--) {
				BlockPos candidate = new BlockPos(candidateX, player.getBlockY() + verticalOffset, candidateZ);
				if (!homeFootprintSafe(level, ashWeaver, candidate)) continue;
				BlockHitResult visible = homeRay(level, ashWeaver, player.getEyePosition(),
						Vec3.atBottomCenterOf(candidate).add(0.0, 1.3, 0.0));
				if (visible == null || visible.getType() == HitResult.Type.BLOCK
						|| !level.getEntitiesOfClass(LivingEntity.class, new AABB(candidate).inflate(0.4),
								living -> living.isAlive() && living != ashWeaver).isEmpty()) continue;
				double targetX = candidate.getX() + 0.5;
				double targetY = candidate.getY();
				double targetZ = candidate.getZ() + 0.5;
				AABB movedBox = ashWeaver.getBoundingBox().move(targetX - ashWeaver.getX(), targetY - ashWeaver.getY(), targetZ - ashWeaver.getZ());
				if (!level.noCollision(ashWeaver, movedBox)) continue;
				ashWeaver.teleportTo(targetX, targetY, targetZ);
				ashWeaver.getNavigation().stop();
				return true;
			}
		}
		return false;
	}

	private static boolean isSafeStandingPosition(ServerLevel level, BlockPos feet) {
		if (level.isOutsideBuildHeight(feet) || level.isOutsideBuildHeight(feet.above())
				|| !level.hasChunkAt(feet) || !level.getWorldBorder().isWithinBounds(feet)) return false;
		BlockState feetState = level.getBlockState(feet);
		BlockState headState = level.getBlockState(feet.above());
		BlockPos floorPos = feet.below();
		BlockState floor = level.getBlockState(floorPos);
		return feetState.isAir() && headState.isAir() && feetState.getFluidState().isEmpty() && headState.getFluidState().isEmpty()
				&& floor.isFaceSturdy(level, floorPos, Direction.UP) && floor.getFluidState().isEmpty() && !isHazardousFloor(floor);
	}

	// Sample the collision surface, not just the block directly below the feet.
	private static double terrainDrop(ServerLevel level, AshWeaverEntity ashWeaver, BlockPos column) {
		for (int down = 0; down <= 6; down++) {
			BlockPos floorPos = column.below(down);
			if (!level.hasChunkAt(floorPos) || !level.getWorldBorder().isWithinBounds(floorPos)) return Double.POSITIVE_INFINITY;
			BlockState floor = level.getBlockState(floorPos);
			if (!floor.getFluidState().isEmpty() || isHazardousFloor(floor)) return Double.POSITIVE_INFINITY;
			VoxelShape shape = floor.getCollisionShape(level, floorPos);
			if (shape.isEmpty()) continue;
			double surfaceY = floorPos.getY() + shape.max(Direction.Axis.Y);
			double drop = ashWeaver.getY() - surfaceY;
			if (drop < -0.3) return -1.0; // An obstacle or an upward step, not a gap.
			AABB standing = ashWeaver.getBoundingBox().move(column.getX() + 0.5 - ashWeaver.getX(),
					surfaceY - ashWeaver.getY(), column.getZ() + 0.5 - ashWeaver.getZ());
			return level.noCollision(ashWeaver, standing) ? Math.max(0.0, drop) : -1.0;
		}
		return Double.POSITIVE_INFINITY;
	}

	private static boolean isNearLedge(ServerLevel level, AshWeaverEntity ashWeaver) {
		if (!ashWeaver.onGround()) return false;
		Vec3 travel = ashWeaver.getDeltaMovement();
		if (travel.horizontalDistanceSqr() < 0.005) {
			Path path = ashWeaver.getNavigation().getPath();
			if (path == null || path.isDone()) return false;
			travel = Vec3.atCenterOf(path.getNextNodePos()).subtract(ashWeaver.position());
		}
		travel = new Vec3(travel.x, 0.0, travel.z);
		if (travel.lengthSqr() < 0.001) return false;
		travel = travel.normalize();
		BlockPos probe = BlockPos.containing(ashWeaver.getX() + travel.x * 0.9, ashWeaver.getY(), ashWeaver.getZ() + travel.z * 0.9);
		if (!level.hasChunkAt(probe)) return false;
		return terrainDrop(level, ashWeaver, probe) > 2.05;
	}

	private static void applySneakEdgeBrake(ServerLevel level, AshWeaverEntity ashWeaver) {
		if (!ashWeaver.isShiftKeyDown() || !ashWeaver.onGround() || ashWeaver.isInWater()) return;
		Vec3 motion = ashWeaver.getDeltaMovement();
		if (motion.horizontalDistanceSqr() < 0.001) return;
		Vec3 direction = new Vec3(motion.x, 0.0, motion.z).normalize();
		double width = ashWeaver.getBbWidth() * 0.45;
		Vec3 side = new Vec3(-direction.z * width, 0.0, direction.x * width);
		for (int sign : new int[] {-1, 0, 1}) {
			BlockPos ahead = BlockPos.containing(ashWeaver.getX() + direction.x * 0.8 + side.x * sign,
					ashWeaver.getY(), ashWeaver.getZ() + direction.z * 0.8 + side.z * sign);
			if (terrainDrop(level, ashWeaver, ahead) > 2.05) {
				ashWeaver.getNavigation().stop();
				ashWeaver.setSprinting(false);
				ashWeaver.setDeltaMovement(0.0, motion.y, 0.0);
				return;
			}
		}
	}

	private static void separateFromOtherWeavers(ServerLevel level, AshWeaverEntity ashWeaver, long gameTime) {
		if (Math.floorMod(ashWeaver.tickCount + ashWeaver.getId(), 20) != 0) return;
		AABB bounds = ashWeaver.getBoundingBox().inflate(1.25, 0.5, 1.25);
		for (AshWeaverEntity other : level.getEntitiesOfClass(AshWeaverEntity.class, bounds, candidate -> candidate != ashWeaver && candidate.isAlive())) {
			Vec3 separation = ashWeaver.position().subtract(other.position());
			if (separation.horizontalDistanceSqr() > 0.001) {
				separation = new Vec3(separation.x, 0.0, separation.z).normalize().scale(0.06);
				ashWeaver.push(separation.x, 0.0, separation.z);
			}
			break;
		}
	}

	private static int countNearbyAshRoses(ServerLevel world, BlockPos center, int radiusXZ, int radiusY) {
		int count = 0;
		BlockPos.MutableBlockPos mutable = new BlockPos.MutableBlockPos();
		for (int offsetX = -radiusXZ; offsetX <= radiusXZ; offsetX++) {
			for (int offsetZ = -radiusXZ; offsetZ <= radiusXZ; offsetZ++) {
				mutable.set(center.getX() + offsetX, center.getY(), center.getZ() + offsetZ);
				if (!world.hasChunkAt(mutable)) continue;
				for (int offsetY = -radiusY; offsetY <= radiusY; offsetY++) {
					mutable.set(center.getX() + offsetX, center.getY() + offsetY, center.getZ() + offsetZ);
					if (isRoseBlock(world.getBlockState(mutable)) && ++count >= MAX_NEARBY_ROSES) return count;
				}
			}
		}
		return count;
	}

	private static boolean hasNearbyAshRose(ServerLevel world, BlockPos center, int radius) {
		BlockPos.MutableBlockPos mutable = new BlockPos.MutableBlockPos();
		for (int offsetX = -radius; offsetX <= radius; offsetX++) {
			for (int offsetZ = -radius; offsetZ <= radius; offsetZ++) {
				mutable.set(center.getX() + offsetX, center.getY(), center.getZ() + offsetZ);
				if (!world.hasChunkAt(mutable)) continue;
				for (int offsetY = -1; offsetY <= 1; offsetY++) {
					mutable.set(center.getX() + offsetX, center.getY() + offsetY, center.getZ() + offsetZ);
					if (isRoseBlock(world.getBlockState(mutable))) return true;
				}
			}
		}
		return false;
	}

	private static void handleDoorInteraction(ServerLevel level, AshWeaverEntity ashWeaver, CompoundTag data, int state, long gameTime) {
		if (!ashWeaver.getNavigation().getNodeEvaluator().canOpenDoors()) {
			ashWeaver.getNavigation().getNodeEvaluator().setCanOpenDoors(true);
		}
		boolean busy = data.getBoolean(KEY_CHAT_STAY) || state == STATE_PLANT || state == STATE_WAIT || state == STATE_WARN
				|| state == STATE_DODGE || state == STATE_DEFEND || state == STATE_ALERT;
		if (busy) return;
		if (data.contains(KEY_DOOR_PENDING)) {
			BlockPos pending = BlockPos.of(dataLong(data, KEY_DOOR_PENDING, 0L));
			if (busy || gameTime > dataLong(data, KEY_DOOR_FACE_AT, 0L) + 20L || !level.hasChunkAt(pending)) {
				data.remove(KEY_DOOR_PENDING);
			} else {
				BlockState pendingState = level.getBlockState(pending);
				if (pendingState.getBlock() instanceof DoorBlock door && door.type().canOpenByHand()
						&& !door.isOpen(pendingState) && ashWeaver.distanceToSqr(Vec3.atCenterOf(pending)) < 2.4 * 2.4) {
					faceDoor(ashWeaver, pending);
					ashWeaver.getNavigation().stop();
					ashWeaver.setSprinting(false);
					if (gameTime >= dataLong(data, KEY_DOOR_FACE_AT, gameTime) + 4L) {
						if (!net.neoforged.neoforge.event.EventHooks.canEntityGrief(level, ashWeaver)) {
							data.remove(KEY_DOOR_PENDING); return;
						}
						swingMainHand(level, ashWeaver);
						door.setOpen(ashWeaver, level, pendingState, pending, true);
						data.putLong(KEY_DOOR_POS, pending.asLong());
						data.putLong(KEY_DOOR_OPENED_AT, gameTime);
						data.remove(KEY_DOOR_PENDING);
					}
					return;
				}
				data.remove(KEY_DOOR_PENDING);
			}
		}
		if (data.contains(KEY_DOOR_POS)) {
			BlockPos previous = BlockPos.of(dataLong(data, KEY_DOOR_POS, 0L));
			if (!level.hasChunkAt(previous)) {
				data.remove(KEY_DOOR_POS);
			} else {
				BlockState previousState = level.getBlockState(previous);
				if (!(previousState.getBlock() instanceof DoorBlock door)) {
					data.remove(KEY_DOOR_POS);
				} else if (gameTime > dataLong(data, KEY_DOOR_OPENED_AT, gameTime) + 100L || !door.isOpen(previousState)) {
					data.remove(KEY_DOOR_POS);
				} else if (ashWeaver.distanceToSqr(Vec3.atCenterOf(previous)) > 3.0 * 3.0) {
					AABB doorway = new AABB(previous).inflate(0.8, 1.0, 0.8);
					if (level.getEntitiesOfClass(LivingEntity.class, doorway, Entity::isAlive).isEmpty()) {
						faceDoor(ashWeaver, previous);
						if (!data.contains(KEY_DOOR_CLOSE_AT)) data.putLong(KEY_DOOR_CLOSE_AT, gameTime);
						if (gameTime < dataLong(data, KEY_DOOR_CLOSE_AT, gameTime) + 4L) {
							ashWeaver.getNavigation().stop();
							return;
						}
						if (!net.neoforged.neoforge.event.EventHooks.canEntityGrief(level, ashWeaver)) {
							data.remove(KEY_DOOR_CLOSE_AT); data.remove(KEY_DOOR_POS); return;
						}
						swingMainHand(level, ashWeaver);
						door.setOpen(ashWeaver, level, previousState, previous, false);
						data.remove(KEY_DOOR_POS);
						data.remove(KEY_DOOR_CLOSE_AT);
					} else {
						data.remove(KEY_DOOR_CLOSE_AT);
					}
				}
			}
		}
		if (data.contains(KEY_DOOR_POS) || busy) return;
		Path path = ashWeaver.getNavigation().getPath();
		if (path == null || path.isDone()) return;
		for (int index = path.getNextNodeIndex(); index < Math.min(path.getNodeCount(), path.getNextNodeIndex() + 2); index++) {
			BlockPos next = path.getNodePos(index);
			if (!level.hasChunkAt(next) || ashWeaver.distanceToSqr(next.getX() + 0.5, next.getY(), next.getZ() + 0.5) > 2.4 * 2.4) continue;
			BlockState nextState = level.getBlockState(next);
			if (nextState.getBlock() instanceof DoorBlock door && door.type().canOpenByHand() && !door.isOpen(nextState)) {
				data.putLong(KEY_DOOR_PENDING, next.asLong());
				data.putLong(KEY_DOOR_FACE_AT, gameTime);
				faceDoor(ashWeaver, next);
				ashWeaver.getNavigation().stop();
				ashWeaver.setSprinting(false);
				return;
			}
		}
	}

	private static void faceDoor(AshWeaverEntity ashWeaver, BlockPos door) {
		double x = door.getX() + 0.5;
		double z = door.getZ() + 0.5;
		ashWeaver.getLookControl().setLookAt(x, door.getY() + 1.0, z, 14.0F, 12.0F);
		float desiredYaw = (float) (Mth.atan2(z - ashWeaver.getZ(), x - ashWeaver.getX()) * (180.0 / Math.PI)) - 90.0F;
		ashWeaver.setYRot(Mth.approachDegrees(ashWeaver.getYRot(), desiredYaw, 12.0F));
		ashWeaver.setYBodyRot(Mth.approachDegrees(ashWeaver.yBodyRot, desiredYaw, 12.0F));
	}

	private static void abortPlanting(AshWeaverEntity ashWeaver, CompoundTag data, long gameTime) {
		ashWeaver.getEntityData().set(AshWeaverEntity.DATA_roseCooldown,
				Math.max(40, ashWeaver.getEntityData().get(AshWeaverEntity.DATA_roseCooldown)));
		clearPlantingIntent(data);
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
		setState(data, STATE_FOLLOW, gameTime, 0L);
	}

	@Nullable
	private static BlockPos getTargetPos(CompoundTag data) {
		return data.contains(KEY_TARGET_POS) ? BlockPos.of(data.getLong(KEY_TARGET_POS)) : null;
	}

	private static void equipRose(AshWeaverEntity ashWeaver) {
		ashWeaver.setDropChance(net.minecraft.world.entity.EquipmentSlot.MAINHAND, 0.0F);
		if (ashWeaver.getItemInHand(InteractionHand.MAIN_HAND).getItem() != TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) {
			ashWeaver.setItemInHand(InteractionHand.MAIN_HAND, new ItemStack(TheBackwoodsModBlocks.ASH_ROSE.get()));
		}
	}

	private static void clearHeldRose(AshWeaverEntity ashWeaver) {
		if (ashWeaver.getItemInHand(InteractionHand.MAIN_HAND).getItem() == TheBackwoodsModBlocks.ASH_ROSE.get().asItem()) {
			ashWeaver.setItemInHand(InteractionHand.MAIN_HAND, ItemStack.EMPTY);
		}
	}

	private static void setSneaking(AshWeaverEntity ashWeaver, boolean sneaking) {
		if (ashWeaver.isShiftKeyDown() != sneaking) ashWeaver.setShiftKeyDown(sneaking);
		if (sneaking) {
			if (ashWeaver.getPose() != Pose.CROUCHING) ashWeaver.setPose(Pose.CROUCHING);
		}
		else if (ashWeaver.getPose() == Pose.CROUCHING) ashWeaver.setPose(Pose.STANDING);
	}

	private static void swingMainHand(ServerLevel level, AshWeaverEntity ashWeaver) {
		// Match the working Splinter tick procedure's vanilla hand-swing path.
		ashWeaver.swing(InteractionHand.MAIN_HAND, true);
	}

	private static void resetVisibleAction(AshWeaverEntity ashWeaver) {
		ashWeaver.getNavigation().stop();
		ashWeaver.setSprinting(false);
		clearHeldRose(ashWeaver);
		setSneaking(ashWeaver, false);
	}
